use crate::util::Rng;
pub const ABSENT: u16 = u16::MAX;
#[derive(Clone, Debug)]
pub struct Canon {
    pub syms: Vec<u16>,
    pub byte_of: Vec<u8>,
    pub sym_of: [u16; 256],
}
impl Canon {
    pub fn new(bytes: &[u8]) -> Canon {
        let mut sym_of = [ABSENT; 256];
        let mut byte_of: Vec<u8> = Vec::new();
        let mut syms: Vec<u16> = Vec::with_capacity(bytes.len());
        for &b in bytes {
            let slot = &mut sym_of[b as usize];
            if *slot == ABSENT {
                *slot = byte_of.len() as u16;
                byte_of.push(b);
            }
            syms.push(*slot);
        }
        Canon { syms, byte_of, sym_of }
    }
    pub fn n(&self) -> usize {
        self.byte_of.len()
    }
    pub fn len(&self) -> usize {
        self.syms.len()
    }
    pub fn is_empty(&self) -> bool {
        self.syms.is_empty()
    }
    pub fn bytes_of(&self, syms: &[u16]) -> Vec<u8> {
        syms.iter().filter_map(|&s| self.byte_of.get(s as usize).copied()).collect()
    }
}
pub fn permute(bytes: &[u8], perm: &[u8; 256]) -> Vec<u8> {
    bytes.iter().map(|&b| perm[b as usize]).collect()
}
pub fn random_perm(rng: &mut Rng) -> [u8; 256] {
    let mut v: Vec<u8> = (0..=255u8).collect();
    rng.shuffle(&mut v);
    let mut p = [0u8; 256];
    p.copy_from_slice(&v);
    p
}
pub fn inverse(perm: &[u8; 256]) -> [u8; 256] {
    let mut inv = [0u8; 256];
    for (i, &p) in perm.iter().enumerate() {
        inv[p as usize] = i as u8;
    }
    inv
}

pub struct Closure {
    pub ptgrs: bool,
    pub phi: bool,
    pub nl_text: bool,
    pub definitions: bool,
    pub self_described_e: bool,
    pub rendered_h: bool,
    pub intentions: bool,
    pub pretrained_prior: bool,
    pub nl_head: bool,
}
impl Closure {
    pub fn closed() -> Self {
        Closure {
            ptgrs: true,
            phi: true,
            nl_text: false,
            definitions: false,
            self_described_e: false,
            rendered_h: false,
            intentions: false,
            pretrained_prior: false,
            nl_head: false,
        }
    }
    pub fn is_closed(&self) -> bool {
        self.ptgrs
            && self.phi
            && !self.nl_text
            && !self.definitions
            && !self.self_described_e
            && !self.rendered_h
            && !self.intentions
            && !self.pretrained_prior
            && !self.nl_head
    }
    pub fn bits(&self) -> u32 {
        let f = [
            self.ptgrs,
            self.phi,
            self.nl_text,
            self.definitions,
            self.self_described_e,
            self.rendered_h,
            self.intentions,
            self.pretrained_prior,
            self.nl_head,
        ];
        let mut x = 0u32;
        for (i, b) in f.iter().enumerate() {
            if *b {
                x |= 1 << i;
            }
        }
        x
    }
    pub fn from_bits(x: u32) -> Self {
        let g = |i: usize| (x >> i) & 1 == 1;
        Closure {
            ptgrs: g(0),
            phi: g(1),
            nl_text: g(2),
            definitions: g(3),
            self_described_e: g(4),
            rendered_h: g(5),
            intentions: g(6),
            pretrained_prior: g(7),
            nl_head: g(8),
        }
    }
}

struct Rd<'a> {
    b: &'a [u8],
    i: usize,
}
impl<'a> Rd<'a> {
    fn take(&mut self, n: usize) -> io::Result<&'a [u8]> {
        if self.i.checked_add(n).map(|e| e > self.b.len()).unwrap_or(true) {
            return Err(io::Error::new(io::ErrorKind::InvalidData, "artefact tronque"));
        }
        let s = &self.b[self.i..self.i + n];
        self.i += n;
        Ok(s)
    }
    fn u64(&mut self) -> io::Result<u64> {
        let s = self.take(8)?;
        let mut a = [0u8; 8];
        a.copy_from_slice(s);
        Ok(u64::from_le_bytes(a))
    }
    fn u32(&mut self) -> io::Result<u32> {
        let s = self.take(4)?;
        let mut a = [0u8; 4];
        a.copy_from_slice(s);
        Ok(u32::from_le_bytes(a))
    }
    fn f32(&mut self) -> io::Result<f32> {
        let s = self.take(4)?;
        let mut a = [0u8; 4];
        a.copy_from_slice(s);
        Ok(f32::from_le_bytes(a))
    }
    fn u8(&mut self) -> io::Result<u8> {
        Ok(self.take(1)?[0])
    }
    fn boolean(&mut self) -> io::Result<bool> {
        Ok(self.u8()? != 0)
    }
    fn string(&mut self) -> io::Result<String> {
        let n = self.u64()? as usize;
        if n > 4096 {
            return Err(io::Error::new(io::ErrorKind::InvalidData, "etiquette trop longue"));
        }
        let s = self.take(n)?;
        String::from_utf8(s.to_vec())
            .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "etiquette non utf8"))
    }
}
pub fn params_fingerprint(p: &Params) -> u64 {
    let mut b = Vec::new();
    pu64(&mut b, p.nblocks() as u64);
    for (i, d) in p.data.iter().enumerate() {
        pu64(&mut b, p.shapes[i].r as u64);
        pu64(&mut b, p.shapes[i].c as u64);
        for x in d.iter() {
            pf32(&mut b, *x);
        }
    }
    fnv1a(&b)
}

pub struct Ledger {
    pub path: String,
    pub rows: Vec<(u64, String)>,
}
impl Ledger {
    pub fn load(path: &str) -> Ledger {
        let mut rows = Vec::new();
        if let Ok(s) = fs::read_to_string(path) {
            for line in s.lines() {
                let mut it = line.split('\t');
                if let (Some(a), Some(b)) = (it.next(), it.next()) {
                    if let Ok(fp) = u64::from_str_radix(a, 16) {
                        rows.push((fp, b.to_string()));
                    }
                }
            }
        }
        Ledger { path: path.to_string(), rows }
    }
    pub fn seen(&self, fp: u64, set: &str) -> bool {
        self.rows.iter().any(|&(f, ref s)| f == fp && s == set)
    }
    pub fn record(&mut self, fp: u64, set: &str) -> io::Result<()> {
        if self.seen(fp, set) {
            return Ok(());
        }
        if let Some(par) = Path::new(&self.path).parent() {
            if !par.as_os_str().is_empty() {
                fs::create_dir_all(par)?;
            }
        }
        let mut f = fs::OpenOptions::new().create(true).append(true).open(&self.path)?;
        writeln!(f, "{:016x}\t{}", fp, set)?;
        self.rows.push((fp, set.to_string()));
        Ok(())
    }
}

#[derive(Clone)]
pub struct Mech {
    pub m: Vec<usize>,
    pub pe: Vec<usize>,
    pub pc: Vec<usize>,
    pub gain: Vec<f32>,
    pub in_m: Vec<bool>,
}
#[inline]
fn mask(n: usize, s: &[usize]) -> Vec<bool> {
    let mut m = vec![false; n];
    for &i in s {
        if i < n {
            m[i] = true;
        }
    }
    m
}
fn topk(s: &[f32], k: usize) -> Vec<usize> {
    let mut idx: Vec<usize> = (0..s.len()).filter(|&i| s[i] > 0.0).collect();
    idx.sort_unstable_by(|&a, &b| {
        s[b].partial_cmp(&s[a]).unwrap_or(std::cmp::Ordering::Equal).then(a.cmp(&b))
    });
    idx.truncate(k);
    idx
}
pub fn build_mech(t: &Tape, f: &Fwd, v: &EffView, w: &Wiring, cfg: &PhiCfg) -> Mech {
    let n = v.n();
    let mut gain = vec![0.0f32; v.edges.len()];
    let mut score = vec![0.0f32; w.na];
    for a in 0..w.na {
        let gate = t.at(f.mech, a);
        let mut s = 0.0f32;
        for k in 0..P {
            let eid = w.ptr_eid[a * P + k];
            if eid == usize::MAX {
                continue;
            }
            let pr = t.at(f.ptr, a * (P + 1) + k + 1);
            let m = pr * gate;
            gain[eid] += m;
            s += m;
        }
        score[a] = s;
    }
    let m = topk(&score, cfg.mech);
    let in_m = mask(n, &m);
    let mut we = vec![0.0f32; n];
    let mut wc = vec![0.0f32; n];
    for (ei, e) in v.edges.iter().enumerate() {
        let g = gain[ei];
        if g <= 0.0 {
            continue;
        }
        if in_m[e.src as usize] {
            we[e.dst as usize] += g;
        }
        if in_m[e.dst as usize] {
            wc[e.src as usize] += g;
        }
    }
    let pe = topk(&we, cfg.purview);
    let pc = topk(&wc, cfg.purview);
    Mech { m, pe, pc, gain, in_m }
}
fn mask_itv(v: &EffView, gain: &[f32], keep: impl Fn(usize, usize) -> bool) -> Itv {
    let mut g = vec![0.0f32; v.edges.len()];
    for (ei, e) in v.edges.iter().enumerate() {
        if keep(e.src as usize, e.dst as usize) {
            g[ei] = gain[ei];
        }
    }
    Itv { edge_gain: Some(g), group_gain: [1.0; substrate::NW], rounds: 1 }
}
fn emd_norm(t: &mut Tape, a: usize, b: usize, k: usize) -> usize {
    let d = t.emd(a, b);
    t.scale(d, 1.0 / ((k.max(1) * (Q - 1)) as f32))
}
#[inline]
fn rep_rel(w: &Wiring, k: usize) -> u64 {
    (k * w.rel_count.max(1)) as u64
}
fn candidates(v: &EffView, srcs: &[usize], tgts: &[usize], mip: usize) -> Vec<Vec<bool>> {
    let n = v.n();
    let mut out: Vec<Vec<bool>> = Vec::new();
    let mut lab = vec![false; n];
    for i in 0..n {
        lab[i] = i % 2 == 1;
    }
    out.push(lab);
    let mut lab = vec![false; n];
    for i in 0..n {
        lab[i] = v.nodes[i].scale > 0;
    }
    out.push(lab);
    let mut uni: Vec<usize> = srcs.to_vec();
    uni.extend_from_slice(tgts);
    uni.sort_unstable();
    uni.dedup();
    let mut lab = vec![false; n];
    for (r, &i) in uni.iter().enumerate() {
        lab[i] = r >= uni.len() / 2;
    }
    out.push(lab);
    for &s in srcs.iter().take(3) {
        if let Some(&tt) = tgts.iter().find(|&&x| x != s) {
            let mut lab = vec![false; n];
            lab[s] = true;
            lab[tt] = true;
            out.push(lab);
        }
    }
    out.truncate(mip.max(1));
    out
}
fn side_phi(
    t: &mut Tape,
    p: &Params,
    h: usize,
    v: &EffView,
    w: &Wiring,
    srcs: &[usize],
    tgts: &[usize],
    gain: &[f32],
    effect: bool,
    cfg: &PhiCfg,
) -> (usize, usize, usize, u64) {
    let zero = t.scalar(0.0);
    let k = tgts.len();
    if srcs.len() < 2 || k < 2 {
        return (zero, zero, zero, 0);
    }
    let in_s = mask(v.n(), srcs);
    let in_t = mask(v.n(), tgts);
    let whole_itv = mask_itv(v, gain, |s, d| in_s[s] && in_t[d]);
    let whole = substrate::local_rep(t, p, h, w, tgts, &whole_itv, effect);
    let mut rel = rep_rel(w, k);
    let u = t.input(k, Q, &vec![1.0 / Q as f32; k * Q]);
    let info = emd_norm(t, whole, u, k);
    let mut best: Option<usize> = None;
    for lab in candidates(v, srcs, tgts, cfg.mip) {
        let s1: Vec<usize> = srcs.iter().cloned().filter(|&i| !lab[i]).collect();
        let s2: Vec<usize> = srcs.iter().cloned().filter(|&i| lab[i]).collect();
        let t1: Vec<usize> = tgts.iter().cloned().filter(|&i| !lab[i]).collect();
        let t2: Vec<usize> = tgts.iter().cloned().filter(|&i| lab[i]).collect();
        if s1.is_empty() || s2.is_empty() || t1.is_empty() || t2.is_empty() {
            continue;
        }
        let i1 = mask_itv(v, gain, |s, d| !lab[s] && !lab[d] && in_s[s] && in_t[d]);
        let i2 = mask_itv(v, gain, |s, d| lab[s] && lab[d] && in_s[s] && in_t[d]);
        let ra = substrate::local_rep(t, p, h, w, &t1, &i1, effect);
        let rb = substrate::local_rep(t, p, h, w, &t2, &i2, effect);
        rel += rep_rel(w, t1.len()) + rep_rel(w, t2.len());
        let cat = t.concat_rows(ra, rb);
        let mut pos = vec![0usize; k];
        let (mut c1, mut c2) = (0usize, 0usize);
        for (i, &nd) in tgts.iter().enumerate() {
            if !lab[nd] {
                pos[i] = c1;
                c1 += 1;
            } else {
                pos[i] = t1.len() + c2;
                c2 += 1;
            }
        }
        let rp = t.gather_rows(cat, &pos);
        let d = emd_norm(t, whole, rp, k);
        best = Some(match best {
            None => d,
            Some(b) => t.min2(b, d),
        });
    }
    let integ = best.unwrap_or_else(|| t.scalar(0.0));
    let ph = t.min2(info, integ);
    (ph, info, integ, rel)
}
fn shift_srcs(v: &EffView, w: &Wiring, m: &[usize], kind: u8, fallback_adj: bool) -> Vec<usize> {
    let mut out = Vec::with_capacity(m.len());
    for &a in m {
        let mut found = None;
        for &ei in w.inc[a].iter() {
            let e = v.edges[ei];
            if e.kind == kind && e.src as usize != a {
                found = Some(e.src as usize);
                break;
            }
        }
        match found {
            Some(x) => out.push(x),
            None => {
                if fallback_adj && w.na > 1 {
                    out.push(if a + 1 < w.na { a + 1 } else { a - 1 });
                } else {
                    out.push(a);
                }
            }
        }
    }
    out.sort_unstable();
    out.dedup();
    out
}
fn stability(
    t: &mut Tape,
    p: &Params,
    h: usize,
    v: &EffView,
    w: &Wiring,
    mech: &Mech,
    whole: usize,
) -> (usize, u64) {
    let k = mech.pe.len();
    if k == 0 || mech.m.is_empty() {
        return (t.scalar(1.0), 0);
    }
    let in_pe = mask(v.n(), &mech.pe);
    let in_m = mask(v.n(), &mech.m);
    let mut rel = 0u64;
    let zero = t.scalar(0.0);
    let one = t.scalar(1.0);
    let mut best = one;
    let mut g2 = mech.gain.clone();
    let mut bi = usize::MAX;
    let mut bv = 0.0f32;
    for (ei, e) in v.edges.iter().enumerate() {
        if in_m[e.src as usize] && in_pe[e.dst as usize] && mech.gain[ei] > bv {
            bv = mech.gain[ei];
            bi = ei;
        }
    }
    if bi != usize::MAX {
        g2[bi] = 0.0;
    }
    let i_ref = mask_itv(v, &g2, |s, d| in_m[s] && in_pe[d]);
    let r_ref = substrate::local_rep(t, p, h, w, &mech.pe, &i_ref, true);
    rel += rep_rel(w, k);
    let d = emd_norm(t, whole, r_ref, k);
    let nd = t.neg(d);
    let s1 = t.add_const(nd, 1.0);
    let s1 = t.max2(zero, s1);
    best = t.min2(best, s1);
    let md = shift_srcs(v, w, &mech.m, REL_ADJ, true);
    let in_md = mask(v.n(), &md);
    let i_dis = mask_itv(v, &mech.gain, |s, d| in_md[s] && in_pe[d]);
    let r_dis = substrate::local_rep(t, p, h, w, &mech.pe, &i_dis, true);
    rel += rep_rel(w, k);
    let d = emd_norm(t, whole, r_dis, k);
    let nd = t.neg(d);
    let s2 = t.add_const(nd, 1.0);
    let s2 = t.max2(zero, s2);
    best = t.min2(best, s2);
    let ma = shift_srcs(v, w, &mech.m, REL_CONTAIN, false);
    let in_ma = mask(v.n(), &ma);
    let i_abs = mask_itv(v, &mech.gain, |s, d| in_ma[s] && in_pe[d]);
    let r_abs = substrate::local_rep(t, p, h, w, &mech.pe, &i_abs, true);
    rel += rep_rel(w, k);
    let d = emd_norm(t, whole, r_abs, k);
    let nd = t.neg(d);
    let s3 = t.add_const(nd, 1.0);
    let s3 = t.max2(zero, s3);
    best = t.min2(best, s3);
    (best, rel)
}

pub fn evaluate(
    t: &mut Tape,
    p: &Params,
    v: &EffView,
    w: &Wiring,
    f: &Fwd,
    mech: &Mech,
    cfg: &PhiCfg,
) -> PhiVal {
    let (ph_e, info_e, integ_e, rel_e) =
        side_phi(t, p, f.h, v, w, &mech.m, &mech.pe, &mech.gain, true, cfg);
    let (ph_c, info_c, integ_c, rel_c) =
        side_phi(t, p, f.h, v, w, &mech.pc, &mech.m, &mech.gain, false, cfg);
    let bigphi = t.min2(ph_e, ph_c);
    let in_m = mask(v.n(), &mech.m);
    let in_pe = mask(v.n(), &mech.pe);
    let witv = mask_itv(v, &mech.gain, |s, d| in_m[s] && in_pe[d]);
    let whole = substrate::local_rep(t, p, f.h, w, &mech.pe, &witv, true);
    let (stab, rel_s) = if cfg.perturb {
        stability(t, p, f.h, v, w, mech, whole)
    } else {
        (t.scalar(1.0), 0)
    };
    let sharp = t.mul(bigphi, stab);
    let mut sel = vec![0.0f32; P + 1];
    sel[0] = 1.0;
    let selv = t.input(P + 1, 1, &sel);
    let nullcol = t.matmul(f.ptr, selv);
    let one = t.input(w.na, 1, &vec![1.0; w.na]);
    let hit = t.sub(one, nullcol);
    let cov = t.mean_all(hit);
    let atoms: Vec<usize> = (0..w.na).collect();
    let res_a = t.gather_rows(f.res, &atoms);
    let res_vec = t.mul(nullcol, res_a);
    let mut res_node = 0usize;
    let mut rv = f32::NEG_INFINITY;
    for a in 0..w.na {
        let x = t.at(res_vec, a);
        if x > rv {
            rv = x;
            res_node = a;
        }
    }
    PhiVal {
        sharp,
        bigphi,
        stab,
        cov,
        res_vec,
        info_e,
        integ_e,
        info_c,
        integ_c,
        sharp_f: t.scalar_of(sharp),
        bigphi_f: t.scalar_of(bigphi),
        stab_f: t.scalar_of(stab),
        cov_f: t.scalar_of(cov),
        res_node,
        rel: rel_e + rel_c + rel_s + rep_rel(w, mech.pe.len()) + (w.na * (P + 1)) as u64,
    }
}
pub fn residual_gain(
    t: &mut Tape,
    p: &Params,
    v: &EffView,
    w: &Wiring,
    f: &Fwd,
    mech: &Mech,
    cfg: &PhiCfg,
    extra: usize,
    base: usize,
) -> usize {
    if extra >= w.na || mech.in_m[extra] {
        return t.scalar(0.0);
    }
    let mut k0 = 0usize;
    let mut bw = f32::NEG_INFINITY;
    for k in 0..P {
        let x = w.ptr_w[extra * P + k];
        if x > bw {
            bw = x;
            k0 = k;
        }
    }
    let eid = w.ptr_eid[extra * P + k0];
    if eid == usize::MAX {
        return t.scalar(0.0);
    }
    let nz = mech.gain.iter().filter(|&&x| x > 0.0).count().max(1);
    let mean = mech.gain.iter().sum::<f32>() / nz as f32;
    let mut g2 = mech.gain.clone();
    if g2[eid] < mean {
        g2[eid] = mean;
    }
    let mut m2 = mech.m.clone();
    m2.push(extra);
    let (a, _, _, _) = side_phi(t, p, f.h, v, w, &m2, &mech.pe, &g2, true, cfg);
    let (b, _, _, _) = side_phi(t, p, f.h, v, w, &mech.pc, &m2, &g2, false, cfg);
    let with = t.min2(a, b);
    t.sub(with, base)
}

    pub root: usize,
    pub weight: f64,
    pub tag: u16,

pub struct Sample {
    pub syms: Vec<Sym>,
    pub spans: Vec<(u16, usize, usize)>,
    pub steps: usize,

#[derive(Clone, Copy, Debug)]
enum Kind {
    Atom(Sym),
    Cat,
    Call(OpId),
}
struct Gn {
    kind: Kind,
    kids: Vec<u32>,
    tag: u16,
    done: bool,

impl Ptgrs {
    pub fn new(n_atoms: usize) -> Ptgrs {
        Ptgrs { n_atoms, ops: Vec::new(), start: 0 }
    }
    pub fn add_op(&mut self) -> OpId {
        self.ops.push(Vec::new());
        (self.ops.len() - 1) as OpId
    }
    pub fn add_rule(&mut self, op: OpId, rule: Rule) {
        self.ops[op as usize].push(rule);
    }
    pub fn validate(&self) -> Result<(), String> {
        if self.ops.is_empty() {
            return Err("aucun op".to_string());
        }
        if (self.start as usize) >= self.ops.len() {
            return Err("start hors bornes".to_string());
        }
        for (o, alts) in self.ops.iter().enumerate() {
            for (r, rule) in alts.iter().enumerate() {
                if rule.root >= rule.nodes.len() {
                    return Err(format!("op {o} règle {r}: racine hors bornes"));
                }
                if !(rule.weight >= 0.0) {
                    return Err(format!("op {o} règle {r}: poids invalide"));
                }
                for (i, nd) in rule.nodes.iter().enumerate() {
                    match nd {
                        RNode::Atom(a) => {
                            if (*a as usize) >= self.n_atoms {
                                return Err(format!("op {o} règle {r}: atome hors bornes"));
                            }
                        }
                        RNode::Call(c) => {
                            if (*c as usize) >= self.ops.len() {
                                return Err(format!("op {o} règle {r}: appel hors bornes"));
                            }
                        }
                        RNode::Cat(ks) => {
                            for &k in ks {
                                if k >= i {
                                    return Err(format!("op {o} règle {r}: cycle ou ordre"));
                                }
                            }
                        }
                    }
                }
            }
        }
        Ok(())
    }
    pub fn relabel(&self, map: &[Sym], n_new: usize) -> Ptgrs {
        let mut g = self.clone();
        g.n_atoms = n_new;
        for alts in g.ops.iter_mut() {
            for r in alts.iter_mut() {
                for nd in r.nodes.iter_mut() {
                    if let RNode::Atom(a) = nd {
                        *a = map[*a as usize];
                    }
                }
            }
        }
        g
    }
    pub fn generate(&self, rng: &mut Rng, lim: &Limits) -> Option<Sample> {
        let mut g: Vec<Gn> = vec![Gn { kind: Kind::Call(self.start), kids: Vec::new(), tag: 0, done: false }];
        let mut steps = 0usize;
        if !self.reduce(&mut g, 0, rng, lim, &mut steps, 0) {
            return None;
        }
        let mut syms: Vec<Sym> = Vec::new();
        let mut spans: Vec<(u16, usize, usize)> = Vec::new();
        if !Self::flatten(&g, 0, &mut syms, &mut spans, lim.max_len) {
            return None;
        }
        Some(Sample { syms, spans, steps })
    }
    fn choose_rule(&self, op: usize, rng: &mut Rng, depth: usize, lim: &Limits) -> usize {
        let alts = &self.ops[op];
        if alts.len() == 1 {
            return 0;
        }
        let mut w: Vec<f64> = alts.iter().map(|r| r.weight.max(0.0)).collect();
        if depth * 2 >= lim.max_depth && alts.iter().any(|r| !r.has_call()) {
            for (i, r) in alts.iter().enumerate() {
                if r.has_call() {
                    w[i] = 0.0;
                }
            }
        }
        rng.weighted(&w)
    }
    fn reduce(&self, g: &mut Vec<Gn>, n: usize, rng: &mut Rng, lim: &Limits, steps: &mut usize, depth: usize) -> bool {
        if g[n].done {
            return true;
        }
        if depth > lim.max_depth {
            return false;
        }
        loop {
            let op = match g[n].kind {
                Kind::Call(op) => op as usize,
                _ => break,
            };
            if *steps >= lim.max_steps || self.ops[op].is_empty() {
                return false;
            }
            *steps += 1;
            let ri = self.choose_rule(op, rng, depth, lim);
            let rule = &self.ops[op][ri];
            let base = g.len();
            for nd in &rule.nodes {
                let (kind, kids) = match nd {
                    RNode::Atom(a) => (Kind::Atom(*a), Vec::new()),
                    RNode::Call(o) => (Kind::Call(*o), Vec::new()),
                    RNode::Cat(ix) => (Kind::Cat, ix.iter().map(|&i| (base + i) as u32).collect()),
                };
                g.push(Gn { kind, kids, tag: 0, done: false });
            }
            let r = base + rule.root;
            let (k, ks) = (g[r].kind, g[r].kids.clone());
            g[n].kind = k;
            g[n].kids = ks;
            if rule.tag != 0 {
                g[n].tag = rule.tag;
            }
        }
        g[n].done = true;
        let kids = g[n].kids.clone();
        for k in kids {
            if !self.reduce(g, k as usize, rng, lim, steps, depth + 1) {
                return false;
            }
        }
        true
    }
    fn flatten(g: &[Gn], n: usize, out: &mut Vec<Sym>, spans: &mut Vec<(u16, usize, usize)>, max_len: usize) -> bool {
        let start = out.len();
        match g[n].kind {
            Kind::Atom(a) => {
                if out.len() >= max_len {
                    return false;
                }
                out.push(a);
            }
            Kind::Cat => {
                for &k in &g[n].kids {
                    if !Self::flatten(g, k as usize, out, spans, max_len) {
                        return false;
                    }
                }
            }
            Kind::Call(_) => return false,
        }
        if g[n].tag != 0 {
            spans.push((g[n].tag, start, out.len()));
        }
        true
    }
    pub fn dump(&self) -> String {
        let mut s = String::new();
        for (o, alts) in self.ops.iter().enumerate() {
            for r in alts {
                let body: Vec<String> = r
                    .nodes
                    .iter()
                    .enumerate()
                    .map(|(i, n)| match n {
                        RNode::Atom(a) => format!("n{i}=a{a}"),
                        RNode::Call(c) => format!("n{i}=@{c}"),
                        RNode::Cat(k) => format!("n{i}=cat{k:?}"),
                    })
                    .collect();
                s.push_str(&format!("op{o} w={:.3} tag={} root=n{} : {}\n", r.weight, r.tag, r.root, body.join(" ")));
            }
        }
        s
    }
}

pub enum Part {
    A(Sym),
    C(OpId),
    S(usize),

pub struct Builder {
    pub g: Ptgrs,
}
impl Builder {
    pub fn new(n_atoms: usize) -> Builder {
        let mut g = Ptgrs::new(n_atoms);
        g.add_op();
        Builder { g }
    }
    pub fn op(&mut self) -> OpId {
        self.g.add_op()
    }
    pub fn word(rng: &mut Rng, n_atoms: usize, lo: usize, hi: usize) -> Vec<Sym> {
        let len = rng.range(lo, hi);
        (0..len).map(|_| rng.below(n_atoms) as Sym).collect()
    }
    pub fn atoms(w: &[Sym]) -> Vec<Part> {
        w.iter().map(|&a| Part::A(a)).collect()
    }
    pub fn rule(&mut self, op: OpId, weight: f64, shared: &[OpId], parts: &[Part]) {
        let mut nodes: Vec<RNode> = shared.iter().map(|&o| RNode::Call(o)).collect();
        let mut kids: Vec<usize> = Vec::new();
        for p in parts {
            match p {
                Part::S(k) => kids.push(*k),
                Part::A(a) => {
                    nodes.push(RNode::Atom(*a));
                    kids.push(nodes.len() - 1);
                }
                Part::C(o) => {
                    nodes.push(RNode::Call(*o));
                    kids.push(nodes.len() - 1);
                }
            }
        }
        nodes.push(RNode::Cat(kids));
        let root = nodes.len() - 1;
        self.g.add_rule(op, Rule { nodes, root, weight, tag: 0 });
    }
    pub fn chunk(&mut self, w: &[Sym]) -> OpId {
        let op = self.op();
        self.rule(op, 1.0, &[], &Self::atoms(w));
        op
    }
    pub fn choice(&mut self, alts: &[Vec<Sym>], weights: &[f64]) -> OpId {
        let op = self.op();
        for (i, w) in alts.iter().enumerate() {
            self.rule(op, weights[i], &[], &Self::atoms(w));
        }
        op
    }
    pub fn frame(&mut self, pre: &[Sym], slot: OpId, post: &[Sym]) -> OpId {
        let op = self.op();
        let mut parts = Self::atoms(pre);
        parts.push(Part::C(slot));
        parts.extend(Self::atoms(post));
        self.rule(op, 1.0, &[], &parts);
        op
    }
    pub fn agree(&mut self, pre: &[Sym], slot: OpId, mid: &[Sym], post: &[Sym]) -> OpId {
        let op = self.op();
        let mut parts = Self::atoms(pre);
        parts.push(Part::S(0));
        parts.extend(Self::atoms(mid));
        parts.push(Part::S(0));
        parts.extend(Self::atoms(post));
        self.rule(op, 1.0, &[slot], &parts);
        op
    }
    pub fn new_slot(&mut self, rng: &mut Rng) -> OpId {
        let na = self.g.n_atoms;
        let k = rng.range(2, 4);
        let alts: Vec<Vec<Sym>> = (0..k).map(|_| Self::word(rng, na, 1, 3)).collect();
        let ws: Vec<f64> = (0..k).map(|_| 0.3 + rng.f64()).collect();
        self.choice(&alts, &ws)
    }
    pub fn finish(mut self, top: &[Part]) -> Ptgrs {
        self.rule(0, 1.0, &[], top);
        self.g
    }
}

fn frag_repeat(b: &mut Builder, rng: &mut Rng) -> Vec<Part> {
    let na = b.g.n_atoms;
    let nm = rng.range(1, 2);
    let mut ms: Vec<OpId> = Vec::new();
    for _ in 0..nm {
        let w = Builder::word(rng, na, 2, 4);
        ms.push(b.chunk(&w));
    }
    let mut top: Vec<Part> = Vec::new();
    for _ in 0..rng.range(3, 7) {
        top.push(Part::C(ms[rng.below(ms.len())]));
        if rng.chance(0.5) {
            top.push(Part::A(rng.below(na) as Sym));
        }
    }
    top
}
fn frag_frame(b: &mut Builder, rng: &mut Rng) -> Vec<Part> {
    let na = b.g.n_atoms;
    let pre = Builder::word(rng, na, 1, 3);
    let post = Builder::word(rng, na, 1, 2);
    let slot = b.new_slot(rng);
    let f = b.frame(&pre, slot, &post);
    (0..rng.range(4, 8)).map(|_| Part::C(f)).collect()
}
fn frag_agree(b: &mut Builder, rng: &mut Rng) -> Vec<Part> {
    let na = b.g.n_atoms;
    let pre = Builder::word(rng, na, 1, 2);
    let mid = Builder::word(rng, na, 1, 2);
    let post = Builder::word(rng, na, 1, 2);
    let slot = b.new_slot(rng);
    let t = b.agree(&pre, slot, &mid, &post);
    (0..rng.range(3, 6)).map(|_| Part::C(t)).collect()
}
fn frag_hier(b: &mut Builder, rng: &mut Rng, depth: usize) -> Vec<Part> {
    let na = b.g.n_atoms;
    let mut level: Vec<OpId> = Vec::new();
    for _ in 0..rng.range(2, 3) {
        let w = Builder::word(rng, na, 2, 3);
        level.push(b.chunk(&w));
    }
    for _ in 0..depth {
        let mut next: Vec<OpId> = Vec::new();
        for _ in 0..rng.range(1, 2) {
            let op = b.op();
            let k = rng.range(2, 3);
            let parts: Vec<Part> = (0..k).map(|_| Part::C(level[rng.below(level.len())])).collect();
            b.rule(op, 1.0, &[], &parts);
            next.push(op);
        }
        level = next;
    }
    (0..rng.range(3, 5)).map(|_| Part::C(level[rng.below(level.len())])).collect()
}
fn frag_list(b: &mut Builder, rng: &mut Rng) -> Vec<Part> {
    let na = b.g.n_atoms;
    let slot = b.new_slot(rng);
    let sep = rng.below(na) as Sym;
    let r = rng.range(4, 8);
    let mut top: Vec<Part> = Vec::new();
    for i in 0..r {
        top.push(Part::C(slot));
        if i + 1 < r {
            top.push(Part::A(sep));
        }
    }
    top
}
fn fragment(b: &mut Builder, rng: &mut Rng, idx: usize) -> Vec<Part> {
    match idx % 5 {
        0 => frag_repeat(b, rng),
        1 => frag_frame(b, rng),
        2 => frag_agree(b, rng),
        3 => {
            let d = rng.range(1, 2);
            frag_hier(b, rng, d)
        }
        _ => frag_list(b, rng),
    }
}
pub fn build_cause(rng: &mut Rng, fam: Family, n_atoms: usize) -> Ptgrs {
    let mut b = Builder::new(n_atoms);
    let top: Vec<Part> = match fam {
        Family::Repeat => fragment(&mut b, rng, 0),
        Family::Frame => fragment(&mut b, rng, 1),
        Family::Agree => fragment(&mut b, rng, 2),
        Family::Hier => fragment(&mut b, rng, 3),
        Family::List => fragment(&mut b, rng, 4),
        Family::Mixed => {
            let i = rng.below(5);
            let j = (i + 1 + rng.below(4)) % 5;
            let mut a = fragment(&mut b, rng, i);
            let c = fragment(&mut b, rng, j);
            if rng.chance(0.5) {
                a.extend(c);
                a
            } else {
                let mut c = c;
                c.extend(a);
                c
            }
        }
    };
    b.finish(&top)
}

    pub spans: Vec<(u16, usize, usize)>,
}
impl Instance {
    pub fn cause_in_canon(&self, canon: &Canon) -> Ptgrs {
        let mut map: Vec<Sym> = vec![0; self.cause.n_atoms];
        let mut extra = canon.n();
        for a in 0..self.cause.n_atoms {
            let s = canon.sym_of[self.atom_byte[a] as usize];
            if s != ABSENT {
                map[a] = s as Sym;
            } else {
                map[a] = extra as Sym;
                extra += 1;
            }
        }
        self.cause.relabel(&map, extra)
    }
}
pub fn random_alphabet(rng: &mut Rng, n: usize) -> Vec<u8> {
    let mut pool: Vec<u8> = if rng.chance(0.5) { (33..=126u8).collect() } else { (0..=255u8).collect() };
    rng.shuffle(&mut pool);
    pool.truncate(n.min(pool.len()));
    pool
}
pub fn draw_instance(rng: &mut Rng, fam: Family, len_lo: usize, len_hi: usize) -> Option<Instance> {
    let n_atoms = rng.range(4, 10);
    let cause = build_cause(rng, fam, n_atoms);
    if cause.validate().is_err() {
        return None;
    }
    let lim = Limits { max_len: len_hi, ..Limits::default() };
    let s = cause.generate(rng, &lim)?;
    if s.syms.len() < len_lo || s.syms.len() > len_hi {
        return None;
    }
    let atom_byte = random_alphabet(rng, n_atoms);
    let e: Vec<u8> = s.syms.iter().map(|&a| atom_byte[a as usize]).collect();
    let mut seen = [false; 256];
    let mut distinct = 0;
    for &b in &e {
        if !seen[b as usize] {
            seen[b as usize] = true;
            distinct += 1;
        }
    }
    if distinct < 3 {
        return None;
    }
    Some(Instance { e, cause, atom_byte, spans: s.spans })
}

pub struct ObsDist {
    pub strings: Option<f64>,
    pub ngrams: f64,
    pub lengths: f64,
}
impl ObsDist {
    pub fn total(&self) -> f64 {
        match self.strings {
            Some(s) => (s + self.ngrams + self.lengths) / 3.0,
            None => (self.ngrams + self.lengths) / 2.0,
        }
    }
    pub fn quality(&self) -> f64 {
        1.0 - self.total()
    }
}
fn bump<K: Ord>(m: &mut BTreeMap<K, f64>, k: K) {
    *m.entry(k).or_insert(0.0) += 1.0;
}
fn js<K: Ord + Clone>(p: &BTreeMap<K, f64>, q: &BTreeMap<K, f64>) -> f64 {
    let sp: f64 = p.values().sum();
    let sq: f64 = q.values().sum();
    if sp <= 0.0 || sq <= 0.0 {
        return 1.0;
    }
    let mut both: BTreeMap<K, (f64, f64)> = BTreeMap::new();
    for (k, v) in p {
        both.entry(k.clone()).or_insert((0.0, 0.0)).0 = v / sp;
    }
    for (k, v) in q {
        both.entry(k.clone()).or_insert((0.0, 0.0)).1 = v / sq;
    }
    let mut d = 0.0;
    for (_, &(a, b)) in both.iter() {
        let m = 0.5 * (a + b);
        if a > 0.0 {
            d += 0.5 * a * (a / m).log2();
        }
        if b > 0.0 {
            d += 0.5 * b * (b / m).log2();
        }
    }
    d.clamp(0.0, 1.0)
}
fn profile(samples: &[Vec<Sym>], n: usize) -> BTreeMap<Vec<Sym>, f64> {
    let mut m: BTreeMap<Vec<Sym>, f64> = BTreeMap::new();
    for s in samples {
        if s.len() >= n {
            for w in s.windows(n) {
                bump(&mut m, w.to_vec());
            }
        }
    }
    m
}
pub fn sample_many(sys: &Ptgrs, rng: &mut Rng, n: usize, lim: &Limits) -> Vec<Vec<Sym>> {
    let mut out: Vec<Vec<Sym>> = Vec::new();
    let mut tries = 0;
    while out.len() < n && tries < n * 4 {
        tries += 1;
        if let Some(s) = sys.generate(rng, lim) {
            out.push(s.syms);
        }
    }
    out
}
pub fn obs_dist(a: &Ptgrs, b: &Ptgrs, rng: &mut Rng, n: usize, lim: &Limits) -> Option<ObsDist> {
    let sa = sample_many(a, rng, n, lim);
    let sb = sample_many(b, rng, n, lim);
    if sa.is_empty() || sb.is_empty() || sa.len() < n / 2 || sb.len() < n / 2 {
        return None;
    }
    let mut ma: BTreeMap<Vec<Sym>, f64> = BTreeMap::new();
    let mut mb: BTreeMap<Vec<Sym>, f64> = BTreeMap::new();
    for s in &sa {
        bump(&mut ma, s.clone());
    }
    for s in &sb {
        bump(&mut mb, s.clone());
    }
    let uniq_a = ma.len() as f64 / sa.len() as f64;
    let uniq_b = mb.len() as f64 / sb.len() as f64;
    let strings = if uniq_a > 0.5 || uniq_b > 0.5 { None } else { Some(js(&ma, &mb)) };
    let mut ng = 0.0;
    for k in 1..=3 {
        ng += js(&profile(&sa, k), &profile(&sb, k));
    }
    ng /= 3.0;
    let mut la: BTreeMap<usize, f64> = BTreeMap::new();
    let mut lb: BTreeMap<usize, f64> = BTreeMap::new();
    for s in &sa {
        bump(&mut la, s.len() / 4);
    }
    for s in &sb {
        bump(&mut lb, s.len() / 4);
    }
    Some(ObsDist { strings, ngrams: ng, lengths: js(&la, &lb) })
}

pub fn train(cfg: &TrainCfg, out_path: &str) -> io::Result<(Model, TrainReport)> {
    let mut rng = Rng::new(cfg.seed);
    let mut p: Params = substrate::init(&mut rng.fork(tag("init")));
    let mut opt = Adam::new(&p, cfg.lr);
    let mut grads: Vec<Vec<f32>> = p.data.iter().map(|b| vec![0.0f32; b.len()]).collect();
    let spec = FamilySpec::train();
    let mut curve: Vec<CurveRow> = Vec::new();
    let mut steps = 0usize;
    let mut nan_steps = 0usize;
    let mut total_rel = 0u64;
    let mut pending = 0usize;
    let batch = cfg.batch.max(1);
    for ep in 0..cfg.episodes {
        let mut er = rng.fork(tag("ptgrs").wrapping_add(ep as u64));
        let draw = ptgrs::episode(&spec, &mut er);
        let e = draw.e;
        drop(draw.cause);
        let v = eff::build(&e, &cfg.view);
        let w = substrate::wire(&v);
        let seed = rng.fork(tag("search")).next_u64();
        let out = abduction::search(&p, &e, &v, &w, &cfg.search, seed, Some(&mut grads));
        total_rel = total_rel.saturating_add(out.rel);
        curve.push(CurveRow { ep, phi: out.phi, cov: out.cov, ops: out.ops, rel: out.rel });
        pending += 1;
        if pending >= batch {
            if finite(&grads) {
                opt.step(&mut p, &grads);
                steps += 1;
            } else {
                nan_steps += 1;
            }
            for b in grads.iter_mut() {
                for x in b.iter_mut() {
                    *x = 0.0;
                }
            }
            pending = 0;
        }
        if cfg.log_every > 0 && (ep + 1) % cfg.log_every == 0 {
            let k = curve.len().min(cfg.log_every);
            let m: f32 = curve[curve.len() - k..].iter().map(|r| r.phi).sum::<f32>() / k as f32;
            eprintln!("ep {:>5}  phi_moy {:.6}  steps {}  rel {}", ep + 1, m, steps, total_rel);
        }
    }
    if pending > 0 {
        if finite(&grads) {
            opt.step(&mut p, &grads);
            steps += 1;
        } else {
            nan_steps += 1;
        }
    }
    let model = Model {
        params: p,
        view: cfg.view,
        search: cfg.infer,
        emit: cfg.emit,
        seed: cfg.seed,
        episodes: cfg.episodes as u64,
        steps: steps as u64,
        closure: Closure::closed(),
        experiment: CLOSED_EXPERIMENT.to_string(),
        fingerprint: 0,
    };
    let fp = model.save(out_path)?;
    let k = curve.len().min(20).max(1);
    let final_phi_mean = if curve.is_empty() {
        0.0
    } else {
        curve[curve.len() - k..].iter().map(|r| r.phi).sum::<f32>() / k as f32
    };
    Ok((
        model,
        TrainReport {
            episodes: cfg.episodes,
            steps,
            nan_steps,
            curve,
            final_phi_mean,
            fingerprint: fp,
            total_rel,
        },
    ))
}

#[derive(Clone, Debug)]
pub struct Rng {
    s: u64,

impl Rng {
    pub fn new(seed: u64) -> Rng {
        Rng { s: seed ^ 0x9E37_79B9_7F4A_7C15 }
    }
    pub fn next_u64(&mut self) -> u64 {
        self.s = self.s.wrapping_add(0x9E37_79B9_7F4A_7C15);
        let mut z = self.s;
        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
        z ^ (z >> 31)
    }
    pub fn f64(&mut self) -> f64 {
        (self.next_u64() >> 11) as f64 * INV53
    }
    pub fn below(&mut self, n: usize) -> usize {
        if n <= 1 { 0 } else { (self.next_u64() % (n as u64)) as usize }
    }
    pub fn range(&mut self, lo: usize, hi: usize) -> usize {
        if hi <= lo { lo } else { lo + self.below(hi - lo + 1) }
    }
    pub fn chance(&mut self, p: f64) -> bool {
        self.f64() < p
    }
    pub fn normal(&mut self) -> f64 {
        let u1 = self.f64().max(1e-300);
        let u2 = self.f64();
        (-2.0 * u1.ln()).sqrt() * (2.0 * std::f64::consts::PI * u2).cos()
    }
    pub fn shuffle<T>(&mut self, v: &mut [T]) {
        let n = v.len();
        for i in (1..n).rev() {
            let j = self.below(i + 1);
            v.swap(i, j);
        }
    }
    pub fn weighted(&mut self, w: &[f64]) -> usize {
        if w.is_empty() {
            return 0;
        }
        let tot: f64 = w.iter().sum();
        if !(tot > 0.0) {
            return self.below(w.len());
        }
        let mut x = self.f64() * tot;
        for (i, &wi) in w.iter().enumerate() {
            if x < wi {
                return i;
            }
            x -= wi;
        }
        w.len() - 1
    }
    pub fn fork(&mut self) -> Rng {
        Rng::new(self.next_u64())
    }
}

pub fn softmax(x: &[f64]) -> Vec<f64> {
    if x.is_empty() {
        return Vec::new();
    }
    let m = x.iter().cloned().fold(f64::NEG_INFINITY, f64::max);
    let e: Vec<f64> = x.iter().map(|v| (v - m).exp()).collect();
    let s: f64 = e.iter().sum();
    e.iter().map(|v| v / s).collect()
}
