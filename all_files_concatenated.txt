use crate::percept::Percept;
use crate::search::{Outcome, Span, Step};
use std::cmp::Ordering;
const NONE: usize = usize::MAX;
fn is_cont(b: u8) -> bool {
    b & 0xC0 == 0x80
}
fn width(b: u8) -> usize {
    match b {
        0x00..=0x7F => 1,
        0xC0..=0xDF => 2,
        0xE0..=0xEF => 3,
        0xF0..=0xF7 => 4,
        _ => 0,
    }
}
fn glyph(eff: &[u8], at: usize) -> Option<(usize, usize)> {
    if at == 0 || at >= eff.len() || !is_cont(eff[at]) {
        return None;
    }
    let mut a = at;
    while a > 0 && at - a < 3 && is_cont(eff[a]) {
        a -= 1;
    }
    let end = a + width(eff[a]);
    if end > at && end <= eff.len() && eff[at..end].iter().all(|&b| is_cont(b)) {
        Some((a, end))
    } else {
        None
    }
}
fn down(eff: &[u8], at: usize) -> usize {
    glyph(eff, at).map_or(at, |g| g.0)
}
fn up(eff: &[u8], at: usize) -> usize {
    glyph(eff, at).map_or(at, |g| g.1)
}
fn cover(p: &Percept, spans: &[Span]) -> Vec<Span> {
    let mut raw: Vec<(usize, usize)> = spans
        .iter()
        .map(|s| (down(&p.eff, s.start), up(&p.eff, s.end)))
        .collect();
    raw.sort_unstable();
    let mut out: Vec<Span> = Vec::with_capacity(raw.len());
    for (a, b) in raw {
        let join = out.last().is_some_and(|l| a <= l.end);
        if join {
            if let Some(l) = out.last_mut() {
                l.end = l.end.max(b);
            }
        } else {
            out.push(Span { start: a, end: b, bits: 0.0 });
        }
    }
    for s in out.iter_mut() {
        s.bits = p.mass(s.start, s.end);
    }
    out
}
fn lacking(p: &Percept, o: &Outcome) -> Vec<Span> {
    let mut v: Vec<Span> = Vec::with_capacity(o.residual.spans.len());
    for s in &o.residual.spans {
        let (a, b) = (up(&p.eff, s.start), down(&p.eff, s.end));
        if a < b {
            v.push(Span { start: a, end: b, bits: p.mass(a, b) });
        }
    }
    v.sort_by(|x, y| y.bits.partial_cmp(&x.bits).unwrap_or(Ordering::Equal));
    v
}
pub fn progress(s: &Step) -> String {
    let act = if s.class == NONE {
        String::from("départ")
    } else if s.added {
        format!("ajout classe {}", s.class)
    } else {
        format!("retrait classe {}", s.class)
    };
    let verdict = if s.accepted { "retenu" } else { "écarté" };
    format!(
        "éval {:>5} | {:<18} | {} | φ_sharp {:>10.6} | meilleur {:>10.6} | taille {:>4}",
        s.eval, act, verdict, s.phi, s.best, s.size
    )
}
pub fn explanation(p: &Percept, o: &Outcome) -> Vec<u8> {
    let mut out: Vec<u8> = Vec::new();
    for (i, s) in cover(p, &o.explanation.spans).iter().enumerate() {
        if i > 0 {
            out.push(b'\n');
        }
        out.extend_from_slice(&p.eff[s.start..s.end]);
    }
    out
}
pub fn residual(p: &Percept, o: &Outcome) -> Vec<u8> {
    let r = &o.residual;
    let mut out = format!(
        "Ce qui manque pour une explication plus claire, plus expressive, plus juste (0 : rien, 1 : tout) :\n  clarté       {:.6}\n  expressivité {:.6}\n  justesse     {:.6}\n",
        r.unclear, r.inexpressive, r.unjust
    )
    .into_bytes();
    let zones = lacking(p, o);
    if !zones.is_empty() {
        let head = format!(
            "Non expliqué, par masse d'information décroissante ({} zones) :\n",
            zones.len()
        );
        out.extend_from_slice(head.as_bytes());
        for z in &zones {
            let tag = format!("[{}..{}] {:.3} bits\n", z.start, z.end, z.bits);
            out.extend_from_slice(tag.as_bytes());
            out.extend_from_slice(&p.eff[z.start..z.end]);
            out.push(b'\n');
        }
    }
    out
}
pub fn report(p: &Percept, o: &Outcome) -> Vec<u8> {
    let mut out = explanation(p, o);
    if out.is_empty() {
        out.extend_from_slice("Aucun sens trouvé.".as_bytes());
    }
    let head = format!("\n\nφ_sharp = {:.6} ({} évaluations)\n", o.phi, o.evals);
    out.extend_from_slice(head.as_bytes());
    out.extend_from_slice(&residual(p, o));
    out
}

#[derive(Clone, Debug)]
pub struct Model {
    pub net: Net,
}
#[derive(Clone, Debug)]
pub struct Answer {
    pub percept: Percept,
    pub outcome: Outcome,
}
pub fn seed(eff: &[u8]) -> u64 {
    let mut h = 0xcbf2_9ce4_8422_2325u64;
    for &b in eff {
        h = (h ^ u64::from(b)).wrapping_mul(0x0000_0100_0000_01b3);
    }
    h ^ eff.len() as u64
}
impl Model {
    pub fn new(net: Net) -> Model {
        Model { net }
    }
    pub fn to_bytes(&self) -> Vec<u8> {
        self.net.to_bytes()
    }
    pub fn from_bytes(b: &[u8]) -> Result<Model, String> {
        Net::from_bytes(b).map(Model::new)
    }
    pub fn save<P: AsRef<Path>>(&self, path: P) -> Result<(), String> {
        let p: &Path = path.as_ref();
        self.net.save(p).map_err(|e| format!("{} : {e}", p.display()))
    }
    pub fn load<P: AsRef<Path>>(path: P) -> Result<Model, String> {
        let p: &Path = path.as_ref();
        Net::load(p)
            .map(Model::new)
            .map_err(|e| format!("{} : {e}", p.display()))
    }
    pub fn ask(&self, eff: &[u8]) -> Answer {
        self.ask_with(eff, &mut |_: &Step| {})
    }
    pub fn ask_with(&self, eff: &[u8], on_step: &mut dyn FnMut(&Step)) -> Answer {
        let percept = Percept::new(eff);
        let mut rng = Rng::new(seed(eff));
        let budget = Budget::measured(&percept);
        let outcome = search::run(&self.net, &percept, &mut rng, budget, on_step);
        Answer { percept, outcome }
    }
    pub fn ask_file<P: AsRef<Path>>(
        &self,
        path: P,
        on_step: &mut dyn FnMut(&Step),
    ) -> Result<Answer, String> {
        let p: &Path = path.as_ref();
        let eff = std::fs::read(p).map_err(|e| format!("{} : {e}", p.display()))?;
        Ok(self.ask_with(&eff, on_step))
    }
}

use crate::phi::Tpm;
use crate::rng::Rng;

pub const CHANS: [Chan; 5] = [
    Chan::Identity,
    Chan::Forward,
    Chan::Backward,
    Chan::Iso,
    Chan::IsoBytes,
];
pub const C: usize = CHANS.len();
pub const F: usize = 3 + 2 * C;
pub const K_MIN: usize = 3;
pub const K_DEFAULT: usize = 7;
pub const K_MAX: usize = 12;
const PMIN: f64 = 1.0 / 256.0;
const GATE_EPS: f64 = 1e-9;
const MAGIC: [u8; 4] = *b"PHIS";
const VERSION: u32 = 1;
const HEADER: usize = 20;
pub fn sigmoid(z: f64) -> f64 {
    1.0 / (1.0 + (-z).exp())
}
fn maxof(v: &[f64]) -> f64 {
    v.iter().copied().fold(0.0f64, f64::max)
}
fn unit(v: f64, mx: f64) -> f64 {
    if mx > 0.0 { v / mx } else { 0.0 }
}
fn strength(c: usize, w: f32) -> f64 {
    let w = f64::from(w).max(0.0);
    match CHANS[c] {
        Chan::Forward | Chan::Backward => 1.0 / (1.0 + w),
        Chan::Identity | Chan::Iso | Chan::IsoBytes => (1.0 + w).ln(),
    }
}
fn orient(k: usize) -> Vec<(usize, usize)> {
    let mut v = Vec::with_capacity(k * (k - 1) / 2);
    for i in 0..k {
        for d in 1..=(k - 1) / 2 {
            v.push((i, (i + d) % k));
        }
    }
    if k % 2 == 0 {
        for i in 0..k / 2 {
            v.push((i, i + k / 2));
        }
    }
    v
}
#[derive(Clone, Debug)]
pub struct Net {
    pub k: usize,
    arrows: Vec<(usize, usize)>,
    g: Vec<f32>,
}
impl Net {
    fn dim(k: usize) -> usize {
        k * F + k + C * (k * (k - 1) / 2) + k + 1
    }
    fn o_bm(&self) -> usize {
        self.k * F
    }
    fn o_w(&self) -> usize {
        self.k * F + self.k
    }
    fn o_th(&self) -> usize {
        self.o_w() + C * self.arrows.len()
    }
    fn o_beta(&self) -> usize {
        self.o_th() + self.k
    }
    pub fn new(k: usize, rng: &mut Rng) -> Net {
        assert!((K_MIN..=K_MAX).contains(&k));
        let mut net = Net { k, arrows: orient(k), g: vec![0.0; Net::dim(k)] };
        let (bm, w, th, be) = (net.o_bm(), net.o_w(), net.o_th(), net.o_beta());
        for i in 0..be {
            let s = if i < bm {
                1.5
            } else if i < w {
                1.0
            } else if i < th {
                5.0
            } else {
                1.0
            };
            net.g[i] = (rng.normal() * s) as f32;
        }
        net
    }
    pub fn params(&self) -> &[f32] {
        &self.g
    }
    pub fn params_mut(&mut self) -> &mut [f32] {
        &mut self.g
    }
    pub fn arrows(&self) -> &[(usize, usize)] {
        &self.arrows
    }
    pub fn matrix(&self, c: usize) -> Vec<f32> {
        assert!(c < C);
        let k = self.k;
        let na = self.arrows.len();
        let ow = self.o_w();
        let mut m = vec![0.0f32; k * k];
        for (ai, &(from, to)) in self.arrows.iter().enumerate() {
            m[from * k + to] = self.g[ow + c * na + ai];
        }
        m
    }
    pub fn sharpness(&self) -> f64 {
        16.0 * sigmoid(f64::from(self.g[self.o_beta()]))
    }
    fn member(&self, i: usize, feat: &[f64]) -> f64 {
        let mut z = f64::from(self.g[self.o_bm() + i]);
        for f in 0..F {
            z += f64::from(self.g[i * F + f]) * feat[f];
        }
        sigmoid(z)
    }
    pub fn tpm(&self, ev: &Evidence) -> Tpm {
        let k = self.k;
        let na = self.arrows.len();
        let ow = self.o_w();
        let th = self.o_th();
        let mut t = vec![0.0f64; k * k];
        for (ai, &(from, to)) in self.arrows.iter().enumerate() {
            let mut s = 0.0f64;
            for c in 0..C {
                s += f64::from(self.g[ow + c * na + ai]) * ev.gate(c, from, to);
            }
            t[from * k + to] = s;
        }
        Tpm::from_fn(k, |x, i| {
            let mut z = f64::from(self.g[th + i]);
            for j in 0..k {
                if x >> j & 1 == 1 {
                    z += t[j * k + i];
                }
            }
            let p = sigmoid(z);
            if p.is_nan() { 0.5 } else { p.clamp(PMIN, 1.0 - PMIN) }
        })
    }
    pub fn to_bytes(&self) -> Vec<u8> {
        let mut b = Vec::with_capacity(HEADER + 4 * self.g.len());
        b.extend_from_slice(&MAGIC);
        for v in [VERSION, self.k as u32, C as u32, F as u32] {
            b.extend_from_slice(&v.to_le_bytes());
        }
        for v in &self.g {
            b.extend_from_slice(&v.to_le_bytes());
        }
        b
    }
    pub fn from_bytes(b: &[u8]) -> Result<Net, String> {
        if b.len() < HEADER || b[..4] != MAGIC[..] {
            return Err("fichier .phis invalide".into());
        }
        let rd = |o: usize| u32::from_le_bytes([b[o], b[o + 1], b[o + 2], b[o + 3]]);
        if rd(4) != VERSION {
            return Err(format!("version {} non supportée", rd(4)));
        }
        let k = rd(8) as usize;
        if rd(12) as usize != C || rd(16) as usize != F || !(K_MIN..=K_MAX).contains(&k) {
            return Err("architecture incompatible".into());
        }
        let n = Net::dim(k);
        if b.len() != HEADER + 4 * n {
            return Err("taille de modèle incohérente".into());
        }
        let mut g = Vec::with_capacity(n);
        for i in 0..n {
            let o = HEADER + 4 * i;
            let v = f32::from_le_bytes([b[o], b[o + 1], b[o + 2], b[o + 3]]);
            if !v.is_finite() {
                return Err("paramètre non fini".into());
            }
            g.push(v);
        }
        Ok(Net { k, arrows: orient(k), g })
    }
    pub fn save<P: AsRef<Path>>(&self, path: P) -> std::io::Result<()> {
        if let Some(dir) = path.as_ref().parent() {
            if !dir.as_os_str().is_empty() {
                std::fs::create_dir_all(dir)?;
            }
        }
        std::fs::write(path, self.to_bytes())
    }
    pub fn load<P: AsRef<Path>>(path: P) -> Result<Net, String> {
        let b = std::fs::read(path.as_ref()).map_err(|e| e.to_string())?;
        Net::from_bytes(&b)
    }
}
#[derive(Clone, Debug)]
pub struct Field {
    pub k: usize,
    pub na: usize,
    pub nc: usize,
    pub class_of: Vec<usize>,
    pub by_class: Vec<Vec<usize>>,
    pub cands: Vec<usize>,
    pub m: Vec<f64>,
    pub mbar: Vec<f64>,
    pub sal: Vec<f64>,
    ostr: Vec<f64>,
    edges: Vec<Vec<(usize, usize, f64)>>,
}
impl Field {
    pub fn new(net: &Net, p: &Percept) -> Field {
        let k = net.k;
        let na = p.anchors.len();
        let nc = p.repeats.len();
        let class_of: Vec<usize> = p.anchors.iter().map(|a| a.class).collect();
        let mut by_class: Vec<Vec<usize>> = vec![Vec::new(); nc];
        for (a, &c) in class_of.iter().enumerate() {
            by_class[c].push(a);
        }
        let cands: Vec<usize> = (0..nc).filter(|&c| !by_class[c].is_empty()).collect();
        let mut edges: Vec<Vec<(usize, usize, f64)>> = Vec::with_capacity(C);
        let mut ostr = vec![0.0f64; C * na];
        let mut lout = vec![0.0f64; C * na];
        let mut lin = vec![0.0f64; C * na];
        for c in 0..C {
            let mut v: Vec<(usize, usize, f64)> = Vec::new();
            for e in p.rel.channel(CHANS[c]) {
                let s = strength(c, e.w);
                v.push((e.from, e.to, s));
                ostr[c * na + e.from] += s;
                lout[c * na + e.from] += 1.0;
                lin[c * na + e.to] += 1.0;
            }
            edges.push(v);
        }
        for v in lout.iter_mut().chain(lin.iter_mut()) {
            *v = (1.0 + *v).ln();
        }
        let mo: Vec<f64> = (0..C).map(|c| maxof(&lout[c * na..(c + 1) * na])).collect();
        let mi: Vec<f64> = (0..C).map(|c| maxof(&lin[c * na..(c + 1) * na])).collect();
        let lb: Vec<f64> = p
            .anchors
            .iter()
            .map(|a| (1.0 + p.repeats[a.class].bits).ln())
            .collect();
        let lc: Vec<f64> = p
            .anchors
            .iter()
            .map(|a| (p.repeats[a.class].count() as f64).ln())
            .collect();
        let ll: Vec<f64> = p.anchors.iter().map(|a| (a.len as f64).ln()).collect();
        let (mb, mc, ml) = (maxof(&lb), maxof(&lc), maxof(&ll));
        let mut m = vec![0.0f64; na * k];
        let mut feat = [0.0f64; F];
        for a in 0..na {
            feat[0] = unit(lb[a], mb);
            feat[1] = unit(lc[a], mc);
            feat[2] = unit(ll[a], ml);
            for c in 0..C {
                feat[3 + c] = unit(lout[c * na + a], mo[c]);
                feat[3 + C + c] = unit(lin[c * na + a], mi[c]);
            }
            for i in 0..k {
                m[a * k + i] = net.member(i, &feat);
            }
        }
        let mut mbar = vec![0.0f64; nc * k];
        for &c in &cands {
            let inv = 1.0 / by_class[c].len() as f64;
            for &a in &by_class[c] {
                for i in 0..k {
                    mbar[c * k + i] += m[a * k + i] * inv;
                }
            }
        }
        Field { k, na, nc, class_of, by_class, cands, m, mbar, sal: lb, ostr, edges }
    }
    pub fn response(&self) -> usize {
        let k = self.k;
        let tot: f64 = self.sal.iter().sum();
        let mut r = vec![0.0f64; k];
        if tot > 0.0 {
            for a in 0..self.na {
                for i in 0..k {
                    r[i] += self.sal[a] * self.m[a * k + i];
                }
            }
            for v in r.iter_mut() {
                *v /= tot;
            }
        }
        let mean = r.iter().sum::<f64>() / k as f64;
        let mut s = 0usize;
        for i in 0..k {
            if r[i] >= mean - 1e-12 {
                s |= 1 << i;
            }
        }
        s
    }
    pub fn align(&self, class: usize, z: usize) -> f64 {
        let k = self.k;
        let mut s = 0.0f64;
        for i in 0..k {
            let u = if z >> i & 1 == 1 { 1.0 } else { -1.0 };
            s += u * (2.0 * self.mbar[class * k + i] - 1.0);
        }
        s / k as f64
    }
    pub fn region(&self, r: usize) -> Vec<bool> {
        let mut sel = vec![false; self.nc];
        if self.cands.is_empty() {
            return sel;
        }
        let al: Vec<f64> = self.cands.iter().map(|&c| self.align(c, r)).collect();
        let mean = al.iter().sum::<f64>() / al.len() as f64;
        for (i, &c) in self.cands.iter().enumerate() {
            if al[i] >= mean - 1e-12 {
                sel[c] = true;
            }
        }
        sel
    }
}
#[derive(Clone, Debug)]
pub struct Evidence {
    k: usize,
    flow: Vec<f64>,
    mass: Vec<f64>,
}
impl Evidence {
    pub fn of(f: &Field, sel: &[bool]) -> Evidence {
        let k = f.k;
        let mut flow = vec![0.0f64; C * k * k];
        let mut mass = vec![0.0f64; C * k];
        for a in 0..f.na {
            if !sel[f.class_of[a]] {
                continue;
            }
            for c in 0..C {
                let s = f.ostr[c * f.na + a];
                if s > 0.0 {
                    for j in 0..k {
                        mass[c * k + j] += f.m[a * k + j] * s;
                    }
                }
            }
        }
        for c in 0..C {
            for &(a, b, s) in &f.edges[c] {
                if !(sel[f.class_of[a]] && sel[f.class_of[b]]) {
                    continue;
                }
                for j in 0..k {
                    let v = s * f.m[a * k + j];
                    for i in 0..k {
                        flow[(c * k + j) * k + i] += v * f.m[b * k + i];
                    }
                }
            }
        }
        Evidence { k, flow, mass }
    }
    pub fn gate(&self, c: usize, from: usize, to: usize) -> f64 {
        self.flow[(c * self.k + from) * self.k + to] / (self.mass[c * self.k + from] + GATE_EPS)
    }
}

const NONE: usize = usize::MAX;
#[derive(Clone, Debug)]
pub struct Null {
    pub n: usize,
    pub count: [usize; 256],
    pub bits: [f64; 256],
}
impl Null {
    pub fn of(eff: &[u8]) -> Null {
        let n = eff.len();
        let mut count = [0usize; 256];
        for &b in eff {
            count[b as usize] += 1;
        }
        let mut bits = [f64::INFINITY; 256];
        for b in 0..256 {
            if count[b] > 0 {
                bits[b] = (n as f64 / count[b] as f64).log2();
            }
        }
        Null { n, count, bits }
    }
}

            Chan::Iso => self.iso.as_slice(),
            Chan::IsoBytes => self.iso_bytes.as_slice(),
        }
    }

#[derive(Clone, Debug)]
pub struct Percept {
    pub eff: Vec<u8>,
    pub null: Null,
    pub surprise: Vec<f32>,
    pub cum: Vec<f64>,
    pub sa: Vec<usize>,
    pub lcp: Vec<usize>,
    pub repeats: Vec<Repeat>,
    pub anchors: Vec<Anchor>,
    pub rel: Relations,
}
impl Percept {
    pub fn new(eff: &[u8]) -> Percept {
        let n = eff.len();
        let null = Null::of(eff);
        let surprise: Vec<f32> = eff.iter().map(|&b| null.bits[b as usize] as f32).collect();
        let mut cum: Vec<f64> = Vec::with_capacity(n + 1);
        let mut acc = 0.0f64;
        cum.push(acc);
        for &b in eff {
            acc += null.bits[b as usize];
            cum.push(acc);
        }
        let sa = suffix_array(eff);
        let lcp = lcp_array(eff, &sa);
        let (repeats, owner) = maximal_repeats(eff, &sa, &lcp, &cum);
        let mut anchors: Vec<Anchor> = Vec::new();
        for (p, &c) in owner.iter().enumerate() {
            if c != NONE {
                anchors.push(Anchor { pos: p, len: repeats[c].len, class: c });
            }
        }
        let rel = relate(eff, &sa, &repeats, &anchors);
        Percept { eff: eff.to_vec(), null, surprise, cum, sa, lcp, repeats, anchors, rel }
    }
    pub fn mass(&self, a: usize, b: usize) -> f64 {
        self.cum[b] - self.cum[a]
    }
    pub fn occurrences(&self, class: usize) -> &[usize] {
        let r = &self.repeats[class];
        &self.sa[r.lb..=r.rb]
    }
    pub fn content(&self, class: usize) -> &[u8] {
        window(&self.eff, &self.sa, &self.repeats[class])
    }
}
fn suffix_array(s: &[u8]) -> Vec<usize> {
    let n = s.len();
    let mut sa: Vec<usize> = (0..n).collect();
    if n < 2 {
        return sa;
    }
    sa.sort_unstable_by_key(|&i| s[i]);
    let mut rank = vec![0usize; n];
    for j in 1..n {
        rank[sa[j]] = rank[sa[j - 1]] + usize::from(s[sa[j]] != s[sa[j - 1]]);
    }
    let mut next = vec![0usize; n];
    let mut by_second = vec![0usize; n];
    let mut k = 1usize;
    while rank[sa[n - 1]] + 1 < n {
        let mut p = 0;
        for i in n.saturating_sub(k)..n {
            by_second[p] = i;
            p += 1;
        }
        for j in 0..n {
            if sa[j] >= k {
                by_second[p] = sa[j] - k;
                p += 1;
            }
        }
        let classes = rank[sa[n - 1]] + 1;
        let mut bucket = vec![0usize; classes + 1];
        for &r in &rank {
            bucket[r + 1] += 1;
        }
        for r in 0..classes {
            bucket[r + 1] += bucket[r];
        }
        for &i in &by_second {
            sa[bucket[rank[i]]] = i;
            bucket[rank[i]] += 1;
        }
        next[sa[0]] = 0;
        for j in 1..n {
            let (a, b) = (sa[j - 1], sa[j]);
            let ka = (rank[a], if a + k < n { rank[a + k] as isize } else { -1 });
            let kb = (rank[b], if b + k < n { rank[b + k] as isize } else { -1 });
            next[b] = next[a] + usize::from(ka != kb);
        }
        std::mem::swap(&mut rank, &mut next);
        k <<= 1;
    }
    sa
}
fn lcp_array(s: &[u8], sa: &[usize]) -> Vec<usize> {
    let n = s.len();
    let mut rank = vec![0usize; n];
    for (r, &i) in sa.iter().enumerate() {
        rank[i] = r;
    }
    let mut lcp = vec![0usize; n];
    let mut h = 0usize;
    for i in 0..n {
        let r = rank[i];
        if r == 0 {
            h = 0;
            continue;
        }
        let j = sa[r - 1];
        while i + h < n && j + h < n && s[i + h] == s[j + h] {
            h += 1;
        }
        lcp[r] = h;
        h = h.saturating_sub(1);
    }
    lcp
}
fn maximal_repeats(
    eff: &[u8],
    sa: &[usize],
    lcp: &[usize],
    cum: &[f64],
) -> (Vec<Repeat>, Vec<usize>) {
    let n = eff.len();
    let mut repeats: Vec<Repeat> = Vec::new();
    let mut owner = vec![NONE; n];
    if n < 2 {
        return (repeats, owner);
    }
    let left: Vec<u16> = sa
        .iter()
        .map(|&p| if p == 0 { 256 } else { u16::from(eff[p - 1]) })
        .collect();
    let mut diff = vec![n; n];
    for k in (0..n - 1).rev() {
        diff[k] = if left[k + 1] != left[k] { k + 1 } else { diff[k + 1] };
    }
    let mut stack: Vec<(usize, usize)> = vec![(0, 0)];
    let mut pending: Vec<usize> = Vec::new();
    for i in 1..=n {
        let cur = if i < n { lcp[i] } else { 0 };
        let mut lb = i - 1;
        loop {
            let (len, b) = stack[stack.len() - 1];
            if cur >= len {
                break;
            }
            stack.pop();
            lb = b;
            let rb = i - 1;
            if diff[b] > rb {
                continue;
            }
            let id = repeats.len();
            let split = pending.partition_point(|&c| repeats[c].lb < b);
            let mut cursor = b;
            for &c in &pending[split..] {
                for r in cursor..repeats[c].lb {
                    owner[sa[r]] = id;
                }
                cursor = repeats[c].rb + 1;
            }
            for r in cursor..=rb {
                owner[sa[r]] = id;
            }
            for &c in &pending[split..] {
                repeats[c].parent = Some(id);
            }
            pending.truncate(split);
            pending.push(id);
            let p = sa[b];
            repeats.push(Repeat { len, lb: b, rb, parent: None, bits: cum[p + len] - cum[p] });
        }
        if cur > stack[stack.len() - 1].0 {
            stack.push((cur, lb));
        }
    }
    (repeats, owner)
}
fn window<'a>(eff: &'a [u8], sa: &[usize], r: &Repeat) -> &'a [u8] {
    let p = sa[r.lb];
    &eff[p..p + r.len]
}
fn code(prev: &[usize], a: usize, t: usize) -> usize {
    let p = prev[t];
    if p != NONE && p >= a { t - p } else { 0 }
}
fn same_cycle(prev: &[usize], a1: usize, a2: usize, len: usize) -> bool {
    (0..len).all(|d| code(prev, a1, a1 + d) == code(prev, a2, a2 + d))
}
fn repeated_in(prev: &[usize], a: usize, b: usize) -> usize {
    (a..b).filter(|&t| code(prev, a, t) != 0).count()
}
fn repeated(w: &[u8]) -> usize {
    let mut seen = [false; 256];
    let mut r = 0;
    for &b in w {
        if seen[b as usize] {
            r += 1;
        } else {
            seen[b as usize] = true;
        }
    }
    r
}
fn pattern_hash(w: &[u8]) -> Option<u64> {
    let mut id = [u16::MAX; 256];
    let mut next = 0u16;
    let mut h = 0xcbf2_9ce4_8422_2325u64;
    for &b in w {
        let slot = &mut id[b as usize];
        if *slot == u16::MAX {
            *slot = next;
            next += 1;
        }
        h = (h ^ u64::from(*slot)).wrapping_mul(0x0000_0100_0000_01b3);
    }
    if usize::from(next) < w.len() { Some(h) } else { None }
}
fn same_pattern(a: &[u8], b: &[u8]) -> bool {
    if a.len() != b.len() {
        return false;
    }
    let mut ia = [u16::MAX; 256];
    let mut ib = [u16::MAX; 256];
    let mut next = 0u16;
    for t in 0..a.len() {
        let (x, y) = (a[t] as usize, b[t] as usize);
        let (new_a, new_b) = (ia[x] == u16::MAX, ib[y] == u16::MAX);
        if new_a != new_b {
            return false;
        }
        if new_a {
            ia[x] = next;
            ib[y] = next;
            next += 1;
        } else if ia[x] != ib[y] {
            return false;
        }
    }
    true
}
fn relate(eff: &[u8], sa: &[usize], repeats: &[Repeat], anchors: &[Anchor]) -> Relations {
    let m = anchors.len();
    let mut rel = Relations::default();
    let mut prev = vec![NONE; m];
    let mut next = vec![NONE; m];
    let mut last = vec![NONE; repeats.len()];
    for (k, a) in anchors.iter().enumerate() {
        let j = last[a.class];
        if j != NONE {
            prev[k] = j;
            next[j] = k;
            let w = repeats[a.class].bits as f32;
            rel.identity.push(Edge { from: j, to: k, w });
            rel.identity.push(Edge { from: k, to: j, w });
        }
        last[a.class] = k;
    }
    for (k, a) in anchors.iter().enumerate() {
        let e = a.end();
        let j = anchors.partition_point(|x| x.pos < e);
        if j < m {
            rel.forward.push(Edge { from: k, to: j, w: (anchors[j].pos - e) as f32 });
        }
    }
    let mut by_end: Vec<usize> = (0..m).collect();
    by_end.sort_unstable_by_key(|&k| (anchors[k].end(), anchors[k].len));
    let mut j = 0;
    for (k, a) in anchors.iter().enumerate() {
        while j < m && anchors[by_end[j]].end() <= a.pos {
            j += 1;
        }
        if j > 0 {
            let b = by_end[j - 1];
            rel.backward.push(Edge { from: k, to: b, w: (a.pos - anchors[b].end()) as f32 });
        }
    }
    rel.iso = iso_edges(&prev, &next);
    rel.iso_bytes = iso_bytes_edges(eff, sa, repeats, anchors);
    rel
}
fn iso_edges(prev: &[usize], next: &[usize]) -> Vec<Edge> {
    const BASE: u64 = 0x9E37_79B9_7F4A_7C15;
    let m = prev.len();
    let mut out: Vec<Edge> = Vec::new();
    let mut q = vec![NONE; m + 1];
    for k in (0..m).rev() {
        q[k] = q[k + 1].min(next[k]);
    }
    let mut inv = BASE;
    for _ in 0..6 {
        inv = inv.wrapping_mul(2u64.wrapping_sub(BASE.wrapping_mul(inv)));
    }
    let mut pw = vec![1u64; m + 1];
    let mut ipw = vec![1u64; m + 1];
    for t in 1..=m {
        pw[t] = pw[t - 1].wrapping_mul(BASE);
        ipw[t] = ipw[t - 1].wrapping_mul(inv);
    }
    let mut fen = vec![0u64; m + 1];
    let mut keyed: Vec<(usize, u64, usize)> = Vec::new();
    for a in (0..m).rev() {
        let b = next[a];
        if b == NONE {
            continue;
        }
        let v = ((b - a) as u64).wrapping_mul(ipw[b]);
        let mut i = b + 1;
        while i <= m {
            fen[i] = fen[i].wrapping_add(v);
            i += i & i.wrapping_neg();
        }
        if q[a] < b {
            let mut s = 0u64;
            let mut j = b;
            while j > 0 {
                s = s.wrapping_add(fen[j]);
                j -= j & j.wrapping_neg();
            }
            keyed.push((b - a, pw[b - 1].wrapping_mul(s), a));
        }
    }
    keyed.sort_unstable();
    let mut i = 0;
    while i < keyed.len() {
        let (len, hash, _) = keyed[i];
        let mut e = i + 1;
        while e < keyed.len() && keyed[e].0 == len && keyed[e].1 == hash {
            e += 1;
        }
        if e - i >= 2 {
            let mut groups: Vec<Vec<usize>> = Vec::new();
            for &(_, _, a) in &keyed[i..e] {
                let home = groups.iter().position(|g| same_cycle(prev, g[0], a, len));
                match home {
                    Some(g) => groups[g].push(a),
                    None => groups.push(vec![a]),
                }
            }
            for g in &groups {
                if g.len() < 2 {
                    continue;
                }
                let w = repeated_in(prev, g[0], g[0] + len) as f32;
                for p in g.windows(2) {
                    out.push(Edge { from: p[0], to: p[1], w });
                    out.push(Edge { from: p[1], to: p[0], w });
                }
            }
        }
        i = e;
    }
    out
}
fn iso_bytes_edges(eff: &[u8], sa: &[usize], repeats: &[Repeat], anchors: &[Anchor]) -> Vec<Edge> {
    let m = anchors.len();
    let mut out: Vec<Edge> = Vec::new();
    let mut first = vec![0usize; repeats.len() + 1];
    for a in anchors {
        first[a.class + 1] += 1;
    }
    for c in 0..repeats.len() {
        first[c + 1] += first[c];
    }
    let mut fill = first.clone();
    let mut by_class = vec![0usize; m];
    for (k, a) in anchors.iter().enumerate() {
        by_class[fill[a.class]] = k;
        fill[a.class] += 1;
    }
    let mut keyed: Vec<(usize, u64, usize)> = Vec::new();
    for c in 0..repeats.len() {
        if first[c] == first[c + 1] {
            continue;
        }
        if let Some(h) = pattern_hash(window(eff, sa, &repeats[c])) {
            keyed.push((repeats[c].len, h, c));
        }
    }
    keyed.sort_unstable();
    let mut i = 0;
    while i < keyed.len() {
        let (len, hash, _) = keyed[i];
        let mut e = i + 1;
        while e < keyed.len() && keyed[e].0 == len && keyed[e].1 == hash {
            e += 1;
        }
        let mut groups: Vec<Vec<usize>> = Vec::new();
        for &(_, _, c) in &keyed[i..e] {
            let wc = window(eff, sa, &repeats[c]);
            let home = groups
                .iter()
                .position(|g| same_pattern(window(eff, sa, &repeats[g[0]]), wc));
            match home {
                Some(g) => groups[g].push(c),
                None => groups.push(vec![c]),
            }
        }
        for g in groups {
            if g.len() < 2 {
                continue;
            }
            let w = repeated(window(eff, sa, &repeats[g[0]])) as f32;
            let mut members: Vec<usize> = Vec::new();
            for &c in &g {
                members.extend_from_slice(&by_class[first[c]..first[c + 1]]);
            }
            members.sort_unstable();
            for p in members.windows(2) {
                out.push(Edge { from: p[0], to: p[1], w });
                out.push(Edge { from: p[1], to: p[0], w });
            }
        }
        i = e;
    }
    out
}

pub fn scatter(mask: usize, x: usize) -> usize {
    let mut out = 0usize;

    while m != 0 {

        if x >> r & 1 == 1 {
            out |= b;
        }
        m ^= b;
        r += 1;
    }

        let cut = sys.cut(a, b);
        let cc = cut.cause(s);
        let links = (a.count_ones() * b.count_ones()) as f64;
        let pcs: Vec<f64> = tc.iter().map(|&z| gap(cause[z], cc[z])).collect();
        let pes: Vec<f64> = te.iter().map(|&z| gap(effect[z], cut.effect_at(s, z))).collect();
        for (i, &pc) in pcs.iter().enumerate() {
            for (j, &pe) in pes.iter().enumerate() {
                let phi = pc.min(pe);
                let norm = phi / links;
                let slot = &mut best[i * width + j];
                if slot.cut == NONE || norm < slot.norm - TIE {
                    *slot = Best { norm, phi, pc, pe, cut: ci };
                }
            }
        }
    }
    let mut pick = 0;
    for i in 1..best.len() {
        if best[i].phi > best[pick].phi + TIE {
            pick = i;
        }
    }
    let top = &best[pick];
    let (a, b) = cuts[top.cut];
    let cut = sys.cut(a, b);
    let part = Partition { from: scatter(units, a), to: scatter(units, b) };
    let phi = if top.phi < EPS { 0.0 } else { top.phi };
    Sia {
        units,
        state: s,
        cause_state: tc[pick / width],
        effect_state: te[pick % width],
        phi,
        phi_cause: top.pc,
        phi_effect: top.pe,
        normalized: phi / part.connections() as f64,
        mip: Some(part),
        cause_cut: cut.cause(s),
        effect_cut: cut.effect(s),
        cause,
        effect,
    }
}
pub fn complexes(tpm: &Tpm, state: usize) -> Vec<Sia> {
    let mut ranked: Vec<(i64, u32, usize)> = Vec::new();
    for units in 1..1usize << tpm.n {
        let phi = analyze(tpm, state, units).phi;
        if phi > 0.0 {
            ranked.push(((phi / EPS).round() as i64, units.count_ones(), units));
        }
    }
    ranked.sort_unstable_by(|a, b| b.0.cmp(&a.0).then(b.1.cmp(&a.1)).then(a.2.cmp(&b.2)));
    let mut covered = 0usize;
    let mut out: Vec<Sia> = Vec::new();
    for &(_, _, units) in &ranked {
        if units & covered == 0 {
            covered |= units;
            out.push(analyze(tpm, state, units));
        }
    }
    out
}
pub fn complex(tpm: &Tpm, state: usize) -> Option<Sia> {
    complexes(tpm, state).into_iter().next()
}
pub fn phi_sharp(tpm: &Tpm, state: usize) -> f64 {
    complex(tpm, state).map_or(0.0, |c| c.phi)
}

fn surface(rng: &mut Rng, cfg: &Cfg) -> (u8, u8, u8, Vec<u8>) {
    let mut cand: Vec<u8> = Vec::with_capacity(96);
    cand.push(b'\n');
    cand.extend(0x20u8..=0x7e);
    let w: u8;
    let h: u8;
    let l: u8;
    if rng.chance(0.5) {
        w = b' ';
        h = b':';
        l = b'\n';
        cand.retain(|&b| b != w && b != h && b != l);
    } else {
        rng.shuffle(&mut cand);
        w = cand[0];
        h = cand[1];
        l = cand[2];
        cand.drain(0..3);
    }
    rng.shuffle(&mut cand);
    let a = rng.range(cfg.alpha_min, cfg.alpha_max + 1).min(cand.len());
    cand.truncate(a);
    (w, h, l, cand)
}
fn build_rhs(
    rng: &mut Rng,
    syms: &[Sym],
    by_level: &[Vec<usize>],
    head: usize,
    cfg: &Cfg,
) -> Vec<Tpl> {
    let m = syms[head].arity;
    let lvl = syms[head].level;
    let mut nodes: Vec<Tpl> = (0..m).map(Tpl::Var).collect();
    let mut unused: Vec<usize> = (0..m).collect();
    let n_app = rng.range(cfg.rhs_min, cfg.rhs_max + 1);
    for _ in 0..n_app {
        let tl = if lvl > 1 && rng.chance(0.7) { lvl - 1 } else { rng.below(lvl) };
        let s = by_level[tl][rng.below(by_level[tl].len())];
        let mut ch = Vec::with_capacity(syms[s].arity);
        for _ in 0..syms[s].arity {
            let c = if !unused.is_empty() && rng.chance(0.8) {
                let u = rng.below(unused.len());
                unused.swap_remove(u)
            } else if !nodes.is_empty() {
                rng.below(nodes.len()) 
            } else {
                let t = by_level[0][rng.below(by_level[0].len())];
                nodes.push(Tpl::App(t, Vec::new()));
                nodes.len() - 1
            };
            ch.push(c);
        }
        nodes.push(Tpl::App(s, ch));
        unused.push(nodes.len() - 1);
    }
    nodes
}
fn reachable(g: &[GNode], root: usize) -> Vec<usize> {
    let mut seen = vec![false; g.len()];
    let mut out = Vec::new();
    let mut stack = vec![root];
    while let Some(n) = stack.pop() {
        if seen[n] {
            continue;
        }
        seen[n] = true;
        out.push(n);
        for &a in &g[n].args {
            stack.push(a);
        }
    }
    out
}
fn unfolded(g: &[GNode], id: usize, memo: &mut [usize]) -> usize {
    if memo[id] != usize::MAX {
        return memo[id];
    }
    let mut n = 1usize;
    for &a in &g[id].args {
        n = n.saturating_add(unfolded(g, a, memo));
    }
    memo[id] = n;
    n
}
fn emit_graph<'a>(g: &[GNode], id: usize, words: &'a [Vec<u8>], out: &mut Vec<&'a [u8]>) {
    out.push(&words[g[id].sym]);
    for &a in &g[id].args {
        emit_graph(g, a, words, out);
    }
}
fn emit_tpl<'a>(
    nodes: &[Tpl],
    idx: usize,
    words: &'a [Vec<u8>],
    vars: &'a [Vec<u8>],
    out: &mut Vec<&'a [u8]>,
) {
    match &nodes[idx] {
        Tpl::Var(i) => out.push(&vars[*i]),
        Tpl::App(s, ch) => {
            out.push(&words[*s]);
            for &c in ch {
                emit_tpl(nodes, c, words, vars, out);
            }
        }
    }
}
fn derive(
    rng: &mut Rng,
    cfg: &Cfg,
    syms: &[Sym],
    rules: &[Rule],
    by_head: &[Vec<usize>],
    terminals: &[usize],
    top_syms: &[usize],
) -> Deriv {
    let top = top_syms[rng.below(top_syms.len())];
    let mut g: Vec<GNode> = Vec::new();
    let mut args: Vec<usize> = Vec::new();
    for _ in 0..syms[top].arity {
        if !args.is_empty() && rng.chance(0.3) {
            let a = args[rng.below(args.len())];
            args.push(a);
        } else {
            let t = terminals[rng.below(terminals.len())];
            g.push(GNode { sym: t, args: Vec::new(), born: None });
            args.push(g.len() - 1);
        }
    }
    g.push(GNode { sym: top, args, born: None });
    let root = g.len() - 1;
    let max_steps = if rng.chance(0.55) {
        usize::MAX
    } else {
        1 + rng.below(2 * cfg.levels + 2)
    };
    let (mut applied, mut prov, mut tops) = (Vec::new(), Vec::new(), Vec::new());
    let mut steps = 0usize;
    while steps < max_steps {
        let redexes: Vec<usize> = reachable(&g, root)
            .into_iter()
            .filter(|&n| !by_head[g[n].sym].is_empty())
            .collect();
        if redexes.is_empty() {
            break;
        }
        let mut memo = vec![usize::MAX; g.len()];
        if unfolded(&g, root, &mut memo) > cfg.max_words {
            break;
        }
        let r = redexes[rng.below(redexes.len())];
        let cands = &by_head[g[r].sym];
        let w: Vec<f64> = cands.iter().map(|&i| rules[i].weight).collect();
        let ri = cands[rng.weighted(&w)];
        match g[r].born {
            Some(p) => prov.push((p, ri)),
            None => tops.push(ri),
        }
        let rule = &rules[ri];
        let bound = g[r].args.clone();
        let last = rule.nodes.len() - 1;
        let mut map = vec![usize::MAX; rule.nodes.len()];
        for (ti, t) in rule.nodes.iter().enumerate() {
            match t {
                Tpl::Var(i) => map[ti] = bound[*i],
                Tpl::App(s, ch) => {
                    let cargs: Vec<usize> = ch.iter().map(|&c| map[c]).collect();
                    if ti == last {
                        g[r] = GNode { sym: *s, args: cargs, born: Some(ri) };
                        map[ti] = r;
                    } else {
                        g.push(GNode { sym: *s, args: cargs, born: Some(ri) });
                        map[ti] = g.len() - 1;
                    }
                }
            }
        }
        applied.push(ri);
        steps += 1;
    }
    Deriv { g, root, applied, prov, tops, steps }
}
fn add_link(links: &mut Vec<(usize, usize)>, a: usize, b: usize) {
    if a == b {
        return;
    }
    let p = if a < b { (a, b) } else { (b, a) };
    if !links.contains(&p) {
        links.push(p);
    }
}

impl Ptgrs {
    pub fn new(cfg: Cfg) -> Self {

    pub fn at_scale(rng: &mut Rng, scale: f64) -> Sample {

    pub fn sample(&self, rng: &mut Rng) -> Sample {
        let cfg = &self.cfg;
        let (w_sep, h_sep, l_sep, alpha) = surface(rng, cfg);
        let mut used_words: HashSet<Vec<u8>> = HashSet::new();
        let mut syms: Vec<Sym> = Vec::new();
        let mut by_level: Vec<Vec<usize>> = vec![Vec::new(); cfg.levels + 1];
        for _ in 0..cfg.terminals {
            syms.push(Sym { level: 0, arity: 0 });
            by_level[0].push(syms.len() - 1);
        }
        let wts: Vec<f64> = (0..=cfg.max_arity)
            .map(|a| if a == 0 { 1.0 } else { 3.0 / a as f64 })
            .collect();
        for lv in 1..=cfg.levels {
            for _ in 0..cfg.width {
                let ar = rng.weighted(&wts);
                syms.push(Sym { level: lv, arity: ar });
                by_level[lv].push(syms.len() - 1);
            }
        }
        let words: Vec<Vec<u8>> = (0..syms.len())
            .map(|_| make_word(rng, &alpha, cfg.word_min, cfg.word_max, &mut used_words))
            .collect();
        let mut rules: Vec<Rule> = Vec::new();
        let mut by_head: Vec<Vec<usize>> = vec![Vec::new(); syms.len()];
        for s in 0..syms.len() {
            if syms[s].level == 0 {
                continue;
            }
            let alts = 1 + rng.below(cfg.alt_max);
            for _ in 0..alts {
                let nodes = build_rhs(rng, &syms, &by_level, s, cfg);
                let mut vars: Vec<Vec<u8>> = Vec::new();
                for _ in 0..syms[s].arity {
                    vars.push(make_word(rng, &alpha, cfg.word_min, cfg.word_max, &mut used_words));
                }
                let weight = 0.2 + rng.unit();
                by_head[s].push(rules.len());
                rules.push(Rule { head: s, nodes, vars, weight });
            }
        }
        let top_syms: Vec<usize> = by_level[cfg.levels].clone();
        let mut derivs: Vec<Deriv> = Vec::new();
        for _ in 0..cfg.instances {
            derivs.push(derive(rng, cfg, &syms, &rules, &by_head, &by_level[0], &top_syms));
        }
        let mut applied = vec![false; rules.len()];
        for d in &derivs {
            for &r in &d.applied {
                applied[r] = true;
            }
        }
        struct Entry {
            text: Vec<u8>,
            role: Role,
            rule: Option<usize>,
            inst: Option<usize>,
        }
        let mut entries: Vec<Entry> = Vec::new();
        for (ri, r) in rules.iter().enumerate() {
            let mut head: Vec<&[u8]> = Vec::new();
            head.push(&words[r.head]);
            for v in &r.vars {
                head.push(v);
            }
            let mut body: Vec<&[u8]> = Vec::new();
            emit_tpl(&r.nodes, r.nodes.len() - 1, &words, &r.vars, &mut body);
            let mut text = join(&head, w_sep);
            text.push(h_sep);
            text.push(w_sep);
            text.extend(join(&body, w_sep));
            let role = if applied[ri] { Role::Used } else { Role::Unused };
            entries.push(Entry { text, role, rule: Some(ri), inst: None });
        }
        for _ in 0..cfg.noise {
            let n = rng.range(4, 11);
            let mut toks: Vec<Vec<u8>> = Vec::new();
            for _ in 0..n {
                toks.push(make_word(rng, &alpha, cfg.word_min, cfg.word_max, &mut used_words));
            }
            let refs: Vec<&[u8]> = toks.iter().map(|t| t.as_slice()).collect();
            entries.push(Entry { text: join(&refs, w_sep), role: Role::Noise, rule: None, inst: None });
        }
        for _ in 0..cfg.decoys {
            let nsh = rng.range(2, 4);
            let lo = cfg.word_max.max(5);
            let mut shared: Vec<Vec<u8>> = Vec::new();
            for _ in 0..nsh {
                shared.push(make_word(rng, &alpha, lo, lo + 2, &mut used_words));
            }
            for _ in 0..2 {
                let mut toks: Vec<Vec<u8>> = Vec::new();
                for w in &shared {
                    if rng.chance(0.6) {
                        toks.push(make_word(rng, &alpha, cfg.word_min, cfg.word_max, &mut used_words));
                    }
                    toks.push(w.clone());
                }
                toks.push(make_word(rng, &alpha, cfg.word_min, cfg.word_max, &mut used_words));
                let refs: Vec<&[u8]> = toks.iter().map(|t| t.as_slice()).collect();
                entries.push(Entry { text: join(&refs, w_sep), role: Role::Decoy, rule: None, inst: None });
            }
        }
        rng.shuffle(&mut entries);
        let marker: Option<Vec<u8>> = if rng.chance(0.5) {
            Some(make_word(rng, &alpha, cfg.word_min, cfg.word_max, &mut used_words))
        } else {
            None
        };
        for (k, d) in derivs.iter().enumerate() {
            let mut toks: Vec<&[u8]> = Vec::new();
            emit_graph(&d.g, d.root, &words, &mut toks);
            let mut text: Vec<u8> = Vec::new();
            if let Some(m) = &marker {
                text.extend_from_slice(m);
                text.push(h_sep);
                text.push(w_sep);
            }
            text.extend(join(&toks, w_sep));
            entries.push(Entry { text, role: Role::Instance, rule: None, inst: Some(k) });
        }
        let mut eff: Vec<u8> = Vec::new();
        let mut lines: Vec<Line> = Vec::new();
        let mut rule_line = vec![usize::MAX; rules.len()];
        let mut inst_line = vec![usize::MAX; derivs.len()];
        for e in &entries {
            let start = eff.len();
            eff.extend_from_slice(&e.text);
            let end = eff.len();
            eff.push(l_sep);
            let li = lines.len();
            lines.push(Line { start, end, role: e.role });
            if let Some(r) = e.rule {
                rule_line[r] = li;
            }
            if let Some(k) = e.inst {
                inst_line[k] = li;
            }
        }
        let mut links: Vec<(usize, usize)> = Vec::new();
        let mut tops: Vec<usize> = Vec::new();
        let mut steps = 0usize;
        for (k, d) in derivs.iter().enumerate() {
            steps += d.steps;
            for &(a, b) in &d.prov {
                add_link(&mut links, rule_line[a], rule_line[b]);
            }
            for &t in &d.tops {
                let l = rule_line[t];
                if !tops.contains(&l) {
                    tops.push(l);
                }
            }
            for n in reachable(&d.g, d.root) {
                if let Some(r) = d.g[n].born {
                    add_link(&mut links, inst_line[k], rule_line[r]);
                }
            }
        }
        links.sort();
        Sample { eff, cause: Cause { lines, links, tops, steps } }
    }
}

#[derive(Clone, Debug)]
pub struct Rng {
    s: [u64; 4],
}
fn splitmix64(state: &mut u64) -> u64 {
    *state = state.wrapping_add(0x9E37_79B9_7F4A_7C15);
    let mut z = *state;
    z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
    z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
    z ^ (z >> 31)
}
impl Rng {
    pub fn new(seed: u64) -> Self {
        let mut st = seed;
        let s = [
            splitmix64(&mut st),
            splitmix64(&mut st),
            splitmix64(&mut st),
            splitmix64(&mut st),
        ];
        Rng { s }
    }
    pub fn next_u64(&mut self) -> u64 {
        let result = self.s[1].wrapping_mul(5).rotate_left(7).wrapping_mul(9);
        let t = self.s[1] << 17;
        self.s[2] ^= self.s[0];
        self.s[3] ^= self.s[1];
        self.s[1] ^= self.s[2];
        self.s[0] ^= self.s[3];
        self.s[2] ^= t;
        self.s[3] = self.s[3].rotate_left(45);
        result
    }
    pub fn unit(&mut self) -> f64 {
        (self.next_u64() >> 11) as f64 * (1.0 / (1u64 << 53) as f64)
    }
    pub fn below(&mut self, n: usize) -> usize {
        ((self.next_u64() as u128 * n as u128) >> 64) as usize
    }
    pub fn range(&mut self, lo: usize, hi: usize) -> usize {
        lo + self.below(hi - lo)
    }
    pub fn chance(&mut self, p: f64) -> bool {
        self.unit() < p
    }
    pub fn normal(&mut self) -> f64 {
        let u1 = self.unit().max(1e-300);
        let u2 = self.unit();
        (-2.0 * u1.ln()).sqrt() * (std::f64::consts::TAU * u2).cos()
    }
    pub fn shuffle<T>(&mut self, v: &mut [T]) {
        for i in (1..v.len()).rev() {
            let j = self.below(i + 1);
            v.swap(i, j);
        }
    }
    pub fn weighted(&mut self, w: &[f64]) -> usize {
        let tot: f64 = w.iter().sum();
        let mut x = self.unit() * tot;
        for (i, &p) in w.iter().enumerate() {
            if x < p {
                return i;
            }
            x -= p;
        }
        w.len() - 1
    }
    pub fn fork(&mut self) -> Rng {
        Rng::new(self.next_u64())
    }
}

use crate::net::{Evidence, Field, Net};
use crate::percept::Percept;
use crate::phi::{complex, Sia, Tpm};
use crate::rng::Rng;
use std::collections::HashMap;
const NONE: usize = usize::MAX;
const TOL: f64 = 1e-9;
#[derive(Clone, Copy, Debug)]
pub struct Budget {
    pub evals: usize,
    pub patience: usize,
}
impl Budget {
    pub fn measured(p: &Percept) -> Budget {
        let evals = p.repeats.len().clamp(32, 1024);
        Budget { evals, patience: (evals / 4).max(8) }
    }
}
#[derive(Clone, Copy, Debug)]
pub struct Span {
    pub start: usize,
    pub end: usize,
    pub bits: f64,
}
#[derive(Clone, Debug, Default)]
pub struct Explanation {
    pub classes: Vec<usize>,
    pub spans: Vec<Span>,
}
impl Explanation {
    pub fn text(&self, eff: &[u8]) -> Vec<u8> {
        let mut out = Vec::new();
        for (i, s) in self.spans.iter().enumerate() {
            if i > 0 {
                out.push(b'\n');
            }
            out.extend_from_slice(&eff[s.start..s.end]);
        }
        out
    }
}
#[derive(Clone, Debug)]
pub struct Residual {
    pub unclear: f64,
    pub inexpressive: f64,
    pub unjust: f64,
    pub spans: Vec<Span>,
}
#[derive(Clone, Debug)]
pub struct Step {
    pub eval: usize,
    pub class: usize,
    pub added: bool,
    pub accepted: bool,
    pub phi: f64,
    pub best: f64,
    pub size: usize,
    pub cause: usize,
}
#[derive(Clone, Debug)]
pub struct Outcome {
    pub r: usize,
    pub phi: f64,
    pub cause: usize,
    pub effect: usize,
    pub sia: Option<Sia>,
    pub explanation: Explanation,
    pub residual: Residual,
    pub evals: usize,
    pub trace: Vec<Step>,
}
struct Judged {
    tpm: Tpm,
    sia: Option<Sia>,
    phi: f64,
}
fn judge(net: &Net, f: &Field, sel: &[bool], state: usize) -> Judged {
    let tpm = net.tpm(&Evidence::of(f, sel));
    let sia = complex(&tpm, state);
    let phi = sia.as_ref().map_or(0.0, |s| s.phi);
    Judged { tpm, sia, phi }
}
fn abduce(rng: &mut Rng, cur: &Judged, state: usize) -> usize {
    match &cur.sia {
        Some(s) => (state & !s.units) | s.global(rng.weighted(&s.cause)),
        None => rng.weighted(&cur.tpm.cause(state)),
    }
}
fn count(f: &Field, sel: &[bool]) -> usize {
    f.cands.iter().filter(|&&c| sel[c]).count()
}
fn unit_gap(p: &Percept) -> usize {
    let mut h: HashMap<usize, usize> = HashMap::new();
    for e in &p.rel.forward {
        let g = e.w as usize;
        if g > 0 {
            *h.entry(g).or_insert(0) += 1;
        }
    }
    h.into_iter()
        .max_by(|a, b| a.1.cmp(&b.1).then(b.0.cmp(&a.0)))
        .map_or(0, |(g, _)| g)
}
fn explain(p: &Percept, f: &Field, sel: &[bool], gap: usize) -> Explanation {
    let classes: Vec<usize> = f.cands.iter().copied().filter(|&c| sel[c]).collect();
    let mut spans: Vec<Span> = Vec::new();
    for a in &p.anchors {
        if !sel[a.class] {
            continue;
        }
        let (s, e) = (a.pos, a.end());
        let join = spans.last().is_some_and(|l| s <= l.end + gap);
        if join {
            if let Some(l) = spans.last_mut() {
                l.end = l.end.max(e);
            }
        } else {
            spans.push(Span { start: s, end: e, bits: 0.0 });
        }
    }
    for sp in spans.iter_mut() {
        sp.bits = p.mass(sp.start, sp.end);
    }
    Explanation { classes, spans }
}
fn leftover(p: &Percept, cur: &Judged, state: usize, ex: &Explanation) -> Residual {
    let n = p.eff.len();
    let total = p.cum[n];
    let (unclear, unjust) = match &cur.sia {
        Some(s) => {
            let z = (state & !s.units) | s.global(s.cause_state);
            (
                (1.0 - s.cause[s.cause_state]).clamp(0.0, 1.0),
                (1.0 - cur.tpm.effect_at(z, state)).clamp(0.0, 1.0),
            )
        }
        None => (1.0, 1.0),
    };
    let covered: f64 = ex.spans.iter().map(|s| s.bits).sum();
    let inexpressive = if total > 0.0 { (1.0 - covered / total).clamp(0.0, 1.0) } else { 0.0 };
    let mut spans: Vec<Span> = Vec::new();
    let mut at = 0usize;
    for s in &ex.spans {
        if s.start > at {
            spans.push(Span { start: at, end: s.start, bits: p.mass(at, s.start) });
        }
        at = at.max(s.end);
    }
    if at < n {
        spans.push(Span { start: at, end: n, bits: p.mass(at, n) });
    }
    spans.sort_by(|a, b| b.bits.partial_cmp(&a.bits).unwrap_or(std::cmp::Ordering::Equal));
    Residual { unclear, inexpressive, unjust, spans }
}
pub fn run(
    net: &Net,
    p: &Percept,
    rng: &mut Rng,
    budget: Budget,
    on_step: &mut dyn FnMut(&Step),
) -> Outcome {
    let field = Field::new(net, p);
    let state = field.response();
    let mut sel = field.region(state);
    let mut cur = judge(net, &field, &sel, state);
    let beta = net.sharpness();
    let mut evals = 1usize;
    let mut since = 0usize;
    let mut tabu = vec![false; field.cands.len()];
    let mut trace: Vec<Step> = Vec::new();
    let first = Step {
        eval: 0,
        class: NONE,
        added: true,
        accepted: true,
        phi: cur.phi,
        best: cur.phi,
        size: count(&field, &sel),
        cause: state,
    };
    on_step(&first);
    trace.push(first);
    while evals < budget.evals && since < budget.patience {
        let z = abduce(rng, &cur, state);
        let mut w = vec![0.0f64; field.cands.len()];
        let mut total = 0.0f64;
        for (i, &c) in field.cands.iter().enumerate() {
            if tabu[i] {
                continue;
            }
            let sgn = if sel[c] { -1.0 } else { 1.0 };
            let v = (beta * sgn * field.align(c, z)).exp();
            w[i] = v;
            total += v;
        }
        if !(total > 0.0) {
            break;
        }
        let i = rng.weighted(&w);
        let c = field.cands[i];
        let adding = !sel[c];
        sel[c] = adding;
        let next = judge(net, &field, &sel, state);
        evals += 1;
        let tried = next.phi;
        let ok = if adding { tried >= cur.phi - TOL } else { tried > cur.phi + TOL };
        if ok {
            cur = next;
            tabu.fill(false);
            since = 0;
        } else {
            sel[c] = !adding;
            tabu[i] = true;
            since += 1;
        }
        let step = Step {
            eval: evals - 1,
            class: c,
            added: adding,
            accepted: ok,
            phi: tried,
            best: cur.phi,
            size: count(&field, &sel),
            cause: z,
        };
        on_step(&step);
        trace.push(step);
    }
    let found = cur.sia.is_some();
    let explanation = if found {
        explain(p, &field, &sel, unit_gap(p))
    } else {
        Explanation::default()
    };
    let residual = leftover(p, &cur, state, &explanation);
    let (cause, effect) = match &cur.sia {
        Some(s) => (s.global(s.cause_state), s.global(s.effect_state)),
        None => (0, 0),
    };
    Outcome {
        r: state,
        phi: cur.phi,
        cause,
        effect,
        sia: cur.sia,
        explanation,
        residual,
        evals,
        trace,
    }
}
pub fn ask(net: &Net, p: &Percept, rng: &mut Rng) -> Outcome {
    run(net, p, rng, Budget::measured(p), &mut |_: &Step| {})
}
