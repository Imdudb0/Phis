const TINY: f64 = 1e-300;
fn sub_index(idx: usize, a: usize, pos: &[usize], apow: &[usize]) -> usize {
    let mut s = 0usize;
    let mut mul = 1usize;
    for &p in pos {
        s += ((idx / apow[p]) % a) * mul;
        mul *= a;
    }
    s
}
fn ids(v: &[usize]) -> String {
    v.iter().map(|j| j.to_string()).collect::<Vec<String>>().join(",")
}
pub struct Dx<'c, 'm> {
    cx: &'c Ctx<'m>,
    cfg: ExactCfg,
    tau: usize,
    state: Vec<u16>,
    mg: Mg<'c, 'm>,
    ct: RefCell<HashMap<(Vec<usize>, Vec<usize>), Rc<Vec<f64>>>>,
    pt: RefCell<HashMap<(usize, usize), Rc<Vec<Vec<u8>>>>>,
}
impl<'c, 'm> Dx<'c, 'm> {
    pub fn new(cx: &'c Ctx<'m>, tau: usize, state: &[u16], cfg: &ExactCfg) -> Dx<'c, 'm> {
        let n = cx.n;
        assert!(state.len() >= n, "state must hold one symbol per node");
        let amax = (cx.a - 1) as u16;
        let st: Vec<u16> = state[..n].iter().map(|&s| s.min(amax)).collect();
        Dx {
            cx,
            cfg: cfg.clone(),
            tau: tau.max(1),
            state: st,
            mg: Mg::new(cx, cfg.backend, tau),
            ct: RefCell::new(HashMap::new()),
            pt: RefCell::new(HashMap::new()),
        }
    }
    pub fn tau(&self) -> usize {
        self.tau
    }
    pub fn inexact(&self) -> bool {
        self.mg.bad.get()
    }
    fn parts(&self, km: usize, kz: usize) -> Rc<Vec<Vec<u8>>> {
        let hit = self.pt.borrow().get(&(km, kz)).cloned();
        if let Some(p) = hit {
            return p;
        }
        let p = Rc::new(set_partitions(km, kz));
        self.pt.borrow_mut().insert((km, kz), Rc::clone(&p));
        p
    }
    pub fn mechanisms(&self) -> Vec<Vec<usize>> {
        let free = self.cx.free_nodes();
        let kmax = if self.cfg.kmax == 0 { free.len() } else { self.cfg.kmax.min(free.len()) };
        let mut all = subsets(free, 1, kmax);
        all.truncate(self.cfg.max_mech);
        all
    }
    pub fn purviews(&self) -> Vec<Vec<usize>> {
        let free = self.cx.free_nodes();
        let a = self.cx.a;
        let mut qm = if self.cfg.qmax == 0 { free.len() } else { self.cfg.qmax.min(free.len()) };
        while qm > 0 && ipow(a, qm) > self.cfg.max_states {
            qm -= 1;
        }
        subsets(free, 1, qm)
    }
    fn cause_table(&self, m: &[usize], z: &[usize]) -> Rc<Vec<f64>> {
        let key = (m.to_vec(), z.to_vec());
        let hit = self.ct.borrow().get(&key).cloned();
        if let Some(t) = hit {
            return t;
        }
        let a = self.cx.a;
        let k = z.len();
        let total = ipow(a, k);
        let sm: Vec<u16> = m.iter().map(|&j| self.state[j]).collect();
        let mut tab = vec![0.0f64; total];
        let mut zs = vec![0u16; k];
        let mut sum = 0.0f64;
        for idx in 0..total {
            decode(idx, a, &mut zs);
            let mut l = 1.0f64;
            for (t, &j) in m.iter().enumerate() {
                let out = self.mg.get(j, z, &zs);
                l *= out[sm[t] as usize];
            }
            tab[idx] = l;
            sum += l;
        }
        if sum > TINY {
            let inv = 1.0 / sum;
            for v in tab.iter_mut() {
                *v *= inv;
            }
        } else {
            tab.fill(1.0 / total as f64);
        }
        let rc = Rc::new(tab);
        self.ct.borrow_mut().insert(key, Rc::clone(&rc));
        rc
    }
    fn partitioned_cause(&self, m: &[usize], z: &[usize], lab: &[u8], apow: &[usize], q: &mut [f64]) {
        let a = self.cx.a;
        let km = m.len();
        let kz = z.len();
        let nb = lab.iter().copied().max().map_or(0, |x| x as usize + 1);
        q.fill(1.0);
        for b in 0..nb {
            let bb = b as u8;
            let mb: Vec<usize> = (0..km).filter(|&i| lab[i] == bb).map(|i| m[i]).collect();
            let zpos: Vec<usize> = (0..kz).filter(|&i| lab[km + i] == bb).collect();
            if zpos.is_empty() {
                continue;
            }
            if mb.is_empty() {
                let f = 1.0 / (ipow(a, zpos.len()) as f64);
                for v in q.iter_mut() {
                    *v *= f;
                }
            } else {
                let zb: Vec<usize> = zpos.iter().map(|&i| z[i]).collect();
                let tab = self.cause_table(&mb, &zb);
                for (idx, v) in q.iter_mut().enumerate() {
                    *v *= tab[sub_index(idx, a, &zpos, apow)];
                }
            }
        }
    }
    fn cause_eval(&self, m: &[usize], z: &[usize]) -> (f64, Vec<Vec<u16>>) {
        let a = self.cx.a;
        let km = m.len();
        let kz = z.len();
        let total = ipow(a, kz);
        let p = self.cause_table(m, z);
        let apow: Vec<usize> = (0..kz).map(|i| ipow(a, i)).collect();
        let parts = self.parts(km, kz);
        let mut q = vec![0.0f64; total];
        let mut best = f64::INFINITY;
        let mut best_q: Vec<f64> = Vec::new();
        for lab in parts.iter() {
            self.partitioned_cause(m, z, lab, &apow, &mut q);
            let (v, _) = id_max(&p, &q);
            if v < best {
                best = v;
                best_q.clear();
                best_q.extend_from_slice(&q);
            }
            if best <= PHI_EPS {
                return (best, Vec::new());
            }
        }
        if !best.is_finite() {
            return (0.0, Vec::new());
        }
        let tol = TIE_REL * best.max(1.0);
        let mut states: Vec<Vec<u16>> = Vec::new();
        for idx in 0..total {
            if id_term(p[idx], best_q[idx]) >= best - tol {
                let mut zs = vec![0u16; kz];
                decode(idx, a, &mut zs);
                states.push(zs);
                if states.len() >= MAX_TIES {
                    break;
                }
            }
        }
        (best, states)
    }
    fn effect_eval(&self, m: &[usize], z: &[usize]) -> (f64, Vec<Vec<u16>>) {
        let a = self.cx.a;
        let km = m.len();
        let kz = z.len();
        let sm: Vec<u16> = m.iter().map(|&j| self.state[j]).collect();
        let pz: Vec<Rc<Vec<f64>>> = z.iter().map(|&y| self.mg.get(y, m, &sm)).collect();
        let fixed: Vec<Option<u16>> = z.iter().map(|&y| self.cx.e_sym(y)).collect();
        let lat: Vec<usize> = (0..kz).filter(|&i| fixed[i].is_none()).collect();
        let nstate = ipow(a, lat.len());
        let mut cand = vec![0u16; nstate * kz];
        let mut pc = vec![1.0f64; nstate];
        let mut digits = vec![0u16; lat.len()];
        for c in 0..nstate {
            decode(c, a, &mut digits);
            let mut t = 0usize;
            for i in 0..kz {
                let s = match fixed[i] {
                    Some(s) => s,
                    None => {
                        let d = digits[t];
                        t += 1;
                        d
                    }
                };
                cand[c * kz + i] = s;
                pc[c] *= pz[i][s as usize];
            }
        }
        let parts = self.parts(km, kz);
        let mut qc = vec![0.0f64; nstate];
        let mut best = f64::INFINITY;
        let mut best_q: Vec<f64> = Vec::new();
        for lab in parts.iter() {
            let mut qy: Vec<Rc<Vec<f64>>> = Vec::with_capacity(kz);
            for i in 0..kz {
                let bb = lab[km + i];
                let mut mb: Vec<usize> = Vec::new();
                let mut smb: Vec<u16> = Vec::new();
                for t in 0..km {
                    if lab[t] == bb {
                        mb.push(m[t]);
                        smb.push(sm[t]);
                    }
                }
                qy.push(self.mg.get(z[i], &mb, &smb));
            }
            let mut v = 0.0f64;
            for c in 0..nstate {
                let mut q = 1.0f64;
                for i in 0..kz {
                    q *= qy[i][cand[c * kz + i] as usize];
                }
                qc[c] = q;
                let t = id_term(pc[c], q);
                if t > v {
                    v = t;
                }
            }
            if v < best {
                best = v;
                best_q.clear();
                best_q.extend_from_slice(&qc);
            }
            if best <= PHI_EPS {
                return (best, Vec::new());
            }
        }
        if !best.is_finite() {
            return (0.0, Vec::new());
        }
        let tol = TIE_REL * best.max(1.0);
        let mut states: Vec<Vec<u16>> = Vec::new();
        for c in 0..nstate {
            if id_term(pc[c], best_q[c]) >= best - tol {
                states.push(cand[c * kz..(c + 1) * kz].to_vec());
                if states.len() >= MAX_TIES {
                    break;
                }
            }
        }
        (best, states)
    }
    pub fn phi_cause(&self, m: &[usize], z: &[usize]) -> f64 {
        let (m, z) = (norm(m), norm(z));
        if m.is_empty() || z.is_empty() {
            return 0.0;
        }
        let v = self.cause_eval(&m, &z).0;
        if v <= PHI_EPS { 0.0 } else { v }
    }
    pub fn phi_effect(&self, m: &[usize], z: &[usize]) -> f64 {
        let (m, z) = (norm(m), norm(z));
        if m.is_empty() || z.is_empty() {
            return 0.0;
        }
        let v = self.effect_eval(&m, &z).0;
        if v <= PHI_EPS { 0.0 } else { v }
    }
    fn best_side(&self, m: &[usize], cands: &[Vec<usize>], dir: Dir) -> (f64, Vec<Face>) {
        let mut best = 0.0f64;
        let mut faces: Vec<Face> = Vec::new();
        for z in cands {
            let (phi, states) = match dir {
                Dir::Cause => self.cause_eval(m, z),
                Dir::Effect => self.effect_eval(m, z),
            };
            if phi <= PHI_EPS {
                continue;
            }
            let tol = TIE_REL * best.max(phi).max(1.0);
            if phi > best + tol {
                best = phi;
                faces.clear();
                faces.push(Face { nodes: z.clone(), phi, states });
            } else if phi >= best - tol {
                if faces.len() < MAX_TIES {
                    faces.push(Face { nodes: z.clone(), phi, states });
                }
                if phi > best {
                    best = phi;
                }
            }
        }
        (best, faces)
    }
    pub fn distinctions(&self) -> Vec<Dist> {
        let cands = self.purviews();
        let mut out: Vec<Dist> = Vec::new();
        for m in self.mechanisms() {
            let (phi_c, fc) = self.best_side(&m, &cands, Dir::Cause);
            let (phi_e, fe) = self.best_side(&m, &cands, Dir::Effect);
            if phi_c <= PHI_EPS && phi_e <= PHI_EPS {
                continue;
            }
            let anchored = fc.iter().chain(fe.iter()).any(|f| self.cx.touches_e(&f.nodes));
            let mech_state: Vec<u16> = m.iter().map(|&j| self.state[j]).collect();
            out.push(Dist {
                mech: m,
                mech_state,
                cause: fc,
                effect: fe,
                phi_c,
                phi_e,
                phi_d: phi_c.min(phi_e),
                anchored,
            });
        }
        out
    }
    pub fn footprint(&self, d: &Dist) -> Vec<usize> {
        let mut v: Vec<usize> = Vec::new();
        for f in d.cause.iter().chain(d.effect.iter()) {
            for &j in &f.nodes {
                if let Some(p) = self.cx.e_pos(j) {
                    v.push(p);
                }
            }
        }
        v.sort_unstable();
        v.dedup();
        v
    }
    pub fn repertoire_marginals(&self, m: &[usize], z: &[usize], dir: Dir) -> Vec<Vec<f64>> {
        let a = self.cx.a;
        let kz = z.len();
        let mut out = vec![vec![0.0f64; a]; kz];
        if m.is_empty() || kz == 0 {
            for row in out.iter_mut() {
                row.fill(1.0 / a as f64);
            }
            return out;
        }
        match dir {
            Dir::Effect => {
                let sm: Vec<u16> = m.iter().map(|&j| self.state[j]).collect();
                for (i, &y) in z.iter().enumerate() {
                    let r = self.mg.get(y, m, &sm);
                    out[i].copy_from_slice(&r);
                }
            }
            Dir::Cause => {
                let p = self.cause_table(m, z);
                let apow: Vec<usize> = (0..kz).map(|i| ipow(a, i)).collect();
                for idx in 0..p.len() {
                    for i in 0..kz {
                        out[i][(idx / apow[i]) % a] += p[idx];
                    }
                }
            }
        }
        out
    }
    fn sym(&self, v: u16) -> String {
        match self.cx.alphabet.get(v as usize) {
            Some(&b) => show_byte(b),
            None => "?".to_string(),
        }
    }
    fn fmt_face(&self, f: &Face) -> String {
        let mut s = format!("{{{}}}=", ids(&f.nodes));
        for (k, st) in f.states.iter().enumerate() {
            if k > 0 {
                s.push('|');
            }
            for &v in st {
                s.push_str(&self.sym(v));
            }
        }
        s.push_str(&format!(" phi={:.4}", f.phi));
        s
    }
    pub fn describe(&self, d: &Dist) -> String {
        let mut s = format!(
            "m{{{}}} phi_d={:.4} (c={:.4} e={:.4}){}",
            ids(&d.mech),
            d.phi_d,
            d.phi_c,
            d.phi_e,
            if d.anchored { " [E]" } else { "" }
        );
        for f in &d.cause {
            s.push_str(&format!("\n  cause  {}", self.fmt_face(f)));
        }
        for f in &d.effect {
            s.push_str(&format!("\n  effect {}", self.fmt_face(f)));
        }
        s
    }
}
pub fn residuals_exact(dx: &Dx<'_, '_>, dists: &[Dist]) -> Vec<Residual> {
    let mut out: Vec<Residual> = Vec::new();
    for d in dists {
        let rho = (d.phi_c - d.phi_e).abs();
        if rho <= PHI_EPS {
            continue;
        }
        let weak = if d.phi_c < d.phi_e { Dir::Cause } else { Dir::Effect };
        let (wf, sf) = match weak {
            Dir::Cause => (&d.cause, &d.effect),
            Dir::Effect => (&d.effect, &d.cause),
        };
        let from_strong = wf.is_empty();
        let faces = if from_strong { sf } else { wf };
        let ties: usize = wf.iter().map(|f| f.states.len()).sum();
        let mut acc: Vec<(usize, f64)> = Vec::new();
        for f in faces {
            let rows = dx.repertoire_marginals(&d.mech, &f.nodes, weak);
            for (i, &j) in f.nodes.iter().enumerate() {
                let h = norm_entropy(&rows[i]);
                match acc.iter().position(|x| x.0 == j) {
                    Some(p) => {
                        if h > acc[p].1 {
                            acc[p].1 = h;
                        }
                    }
                    None => acc.push((j, h)),
                }
            }
        }
        acc.sort_by_key(|x| x.0);
        let tot: f64 = acc.iter().map(|x| x.1).sum();
        let k = acc.len().max(1) as f64;
        let nodes: Vec<(usize, f64)> = acc
            .iter()
            .map(|&(j, h)| (j, rho * if tot > MASS_EPS { h / tot } else { 1.0 / k }))
            .collect();
        out.push(Residual { mech: d.mech.clone(), weak, rho, ties, nodes, from_strong });
    }
    out
}

pub fn tractable(m: &Model, e: &Percept, cfg: &ExactCfg) -> Result<(), String> {
    let n = m.n();
    let top = cfg.max_nodes.min(30);
    if n > top {
        return Err(format!("{} nodes: above max_nodes = {}", n, top));
    }
    let kk = if cfg.kmax == 0 { n } else { cfg.kmax.min(n) };
    let qq = if cfg.qmax == 0 { n } else { cfg.qmax.min(n) };
    if kk + qq > cfg.max_part {
        return Err(format!(
            "partitions of {}+{} elements: above max_part = {}",
            kk, qq, cfg.max_part
        ));
    }
    if cfg.backend == Backend::Exact {
        let a = e.a.max(1);
        let joint = ipow(a, n);
        let lim = if cfg.grain_max > 1 { 2_048 } else { 4_000_000 };
        let lim = lim.min(cfg.max_joint);
        if joint > lim {
            return Err(format!(
                "|A(E)|^n = {}^{} = {}: above the exact limit {}",
                a, n, joint, lim
            ));
        }
    }
    Ok(())
}
fn comb_rec(nodes: &[usize], start: usize, size: usize, cur: &mut Vec<usize>, out: &mut Vec<Vec<usize>>) {
    if cur.len() == size {
        out.push(cur.clone());
        return;
    }
    for i in start..nodes.len() {
        if nodes.len() - i < size - cur.len() {
            break;
        }
        cur.push(nodes[i]);
        comb_rec(nodes, i + 1, size, cur, out);
        cur.pop();
    }
}
pub fn subsets(nodes: &[usize], min_size: usize, max_size: usize) -> Vec<Vec<usize>> {
    let mut out: Vec<Vec<usize>> = Vec::new();
    let mut cur: Vec<usize> = Vec::new();
    for size in min_size.max(1)..=max_size.min(nodes.len()) {
        comb_rec(nodes, 0, size, &mut cur, &mut out);
    }
    out
}

struct Mg<'c, 'm> {
    cx: &'c Ctx<'m>,
    be: Backend,
    tau: usize,
    cache: RefCell<HashMap<Vec<usize>, Rc<Vec<f64>>>>,
    bad: Cell<bool>,
}
impl<'c, 'm> Mg<'c, 'm> {
    fn new(cx: &'c Ctx<'m>, be: Backend, tau: usize) -> Mg<'c, 'm> {
        Mg { cx, be, tau: tau.max(1), cache: RefCell::new(HashMap::new()), bad: Cell::new(false) }
    }
    fn get(&self, j: usize, clamp: &[usize], syms: &[u16]) -> Rc<Vec<f64>> {
        let mut pr: Vec<(usize, u16)> = clamp.iter().copied().zip(syms.iter().copied()).collect();
        pr.sort_unstable();
        let mut key: Vec<usize> = Vec::with_capacity(2 + 2 * pr.len());
        key.push(j);
        key.push(self.tau);
        for &(c, s) in &pr {
            key.push(c);
            key.push(s as usize);
        }
        let hit = self.cache.borrow().get(&key).cloned();
        if let Some(v) = hit {
            return v;
        }
        let cl: Vec<usize> = pr.iter().map(|p| p.0).collect();
        let sy: Vec<u16> = pr.iter().map(|p| p.1).collect();
        let mut out = vec![0.0f64; self.cx.a];
        let done = match self.be {
            Backend::Exact => self.cx.marg_exact(j, &cl, self.tau, &sy, &mut out),
            Backend::Mc => false,
        };
        if !done {
            if self.be == Backend::Exact {
                self.bad.set(true);
            }
            self.cx.marg(j, &cl, self.tau, &sy, &mut out);
        }
        let rc = Rc::new(out);
        self.cache.borrow_mut().insert(key, Rc::clone(&rc));
        rc
    }
}

fn max_prod_all(p: &[Rc<Vec<f64>>], r: &[Vec<f64>], vals: &[Vec<u16>]) -> f64 {
    let d = p.len();
    if d == 0 {
        return 0.0;
    }
    let mut idx = vec![0usize; d];
    let mut best = 0.0f64;
    loop {
        let mut pr = 1.0f64;
        let mut s = 0.0f64;
        for y in 0..d {
            let v = vals[y][idx[y]] as usize;
            pr *= p[y][v];
            s += r[y][v];
        }
        let val = pr * s.abs();
        if val > best {
            best = val;
        }
        let mut y = 0usize;
        while y < d {
            idx[y] += 1;
            if idx[y] < vals[y].len() {
                break;
            }
            idx[y] = 0;
            y += 1;
        }
        if y == d {
            break;
        }
    }
    best
}
fn cut_value_exact(
    mg: &Mg<'_, '_>,
    k: &[usize],
    sk: &[u16],
    p: &[Rc<Vec<f64>>],
    vals: &[Vec<u16>],
    pmax: &[f64],
    mask: u32,
) -> f64 {
    let m = k.len();
    let a = mg.cx.a;
    let mut fa = 1.0f64;
    let mut bpos: Vec<usize> = Vec::new();
    for i in 0..m {
        if ((mask >> i) & 1) == 1 {
            fa *= pmax[i];
        } else {
            bpos.push(i);
        }
    }
    let bnodes: Vec<usize> = bpos.iter().map(|&i| k[i]).collect();
    let bsyms: Vec<u16> = bpos.iter().map(|&i| sk[i]).collect();
    let mut pb: Vec<Rc<Vec<f64>>> = Vec::with_capacity(bpos.len());
    let mut rb: Vec<Vec<f64>> = Vec::with_capacity(bpos.len());
    let mut vb: Vec<Vec<u16>> = Vec::with_capacity(bpos.len());
    let mut changed = false;
    for &i in &bpos {
        let q = mg.get(k[i], &bnodes, &bsyms);
        let mut r = vec![0.0f64; a];
        for v in 0..a {
            r[v] = (p[i][v] / q[v]).log2();
            if r[v].abs() > 1e-12 {
                changed = true;
            }
        }
        pb.push(Rc::clone(&p[i]));
        rb.push(r);
        vb.push(vals[i].clone());
    }
    if changed { fa * max_prod_all(&pb, &rb, &vb) } else { 0.0 }
}
pub fn system_phi_exact(cx: &mut Ctx<'_>, be: Backend, state: &[u16], tau: usize, k: &[usize]) -> SysX {
    let m = k.len();
    if m < 2 {
        return SysX { phi: 0.0, src: 0, ok: true };
    }
    assert!(m <= 30, "a candidate system has at most 30 nodes");
    let n = cx.n;
    let mut inside = vec![false; n];
    for &j in k {
        inside[j] = true;
    }
    let mut bn: Vec<usize> = Vec::new();
    let mut bs: Vec<u16> = Vec::new();
    for j in 0..n {
        if !inside[j] {
            bn.push(j);
            bs.push(state[j]);
        }
    }
    cx.set_background(&bn, &bs);
    let cx: &Ctx<'_> = &*cx;
    let a = cx.a;
    let amax = (a - 1) as u16;
    let mg = Mg::new(cx, be, tau);
    let sk: Vec<u16> = k.iter().map(|&j| state[j].min(amax)).collect();
    let mut p: Vec<Rc<Vec<f64>>> = Vec::with_capacity(m);
    let mut vals: Vec<Vec<u16>> = Vec::with_capacity(m);
    let mut pmax: Vec<f64> = Vec::with_capacity(m);
    for &y in k {
        let py = mg.get(y, k, &sk);
        let vs = all_vals(cx, y);
        let mut mx = 0.0f64;
        for &v in &vs {
            if py[v as usize] > mx {
                mx = py[v as usize];
            }
        }
        p.push(py);
        vals.push(vs);
        pmax.push(mx);
    }
    let full: u32 = (1u32 << m) - 1;
    let mut best = f64::INFINITY;
    let mut bm = 0u32;
    for mask in 1..full {
        let v = cut_value_exact(&mg, k, &sk, &p, &vals, &pmax, mask);
        if v < best {
            best = v;
            bm = mask;
        }
        if best <= PHI_EPS {
            break;
        }
    }
    let phi = if best.is_finite() { best.max(0.0) } else { 0.0 };
    SysX { phi, src: bm, ok: !mg.bad.get() }
}
#[derive(Clone, Debug)]
pub struct FoundX {
    pub tau: usize,
    pub systems: Vec<System>,
    pub ok: bool,
}
fn best_system_exact(
    cx: &mut Ctx<'_>,
    cfg: &ExactCfg,
    state: &[u16],
    tau: usize,
    rem: &[usize],
    ok: &mut bool,
) -> Option<(Vec<usize>, f64, u32)> {
    if rem.len() < 2 {
        return None;
    }
    let mut best: Option<(Vec<usize>, f64, u32)> = None;
    for k in subsets(rem, 2, rem.len()) {
        let s = system_phi_exact(cx, cfg.backend, state, tau, &k);
        if !s.ok {
            *ok = false;
        }
        let better = match &best {
            None => true,
            Some(b) => s.phi > b.1 * (1.0 + REL_TOL),
        };
        if better {
            best = Some((k, s.phi, s.src));
        }
    }
    match best {
        Some(b) if b.1 > PHI_EPS => Some(b),
        _ => None,
    }
}
pub fn find_complexes_exact(cx: &mut Ctx<'_>, state: &[u16], cfg: &ExactCfg) -> FoundX {
    let n = cx.n;
    let all: Vec<usize> = (0..n).collect();
    let mut ok = true;
    let mut tau_star = 1usize;
    let mut first: Option<(Vec<usize>, f64, u32)> = None;
    for tau in 1..=cfg.grain_max.max(1) {
        if let Some(c) = best_system_exact(cx, cfg, state, tau, &all, &mut ok) {
            let better = match &first {
                None => true,
                Some(f) => c.1 > f.1 * (1.0 + REL_TOL),
            };
            if better {
                tau_star = tau;
                first = Some(c);
            }
        }
    }
    let mut chosen: Vec<(Vec<usize>, f64, u32)> = Vec::new();
    if let Some(f) = first {
        let mut remaining: Vec<usize> = all.iter().copied().filter(|j| !f.0.contains(j)).collect();
        chosen.push(f);
        while chosen.len() < cfg.max_cx.max(1) && remaining.len() >= 2 {
            match best_system_exact(cx, cfg, state, tau_star, &remaining, &mut ok) {
                Some(c) => {
                    remaining.retain(|j| !c.0.contains(j));
                    chosen.push(c);
                }
                None => break,
            }
        }
    }
    let systems: Vec<System> = chosen
        .into_iter()
        .map(|(k, phi, src)| {
            let (a, b) = split_mask(&k, src);
            System { nodes: k, phi_s: phi, cut_src: a, cut_dst: b, exact: true }
        })
        .collect();
    FoundX { tau: tau_star, systems, ok }
}

pub struct Model {
    pub np: usize,
    pub nl: usize,
    pub w: Vec<f64>,
    pub hold: f64,
    pub beta: f64,
    pub hy: Hyper,
    pub iters: u64,
    pub best: f64,

impl Model {
    pub fn n(&self) -> usize { self.np + self.nl }
    pub fn random(np: usize, nl: usize, hy: Hyper, rng: &mut Rng) -> Model {
        let n = np + nl;
        assert!(n >= 2, "need at least two nodes");
        let mut w = vec![0.0f64; n * n];
        for i in 0..n {
            for j in 0..n {
                if i != j {
                    w[i * n + j] = 0.7 * rng.normal();
                }
            }
        }
        let mut m = Model { np, nl, w, hold: 2.0, beta: 1.0, hy, iters: 0, best: 0.0 };
        m.enforce_nonreciprocal(NONRECIP);
        m
    }
    pub fn enforce_nonreciprocal(&mut self, delta: f64) {
        let n = self.n();
        for i in 0..n {
            self.w[i * n + i] = 0.0;
            for j in (i + 1)..n {
                let a = self.w[i * n + j];
                let b = self.w[j * n + i];
                let d = a - b;
                if d.abs() < delta {
                    let push = (delta - d.abs()) * 0.5;
                    let sgn = if d > 0.0 || (d == 0.0 && (i + j) % 2 == 0) { 1.0 } else { -1.0 };
                    self.w[i * n + j] = a + sgn * push;
                    self.w[j * n + i] = b - sgn * push;
                }
            }
        }
    }
    pub fn nonreciprocity(&self) -> f64 {
        let n = self.n();
        let mut m = f64::INFINITY;
        for i in 0..n {
            for j in (i + 1)..n {
                m = m.min((self.w[i * n + j] - self.w[j * n + i]).abs());
            }
        }
        m
    }
    pub fn n_params(&self) -> usize {
        let n = self.n();
        n * (n - 1) + 2
    }
    pub fn params(&self) -> Vec<f64> {
        let n = self.n();
        let mut p = Vec::with_capacity(self.n_params());
        for i in 0..n {
            for j in 0..n {
                if i != j {
                    p.push(self.w[i * n + j]);
                }
            }
        }
        p.push(self.hold);
        p.push(self.beta.ln());
        p
    }
    pub fn set_params(&mut self, p: &[f64]) {
        let n = self.n();
        let mut k = 0usize;
        for i in 0..n {
            for j in 0..n {
                if i != j {
                    self.w[i * n + j] = p[k].clamp(-WMAX, WMAX);
                    k += 1;
                } else {
                    self.w[i * n + j] = 0.0;
                }
            }
        }
        self.hold = p[k].clamp(0.0, 8.0);
        self.beta = p[k + 1].clamp(-2.0, 2.0).exp();
        self.enforce_nonreciprocal(NONRECIP);
    }
    pub fn describe(&self) -> String {
        format!(
            "np={} nl={} beta={:.3} hold={:.3} nonrecip>={:.3} iters={} best={:.4} B={} R={} kmax={} qmax={} grains=1..{}",
            self.np, self.nl, self.beta, self.hold, self.nonreciprocity(),
            self.iters, self.best, self.hy.budget, self.hy.draws, self.hy.kmax,
            self.hy.qmax, self.hy.grain_max
        )
    }
    pub fn save(&self, path: &str) -> IoResult<()> {
        let mut b: Vec<u8> = Vec::new();
        b.extend_from_slice(MAGIC);
        b.extend_from_slice(&VERSION.to_le_bytes());
        let ints = [
            self.np, self.nl, self.hy.kmax, self.hy.qmax, self.hy.nbr,
            self.hy.budget, self.hy.draws, self.hy.ncuts, self.hy.grain_max, self.hy.max_cx,
        ];
        for v in ints {
            b.extend_from_slice(&(v as u32).to_le_bytes());
        }
        b.extend_from_slice(&self.hy.seed.to_le_bytes());
        b.extend_from_slice(&self.iters.to_le_bytes());
        for v in [self.hold, self.beta, self.best] {
            b.extend_from_slice(&v.to_le_bytes());
        }
        for &x in &self.w {
            b.extend_from_slice(&x.to_le_bytes());
        }
        if let Some(dir) = std::path::Path::new(path).parent() {
            if !dir.as_os_str().is_empty() {
                std::fs::create_dir_all(dir)?;
            }
        }
        std::fs::write(path, b)
    }
    pub fn load(path: &str) -> IoResult<Model> {
        let bytes = std::fs::read(path)?;
        let mut r = Rd { b: &bytes, p: 0 };
        if r.take(8)? != &MAGIC[..] {
            return Err(Error::new(ErrorKind::InvalidData, "not a .phis file"));
        }
        if r.u32()? != VERSION {
            return Err(Error::new(ErrorKind::InvalidData, "unsupported .phis version"));
        }
        let np = r.u32()? as usize;
        let nl = r.u32()? as usize;
        let kmax = r.u32()? as usize;
        let qmax = r.u32()? as usize;
        let nbr = r.u32()? as usize;
        let budget = r.u32()? as usize;
        let draws = r.u32()? as usize;
        let ncuts = r.u32()? as usize;
        let grain_max = r.u32()? as usize;
        let max_cx = r.u32()? as usize;
        let seed = r.u64()?;
        let iters = r.u64()?;
        let hold = r.f64()?;
        let beta = r.f64()?;
        let best = r.f64()?;
        let n = np + nl;
        if n < 2 || n > 4096 {
            return Err(Error::new(ErrorKind::InvalidData, "bad node count"));
        }
        let mut w = Vec::with_capacity(n * n);
        for _ in 0..n * n {
            w.push(r.f64()?);
        }
        let hy = Hyper { kmax, qmax, nbr, budget, draws, ncuts, grain_max, max_cx, seed };
        Ok(Model { np, nl, w, hold, beta, hy, iters, best })
    }
}
struct Rd<'a> {
    b: &'a [u8],
    p: usize,
}
impl<'a> Rd<'a> {
    fn take(&mut self, k: usize) -> IoResult<&'a [u8]> {
        if self.p + k > self.b.len() {
            return Err(Error::new(ErrorKind::UnexpectedEof, "truncated .phis file"));
        }
        let s = &self.b[self.p..self.p + k];
        self.p += k;
        Ok(s)
    }
    fn u32(&mut self) -> IoResult<u32> {
        let s = self.take(4)?;
        Ok(u32::from_le_bytes([s[0], s[1], s[2], s[3]]))
    }
    fn u64(&mut self) -> IoResult<u64> {
        let s = self.take(8)?;
        Ok(u64::from_le_bytes([s[0], s[1], s[2], s[3], s[4], s[5], s[6], s[7]]))
    }
    fn f64(&mut self) -> IoResult<f64> {
        let s = self.take(8)?;
        Ok(f64::from_le_bytes([s[0], s[1], s[2], s[3], s[4], s[5], s[6], s[7]]))
    }
}

        let a = alphabet.len();

    }

        let mut v = Vec::new();
        let mut s = 0usize;
        while s < self.len() {
            let l = np.min(self.len() - s);
            v.push((s, l));
            s += l;
        }
        v
    }

fn combos(u: &[usize], size: usize, start: usize, cur: &mut Vec<usize>, out: &mut Vec<Vec<usize>>) {
    if cur.len() == size {
        out.push(cur.clone());
        return;

    for i in start..u.len() {
        if u.len() - i < size - cur.len() {
            break;
        }
        cur.push(u[i]);

        cur.pop();
    }
}

    ctab_sz: Cell<usize>,
}
impl<'c, 'm> Phi<'c, 'm> {
    pub fn new(cx: &'c Ctx<'m>, tau: usize, state: &[u16], cfg: PhiCfg) -> Phi<'c, 'm> {
        let n = cx.n;
        assert!(state.len() >= n, "state must hold one symbol per node");
        let amax = (cx.a - 1) as u16;
        let st: Vec<u16> = state[..n].iter().map(|&s| s.min(amax)).collect();
        let mut nin: Vec<Vec<usize>> = vec![Vec::new(); n];
        let mut nout: Vec<Vec<usize>> = vec![Vec::new(); n];
        for &j in cx.free_nodes() {
            let mut vi: Vec<(usize, f64)> = Vec::new();
            let mut vo: Vec<(usize, f64)> = Vec::new();
            for &i in cx.free_nodes() {
                let wi = cx.weight(i, j).abs(); 
                if wi > 0.0 {
                    vi.push((i, wi));
                }
                let wo = cx.weight(j, i).abs(); 
                if wo > 0.0 {
                    vo.push((i, wo));
                }
            }
            nin[j] = top_ids(vi, cfg.nbr);
            nout[j] = top_ids(vo, cfg.nbr);
        }
        Phi {
            cx,
            tau: tau.max(1),
            state: st,
            cfg,
            nin,
            nout,
            ctab: RefCell::new(HashMap::default()),
            ctab_sz: Cell::new(0),
        }
    }
    pub fn standard(cx: &'c Ctx<'m>, tau: usize, state: &[u16]) -> Phi<'c, 'm> {
        Phi::new(cx, tau, state, PhiCfg::from_model(cx.m))
    }
    pub fn mechanisms(&self) -> Vec<Vec<usize>> {
        let n = self.cx.n;
        let free = self.cx.free_nodes();
        let mut adj: Vec<Vec<usize>> = vec![Vec::new(); n];
        for &i in free {
            for &j in self.nin[i].iter().chain(self.nout[i].iter()) {
                if i != j {
                    if !adj[i].contains(&j) {
                        adj[i].push(j);
                    }
                    if !adj[j].contains(&i) {
                        adj[j].push(i);
                    }
                }
            }
        }
        let mut all: Vec<Vec<usize>> = free.iter().map(|&i| vec![i]).collect();
        let mut layer: Vec<Vec<usize>> = all.clone();
        for _ in 2..=self.cfg.kmax {
            let mut seen: HashSet<Vec<usize>> = HashSet::new();
            let mut next: Vec<Vec<usize>> = Vec::new();
            for ms in &layer {
                for &x in ms {
                    for &y in &adj[x] {
                        if ms.contains(&y) {
                            continue;
                        }
                        let mut v = ms.clone();
                        v.push(y);
                        v.sort_unstable();
                        if seen.insert(v.clone()) {
                            next.push(v);
                        }
                    }
                }
            }
            next.sort();
            all.extend(next.iter().cloned());
            layer = next;
            if layer.is_empty() || all.len() >= self.cfg.max_mech {
                break;
            }
        }
        all.sort_by(|x, y| x.len().cmp(&y.len()).then_with(|| x.cmp(y)));
        all.truncate(self.cfg.max_mech);
        all
    }
    pub fn universe(&self, m: &[usize], dir: Dir) -> Vec<usize> {
        let lists = match dir {
            Dir::Cause => &self.nin,
            Dir::Effect => &self.nout,
        };
        let cap = 2 * self.cfg.nbr;
        let mut seen = vec![false; self.cx.n];
        let mut out: Vec<usize> = Vec::new();
        let mut front: Vec<usize> = m.to_vec();
        'hops: for _ in 0..self.tau {
            let mut next: Vec<usize> = Vec::new();
            for r in 0..self.cfg.nbr {
                for &x in &front {
                    if let Some(&y) = lists[x].get(r) {
                        if !seen[y] {
                            seen[y] = true;
                            next.push(y);
                            out.push(y);
                            if out.len() >= cap {
                                break 'hops;
                            }
                        }
                    }
                }
            }
            if next.is_empty() {
                break;
            }
            front = next;
        }
        out.sort_unstable();
        out
    }
    pub fn purviews(&self, u: &[usize]) -> Vec<Vec<usize>> {
        let a = self.cx.a;
        let mut out: Vec<Vec<usize>> = Vec::new();
        for size in 1..=self.cfg.qmax.min(u.len()) {
            if ipow(a, size) > self.cfg.max_states {
                break;
            }
            let mut cur: Vec<usize> = Vec::new();
            combos(u, size, 0, &mut cur, &mut out);
        }
        out
    }
    pub(crate) fn cause_table(&self, m: &[usize], z: &[usize]) -> Rc<Vec<f64>> {
        let mut ks: Vec<usize> = Vec::with_capacity(m.len() + z.len() + 1);
        ks.extend_from_slice(m);
        ks.push(SEP);
        ks.extend_from_slice(z);
        let key = hash_usize(TAG_CTAB, &ks);
        let hit = self.ctab.borrow().get(&key).cloned();
        if let Some(t) = hit {
            return t;
        }
        let a = self.cx.a;
        let k = z.len();
        let total = ipow(a, k);
        let sm: Vec<u16> = m.iter().map(|&j| self.state[j]).collect();
        let mut tab = vec![0.0f64; total];
        let mut zs = vec![0u16; k];
        let mut out = vec![0.0f64; a];
        let mut sum = 0.0f64;
        for idx in 0..total {
            decode(idx, a, &mut zs);
            let mut l = 1.0f64;
            for (t, &j) in m.iter().enumerate() {
                self.cx.marg(j, z, self.tau, &zs, &mut out);
                l *= out[sm[t] as usize];
            }
            tab[idx] = l;
            sum += l;
        }
        if sum > TINY {
            let inv = 1.0 / sum;
            for v in tab.iter_mut() {
                *v *= inv;
            }
        } else {
            tab.fill(1.0 / total as f64);
        }
        let rc = Rc::new(tab);
        let mut c = self.ctab.borrow_mut();
        if self.ctab_sz.get() + total > CACHE_CAP {
            c.clear();
            self.ctab_sz.set(0);
        }
        self.ctab_sz.set(self.ctab_sz.get() + total);
        c.insert(key, Rc::clone(&rc));
        rc
    }
    fn partitioned_cause(&self, m: &[usize], z: &[usize], lab: &[u8], apow: &[usize], q: &mut [f64]) {
        let a = self.cx.a;
        let km = m.len();
        let kz = z.len();
        let nb = lab.iter().copied().max().map_or(0, |x| x as usize + 1);
        q.fill(1.0);
        for b in 0..nb {
            let bb = b as u8;
            let mb: Vec<usize> = (0..km).filter(|&i| lab[i] == bb).map(|i| m[i]).collect();
            let zpos: Vec<usize> = (0..kz).filter(|&i| lab[km + i] == bb).collect();
            if zpos.is_empty() {
                continue; 
            }
            if mb.is_empty() {
                let f = 1.0 / (ipow(a, zpos.len()) as f64);
                for v in q.iter_mut() {
                    *v *= f;
                }
            } else {
                let zb: Vec<usize> = zpos.iter().map(|&i| z[i]).collect();
                let tab = self.cause_table(&mb, &zb);
                for (idx, v) in q.iter_mut().enumerate() {
                    *v *= tab[sub_index(idx, a, &zpos, apow)];
                }
            }
        }
    }
    fn cause_eval(&self, m: &[usize], z: &[usize], floor: f64) -> SideEval {
        let a = self.cx.a;
        let km = m.len();
        let kz = z.len();
        let total = ipow(a, kz);
        let p = self.cause_table(m, z);
        let apow: Vec<usize> = (0..kz).map(|i| ipow(a, i)).collect();
        let parts = set_partitions(km, kz);
        let mut q = vec![0.0f64; total];
        let mut best = f64::INFINITY;
        let mut best_q: Vec<f64> = Vec::new();
        for lab in &parts {
            self.partitioned_cause(m, z, lab, &apow, &mut q);
            let (v, _) = id_max(&p, &q);
            if v < best {
                best = v;
                best_q.clear();
                best_q.extend_from_slice(&q);
            }
            if best <= PHI_EPS {
                return SideEval { phi: best, states: Vec::new(), exact: true };
            }
            if best < floor {
                return SideEval { phi: best, states: Vec::new(), exact: false };
            }
        }
        if !best.is_finite() {
            return SideEval { phi: 0.0, states: Vec::new(), exact: true };
        }
        let tol = TIE_REL * best.max(1.0);
        let mut states: Vec<Vec<u16>> = Vec::new();
        for idx in 0..total {
            if id_term(p[idx], best_q[idx]) >= best - tol {
                let mut zs = vec![0u16; kz];
                decode(idx, a, &mut zs);
                states.push(zs);
                if states.len() >= MAX_TIES {
                    break;
                }
            }
        }
        SideEval { phi: best, states, exact: true }
    }
    fn effect_eval(&self, m: &[usize], z: &[usize], floor: f64) -> SideEval {
        let a = self.cx.a;
        let km = m.len();
        let kz = z.len();
        let sm: Vec<u16> = m.iter().map(|&j| self.state[j]).collect();
        let pz: Vec<Vec<f64>> = z
            .iter()
            .map(|&y| self.cx.marg_vec(y, m, self.tau, &sm))
            .collect();
        let fixed: Vec<Option<u16>> = z.iter().map(|&y| self.cx.e_sym(y)).collect();
        let lat: Vec<usize> = (0..kz).filter(|&i| fixed[i].is_none()).collect();
        let nstate = ipow(a, lat.len());
        let mut cand = vec![0u16; nstate * kz];
        let mut pc = vec![1.0f64; nstate];
        let mut digits = vec![0u16; lat.len()];
        for c in 0..nstate {
            decode(c, a, &mut digits);
            let mut t = 0usize;
            for i in 0..kz {
                let s = match fixed[i] {
                    Some(s) => s,
                    None => {
                        let d = digits[t];
                        t += 1;
                        d
                    }
                };
                cand[c * kz + i] = s;
                pc[c] *= pz[i][s as usize];
            }
        }
        let parts = set_partitions(km, kz);
        let mut qy: Vec<Vec<f64>> = vec![vec![0.0f64; a]; kz];
        let mut qc = vec![0.0f64; nstate];
        let mut best = f64::INFINITY;
        let mut best_q: Vec<f64> = Vec::new();
        for lab in &parts {
            for i in 0..kz {
                let bb = lab[km + i];
                let mut mb: Vec<usize> = Vec::new();
                let mut smb: Vec<u16> = Vec::new();
                for t in 0..km {
                    if lab[t] == bb {
                        mb.push(m[t]);
                        smb.push(sm[t]);
                    }
                }
                self.cx.marg(z[i], &mb, self.tau, &smb, &mut qy[i]);
            }
            let mut v = 0.0f64;
            for c in 0..nstate {
                let mut q = 1.0f64;
                for i in 0..kz {
                    q *= qy[i][cand[c * kz + i] as usize];
                }
                qc[c] = q;
                let t = id_term(pc[c], q);
                if t > v {
                    v = t;
                }
            }
            if v < best {
                best = v;
                best_q.clear();
                best_q.extend_from_slice(&qc);
            }
            if best <= PHI_EPS {
                return SideEval { phi: best, states: Vec::new(), exact: true };
            }
            if best < floor {
                return SideEval { phi: best, states: Vec::new(), exact: false };
            }
        }
        if !best.is_finite() {
            return SideEval { phi: 0.0, states: Vec::new(), exact: true };
        }
        let tol = TIE_REL * best.max(1.0);
        let mut states: Vec<Vec<u16>> = Vec::new();
        for c in 0..nstate {
            if id_term(pc[c], best_q[c]) >= best - tol {
                states.push(cand[c * kz..(c + 1) * kz].to_vec());
                if states.len() >= MAX_TIES {
                    break;
                }
            }
        }
        SideEval { phi: best, states, exact: true }
    }
    pub fn phi_cause(&self, m: &[usize], z: &[usize]) -> (f64, Vec<Vec<u16>>) {
        let (m, z) = (norm(m), norm(z));
        if m.is_empty() || z.is_empty() {
            return (0.0, Vec::new());
        }
        debug_assert!(m.iter().chain(z.iter()).all(|&i| self.cx.pinned(i).is_none()));
        let ev = self.cause_eval(&m, &z, -1.0);
        if ev.phi <= PHI_EPS { (0.0, Vec::new()) } else { (ev.phi, ev.states) }
    }
    pub fn phi_effect(&self, m: &[usize], z: &[usize]) -> (f64, Vec<Vec<u16>>) {
        let (m, z) = (norm(m), norm(z));
        if m.is_empty() || z.is_empty() {
            return (0.0, Vec::new());
        }
        debug_assert!(m.iter().chain(z.iter()).all(|&i| self.cx.pinned(i).is_none()));
        let ev = self.effect_eval(&m, &z, -1.0);
        if ev.phi <= PHI_EPS { (0.0, Vec::new()) } else { (ev.phi, ev.states) }
    }
    fn best_side(&self, m: &[usize], cands: &[Vec<usize>], dir: Dir) -> (f64, Vec<Face>) {
        let mut best = 0.0f64;
        let mut faces: Vec<Face> = Vec::new();
        for z in cands {
            let floor = if best > PHI_EPS { best - TIE_REL * best.max(1.0) } else { -1.0 };
            let ev = match dir {
                Dir::Cause => self.cause_eval(m, z, floor),
                Dir::Effect => self.effect_eval(m, z, floor),
            };
            if !ev.exact || ev.phi <= PHI_EPS {
                continue;
            }
            let tol = TIE_REL * best.max(ev.phi).max(1.0);
            if ev.phi > best + tol {
                best = ev.phi;
                faces.clear();
                faces.push(Face { nodes: z.clone(), phi: ev.phi, states: ev.states });
            } else if ev.phi >= best - tol {
                if faces.len() < MAX_TIES {
                    faces.push(Face { nodes: z.clone(), phi: ev.phi, states: ev.states });
                }
                if ev.phi > best {
                    best = ev.phi;
                }
            }
        }
        (best, faces)
    }
    pub fn distinctions(&self, only_anchored: bool) -> Vec<Dist> {
        let mut out: Vec<Dist> = Vec::new();
        for m in self.mechanisms() {
            let uc = self.universe(&m, Dir::Cause);
            let ue = self.universe(&m, Dir::Effect);
            if only_anchored && !(self.cx.touches_e(&uc) || self.cx.touches_e(&ue)) {
                continue;
            }
            let cc = self.purviews(&uc);
            let ce = self.purviews(&ue);
            let (phi_c, fc) = self.best_side(&m, &cc, Dir::Cause);
            let (phi_e, fe) = self.best_side(&m, &ce, Dir::Effect);
            if phi_c <= PHI_EPS && phi_e <= PHI_EPS {
                continue;
            }
            let anchored = fc.iter().chain(fe.iter()).any(|f| self.cx.touches_e(&f.nodes));
            let mech_state: Vec<u16> = m.iter().map(|&j| self.state[j]).collect();
            out.push(Dist {
                mech: m,
                mech_state,
                cause: fc,
                effect: fe,
                phi_c,
                phi_e,
                phi_d: phi_c.min(phi_e),
                anchored,
            });
        }
        out
    }
    pub fn dist_e(&self) -> Vec<Dist> {
        self.distinctions(true)
            .into_iter()
            .filter(|d| d.anchored && d.is_distinction())
            .collect()
    }
    pub fn footprint(&self, d: &Dist) -> Vec<usize> {
        let mut v: Vec<usize> = Vec::new();
        for f in d.cause.iter().chain(d.effect.iter()) {
            for &j in &f.nodes {
                if let Some(p) = self.cx.e_pos(j) {
                    v.push(p);
                }
            }
        }
        v.sort_unstable();
        v.dedup();
        v
    }
    pub fn repertoire_marginals(&self, m: &[usize], z: &[usize], dir: Dir) -> Vec<Vec<f64>> {
        let a = self.cx.a;
        let (m, z) = (norm(m), norm(z));
        let kz = z.len();
        let mut out = vec![vec![0.0f64; a]; kz];
        if m.is_empty() || kz == 0 {
            for row in out.iter_mut() {
                row.fill(1.0 / a as f64);
            }
            return out;
        }
        match dir {
            Dir::Effect => {
                let sm: Vec<u16> = m.iter().map(|&j| self.state[j]).collect();
                for (i, &y) in z.iter().enumerate() {
                    self.cx.marg(y, &m, self.tau, &sm, &mut out[i]);
                }
            }
            Dir::Cause => {
                let p = self.cause_table(&m, &z);
                let apow: Vec<usize> = (0..kz).map(|i| ipow(a, i)).collect();
                for idx in 0..p.len() {
                    for i in 0..kz {
                        out[i][(idx / apow[i]) % a] += p[idx];
                    }
                }
            }
        }
        out
    }
    fn fmt_face(&self, f: &Face) -> String {
        let mut s = format!("{{{}}}=", join_ids(&f.nodes));
        for (k, st) in f.states.iter().enumerate() {
            if k > 0 {
                s.push('|');
            }
            for &v in st {
                match self.cx.alphabet.get(v as usize) {
                    Some(&b) => s.push_str(&show_byte(b)),
                    None => s.push('?'),
                }
            }
        }
        s.push_str(&format!(" phi={:.4}", f.phi));
        s
    }
    pub fn describe(&self, d: &Dist) -> String {
        let mut s = format!(
            "m{{{}}} phi_d={:.4} (c={:.4} e={:.4}){}",
            join_ids(&d.mech),
            d.phi_d,
            d.phi_c,
            d.phi_e,
            if d.anchored { " [E]" } else { "" }
        );
        for f in &d.cause {
            s.push_str(&format!("\n  cause  {}", self.fmt_face(f)));
        }
        for f in &d.effect {
            s.push_str(&format!("\n  effect {}", self.fmt_face(f)));
        }
        s
    }

        }
    }
}
impl Cfg {
    pub fn level(k: usize) -> Cfg {
        let mut c = Cfg::default();
        match k {
            0 => {
                c.funs = (2, 2);
                c.cons = (2, 2);
                c.vars = 2;
                c.rules = (1, 1);
                c.nest = 0.0;
                c.lit = 0.0;
                c.rhs = 3;
                c.qdepth = 1;
                c.share = 0.0;
                c.steps = (1, 1);
                c.rec_max = 6;
                c.term_max = 6;
                c.style = Some(Style::QA);
            }
            1 => {
                c.funs = (2, 3);
                c.cons = (2, 3);
                c.vars = 3;
                c.rules = (1, 2);
                c.nest = 0.0;
                c.lit = 0.0;
                c.rhs = 4;
                c.qdepth = 1;
                c.share = 0.1;
                c.steps = (1, 2);
                c.rec_max = 7;
                c.term_max = 7;
            }
            2 => {
                c.rules = (2, 2);
                c.nest = 0.15;
                c.lit = 0.05;
            }
            3 => {}
            _ => {
                c.funs = (3, 4);
                c.cons = (3, 4);
                c.rules = (3, 4);
                c.rhs = 6;
                c.steps = (2, 4);
                c.nest = 0.3;
                c.share = 0.2;
                c.rec_max = 10;
                c.term_max = 12;
            }
        }
        c
    }
    pub fn fit(mut self, np: usize, aligned: bool) -> Cfg {
        let h = (np.saturating_sub(2) / 2).max(3);
        self.rec_max = self.rec_max.min(h);
        self.term_max = self.term_max.min(h).max(4);
        self.rhs = self.rhs.min(h);
        self.align = if aligned { np } else { 0 };
        self
    }
}
#[derive(Clone, Debug)]
pub struct Sig {
    pub arity: Vec<usize>,
    pub name: Vec<char>,
    pub arrow: u16,
    pub sep: u16,
    pub pad: u16,
    pub vars: Vec<u16>,
    pub cons: Vec<u16>,
    pub funs: Vec<u16>,
}
impl Sig {
    pub fn n(&self) -> usize {
        self.arity.len()
    }
    fn add(&mut self, a: usize, c: char) -> u16 {
        self.arity.push(a);
        self.name.push(c);
        (self.arity.len() - 1) as u16
    }
}
fn sample_sig(rng: &mut Rng, cfg: &Cfg) -> Sig {
    let nf = rr(rng, cfg.funs).clamp(2, FUNN.len());
    let nc = rr(rng, cfg.cons).clamp(1, CONN.len());
    let nv = cfg.vars.clamp(1, VARN.len());
    let am = cfg.arity.clamp(1, 3);
    let mut s = Sig {
        arity: vec![0; 3],
        name: vec!['>', ';', '_'],
        arrow: 0,
        sep: 1,
        pad: 2,
        vars: Vec::new(),
        cons: Vec::new(),
        funs: Vec::new(),
    };
    for i in 0..nv {
        let id = s.add(0, VARN[i]);
        s.vars.push(id);
    }
    for i in 0..nc {
        let id = s.add(0, CONN[i]);
        s.cons.push(id);
    }
    for i in 0..nf {
        let a = if am >= 2 && rng.chance(0.65) { 2 } else { rng.range(1, am) };
        let id = s.add(a, FUNN[i]);
        s.funs.push(id);
    }
    s
}
#[derive(Clone, Debug, PartialEq)]
pub enum Pat {
    Var(usize),
    App(u16, Vec<Pat>),
}
#[derive(Clone, Debug)]
pub struct Rule {
    pub lhs: Pat,
    pub rhs: Pat,
    pub w: f64,
    pub nv: usize,
}
impl Rule {
    pub fn head(&self) -> u16 {
        match &self.lhs {
            Pat::App(s, _) => *s,
            Pat::Var(_) => IND,
        }
    }
}
fn has_var(p: &Pat) -> bool {
    match p {
        Pat::Var(_) => true,
        Pat::App(_, ks) => ks.iter().any(has_var),
    }
}
fn lhs_arg(rng: &mut Rng, sig: &Sig, cfg: &Cfg, lvl: usize, next: &mut usize) -> Pat {
    let nv = sig.vars.len();
    if lvl >= 1 && rng.chance(cfg.nest) {
        let h = sig.funs[rng.below(lvl)];
        let ah = sig.arity[h as usize];
        if *next + ah <= nv {
            let mut vs = Vec::with_capacity(ah);
            for _ in 0..ah {
                vs.push(Pat::Var(*next));
                *next += 1;
            }
            return Pat::App(h, vs);
        }
    }
    if *next >= nv || rng.chance(cfg.lit) {
        return Pat::App(sig.cons[rng.below(sig.cons.len())], Vec::new());
    }
    *next += 1;
    Pat::Var(*next - 1)
}
fn rhs_app(rng: &mut Rng, sig: &Sig, cfg: &Cfg, lvl: usize, nv: usize, left: &mut usize) -> Pat {
    let f = sig.funs[rng.below(lvl)];
    *left = (*left).saturating_sub(1);
    let mut ks = Vec::with_capacity(sig.arity[f as usize]);
    for _ in 0..sig.arity[f as usize] {
        ks.push(rhs_node(rng, sig, cfg, lvl, nv, left));
    }
    Pat::App(f, ks)
}
fn rhs_node(rng: &mut Rng, sig: &Sig, cfg: &Cfg, lvl: usize, nv: usize, left: &mut usize) -> Pat {
    if *left > 1 && !rng.chance(cfg.leaf) {
        return rhs_app(rng, sig, cfg, lvl, nv, left);
    }
    *left = (*left).saturating_sub(1);
    if nv > 0 && rng.chance(0.7) {
        Pat::Var(rng.below(nv))
    } else {
        Pat::App(sig.cons[rng.below(sig.cons.len())], Vec::new())
    }
}
fn make_rule(rng: &mut Rng, sig: &Sig, cfg: &Cfg, lvl: usize) -> Rule {
    let head = sig.funs[lvl];
    let mut next = 0usize;
    let mut kids = Vec::new();
    for _ in 0..sig.arity[head as usize] {
        kids.push(lhs_arg(rng, sig, cfg, lvl, &mut next));
    }
    let lhs = Pat::App(head, kids);
    let nv = next;
    let mut rhs = Pat::App(sig.cons[0], Vec::new());
    for _ in 0..16 {
        let mut left = cfg.rhs.max(1);
        rhs = if left >= 2 && rng.chance(0.85) {
            rhs_app(rng, sig, cfg, lvl, nv, &mut left)
        } else {
            rhs_node(rng, sig, cfg, lvl, nv, &mut left)
        };
        if nv == 0 || has_var(&rhs) {
            break;
        }
    }
    Rule { lhs, rhs, w: 0.5 + rng.f64(), nv }
}
fn sample_rules(rng: &mut Rng, sig: &Sig, cfg: &Cfg) -> Vec<Rule> {
    let nr = rr(rng, cfg.rules).max(1);
    let mut heads: Vec<usize> = (1..sig.funs.len()).collect();
    rng.shuffle(&mut heads);
    let mut out = Vec::with_capacity(nr);
    for i in 0..nr {
        let lvl = if i < heads.len() { heads[i] } else { heads[rng.below(heads.len())] };
        out.push(make_rule(rng, sig, cfg, lvl));
    }
    out
}
#[derive(Clone, Copy, Debug)]
pub struct Node {
    pub sym: u16,
    pub kids: [usize; 3],
    pub uid: u32,
}
#[derive(Clone, Debug, Default)]
pub struct Graph {
    pub nodes: Vec<Node>,
}
#[derive(Clone, Debug)]
pub struct Redex {
    pub node: usize,
    pub rule: usize,
    pub bind: Vec<usize>,
}
#[derive(Clone, Copy, Debug)]
pub struct Hot<'a> {
    pub node: usize,
    pub rule: usize,
    pub side: u32,
    pub pat: &'a Pat,
}
impl Graph {
    pub fn add(&mut self, sym: u16, kids: &[usize]) -> usize {
        let mut k = [NIL; 3];
        for (j, &x) in kids.iter().take(3).enumerate() {
            k[j] = x;
        }
        let id = self.nodes.len();
        self.nodes.push(Node { sym, kids: k, uid: id as u32 });
        id
    }
    pub fn deref(&self, mut i: usize) -> usize {
        while self.nodes[i].sym == IND {
            i = self.nodes[i].kids[0];
        }
        i
    }
    fn chain(&self, mut i: usize, via: usize) -> (usize, bool) {
        let mut hit = false;
        loop {
            if i == via {
                hit = true;
            }
            if self.nodes[i].sym != IND {
                return (i, hit);
            }
            i = self.nodes[i].kids[0];
        }
    }
    pub fn size(&self, root: usize, cap: usize) -> usize {
        let mut n = 0usize;
        let mut st: Vec<usize> = vec![root];
        while let Some(i) = st.pop() {
            n += 1;
            if n > cap {
                return n;
            }
            let d = self.deref(i);
            for k in 0..3 {
                let c = self.nodes[d].kids[k];
                if c != NIL {
                    st.push(c);
                }
            }
        }
        n
    }
    pub fn matches(&self, p: &Pat, i: usize, bind: &mut [usize]) -> bool {
        let d = self.deref(i);
        match p {
            Pat::Var(v) => {
                if bind[*v] == NIL {
                    bind[*v] = d;
                    true
                } else {
                    bind[*v] == d
                }
            }
            Pat::App(s, ks) => {
                let n = self.nodes[d];
                if n.sym != *s {
                    return false;
                }
                for (k, q) in ks.iter().enumerate() {
                    if k >= 3 || n.kids[k] == NIL || !self.matches(q, n.kids[k], bind) {
                        return false;
                    }
                }
                true
            }
        }
    }
    pub fn redexes(&self, rules: &[Rule], root: usize) -> Vec<Redex> {
        let mut out: Vec<Redex> = Vec::new();
        let mut seen = vec![false; self.nodes.len()];
        let mut st: Vec<usize> = vec![root];
        while let Some(i) = st.pop() {
            let d = self.deref(i);
            if seen[d] {
                continue;
            }
            seen[d] = true;
            for (r, rule) in rules.iter().enumerate() {
                let mut bind = vec![NIL; rule.nv];
                if self.matches(&rule.lhs, d, &mut bind) {
                    out.push(Redex { node: d, rule: r, bind });
                }
            }
            let n = self.nodes[d];
            for k in (0..3).rev() {
                if n.kids[k] != NIL {
                    st.push(n.kids[k]);
                }
            }
        }
        out
    }
    pub fn build(&mut self, p: &Pat, bind: &[usize]) -> usize {
        match p {
            Pat::Var(v) => bind[*v],
            Pat::App(s, ks) => {
                let n = ks.len().min(3);
                let mut kids = [NIL; 3];
                for k in 0..n {
                    kids[k] = self.build(&ks[k], bind);
                }
                self.add(*s, &kids[..n])
            }
        }
    }
    pub fn apply(&mut self, rules: &[Rule], rd: &Redex) -> usize {
        let res = self.build(&rules[rd.rule].rhs, &rd.bind);
        if res != rd.node {
            let n = &mut self.nodes[rd.node];
            n.sym = IND;
            n.kids = [res, NIL, NIL];
        }
        res
    }
    pub fn emit(&self, i: usize, hot: Option<&Hot<'_>>, out: &mut Vec<Cell>) {
        let via = match hot {
            Some(h) => h.node,
            None => NIL,
        };
        let (d, hit) = self.chain(i, via);
        if hit {
            if let Some(h) = hot {
                let mut pos = 0usize;
                self.walk(d, h, h.pat, &mut pos, out);
                return;
            }
        }
        let n = self.nodes[d];
        out.push(Cell { sym: n.sym, uid: n.uid, ent: NONE });
        for k in 0..3 {
            if n.kids[k] != NIL {
                self.emit(n.kids[k], hot, out);
            }
        }
    }
    fn walk(&self, i: usize, h: &Hot<'_>, p: &Pat, pos: &mut usize, out: &mut Vec<Cell>) {
        let at = *pos;
        *pos += 1;
        match p {
            Pat::Var(v) => {
                let s = out.len();
                self.emit(i, None, out);
                out[s].ent = ent(CL_VAR, h.rule, *v);
            }
            Pat::App(_, ks) => {
                let d = self.deref(i);
                let n = self.nodes[d];
                out.push(Cell { sym: n.sym, uid: n.uid, ent: ent(h.side, h.rule, at) });
                for (k, q) in ks.iter().enumerate() {
                    if k < 3 && n.kids[k] != NIL {
                        self.walk(n.kids[k], h, q, pos, out);
                    }
                }
            }
        }
    }
    pub fn emit_lhs(&self, root: usize, rules: &[Rule], rd: &Redex, out: &mut Vec<Cell>) {
        let hot = Hot { node: rd.node, rule: rd.rule, side: CL_LHS, pat: &rules[rd.rule].lhs };
        self.emit(root, Some(&hot), out);
    }
    pub fn emit_rhs(&self, root: usize, rules: &[Rule], rd: &Redex, out: &mut Vec<Cell>) {
        let hot = Hot { node: rd.node, rule: rd.rule, side: CL_RHS, pat: &rules[rd.rule].rhs };
        self.emit(root, Some(&hot), out);
    }
}
#[derive(Clone, Debug)]
pub struct Step {
    pub rule: usize,
    pub before: Vec<Cell>,
    pub after: Vec<Cell>,
}
#[derive(Clone, Debug)]
pub struct Sim {
    pub steps: Vec<Step>,
    pub first: Vec<Cell>,
    pub last: Vec<Cell>,
    pub normal: bool,
}
struct QGen<'a> {
    sig: &'a Sig,
    rules: &'a [Rule],
    cfg: &'a Cfg,
    pool: Vec<usize>,
}
impl<'a> QGen<'a> {
    fn leaf(&mut self, rng: &mut Rng, g: &mut Graph) -> usize {
        let c = self.sig.cons[rng.below(self.sig.cons.len())];
        let id = g.add(c, &[]);
        self.pool.push(id);
        id
    }
    fn node(&mut self, rng: &mut Rng, g: &mut Graph, depth: usize, top: bool) -> usize {
        if !top && !self.pool.is_empty() && rng.chance(self.cfg.share) {
            return self.pool[rng.below(self.pool.len())];
        }
        let p_inst = if top { 0.8 } else { 0.4 };
        if depth > 0 && !self.rules.is_empty() && rng.chance(p_inst) {
            let rules = self.rules;
            let lhs = &rules[rng.below(rules.len())].lhs;
            return self.inst(rng, g, lhs, depth);
        }
        if depth == 0 || (!top && rng.chance(0.2)) {
            return self.leaf(rng, g);
        }
        let f = self.sig.funs[rng.below(self.sig.funs.len())];
        let ar = self.sig.arity[f as usize].min(3);
        let mut kids = [NIL; 3];
        for k in 0..ar {
            kids[k] = self.node(rng, g, depth - 1, false);
        }
        let id = g.add(f, &kids[..ar]);
        self.pool.push(id);
        id
    }
    fn inst(&mut self, rng: &mut Rng, g: &mut Graph, p: &Pat, depth: usize) -> usize {
        match p {
            Pat::Var(_) => self.node(rng, g, depth.saturating_sub(1), false),
            Pat::App(s, ks) => {
                let n = ks.len().min(3);
                let mut kids = [NIL; 3];
                for k in 0..n {
                    kids[k] = self.inst(rng, g, &ks[k], depth);
                }
                let id = g.add(*s, &kids[..n]);
                self.pool.push(id);
                id
            }
        }
    }
}
pub fn sample_query(rng: &mut Rng, sig: &Sig, rules: &[Rule], cfg: &Cfg, g: &mut Graph) -> usize {
    let mut q = QGen { sig, rules, cfg, pool: Vec::new() };
    q.node(rng, g, cfg.qdepth.max(1), true)
}
pub fn simulate(rng: &mut Rng, g: &mut Graph, rules: &[Rule], root: usize, cfg: &Cfg, limit: usize) -> Option<Sim> {
    let cap = cfg.term_max;
    if g.size(root, cap) > cap {
        return None;
    }
    let mut first: Vec<Cell> = Vec::new();
    g.emit(root, None, &mut first);
    let mut steps: Vec<Step> = Vec::new();
    for _ in 0..limit {
        let mut cand = g.redexes(rules, root);
        let mut pick: Option<(Graph, Redex)> = None;
        while !cand.is_empty() {
            let w: Vec<f64> = cand.iter().map(|r| rules[r.rule].w).collect();
            let rd = cand.swap_remove(rng.weighted(&w));
            let mut g2 = g.clone();
            g2.apply(rules, &rd);
            if g2.size(root, cap) <= cap {
                pick = Some((g2, rd));
                break;
            }
        }
        let (g2, rd) = match pick {
            Some(x) => x,
            None => break,
        };
        let mut before: Vec<Cell> = Vec::new();
        g.emit_lhs(root, rules, &rd, &mut before);
        let mut after: Vec<Cell> = Vec::new();
        g2.emit_rhs(root, rules, &rd, &mut after);
        steps.push(Step { rule: rd.rule, before, after });
        *g = g2;
    }
    let mut last: Vec<Cell> = Vec::new();
    g.emit(root, None, &mut last);
    let normal = g.redexes(rules, root).is_empty();
    Some(Sim { steps, first, last, normal })
}
fn emit_pat(sig: &Sig, p: &Pat, rule: usize, side: u32, pos: &mut usize, out: &mut Vec<Cell>) {
    let at = *pos;
    *pos += 1;
    match p {
        Pat::Var(v) => out.push(Cell { sym: sig.vars[*v], uid: NONE, ent: ent(CL_VAR, rule, *v) }),
        Pat::App(s, ks) => {
            out.push(Cell { sym: *s, uid: NONE, ent: ent(side, rule, at) });
            for q in ks {
                emit_pat(sig, q, rule, side, pos, out);
            }
        }
    }
}
struct Asm<'a> {
    sig: &'a Sig,
    cells: Vec<Cell>,
    blocks: Vec<Block>,
    align: usize,
}
impl<'a> Asm<'a> {
    fn pad_to_frame(&mut self) {
        if self.align > 0 {
            while self.cells.len() % self.align != 0 {
                self.cells.push(tok(self.sig.pad));
            }
        }
    }
    fn put(&mut self, kind: BlockKind, rule: usize, step: usize, body: Vec<Cell>) {
        let used = if self.align > 0 { self.cells.len() % self.align } else { 0 };
        if used > 0 && used + body.len() > self.align {
            self.pad_to_frame();
        }
        let start = self.cells.len();
        self.cells.extend(body);
        self.blocks.push(Block { kind, rule, step, start, end: self.cells.len() });
    }
    fn rule(&mut self, rules: &[Rule], r: usize, step: usize) {
        let mut body: Vec<Cell> = Vec::new();
        let mut pos = 0usize;
        emit_pat(self.sig, &rules[r].lhs, r, CL_LHS, &mut pos, &mut body);
        body.push(tok(self.sig.arrow));
        let mut pos = 0usize;
        emit_pat(self.sig, &rules[r].rhs, r, CL_RHS, &mut pos, &mut body);
        body.push(tok(self.sig.sep));
        self.put(BlockKind::Rule, r, step, body);
    }
    fn line(&mut self, kind: BlockKind, rule: usize, step: usize, l: &[Cell], r: &[Cell]) {
        let mut body: Vec<Cell> = Vec::with_capacity(l.len() + r.len() + 2);
        body.extend_from_slice(l);
        body.push(tok(self.sig.arrow));
        body.extend_from_slice(r);
        body.push(tok(self.sig.sep));
        self.put(kind, rule, step, body);
    }
}
pub fn assemble(sig: &Sig, rules: &[Rule], style: Style, sim: &Sim, show: usize, align: usize) -> (Vec<Cell>, Vec<Block>) {
    let mut a = Asm { sig, cells: Vec::new(), blocks: Vec::new(), align };
    if style.spec == Spec::Head {
        for r in 0..rules.len() {
            a.rule(rules, r, NIL);
        }
    }
    let n = if style.steps { show.min(sim.steps.len()) } else { 0 };
    for (s, st) in sim.steps.iter().take(n).enumerate() {
        if style.spec == Spec::Inline {
            a.rule(rules, st.rule, s);
        }
        a.line(BlockKind::Step, st.rule, s, &st.before, &st.after);
    }
    if style.eval {
        match (sim.steps.first(), sim.steps.last()) {
            (Some(f), Some(l)) if !style.steps => a.line(BlockKind::Eval, NIL, NIL, &f.before, &l.after),
            _ => a.line(BlockKind::Eval, NIL, NIL, &sim.first, &sim.last),
        }
    }
    a.pad_to_frame();
    (a.cells, a.blocks)
}
pub fn make_links(cells: &[Cell], blocks: &[Block]) -> Vec<Link> {
    let mut by_ent: HashMap<u32, usize> = HashMap::new();
    let mut by_uid: HashMap<u32, usize> = HashMap::new();
    let mut out: Vec<Link> = Vec::new();
    for (p, c) in cells.iter().enumerate() {
        if c.ent != NONE {
            let first = by_ent.get(&c.ent).copied();
            match first {
                Some(f) => out.push(Link { to: p, from: f, kind: kind_of(c.ent) }),
                None => {
                    by_ent.insert(c.ent, p);
                }
            }
        }
        if c.uid != NONE {
            let first = by_uid.get(&c.uid).copied();
            match first {
                Some(f) => out.push(Link { to: p, from: f, kind: Kind::Share }),
                None => {
                    by_uid.insert(c.uid, p);
                }
            }
        }
    }
    for b in blocks {
        let end = b.end.min(cells.len());
        let arrow = match (b.start..end).find(|&p| (cells[p].ent >> 24) == CL_FMT) {
            Some(a) => a,
            None => continue,
        };
        let from = match b.kind {
            BlockKind::Step => (b.start..arrow).find(|&p| cells[p].ent == ent(CL_LHS, b.rule, 0)),
            _ => Some(b.start),
        };
        if let Some(f) = from {
            let to = arrow + 1 + (f - b.start);
            if to + 1 < end {
                out.push(Link { to, from: f, kind: Kind::Rule });
            }
        }
    }
    out.sort_by_key(|l| (l.to, l.from, l.kind as usize));
    out.dedup_by(|x, y| x.to == y.to && x.from == y.from && x.kind == y.kind);
    out
}
#[derive(Clone, Debug)]
pub struct Abs {
    pub sig: Sig,
    pub rules: Vec<Rule>,
    pub style: Style,
    pub cells: Vec<Cell>,
    pub blocks: Vec<Block>,
    pub links: Vec<Link>,
    pub steps: usize,
    pub deriv: usize,
    pub normal: bool,
    pub align: usize,
}
fn fits(blocks: &[Block], align: usize) -> bool {
    align == 0 || blocks.iter().all(|b| b.end - b.start <= align)
}
fn finish(
    rng: &mut Rng,
    cfg: &Cfg,
    sig: Sig,
    rules: Vec<Rule>,
    style: Style,
    mut g: Graph,
    root: usize,
    strict: bool,
) -> Option<Abs> {
    let n_want = rr(rng, cfg.steps).max(cfg.min_steps).max(1);
    let limit = if style.steps { n_want } else { cfg.rec_max.max(1) };
    let sim = simulate(rng, &mut g, &rules, root, cfg, limit)?;
    let l = sim.steps.len();
    if l < cfg.min_steps.max(1) || (style.eval && !sim.normal) {
        return None;
    }
    let show = if style.steps { l } else { 0 };
    let (cells, blocks) = assemble(&sig, &rules, style, &sim, show, cfg.align);
    if strict && !fits(&blocks, cfg.align) {
        return None;
    }
    let links = make_links(&cells, &blocks);
    Some(Abs { sig, rules, style, cells, blocks, links, steps: show, deriv: l, normal: sim.normal, align: cfg.align })
}
pub fn sample_once(rng: &mut Rng, cfg: &Cfg) -> Option<Abs> {
    let mut sr = rng.fork(TAG_SCHEMA);
    let sig = sample_sig(&mut sr, cfg);
    let rules = sample_rules(&mut sr, &sig, cfg);
    let mut style = match cfg.style {
        Some(s) => s,
        None => *rng.pick(&Style::ALL),
    }
    .norm();
    if !style.steps && !style.eval {
        style.eval = true;
    }
    let mut g = Graph::default();
    let root = sample_query(rng, &sig, &rules, cfg, &mut g);
    finish(rng, cfg, sig, rules, style, g, root, true)
}
fn fallback(rng: &mut Rng, cfg: &Cfg) -> Abs {
    let mut c = cfg.clone();
    c.term_max = c.term_max.max(4);
    c.rec_max = c.rec_max.max(1);
    c.min_steps = 1;
    c.steps = (1, c.steps.1.max(c.steps.0).max(1));
    let mut sr = rng.fork(TAG_SCHEMA);
    let sig = sample_sig(&mut sr, &c);
    let top = sig.funs[1];
    let ar = sig.arity[top as usize].clamp(1, 3);
    let nv = ar.min(sig.vars.len());
    let kp: Vec<Pat> = (0..ar)
        .map(|i| if i < nv { Pat::Var(i) } else { Pat::App(sig.cons[0], Vec::new()) })
        .collect();
    let rules = vec![Rule { lhs: Pat::App(top, kp), rhs: Pat::Var(0), w: 1.0, nv }];
    let mut g = Graph::default();
    let leaf = g.add(sig.cons[0], &[]);
    let root = g.add(top, &vec![leaf; ar]);
    let mut style = c.style.unwrap_or(Style::TRACE).norm();
    if !style.steps && !style.eval {
        style.eval = true;
    }
    finish(rng, &c, sig, rules, style, g, root, false).expect("ptgrs: fallback instance")
}
pub fn sample(rng: &mut Rng, cfg: &Cfg) -> Abs {
    for _ in 0..cfg.tries.max(1) {
        if let Some(a) = sample_once(rng, cfg) {
            return a;
        }
    }
    fallback(rng, cfg)
}
fn prefix_ok(sig: &Sig, cells: &[Cell], end: usize, i: &mut usize) -> bool {
    if *i >= end {
        return false;
    }
    let s = cells[*i].sym as usize;
    if s < 3 || s >= sig.n() {
        return false;
    }
    *i += 1;
    for _ in 0..sig.arity[s] {
        if !prefix_ok(sig, cells, end, i) {
            return false;
        }
    }
    true
}
pub fn verify(a: &Abs) -> Result<(), String> {
    let n = a.cells.len();
    if n == 0 {
        return Err("empty text".to_string());
    }
    for (p, c) in a.cells.iter().enumerate() {
        if (c.sym as usize) >= a.sig.n() {
            return Err(format!("cell {}: unknown symbol {}", p, c.sym));
        }
    }
    if a.align > 0 && n % a.align != 0 {
        return Err(format!("length {} is not a multiple of {}", n, a.align));
    }
    let mut prev = 0usize;
    for (k, b) in a.blocks.iter().enumerate() {
        if b.start < prev || b.end <= b.start || b.end > n {
            return Err(format!("block {}: bad range {}..{}", k, b.start, b.end));
        }
        if a.align > 0 {
            let len = b.end - b.start;
            if len <= a.align && b.start / a.align != (b.end - 1) / a.align {
                return Err(format!("block {} straddles two frames", k));
            }
            if len > a.align && b.start % a.align != 0 {
                return Err(format!("long block {} does not start on a frame", k));
            }
        }
        let mut i = b.start;
        let mut ok = prefix_ok(&a.sig, &a.cells, b.end, &mut i);
        if ok && i < b.end && a.cells[i].sym == a.sig.arrow {
            i += 1;
            ok = prefix_ok(&a.sig, &a.cells, b.end, &mut i)
                && i + 1 == b.end
                && a.cells[i].sym == a.sig.sep;
        } else {
            ok = false;
        }
        if !ok {
            return Err(format!("block {}: not `term > term ;`", k));
        }
        prev = b.end;
    }
    for l in &a.links {
        if l.from >= l.to || l.to >= n {
            return Err(format!("link {:?}: bad positions", l));
        }
        let (x, y) = (&a.cells[l.to], &a.cells[l.from]);
        let ok = match l.kind {
            Kind::Share => x.uid != NONE && x.uid == y.uid && x.sym == y.sym,
            Kind::Format => x.ent == y.ent && x.sym == y.sym,
            Kind::Var => x.ent == y.ent && (x.ent >> 24) == CL_VAR,
            Kind::Rule | Kind::Schema => true,
        };
        if !ok {
            return Err(format!("link {:?}: inconsistent cells", l));
        }
    }
    Ok(())
}
pub fn show_cells(sig: &Sig, cells: &[Cell]) -> String {
    cells
        .iter()
        .map(|c| sig.name.get(c.sym as usize).copied().unwrap_or('?'))
        .collect()
}
pub fn show_abs(a: &Abs) -> String {
    let mut s = String::new();
    for b in &a.blocks {
        let kind = match b.kind {
            BlockKind::Rule => "rule",
            BlockKind::Step => "step",
            BlockKind::Eval => "eval",
        };
        let end = b.end.min(a.cells.len());
        s.push_str(&format!("{:>4} {:<4} {}\n", b.start, kind, show_cells(&a.sig, &a.cells[b.start..end])));
    }
    s
}

fn key(p: usize, q: usize) -> (usize, usize) {
    if p < q { (p, q) } else { (q, p) }
}
fn find(par: &mut [usize], mut x: usize) -> usize {
    while par[x] != x {
        par[x] = par[par[x]];
        x = par[x];
    }
    x
}
fn canon_by<T: Copy + Eq + std::hash::Hash>(xs: &[T]) -> Vec<u16> {
    let mut seen: HashMap<T, u16> = HashMap::new();
    let mut out: Vec<u16> = Vec::with_capacity(xs.len());
    for &x in xs {
        let next = seen.len() as u16;
        out.push(*seen.entry(x).or_insert(next));
    }
    out
}
pub fn canon(raw: &[u8]) -> Vec<u16> {
    canon_by(raw)
}
fn pat_hash(h: u64, p: &Pat) -> u64 {
    match p {
        Pat::Var(v) => mix3(h, 1, *v as u64),
        Pat::App(s, ks) => {
            let mut x = mix3(h, 2, *s as u64);
            for k in ks {
                x = pat_hash(x, k);
            }
            mix(x, 3)
        }
    }
}
pub fn schema_hash(sig: &Sig, rules: &[Rule]) -> u64 {
    let mut h = mix(TAG_SCHEMA, sig.n() as u64);
    for &a in &sig.arity {
        h = mix(h, a as u64 + 1);
    }
    for r in rules {
        h = pat_hash(h, &r.lhs);
        h = pat_hash(h, &r.rhs);
        h = mix(h, r.nv as u64 + 1);
    }
    h
}
impl Abs {
    pub fn encode(&self, perm: &[u8; 256]) -> Vec<u8> {
        self.cells.iter().map(|c| perm[c.sym as usize]).collect()
    }
    pub fn canon(&self) -> Vec<u16> {
        let s: Vec<u16> = self.cells.iter().map(|c| c.sym).collect();
        canon_by(&s)
    }
    pub fn schema(&self) -> u64 {
        schema_hash(&self.sig, &self.rules)
    }
    pub fn lines(&self) -> Vec<Range<usize>> {
        self.blocks.iter().map(|b| b.start..b.end).collect()
    }
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Group {
    pub kind: Kind,
    pub pos: Vec<usize>,
}
impl Group {
    pub fn w(&self) -> f64 {
        self.kind.weight()
    }
    pub fn ties(&self) -> usize {
        self.pos.len() * self.pos.len().saturating_sub(1) / 2
    }
}
pub fn groups(links: &[Link], n: usize) -> Vec<Group> {
    let mut out: Vec<Group> = Vec::new();
    for kind in Kind::ALL {
        let mut par: Vec<usize> = (0..n).collect();
        let mut used = vec![false; n];
        for l in links {
            if l.kind != kind || l.to >= n || l.from >= n {
                continue;
            }
            let (x, y) = (find(&mut par, l.from), find(&mut par, l.to));
            if x != y {
                par[x.max(y)] = x.min(y);
            }
            used[l.from] = true;
            used[l.to] = true;
        }
        let mut by_root: BTreeMap<usize, Vec<usize>> = BTreeMap::new();
        for p in 0..n {
            if used[p] {
                let r = find(&mut par, p);
                by_root.entry(r).or_default().push(p);
            }
        }
        let mut gs: Vec<Group> = by_root
            .into_values()
            .filter(|v| v.len() >= 2)
            .map(|pos| Group { kind, pos })
            .collect();
        gs.sort_by(|x, y| x.pos.cmp(&y.pos));
        out.extend(gs);
    }
    out
}
fn firsts(a: &Abs) -> BTreeMap<u32, usize> {
    let mut m: BTreeMap<u32, usize> = BTreeMap::new();
    for (p, c) in a.cells.iter().enumerate() {
        if c.ent != NONE && (c.ent >> 24) != CL_FMT {
            m.entry(c.ent).or_insert(p);
        }
    }
    m
}
fn schema_ties(a: &Abs, b: &Abs, off: usize) -> Vec<Group> {
    let (fa, fb) = (firsts(a), firsts(b));
    let mut out: Vec<Group> = Vec::new();
    for (k, &p) in &fa {
        if let Some(&q) = fb.get(k) {
            out.push(Group { kind: Kind::Schema, pos: vec![p, off + q] });
        }
    }
    out
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Cause {
    pub groups: Vec<Group>,
    pub len: usize,
    pub schema: u64,
    pub parts: usize,
    pub deriv: usize,
}
impl Cause {
    pub fn new(groups: Vec<Group>, len: usize) -> Cause {
        Cause { groups, len, schema: 0, parts: 1, deriv: 0 }
    }
    pub fn of(a: &Abs) -> Cause {
        Cause {
            groups: groups(&a.links, a.cells.len()),
            len: a.cells.len(),
            schema: a.schema(),
            parts: 1,
            deriv: a.deriv,
        }
    }
    pub fn of_pair(a: &Abs, b: &Abs, off: usize) -> Cause {
        let mut gs = groups(&a.links, a.cells.len());
        for g in groups(&b.links, b.cells.len()) {
            let pos: Vec<usize> = g.pos.iter().map(|&p| p + off).collect();
            gs.push(Group { kind: g.kind, pos });
        }
        gs.extend(schema_ties(a, b, off));
        Cause {
            groups: gs,
            len: off + b.cells.len(),
            schema: a.schema(),
            parts: 2,
            deriv: a.deriv + b.deriv,
        }
    }
    pub fn weight(&self) -> f64 {
        self.groups.iter().map(|g| g.w()).sum()
    }
    pub fn ties(&self) -> usize {
        self.groups.iter().map(|g| g.ties()).sum()
    }
    pub fn count(&self, k: Kind) -> usize {
        self.groups.iter().filter(|g| g.kind == k).count()
    }
    pub fn touching(&self, seg: &Range<usize>) -> f64 {
        let (mut hit, mut all) = (0.0f64, 0.0f64);
        for g in &self.groups {
            all += g.w();
            if g.pos.iter().any(|p| seg.contains(p)) {
                hit += g.w();
            }
        }
        if all > 0.0 { hit / all } else { 0.0 }
    }
    pub fn summary(&self) -> String {
        let mut s = format!(
            "cause: {} group(s), {} tie(s), weight {:.2}, {} part(s), derivation {}",
            self.groups.len(),
            self.ties(),
            self.weight(),
            self.parts,
            self.deriv
        );
        for k in Kind::ALL {
            s.push_str(&format!("  {} {}", k.name(), self.count(k)));
        }
        s
    }
}
#[derive(Clone, Debug, Default)]
pub struct Cover {
    pub local: HashSet<(usize, usize)>,
    pub cross: HashSet<(usize, usize)>,
    pub spans: Vec<(usize, usize)>,
}
impl Cover {
    pub fn feed(&mut self, foot: &[Vec<usize>], live: &[bool], rels: &[Vec<usize>]) {
        for (i, f) in foot.iter().enumerate() {
            if !live.get(i).copied().unwrap_or(false) {
                continue;
            }
            for x in 0..f.len() {
                for y in (x + 1)..f.len() {
                    self.local.insert(key(f[x], f[y]));
                }
            }
        }
        for r in rels {
            for x in 0..r.len() {
                for y in (x + 1)..r.len() {
                    let (a, b) = (r[x], r[y]);
                    if a == b {
                        continue;
                    }
                    if let (Some(fa), Some(fb)) = (foot.get(a), foot.get(b)) {
                        for &p in fa {
                            for &q in fb {
                                if p != q {
                                    self.cross.insert(key(p, q));
                                }
                            }
                        }
                    }
                }
            }
        }
    }
    pub fn of(an: &Analysis) -> Cover {
        let mut cv = Cover::default();
        for fr in &an.frames {
            cv.spans.push((fr.start, fr.len));
            for c in &fr.st.complexes {
                let live: Vec<bool> = c.dists.iter().map(|d| d.is_distinction()).collect();
                let rels: Vec<Vec<usize>> = c.rels.iter().map(|r| r.members.clone()).collect();
                cv.feed(&c.foot, &live, &rels);
            }
        }
        cv
    }
    pub fn reach(&self, p: usize, q: usize) -> bool {
        self.spans.is_empty()
            || self.spans.iter().any(|&(s, l)| p >= s && p < s + l && q >= s && q < s + l)
    }
}
#[derive(Clone, Debug, Default)]
pub struct Inclusion {
    pub score: f64,
    pub local: f64,
    pub cross: f64,
    pub reach: f64,
    pub groups: usize,
    pub hit: usize,
    pub kind: [f64; 5],
}
impl Inclusion {
    pub fn measure(cause: &Cause, an: &Analysis) -> Inclusion {
        Inclusion::against(cause, &Cover::of(an))
    }
    pub fn against(cause: &Cause, cv: &Cover) -> Inclusion {
        let mut inc = Inclusion::default();
        let (mut w, mut lo, mut cr, mut rc) = (0.0f64, 0.0f64, 0.0f64, 0.0f64);
        let mut ks = [0.0f64; 5];
        let mut kn = [0usize; 5];
        for g in &cause.groups {
            let t = g.ties();
            if t == 0 {
                continue;
            }
            let (mut nl, mut nc, mut nr) = (0usize, 0usize, 0usize);
            for x in 0..g.pos.len() {
                for y in (x + 1)..g.pos.len() {
                    let (p, q) = (g.pos[x], g.pos[y]);
                    if cv.reach(p, q) {
                        nr += 1;
                    }
                    let tie = key(p, q);
                    if cv.local.contains(&tie) {
                        nl += 1;
                    } else if cv.cross.contains(&tie) {
                        nc += 1;
                    }
                }
            }
            let (tf, wg) = (t as f64, g.w());
            w += wg;
            lo += wg * nl as f64 / tf;
            cr += wg * nc as f64 / tf;
            rc += wg * nr as f64 / tf;
            let kk = g.kind as usize;
            ks[kk] += (nl + nc) as f64 / tf;
            kn[kk] += 1;
            inc.groups += 1;
            if nl + nc > 0 {
                inc.hit += 1;
            }
        }
        if w > 0.0 {
            inc.local = lo / w;
            inc.cross = cr / w;
            inc.score = (lo + cr) / w;
            inc.reach = rc / w;
        }
        for kk in 0..5 {
            if kn[kk] > 0 {
                inc.kind[kk] = ks[kk] / kn[kk] as f64;
            }
        }
        inc
    }
    pub fn rel(&self) -> f64 {
        if self.reach > 1e-12 { (self.score / self.reach).clamp(0.0, 1.0) } else { 0.0 }
    }
    pub fn line(&self) -> String {
        format!(
            "inclusion {:.4} (local {:.4} cross {:.4})  reach {:.4}  rel {:.4}  groups {} hit {}",
            self.score, self.local, self.cross, self.reach, self.rel(), self.groups, self.hit
        )
    }
}
#[derive(Clone, Debug)]
pub struct Part {
    pub abs: Abs,
    pub perm: [u8; 256],
    pub start: usize,
}
#[derive(Clone, Debug)]
pub struct Draw {
    pub parts: Vec<Part>,
    pub e: Percept,
    pub cause: Cause,
}
impl Draw {
    pub fn first(&self) -> &Abs {
        &self.parts[0].abs
    }
    pub fn blocks(&self) -> Vec<Range<usize>> {
        let mut v: Vec<Range<usize>> = Vec::new();
        for p in &self.parts {
            for b in &p.abs.blocks {
                v.push(p.start + b.start..p.start + b.end);
            }
        }
        v
    }
    pub fn masked(&self, rng: &mut Rng, seg: Range<usize>, fill: Fill) -> Masked {
        mask(rng, &self.e, seg, fill)
    }
    pub fn included(&self, an: &Analysis) -> Inclusion {
        Inclusion::measure(&self.cause, an)
    }
    pub fn show(&self) -> String {
        let mut s = String::new();
        for (i, p) in self.parts.iter().enumerate() {
            s.push_str(&format!(
                "part {} @{}: schema {:016x}, {} cell(s), derivation {} step(s)\n",
                i,
                p.start,
                p.abs.schema(),
                p.abs.cells.len(),
                p.abs.deriv
            ));
            s.push_str(&show_abs(&p.abs));
        }
        s.push_str(&self.cause.summary());
        s.push('\n');
        s
    }
}
pub fn draw(rng: &mut Rng, cfg: &Cfg) -> Draw {
    let abs = sample(rng, cfg);
    let perm = random_renaming(rng);
    let raw = abs.encode(&perm);
    let e = Percept::from_bytes(&raw);
    let cause = Cause::of(&abs);
    Draw { parts: vec![Part { abs, perm, start: 0 }], e, cause }
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Layout {
    Joint,
    Apart,
}
impl Layout {
    pub fn cfg(self, base: &Cfg, np: usize) -> Cfg {
        match self {
            Layout::Joint => base.clone().fit((np / 2).max(8), false),
            Layout::Apart => base.clone().fit(np, true),
        }
    }
}
fn again(rng: &mut Rng, cfg: &Cfg, like: &Abs) -> Option<Abs> {
    for _ in 0..cfg.tries.max(1) {
        let mut g = Graph::default();
        let root = sample_query(rng, &like.sig, &like.rules, cfg, &mut g);
        if let Some(b) = finish(rng, cfg, like.sig.clone(), like.rules.clone(), like.style, g, root, true) {
            return Some(b);
        }
    }
    None
}
pub fn pair(rng: &mut Rng, base: &Cfg, np: usize, layout: Layout) -> Option<Draw> {
    let c = layout.cfg(base, np);
    for _ in 0..base.tries.max(1) {
        let a = sample(rng, &c);
        let b = match again(rng, &c, &a) {
            Some(b) => b,
            None => continue,
        };
        if a.canon() == b.canon() {
            continue;
        }
        let (la, lb) = (a.cells.len(), b.cells.len());
        if layout == Layout::Joint && la + lb > np {
            continue;
        }
        let p = random_renaming(rng);
        let n = a.sig.n();
        let mut q = [0u8; 256];
        for k in 0..256usize {
            q[k] = p[(k + n) % 256];
        }
        let mut raw = a.encode(&p);
        raw.extend(b.encode(&q));
        let cause = Cause::of_pair(&a, &b, la);
        let e = Percept::from_bytes(&raw);
        let parts = vec![Part { abs: a, perm: p, start: 0 }, Part { abs: b, perm: q, start: la }];
        return Some(Draw { parts, e, cause });
    }
    None
}
pub fn shuffled(rng: &mut Rng, e: &Percept) -> Percept {
    let mut raw = e.raw.clone();
    rng.shuffle(&mut raw);
    Percept::from_bytes(&raw)
}
pub fn iid(rng: &mut Rng, len: usize, a: usize) -> Percept {
    let a = a.clamp(1, 256);
    let mut pool: Vec<u8> = (0..=255u8).collect();
    rng.shuffle(&mut pool);
    pool.truncate(a);
    let mut raw: Vec<u8> = Vec::with_capacity(len);
    for i in 0..len {
        let b = if i < a { pool[i] } else { pool[rng.below(a)] };
        raw.push(b);
    }
    rng.shuffle(&mut raw);
    Percept::from_bytes(&raw)
}
pub fn iid_like(rng: &mut Rng, e: &Percept) -> Percept {
    iid(rng, e.len(), e.a)
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Fill {
    Noise,
    Fresh,
    Byte(u8),
}
#[derive(Clone, Debug)]
pub struct Masked {
    pub e: Percept,
    pub seg: Range<usize>,
    pub fill: Fill,
}
impl Masked {
    pub fn hidden(&self, cause: &Cause) -> f64 {
        cause.touching(&self.seg)
    }
    pub fn localization(&self, by_pos: &[f64]) -> Localization {
        localization(by_pos, &self.seg)
    }
}
pub fn mask(rng: &mut Rng, e: &Percept, seg: Range<usize>, fill: Fill) -> Masked {
    let n = e.len();
    let lo = seg.start.min(n);
    let hi = seg.end.min(n).max(lo);
    let fresh = (0..=255u8).find(|b| !e.alphabet.contains(b)).unwrap_or(0);
    let mut raw = e.raw.clone();
    for p in lo..hi {
        raw[p] = match fill {
            Fill::Noise if e.a >= 2 => {
                let o = e.codes[p] as usize;
                let c = rng.below(e.a - 1);
                e.alphabet[if c >= o { c + 1 } else { c }]
            }
            Fill::Noise => e.alphabet[0],
            Fill::Fresh => fresh,
            Fill::Byte(b) => b,
        };
    }
    Masked { e: Percept::from_bytes(&raw), seg: lo..hi, fill }
}
#[derive(Clone, Copy, Debug, Default)]
pub struct Localization {
    pub inside: f64,
    pub total: f64,
    pub share: f64,
    pub lift: f64,
}
pub fn localization(by_pos: &[f64], seg: &Range<usize>) -> Localization {
    let n = by_pos.len();
    let lo = seg.start.min(n);
    let hi = seg.end.min(n).max(lo);
    let total: f64 = by_pos.iter().sum();
    let inside: f64 = by_pos[lo..hi].iter().sum();
    let share = if total > MASS_EPS { inside / total } else { 0.0 };
    let frac = if n > 0 { (hi - lo) as f64 / n as f64 } else { 0.0 };
    let lift = if frac > 0.0 { share / frac } else { 0.0 };
    Localization { inside, total, share, lift }
}

use crate::phi::Dist;
#[derive(Clone, Debug)]
pub struct Relation {
    pub members: Vec<usize>,
    pub overlap: Vec<usize>,
    pub phi: f64,
    pub cross: bool,
}
struct Relatum {
    d: usize,
    z: Vec<usize>,
    lat: Vec<bool>,
    states: Vec<Vec<u16>>,
    unit: f64,
}
fn first_common(a: &Relatum, b: &Relatum) -> Option<usize> {
    for (i, &x) in a.z.iter().enumerate() {
        if a.lat[i] {
            if let Ok(k) = b.z.binary_search(&x) {
                if b.lat[k] {
                    return Some(x);
                }
            }
        }
    }
    None
}
fn congruent(a: &Relatum, b: &Relatum) -> Vec<usize> {
    let mut best: Vec<usize> = Vec::new();
    for sa in &a.states {
        for sb in &b.states {
            let mut o: Vec<usize> = Vec::new();
            for (i, &x) in a.z.iter().enumerate() {
                if !a.lat[i] {
                    continue;
                }
                if let Ok(k) = b.z.binary_search(&x) {
                    if b.lat[k] && sa[i] == sb[k] {
                        o.push(x);
                    }
                }
            }
            if o.len() > best.len() {
                best = o;
            }
        }
    }
    best
}
fn common_overlap(rel: &[Relatum], pick: &[(usize, usize)], x: usize, v: u16) -> Option<(Vec<usize>, f64)> {
    let mut sel: Vec<(&Relatum, &Vec<u16>)> = Vec::with_capacity(pick.len());
    for &(_, ri) in pick {
        let r = &rel[ri];
        let i = r.z.binary_search(&x).ok()?;
        let s = r.states.iter().find(|s| s[i] == v)?;
        sel.push((r, s));
    }
    let (r0, s0) = sel[0];
    let mut o: Vec<usize> = Vec::new();
    for (i, &y) in r0.z.iter().enumerate() {
        if !r0.lat[i] {
            continue;
        }
        let ok = sel[1..].iter().all(|(r, s)| match r.z.binary_search(&y) {
            Ok(k) => r.lat[k] && s[k] == s0[i],
            Err(_) => false,
        });
        if ok {
            o.push(y);
        }
    }
    if o.is_empty() {
        return None;
    }
    let u = sel.iter().map(|(r, _)| r.unit).fold(f64::INFINITY, f64::min);
    let phi = o.len() as f64 * u;
    Some((o, phi))
}
fn disjoint(foot: &[Vec<usize>], mem: &[usize]) -> bool {
    for p in 0..mem.len() {
        for q in (p + 1)..mem.len() {
            let (a, b) = (&foot[mem[p]], &foot[mem[q]]);
            if a.iter().any(|x| b.binary_search(x).is_ok()) {
                return false;
            }
        }
    }
    true
}
pub fn relations(dists: &[Dist], latent: &[bool], foot: &[Vec<usize>]) -> Vec<Relation> {
    let n = latent.len();
    let mut rel: Vec<Relatum> = Vec::new();
    for (di, d) in dists.iter().enumerate() {
        if !d.is_distinction() {
            continue;
        }
        for f in d.cause.iter().chain(d.effect.iter()) {
            let lat: Vec<bool> = f.nodes.iter().map(|&j| latent[j]).collect();
            if f.states.is_empty() || !lat.iter().any(|&b| b) {
                continue;
            }
            rel.push(Relatum {
                d: di,
                z: f.nodes.clone(),
                lat,
                states: f.states.clone(),
                unit: d.phi_d / f.nodes.len() as f64,
            });
        }
    }
    let mut by_node: Vec<Vec<usize>> = vec![Vec::new(); n];
    for (ri, r) in rel.iter().enumerate() {
        for (i, &x) in r.z.iter().enumerate() {
            if r.lat[i] {
                by_node[x].push(ri);
            }
        }
    }
    let mut pairs: Vec<(usize, usize, f64, Vec<usize>)> = Vec::new();
    for x in 0..n {
        let lst = &by_node[x];
        for p in 0..lst.len() {
            for q in (p + 1)..lst.len() {
                let (a, b) = (&rel[lst[p]], &rel[lst[q]]);
                if a.d == b.d || first_common(a, b) != Some(x) {
                    continue;
                }
                let o = congruent(a, b);
                if o.is_empty() {
                    continue;
                }
                let phi = o.len() as f64 * a.unit.min(b.unit);
                pairs.push((a.d.min(b.d), a.d.max(b.d), phi, o));
            }
        }
    }
    pairs.sort_by(|u, v| {
        u.0.cmp(&v.0)
            .then(u.1.cmp(&v.1))
            .then(v.2.total_cmp(&u.2))
            .then_with(|| u.3.cmp(&v.3))
    });
    pairs.dedup_by(|later, first| later.0 == first.0 && later.1 == first.1);
    let mut cl: Vec<(Vec<usize>, f64, Vec<usize>)> = Vec::new();
    let mut trip: Vec<(u16, usize, usize)> = Vec::new();
    for x in 0..n {
        if by_node[x].len() < 3 {
            continue;
        }
        trip.clear();
        for &ri in &by_node[x] {
            let r = &rel[ri];
            if let Ok(i) = r.z.binary_search(&x) {
                for s in &r.states {
                    trip.push((s[i], r.d, ri));
                }
            }
        }
        trip.sort_unstable();
        let mut g = 0usize;
        while g < trip.len() {
            let v = trip[g].0;
            let mut h = g;
            while h < trip.len() && trip[h].0 == v {
                h += 1;
            }
            let mut pick: Vec<(usize, usize)> = Vec::new();
            for &(_, d, ri) in &trip[g..h] {
                let same = matches!(pick.last(), Some(c) if c.0 == d);
                if same {
                    if let Some(c) = pick.last_mut() {
                        if rel[ri].unit > rel[c.1].unit {
                            c.1 = ri;
                        }
                    }
                } else {
                    pick.push((d, ri));
                }
            }
            if pick.len() >= 3 {
                if let Some((o, phi)) = common_overlap(&rel, &pick, x, v) {
                    cl.push((pick.iter().map(|p| p.0).collect(), phi, o));
                }
            }
            g = h;
        }
    }
    cl.sort_by(|u, v| u.0.cmp(&v.0).then(v.1.total_cmp(&u.1)).then_with(|| u.2.cmp(&v.2)));
    cl.dedup_by(|later, first| later.0 == first.0);
    let mut out: Vec<Relation> = Vec::with_capacity(pairs.len() + cl.len());
    for (a, b, phi, o) in pairs {
        out.push(Relation { members: vec![a, b], overlap: o, phi, cross: disjoint(foot, &[a, b]) });
    }
    for (mem, phi, o) in cl {
        let cross = disjoint(foot, &mem);
        out.push(Relation { members: mem, overlap: o, phi, cross });
    }
    out
}

fn agreed(dists: &[Dist], members: &[usize], x: usize) -> Vec<u16> {
    let mut common: Option<Vec<u16>> = None;
    for &mi in members {
        let mut here: Vec<u16> = Vec::new();
        if let Some(d) = dists.get(mi) {
            for f in d.cause.iter().chain(d.effect.iter()) {
                if let Ok(i) = f.nodes.binary_search(&x) {
                    for s in &f.states {
                        if let Some(&v) = s.get(i) {
                            if !here.contains(&v) {
                                here.push(v);
                            }
                        }
                    }
                }
            }
        }
        common = Some(match common {
            None => here,
            Some(c) => c.into_iter().filter(|v| here.contains(v)).collect(),
        });
    }
    let mut v = common.unwrap_or_default();
    v.sort_unstable();
    v
}

    v
}

pub fn strip(an: &Analysis, e: &Percept, width: usize) -> String {
    let n = e.len();
    if n == 0 {
        return "  (E is empty)\n".to_string();
    }
    let w = width.max(24);
    let mut read: Vec<Option<u16>> = vec![None; n];
    for fr in &an.frames {
        for k in 0..fr.len {
            let p = fr.start + k;
            if p < n {
                read[p] = fr.state.get(k).copied();
            }
        }
    }
    let mx = an.resid.by_pos.iter().cloned().fold(0.0f64, f64::max);
    let mut out = String::new();
    let mut p = 0usize;
    while p < n {
        let first = p;
        let (mut re, mut rs, mut rr) = (String::new(), String::new(), String::new());
        let mut used = 0usize;
        while p < n {
            let ce = show_byte(e.raw[p]);
            let cs = match read[p] {
                Some(v) if v == e.codes[p] => "\u{b7}".to_string(),
                Some(v) => match e.alphabet.get(v as usize) {
                    Some(&b) => show_byte(b),
                    None => "?".to_string(),
                },
                None => "?".to_string(),
            };
            let cw = ce.chars().count().max(cs.chars().count()).max(1) + 1;
            if used > 0 && used + cw > w {
                break;
            }
            let mass = an.resid.by_pos.get(p).copied().unwrap_or(0.0);
            re.push_str(&pad(&ce, cw));
            rs.push_str(&pad(&cs, cw));
            rr.push_str(&pad(&glyph(mass, mx).to_string(), cw));
            used += cw;
            p += 1;
        }
        let num = first.to_string();
        let rows = [(num.as_str(), "E", &re), ("", "s*", &rs), ("", "rho", &rr)];
        for (n1, label, row) in rows {
            let line = format!("{:>6} {:<4} {}", n1, label, row);
            out.push_str(line.trim_end());
            out.push('\n');
        }
    }
    out
}
pub fn curve(fr: &FrameResult, scale: f64, rows: usize) -> String {
    const BAR: usize = 24;
    let mut s = String::from("  phi_sharp(t):\n");
    let mx = fr.steps.iter().map(|x| x.phi).fold(0.0f64, f64::max);
    let mut prev: Option<usize> = None;
    for t in pick_rows(fr.steps.len(), fr.t_star, rows) {
        if let Some(p) = prev {
            if t > p + 1 {
                s.push_str("      ...\n");
            }
        }
        prev = Some(t);
        let st = &fr.steps[t];
        let filled = if mx > PHI_EPS {
            (((st.phi / mx) * BAR as f64).round() as usize).min(BAR)
        } else {
            0
        };
        let bar: String = (0..BAR).map(|k| if k < filled { '#' } else { '.' }).collect();
        s.push_str(&format!(
            "    t={:>3} |{}| {:>9.4}  d {:.4} r {:.4}  tau {} cx {}{}\n",
            t,
            bar,
            st.phi * scale,
            st.sum_d * scale,
            st.sum_r * scale,
            st.tau,
            st.n_cx,
            if t == fr.t_star { "   <- t*" } else { "" }
        ));
    }
    s
}

                self.with_state(&d.mech, &d.mech_state),
                side,
                self.num(val),
                self.foot_of(c, i)
            ));
        }
        if shown < half.len() {
            o.ln(&format!(
                "       ... {} more half-distinction(s) not shown (list_cap = {})",
                half.len() - shown,
                self.cfg.list_cap
            ));
        }
    }
    fn rel_list(&self, o: &mut Out, c: &ComplexStruct) {
        let mut order: Vec<usize> = (0..c.rels.len()).collect();
        order.sort_by(|&a, &b| c.rels[b].phi.total_cmp(&c.rels[a].phi).then(a.cmp(&b)));
        o.ln(&format!(
            "     Rel_E: {} relation(s), sum phi_r {}, of which between disjoint footprints {}",
            c.rels.len(),
            self.num(c.sum_r),
            self.num(c.sum_r_cross)
        ));
        let shown = self.cap(order.len());
        for &i in order.iter().take(shown) {
            let r = &c.rels[i];
            let mem: Vec<String> = r.members.iter().map(|&k| format!("d{}", k)).collect();
            let feet: Vec<String> =
                r.members.iter().map(|&k| format!("[{}]", self.foot_of(c, k))).collect();
            let ov: Vec<String> = r
                .overlap
                .iter()
                .map(|&x| {
                    let names: Vec<String> =
                        agreed(&c.dists, &r.members, x).iter().map(|&v| self.sym(v)).collect();
                    if names.is_empty() {
                        format!("{}=?", self.label(x))
                    } else {
                        format!("{}={}", self.label(x), names.join("|"))
                    }
                })
                .collect();
            o.ln(&format!(
                "       r{}  phi_r {}  {}  {}  overlap {}  foot {}",
                i,
                self.num(r.phi),
                if r.cross { "cross" } else { "local" },
                mem.join("+"),
                ov.join(" "),
                feet.join(" ")
            ));
        }
        if shown < order.len() {
            o.ln(&format!(
                "       ... {} more relation(s) not shown (list_cap = {})",
                order.len() - shown,
                self.cfg.list_cap
            ));
        }
    }
    fn resid_list(&self, o: &mut Out, c: &ComplexStruct) {
        if !self.full {
            return;
        }
        let (mut rc, mut re) = (0.0f64, 0.0f64);
        for r in &c.resid {
            match r.weak {
                Dir::Cause => rc += r.rho,
                Dir::Effect => re += r.rho,
            }
        }
        o.ln(&format!(
            "     residual: {} mechanism(s), rho_cause {} (clarté) + rho_effet {} (expressivité)",
            c.resid.len(),
            self.num(rc),
            self.num(re)
        ));
        let mut order: Vec<usize> = (0..c.resid.len()).collect();
        order.sort_by(|&a, &b| c.resid[b].rho.total_cmp(&c.resid[a].rho).then(a.cmp(&b)));
        let shown = self.cap(order.len());
        for &i in order.iter().take(shown) {
            self.resid_entry(o, c, &c.resid[i]);
        }
        if shown < order.len() {
            o.ln(&format!(
                "       ... {} more mechanism(s) not shown (list_cap = {})",
                order.len() - shown,
                self.cfg.list_cap
            ));
        }
    }
    fn resid_entry(&self, o: &mut Out, c: &ComplexStruct, r: &Residual) {
        let ms = match c.dists.iter().find(|d| d.mech == r.mech) {
            Some(d) => self.with_state(&r.mech, &d.mech_state),
            None => self.plain(&r.mech),
        };
        let kind = match r.weak {
            Dir::Cause => "rho_cause",
            Dir::Effect => "rho_effet",
        };
        let mut extra = String::new();
        if r.from_strong {
            extra.push_str("  [weak side has no irreducible purview: nodes of the strong side]");
        }
        if r.ties > 1 {
            extra.push_str(&format!("  [{} tied states]", r.ties));
        }
        o.ln(&format!(
            "       m {}  {} {}  needs {}{}",
            ms,
            kind,
            self.num(r.rho),
            r.need(),
            extra
        ));
        let parts: Vec<String> = r
            .nodes
            .iter()
            .map(|&(j, share)| format!("{} {}", self.label_pos(j), self.num(share)))
            .collect();
        o.ln(&format!("            on {}", parts.join(" | ")));
    }
}
fn is_full(an: &Analysis) -> bool {
    an.frames
        .iter()
        .all(|f| f.st.complexes.iter().all(|c| c.text.len() == c.dists.len()))
}
fn counts(fr: &FrameResult) -> (usize, usize) {
    let mut nd = 0usize;
    let mut nr = 0usize;
    for c in &fr.st.complexes {
        nd += c.dists.iter().filter(|d| d.is_distinction()).count();
        nr += c.rels.len();
    }
    (nd, nr)
}
fn plateau_word(fr: &FrameResult) -> &'static str {
    if fr.phi <= PHI_EPS {

    } else if fr.plateau {
        "yes"
    } else {
        "no"
    }

    o.s

use crate::phi::{Dir, Dist, Phi, PHI_EPS};
pub const MASS_EPS: f64 = 1e-12;
#[derive(Clone, Debug)]
pub struct Residual {
    pub mech: Vec<usize>,
    pub weak: Dir,
    pub rho: f64,
    pub ties: usize,
    pub nodes: Vec<(usize, f64)>,
    pub from_strong: bool,
}
impl Residual {
    pub fn need(&self) -> &'static str {
        match self.weak {
            Dir::Cause => "a discriminant (clarté)",
            Dir::Effect => "reach or context (expressivité)",
        }
    }
}
pub fn norm_entropy(p: &[f64]) -> f64 {
    if p.len() < 2 {
        return 0.0;
    }
    let mut h = 0.0f64;
    for &x in p {
        if x > 0.0 {
            h -= x * x.ln();
        }
    }
    (h / (p.len() as f64).ln()).clamp(0.0, 1.0)
}
pub fn residuals(ph: &Phi<'_, '_>, dists: &[Dist]) -> Vec<Residual> {
    let mut out: Vec<Residual> = Vec::new();
    for d in dists {
        let rho = (d.phi_c - d.phi_e).abs();
        if rho <= PHI_EPS {
            continue;
        }
        let weak = if d.phi_c < d.phi_e { Dir::Cause } else { Dir::Effect };
        let (wf, sf) = match weak {
            Dir::Cause => (&d.cause, &d.effect),
            Dir::Effect => (&d.effect, &d.cause),
        };
        let from_strong = wf.is_empty();
        let faces = if from_strong { sf } else { wf };
        let ties: usize = wf.iter().map(|f| f.states.len()).sum();
        let mut acc: Vec<(usize, f64)> = Vec::new();
        for f in faces {
            let rows = ph.repertoire_marginals(&d.mech, &f.nodes, weak);
            for (i, &j) in f.nodes.iter().enumerate() {
                let h = norm_entropy(&rows[i]);
                match acc.iter().position(|x| x.0 == j) {
                    Some(p) => {
                        if h > acc[p].1 {
                            acc[p].1 = h;
                        }
                    }
                    None => acc.push((j, h)),
                }
            }
        }
        acc.sort_by_key(|x| x.0);
        let tot: f64 = acc.iter().map(|x| x.1).sum();
        let k = acc.len().max(1) as f64;
        let nodes: Vec<(usize, f64)> = acc
            .iter()
            .map(|&(j, h)| (j, rho * if tot > MASS_EPS { h / tot } else { 1.0 / k }))
            .collect();
        out.push(Residual { mech: d.mech.clone(), weak, rho, ties, nodes, from_strong });
    }
    out
}
#[derive(Clone, Debug)]
pub struct ResidualMap {
    pub by_pos: Vec<f64>,
    pub latent: f64,
    pub cause: f64,
    pub effect: f64,
}
impl ResidualMap {
    pub fn total(&self) -> f64 {
        self.cause + self.effect
    }
    pub fn top(&self, k: usize) -> Vec<(usize, f64)> {
        let mut v: Vec<(usize, f64)> = self
            .by_pos
            .iter()
            .copied()
            .enumerate()
            .filter(|&(_, m)| m > MASS_EPS)
            .collect();
        v.sort_by(|a, b| b.1.total_cmp(&a.1).then(a.0.cmp(&b.0)));
        if k > 0 {
            v.truncate(k);
        }
        v
    }
}

pub fn structure(cx: &mut Ctx<'_>, state: &[u16], cfg: &SharpCfg, detail: Detail) -> Structure {
    let n = cx.n;
    let found = find_complexes(cx, state, cfg);
    let tau = found.tau;
    let latent: Vec<bool> = (0..n).map(|j| !cx.anchored(j)).collect();
    let mut complexes: Vec<ComplexStruct> = Vec::with_capacity(found.systems.len());
    for sys in found.systems {
        let mut inside = vec![false; n];
        for &j in &sys.nodes {
            inside[j] = true;
        }
        let mut bn: Vec<usize> = Vec::new();
        let mut bs: Vec<u16> = Vec::new();
        for j in 0..n {
            if !inside[j] {
                bn.push(j);
                bs.push(state[j]);
            }
        }
        cx.set_background(&bn, &bs);
        let (dists, foot, rels, resid, text) = {
            let ph = Phi::new(&*cx, tau, state, cfg.phi.clone());
            let dists: Vec<Dist> = ph
                .distinctions(true)
                .into_iter()
                .filter(|d| d.anchored)
                .collect();
            let foot: Vec<Vec<usize>> = dists.iter().map(|d| ph.footprint(d)).collect();
            let rels = relations(&dists, &latent, &foot);
            let (resid, text) = match detail {
                Detail::Full => (
                    residuals(&ph, &dists),
                    dists.iter().map(|d| ph.describe(d)).collect::<Vec<String>>(),
                ),
                Detail::Lean => (Vec::new(), Vec::new()),
            };
            (dists, foot, rels, resid, text)
        };
        let sum_d = sum_phi_d(&dists);
        let sum_r: f64 = rels.iter().map(|r| r.phi).sum();
        let sum_r_cross: f64 = rels.iter().filter(|r| r.cross).map(|r| r.phi).sum();
        complexes.push(ComplexStruct { sys, dists, foot, rels, sum_d, sum_r, sum_r_cross, resid, text });
    }
    cx.clear_background();
    Structure { tau, complexes }
}
pub fn first_max(v: &[f64]) -> usize {
    let mut best = -1.0f64;
    let mut at = 0usize;
    for (t, &x) in v.iter().enumerate() {
        if x > best + T_TOL * best.max(1.0) {
            best = x;
            at = t;
        }
    }
    at
}
pub fn running_area(v: &[f64]) -> f64 {
    if v.is_empty() {
        return 0.0;
    }
    let mut run = v[0];
    let mut s = 0.0f64;
    for &x in v {
        run = run.max(x);
        s += run;
    }
    s / v.len() as f64
}

fn scan(cx: &mut Ctx<'_>, cfg: &SharpCfg) -> Scan {
    let traj = cx.trajectory();
    let mut seen: HashMap<Vec<u16>, usize> = HashMap::new();
    let mut steps: Vec<Step> = Vec::with_capacity(traj.len());
    for (t, s) in traj.iter().enumerate() {
        let known: Option<usize> = seen.get(s).copied();
        let step = match known {
            Some(k) => Step { t, ..steps[k] },
            None => {
                let st = structure(cx, s, cfg, Detail::Lean);
                seen.insert(s.clone(), t);
                Step {
                    t,
                    phi: st.phi(),
                    sum_d: st.sum_d(),
                    sum_r: st.sum_r(),
                    sum_r_cross: st.sum_r_cross(),
                    tau: st.tau,
                    n_cx: st.complexes.len(),
                }
            }
        };
        steps.push(step);
    }
    let phis: Vec<f64> = steps.iter().map(|s| s.phi).collect();
    let t_star = first_max(&phis);
    let area = running_area(&phis);
    Scan { steps, traj, t_star, area }
}

        let sc = scan(&mut cx, &cfg);
        phi += sc.steps[sc.t_star].phi;
        cross += sc.steps[sc.t_star].sum_r_cross;
        area += sc.area;
    }
    Score { phi: phi * scale, area: area * scale, abstraction: ratio(cross, phi) }
}
pub fn analyze(m: &Model, e: &Percept, detail: Detail, unit: Unit, draws: Option<usize>) -> Analysis {
    let cfg = SharpCfg::from_model(m);
    let scale = unit_scale(unit, e.a);
    let nf = Ctx::n_frames(m, e);
    let mut frames: Vec<FrameResult> = Vec::with_capacity(nf);
    let mut rm = ResidualMap { by_pos: vec![0.0; e.len()], latent: 0.0, cause: 0.0, effect: 0.0 };
    for f in 0..nf {
        let mut cx = Ctx::new(m, e, f);
        if let Some(r) = draws {
            cx.set_draws(r);
        }
        let fr = frame_result(&mut cx, f, &cfg, detail);
        add_residuals(&mut rm, &cx, &fr.st);
        frames.push(fr);
    }
    let (mut phi_bits, mut sum_d, mut sum_r, mut cross, mut area) = (0.0f64, 0.0f64, 0.0f64, 0.0f64, 0.0f64);
    for fr in &frames {
        let s = &fr.steps[fr.t_star];
        phi_bits += s.phi;
        sum_d += s.sum_d;
        sum_r += s.sum_r;
        cross += s.sum_r_cross;
        area += fr.area;
    }
    rm.cause *= scale;
    rm.effect *= scale;
    rm.latent *= scale;
    for v in rm.by_pos.iter_mut() {
        *v *= scale;
    }
    Analysis {
        unit,
        scale,
        a: e.a,
        len: e.len(),
        phi: phi_bits * scale,
        phi_bits,
        sum_d: sum_d * scale,
        sum_r: sum_r * scale,
        abstraction: ratio(cross, phi_bits),
        area: area * scale,
        resid: rm,
        frames,
    }
}

    }

pub fn softmax_into(f: &[f64], beta: f64, p: &mut [f64]) {
    let mut mx = f[0];
    for &v in &f[1..] {
        if v > mx {
            mx = v;
        }
    }
    let mut s = 0.0f64;
    for k in 0..f.len() {
        let e = (beta * (f[k] - mx)).exp();
        p[k] = e;
        s += e;
    }
    let inv = 1.0 / s;
    let mut s2 = 0.0f64;
    for k in 0..f.len() {
        let v = (p[k] * inv).max(PFLOOR);
        p[k] = v;
        s2 += v;
    }
    let inv2 = 1.0 / s2;
    for k in 0..f.len() {
        p[k] *= inv2;
    }
}
#[inline]
fn sample(p: &[f64], u: f64) -> u16 {
    let mut x = u;
    for (k, &q) in p.iter().enumerate() {
        if x < q {
            return k as u16;
        }
        x -= q;
    }
    (p.len() - 1) as u16
}
fn top_by_weight(mut v: Vec<(usize, f64)>, cap: usize) -> Vec<usize> {
    v.sort_by(|x, y| {
        y.1.partial_cmp(&x.1)
            .unwrap_or(std::cmp::Ordering::Equal)
            .then(x.0.cmp(&y.0))
    });
    v.truncate(cap);
    v.iter().map(|p| p.0).collect()
}

                }
            }

        for v in 0..a {
            out[v] *= inv;

            let cap = ((64usize << 20) / (8 * a)).max(1024);
            if cache.len() >= cap {
                cache.clear();

            cache.insert(key, out[..a].to_vec().into_boxed_slice());
        }
    }

    pub fn marg_exact(&self, j: usize, clamp: &[usize], tau: usize, syms: &[u16], out: &mut [f64]) -> bool {
        let n = self.n;
        let a = self.a;
        if let Some(s) = self.pin[j] {
            out[..a].fill(0.0);
            out[s as usize] = 1.0;
            return true;
        }
        let tau = tau.max(1);
        let k = self.free.len();
        let mut total: usize = 1;
        for _ in 0..k {
            total = match total.checked_mul(a) {
                Some(t) if t <= 4_000_000 => t,
                _ => return false,
            };
        }
        if tau > 1 && total > 2_048 {
            return false;
        }
        let amax = (a - 1) as u16;
        let mut fixed: Vec<Option<u16>> = vec![None; n];
        for (&c, &s) in clamp.iter().zip(syms.iter()) {
            if c < n && self.pin[c].is_none() {
                fixed[c] = Some(s.min(amax));
            }
        }
        let mut x = vec![0u16; n];
        for i in 0..n {
            if let Some(s) = self.pin[i] {
                x[i] = s;
            }
        }
        let mut f = vec![0.0f64; a];
        let mut pj = vec![0.0f64; a];
        let mut dist = vec![0.0f64; k * a];
        let mut p = vec![0.0f64; total];
        for idx in 0..total {
            self.decode_free(idx, &mut x);
            let mut w = 1.0f64;
            for &i in &self.free {
                w *= match fixed[i] {
                    Some(s) => {
                        if x[i] == s {
                            1.0
                        } else {
                            0.0
                        }
                    }
                    None => 1.0 / a as f64,
                };
            }
            p[idx] = w;
        }
        for _ in 1..tau {
            let mut q = vec![0.0f64; total];
            for src in 0..total {
                if p[src] == 0.0 {
                    continue;
                }
                self.decode_free(src, &mut x);
                for (t, &i) in self.free.iter().enumerate() {
                    self.field_all(i, &x, &mut f);
                    softmax_into(&f, self.m.beta, &mut dist[t * a..(t + 1) * a]);
                }
                for dst in 0..total {
                    let mut pr = p[src];
                    let mut d = dst;
                    for t in 0..k {
                        pr *= dist[t * a + d % a];
                        d /= a;
                    }
                    q[dst] += pr;
                }
            }
            p = q;
        }
        out[..a].fill(0.0);
        for idx in 0..total {
            if p[idx] == 0.0 {
                continue;
            }
            self.decode_free(idx, &mut x);
            self.field_all(j, &x, &mut f);
            softmax_into(&f, self.m.beta, &mut pj);
            for v in 0..a {
                out[v] += p[idx] * pj[v];
            }
        }
        true
    }
}

fn max_flow(cap0: &[f64], m: usize, s: usize, t: usize) -> (f64, u32) {
    let mut r = cap0.to_vec();
    let mut flow = 0.0f64;
    let mut par = vec![usize::MAX; m];
    let mut queue: Vec<usize> = Vec::with_capacity(m);
    loop {
        par.fill(usize::MAX);
        par[s] = s;
        queue.clear();
        queue.push(s);
        let mut head = 0usize;
        while head < queue.len() && par[t] == usize::MAX {
            let u = queue[head];
            head += 1;
            for v in 0..m {
                if par[v] == usize::MAX && r[u * m + v] > 1e-12 {
                    par[v] = u;
                    queue.push(v);
                }
            }
        }
        if par[t] == usize::MAX {
            let mut mask = 0u32;
            for v in 0..m {
                if par[v] != usize::MAX {
                    mask |= 1u32 << v;
                }
            }
            return (flow, mask);
        }
        let mut b = f64::INFINITY;
        let mut v = t;
        while v != s {
            let u = par[v];
            b = b.min(r[u * m + v]);
            v = u;
        }
        v = t;
        while v != s {
            let u = par[v];
            r[u * m + v] -= b;
            r[v * m + u] += b;
            v = u;
        }
        flow += b;
    }
}
fn max_prod_abs(p: &[Vec<f64>], r: &[Vec<f64>], vals: &[Vec<u16>], lim: usize) -> f64 {
    let d = p.len();
    if d == 0 {
        return 0.0;
    }
    let mut total = 1usize;
    for v in vals {
        total = total.saturating_mul(v.len());
    }
    let mut best = 0.0f64;
    if total <= lim {
        let mut idx = vec![0usize; d];
        loop {
            let mut pr = 1.0f64;
            let mut s = 0.0f64;
            for y in 0..d {
                let v = vals[y][idx[y]] as usize;
                pr *= p[y][v];
                s += r[y][v];
            }
            let val = pr * s.abs();
            if val > best {
                best = val;
            }
            let mut y = 0usize;
            while y < d {
                idx[y] += 1;
                if idx[y] < vals[y].len() {
                    break;
                }
                idx[y] = 0;
                y += 1;
            }
            if y == d {
                break;
            }
        }
        return best;
    }
    let mut z = vec![0usize; d];
    for start in 0..4usize {
        for y in 0..d {
            let mut pick = 0usize;
            let mut bs = f64::NEG_INFINITY;
            for (k, &v) in vals[y].iter().enumerate() {
                let v = v as usize;
                let sc = match start {
                    0 => p[y][v],
                    1 => r[y][v],
                    2 => -r[y][v],
                    _ => unit(mix3(0xC0FF_EE, y as u64, k as u64)),
                };
                if sc > bs {
                    bs = sc;
                    pick = k;
                }
            }
            z[y] = pick;
        }
        for _sweep in 0..32 {
            let mut improved = false;
            for y in 0..d {
                let mut pr = 1.0f64;
                let mut sr = 0.0f64;
                for u in 0..d {
                    if u != y {
                        let v = vals[u][z[u]] as usize;
                        pr *= p[u][v];
                        sr += r[u][v];
                    }
                }
                let cur = vals[y][z[y]] as usize;
                let mut bv = z[y];
                let mut bval = pr * p[y][cur] * (sr + r[y][cur]).abs();
                for (k, &v) in vals[y].iter().enumerate() {
                    let v = v as usize;
                    let val = pr * p[y][v] * (sr + r[y][v]).abs();
                    if val > bval * (1.0 + REL_TOL) {
                        bval = val;
                        bv = k;
                    }
                }
                if bv != z[y] {
                    z[y] = bv;
                    improved = true;
                }
            }
            if !improved {
                break;
            }
        }
        let mut pr = 1.0f64;
        let mut sr = 0.0f64;
        for y in 0..d {
            let v = vals[y][z[y]] as usize;
            pr *= p[y][v];
            sr += r[y][v];
        }
        let val = pr * sr.abs();
        if val > best {
            best = val;
        }
    }
    best
}
struct SysEval<'c, 'm> {
    cx: &'c Ctx<'m>,
    tau: usize,
    k: Vec<usize>,
    sk: Vec<u16>,
    p: Vec<Vec<f64>>,
    vals: Vec<Vec<u16>>,
    pmax: Vec<f64>,
    lim: usize,
    memo: HashMap<u32, f64>,
}
impl<'c, 'm> SysEval<'c, 'm> {
    fn new(cx: &'c Ctx<'m>, tau: usize, state: &[u16], k: &[usize], lim: usize) -> SysEval<'c, 'm> {
        let a = cx.a;
        let amax = (a - 1) as u16;
        let sk: Vec<u16> = k.iter().map(|&j| state[j].min(amax)).collect();
        let mut p: Vec<Vec<f64>> = Vec::with_capacity(k.len());
        let mut vals: Vec<Vec<u16>> = Vec::with_capacity(k.len());
        let mut pmax: Vec<f64> = Vec::with_capacity(k.len());
        for &y in k {
            let py = cx.marg_vec(y, k, tau, &sk);
            let vs: Vec<u16> = match cx.e_sym(y) {
                Some(s) => vec![s],
                None => (0..a as u16).collect(),
            };
            let mut mx = 0.0f64;
            for &v in &vs {
                if py[v as usize] > mx {
                    mx = py[v as usize];
                }
            }
            p.push(py);
            vals.push(vs);
            pmax.push(mx);
        }
        SysEval { cx, tau, k: k.to_vec(), sk, p, vals, pmax, lim, memo: HashMap::new() }
    }
    fn cut_value(&mut self, mask: u32) -> f64 {
        if let Some(&v) = self.memo.get(&mask) {
            return v;
        }
        let m = self.k.len();
        let a = self.cx.a;
        let mut fa = 1.0f64;
        let mut bpos: Vec<usize> = Vec::new();
        for i in 0..m {
            if ((mask >> i) & 1) == 1 {
                fa *= self.pmax[i];
            } else {
                bpos.push(i);
            }
        }
        let bnodes: Vec<usize> = bpos.iter().map(|&i| self.k[i]).collect();
        let bsyms: Vec<u16> = bpos.iter().map(|&i| self.sk[i]).collect();
        let mut pb: Vec<Vec<f64>> = Vec::with_capacity(bpos.len());
        let mut rb: Vec<Vec<f64>> = Vec::with_capacity(bpos.len());
        let mut vb: Vec<Vec<u16>> = Vec::with_capacity(bpos.len());
        let mut changed = false;
        let mut q = vec![0.0f64; a];
        for (t, &i) in bpos.iter().enumerate() {
            self.cx.marg(bnodes[t], &bnodes, self.tau, &bsyms, &mut q);
            let mut r = vec![0.0f64; a];
            for v in 0..a {
                r[v] = (self.p[i][v] / q[v]).log2();
                if r[v].abs() > 1e-12 {
                    changed = true;
                }
            }
            pb.push(self.p[i].clone());
            rb.push(r);
            vb.push(self.vals[i].clone());
        }
        let val = if changed { fa * max_prod_abs(&pb, &rb, &vb, self.lim) } else { 0.0 };
        self.memo.insert(mask, val);
        val
    }
    fn eval_cut(&mut self, mk: u32, evald: &mut Vec<(f64, u32)>) -> f64 {
        let fresh = !self.memo.contains_key(&mk);
        let v = self.cut_value(mk);
        if fresh {
            evald.push((v, mk));
        }
        v
    }
    fn min_cuts(&self) -> Vec<u32> {
        let m = self.k.len();
        let mut cap = vec![0.0f64; m * m];
        for i in 0..m {
            for j in 0..m {
                if i != j {
                    cap[i * m + j] = self.cx.infl(self.k[i], self.k[j]);
                }
            }
        }
        let mut found: Vec<(f64, u32)> = Vec::new();
        for t in 1..m {
            found.push(max_flow(&cap, m, 0, t));
            found.push(max_flow(&cap, m, t, 0));
        }
        found.sort_by(|x, y| x.0.partial_cmp(&y.0).unwrap_or(Ordering::Equal).then(x.1.cmp(&y.1)));
        let mut out: Vec<u32> = Vec::new();
        for (_, mk) in found {
            if !out.contains(&mk) {
                out.push(mk);
                if out.len() >= 6 {
                    break;
                }
            }
        }
        out
    }
    fn search(&mut self, seed: u64, ncuts: usize, exh: usize) -> (f64, u32, bool) {
        let m = self.k.len();
        let full: u32 = (1u32 << m) - 1;
        let total = (1usize << m) - 2;
        let mut best = f64::INFINITY;
        let mut bm = 0u32;
        if total <= exh {
            for mk in 1..full {
                let v = self.cut_value(mk);
                if v < best {
                    best = v;
                    bm = mk;
                }
                if best <= PHI_EPS {
                    break;
                }
            }
            return (best, bm, true);
        }
        let mut evald: Vec<(f64, u32)> = Vec::new();
        let mut cand: Vec<u32> = Vec::new();
        for i in 0..m {
            cand.push(1u32 << i);
            cand.push(full ^ (1u32 << i));
        }
        cand.extend(self.min_cuts());
        let mut rng = Rng::new(seed);
        for _ in 0..ncuts {
            let mut mk = 0u32;
            for i in 0..m {
                if rng.chance(0.5) {
                    mk |= 1u32 << i;
                }
            }
            if mk != 0 && mk != full {
                cand.push(mk);
            }
        }
        for mk in cand {
            let v = self.eval_cut(mk, &mut evald);
            if v < best {
                best = v;
                bm = mk;
            }
            if best <= PHI_EPS {
                return (best, bm, false);
            }
        }
        evald.sort_by(|x, y| x.0.partial_cmp(&y.0).unwrap_or(Ordering::Equal).then(x.1.cmp(&y.1)));
        let starts: Vec<u32> = evald.iter().take(3).map(|e| e.1).collect();
        let mut budget = 4 * m + 32;
        for s0 in starts {
            let mut cur = s0;
            let mut cv = self.cut_value(cur);
            loop {
                let mut improved = false;
                for i in 0..m {
                    if budget == 0 {
                        break;
                    }
                    let nm = cur ^ (1u32 << i);
                    if nm == 0 || nm == full {
                        continue;
                    }
                    budget -= 1;
                    let v = self.cut_value(nm);
                    if v < cv * (1.0 - REL_TOL) {
                        cur = nm;
                        cv = v;
                        improved = true;
                    }
                    if cv <= PHI_EPS {
                        break;
                    }
                }
                if !improved || budget == 0 || cv <= PHI_EPS {
                    break;
                }
            }
            if cv < best {
                best = cv;
                bm = cur;
            }
            if best <= PHI_EPS {
                break;
            }
        }
        (best, bm, false)
    }
}
pub fn system_phi(cx: &mut Ctx<'_>, state: &[u16], tau: usize, cfg: &SharpCfg, k: &[usize]) -> SysPhi {
    assert!(k.len() <= 31, "a candidate system has at most 31 nodes");
    if k.len() < 2 {
        return SysPhi { phi: 0.0, src: 0, exact: true };
    }
    let n = cx.n;
    let mut inside = vec![false; n];
    for &j in k {
        inside[j] = true;
    }
    let mut bn: Vec<usize> = Vec::new();
    let mut bs: Vec<u16> = Vec::new();
    for j in 0..n {
        if !inside[j] {
            bn.push(j);
            bs.push(state[j]);
        }
    }
    cx.set_background(&bn, &bs);
    let seed = hash_usize(mix(cx.seed(), tau as u64), k);
    let (phi, src, exact) = {
        let mut ev = SysEval::new(&*cx, tau, state, k, cfg.enum_lim);
        ev.search(seed, cfg.ncuts, cfg.exh_cuts)
    };
    SysPhi { phi: if phi.is_finite() { phi.max(0.0) } else { 0.0 }, src, exact }
}
pub fn split_mask(k: &[usize], mask: u32) -> (Vec<usize>, Vec<usize>) {
    let mut a: Vec<usize> = Vec::new();
    let mut b: Vec<usize> = Vec::new();
    for (i, &j) in k.iter().enumerate() {
        if ((mask >> i) & 1) == 1 {
            a.push(j);
        } else {
            b.push(j);
        }
    }
    (a, b)
}
fn consider(
    cx: &mut Ctx<'_>,
    state: &[u16],
    tau: usize,
    cfg: &SharpCfg,
    cache: &mut SysCache,
    seen: &mut Vec<(Vec<usize>, SysPhi)>,
    k: Vec<usize>,
) -> f64 {
    if k.len() < 2 {
        return 0.0;
    }
    if let Some(s) = seen.iter().find(|s| s.0 == k) {
        return s.1.phi;
    }
    let cached: Option<SysPhi> = cache.get(&k).copied();
    let sp = match cached {
        Some(v) => v,
        None => {
            let v = system_phi(cx, state, tau, cfg, &k);
            cache.insert(k.clone(), v);
            v
        }
    };
    seen.push((k, sp));
    sp.phi
}
fn scc_candidates(cx: &Ctx<'_>, rem: &[usize], cap: usize) -> Vec<Vec<usize>> {
    let m = rem.len();
    let mut vals: Vec<f64> = Vec::new();
    for x in 0..m {
        for y in 0..m {
            if x != y {
                let c = cx.infl(rem[x], rem[y]);
                if c > 0.0 {
                    vals.push(c);
                }
            }
        }
    }
    let mut out: Vec<Vec<usize>> = Vec::new();
    if vals.is_empty() {
        return out;
    }
    vals.sort_by(|a, b| a.partial_cmp(b).unwrap_or(Ordering::Equal));
    for &q in &[0.0f64, 0.25, 0.5, 0.75] {
        let thr = vals[((vals.len() - 1) as f64 * q) as usize];
        let mut reach = vec![false; m * m];
        for x in 0..m {
            reach[x * m + x] = true;
            for y in 0..m {
                if x != y {
                    let c = cx.infl(rem[x], rem[y]);
                    if c > 0.0 && c >= thr {
                        reach[x * m + y] = true;
                    }
                }
            }
        }
        for k in 0..m {
            for x in 0..m {
                if reach[x * m + k] {
                    for y in 0..m {
                        if reach[k * m + y] {
                            reach[x * m + y] = true;
                        }
                    }
                }
            }
        }
        let mut done = vec![false; m];
        for x in 0..m {
            if done[x] {
                continue;
            }
            let mut comp: Vec<usize> = Vec::new();
            for y in 0..m {
                if reach[x * m + y] && reach[y * m + x] {
                    comp.push(y);
                    done[y] = true;
                }
            }
            if comp.len() < 2 {
                continue;
            }
            if comp.len() > cap {
                let mut sc: Vec<(usize, f64)> = Vec::new();
                for &y in &comp {
                    let mut s = 0.0f64;
                    for &z in &comp {
                        if z != y {
                            s += cx.infl(rem[y], rem[z]) + cx.infl(rem[z], rem[y]);
                        }
                    }
                    sc.push((y, s));
                }
                sc.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(Ordering::Equal).then(a.0.cmp(&b.0)));
                sc.truncate(cap);
                comp = sc.iter().map(|p| p.0).collect();
                comp.sort_unstable();
            }
            out.push(comp.iter().map(|&y| rem[y]).collect());
        }
    }
    out
}
fn best_system(
    cx: &mut Ctx<'_>,
    state: &[u16],
    tau: usize,
    cfg: &SharpCfg,
    cache: &mut SysCache,
    rem: &[usize],
) -> Option<(Vec<usize>, SysPhi)> {
    if rem.len() < 2 {
        return None;
    }
    let cap = cfg.sys_cap.clamp(2, 24);
    let mut seen: Vec<(Vec<usize>, SysPhi)> = Vec::new();
    for x in 0..rem.len() {
        for y in (x + 1)..rem.len() {
            let (i, j) = (rem[x], rem[y]);
            if cx.weight(i, j) != 0.0 && cx.weight(j, i) != 0.0 {
                consider(cx, state, tau, cfg, cache, &mut seen, vec![i, j]);
            }
        }
    }
    let clusters = scc_candidates(&*cx, rem, cap);
    for k in clusters {
        consider(cx, state, tau, cfg, cache, &mut seen, k);
    }
    let mut starts: Vec<(Vec<usize>, f64)> = seen.iter().map(|s| (s.0.clone(), s.1.phi)).collect();
    starts.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(Ordering::Equal).then_with(|| a.0.cmp(&b.0)));
    starts.truncate(cfg.seeds.max(1));
    for (k0, v0) in starts {
        if v0 <= PHI_EPS {
            continue;
        }
        let mut cur = k0;
        let mut cv = v0;
        for _ in 0..cap {
            let mut moves: Vec<Vec<usize>> = Vec::new();
            if cur.len() < cap {
                for &x in rem {
                    if cur.contains(&x) {
                        continue;
                    }
                    let into = cur.iter().any(|&c| cx.weight(c, x) != 0.0);
                    let outof = cur.iter().any(|&c| cx.weight(x, c) != 0.0);
                    if into && outof {
                        let mut nk = cur.clone();
                        nk.push(x);
                        nk.sort_unstable();
                        moves.push(nk);
                    }
                }
            }
            if cur.len() > 2 {
                for i in 0..cur.len() {
                    let mut nk = cur.clone();
                    nk.remove(i);
                    moves.push(nk);
                }
            }
            let mut bm: Option<(Vec<usize>, f64)> = None;
            for nk in moves {
                let v = consider(cx, state, tau, cfg, cache, &mut seen, nk.clone());
                let better = match &bm {
                    None => true,
                    Some(b) => v > b.1,
                };
                if better {
                    bm = Some((nk, v));
                }
            }
            match bm {
                Some((nk, v)) if v > cv * (1.0 + REL_TOL) => {
                    cur = nk;
                    cv = v;
                }
                _ => break,
            }
        }
    }
    let mut best: Option<(Vec<usize>, SysPhi)> = None;
    for s in seen {
        let better = match &best {
            None => true,
            Some(b) => s.1.phi > b.1.phi * (1.0 + REL_TOL),
        };
        if better {
            best = Some(s);
        }
    }
    match best {
        Some(b) if b.1.phi > PHI_EPS => Some(b),
        _ => None,
    }
}
pub fn find_complexes(cx: &mut Ctx<'_>, state: &[u16], cfg: &SharpCfg) -> Found {
    let n = cx.n;
    let gm = cfg.grain_max.max(1);
    let all: Vec<usize> = (0..n).collect();
    let mut caches: Vec<SysCache> = (0..=gm).map(|_| HashMap::new()).collect();
    let mut tau_star = 1usize;
    let mut first: Option<(Vec<usize>, SysPhi)> = None;
    for tau in 1..=gm {
        if let Some(c) = best_system(cx, state, tau, cfg, &mut caches[tau], &all) {
            let better = match &first {
                None => true,
                Some(f) => c.1.phi > f.1.phi * (1.0 + REL_TOL),
            };
            if better {
                tau_star = tau;
                first = Some(c);
            }
        }
    }
    let mut chosen: Vec<(Vec<usize>, SysPhi)> = Vec::new();
    if let Some(f) = first {
        let mut remaining: Vec<usize> = all.iter().copied().filter(|j| !f.0.contains(j)).collect();
        chosen.push(f);
        while chosen.len() < cfg.max_cx.max(1) && remaining.len() >= 2 {
            match best_system(cx, state, tau_star, cfg, &mut caches[tau_star], &remaining) {
                Some(c) => {
                    remaining.retain(|j| !c.0.contains(j));
                    chosen.push(c);
                }
                None => break,
            }
        }
    }
    let systems: Vec<System> = chosen
        .into_iter()
        .map(|(k, sp)| {
            let (a, b) = split_mask(&k, sp.src);
            System { nodes: k, phi_s: sp.phi, cut_src: a, cut_dst: b, exact: sp.exact }
        })
        .collect();
    Found { tau: tau_star, systems }
}

    }

    pub fn weighted(&mut self, w: &[f64]) -> usize {
        let tot: f64 = w.iter().sum();
        let mut x = self.f64() * tot;
        for (k, &p) in w.iter().enumerate() {
            if x < p {
                return k;
            }
            x -= p;
        }
        w.len().saturating_sub(1)
    }
    pub fn fork(&mut self, tag: u64) -> Rng { Rng::new(mix(self.next_u64(), tag)) }
}
pub fn mean(x: &[f64]) -> f64 {
    if x.is_empty() { 0.0 } else { x.iter().sum::<f64>() / x.len() as f64 }
}
pub fn pearson(x: &[f64], y: &[f64]) -> f64 {
    let n = x.len().min(y.len());
    if n < 2 {
        return 0.0;
    }
    let mx = mean(&x[..n]);
    let my = mean(&y[..n]);
    let (mut sxy, mut sxx, mut syy) = (0.0f64, 0.0f64, 0.0f64);
    for k in 0..n {
        let dx = x[k] - mx;
        let dy = y[k] - my;
        sxy += dx * dy;
        sxx += dx * dx;
        syy += dy * dy;
    }
    if sxx <= 0.0 || syy <= 0.0 { 0.0 } else { sxy / (sxx * syy).sqrt() }
}
pub fn show_byte(b: u8) -> String {
    match b {
        b' ' => "\u{2423}".to_string(),
        b'\n' => "\\n".to_string(),
        b'\t' => "\\t".to_string(),
        0x21..=0x7E => (b as char).to_string(),
        _ => format!("\\x{:02x}", b),
    }
}
pub fn show_bytes(bs: &[u8]) -> String {
    let mut s = String::new();
    for &b in bs {
        s.push_str(&show_byte(b));
    }
    s
}
