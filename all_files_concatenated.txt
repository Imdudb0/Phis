use crate::core::phi_sharp::{
    evaluate_phi_sharp, EffectSummary, ModelParams, PhiSharpConfig, PhiSharpEval, StabilityProbes,

use crate::core::tensor::{Mat, RelTensor};
#[derive(Debug, Clone)]
pub struct FlowConfig {
    pub max_steps: usize,
    pub dt_init: f64,
    pub dt_min: f64,
    pub dt_max: f64,
    pub eta: f64,
    pub omega: f64,
    pub omega_perp: f64,
    pub max_perp_norm: f64,
    pub gamma_abs: f64,
    pub abs_eps: f64,
    pub ball_radius_k: f64,
    pub lyapunov_tol: f64,
}
impl Default for FlowConfig {
    fn default() -> Self {
        Self {
            max_steps: 40,
            dt_init: 0.25,
            dt_min: 1e-5,
            dt_max: 8.0,
            eta: 1.0,
            omega: 0.5,
            omega_perp: 0.1,
            max_perp_norm: 0.5,
            gamma_abs: 0.02,
            abs_eps: 1e-2,
            ball_radius_k: 8.0,
            lyapunov_tol: 1e-8,
        }
    }
}
#[derive(Debug, Clone)]
pub struct FlowStepTrace {
    pub step: usize,
    pub t: f64,
    pub dt: f64,
    pub lyapunov: f64,

    pub lambda2: f64,
    pub residual: f64,

    pub active_causal_edges: usize,
    pub state_norm: f64,
    pub lyapunov_rate_pred: f64,

#[derive(Debug, Clone)]
pub struct FlowTrajectory {
    pub r_init: RelTensor,
    pub r_final: RelTensor,
    pub final_eval: PhiSharpEval,
    pub final_lyapunov: f64,
    pub trace: Vec<FlowStepTrace>,
    pub converged_to_critical_manifold: bool,
}
impl FlowTrajectory {
    pub fn is_monotone_lyapunov(&self, tol: f64) -> bool {
        self.trace
            .windows(2)
            .all(|w| w[1].lyapunov + tol >= w[0].lyapunov)
    }
}
fn eval_lyapunov(
    params: &ModelParams,
    r: &RelTensor,
    effect: &EffectSummary,
    probes: &StabilityProbes,
    phi_cfg: &PhiSharpConfig,
    flow_cfg: &FlowConfig,
    compute_grad: bool,
) -> (f64, f64, PhiSharpEval, Option<RelTensor>) {
    let (eval, grad_phi_opt) =
        evaluate_phi_sharp(params, r, effect, probes, phi_cfg, compute_grad);
    let (r_abs, grad_abs_opt) = if flow_cfg.gamma_abs > 0.0 {
        let (v, g) = r.spectral_abstraction_reg(flow_cfg.abs_eps);
        (v, if compute_grad { Some(g) } else { None })
    } else {
        (0.0, None)
    };
    let lyap = eval.phi_sharp - flow_cfg.gamma_abs * r_abs;
    let grad = match (grad_phi_opt, grad_abs_opt) {
        (Some(mut gp), Some(ga)) => {
            gp.axpy(-flow_cfg.gamma_abs, &ga);
            Some(gp)
        }
        (Some(gp), None) => Some(gp),
        _ => None,
    };
    (lyap, r_abs, eval, grad)
}
pub fn build_orthogonal_solenoidal(
    params: &ModelParams,
    r: &RelTensor,
    a_asc: &Mat,
    omega_perp: f64,
    max_norm: f64,
) -> Mat {
    if omega_perp <= 0.0 {
        return Mat::zeros(r.n);
    }
    let mut raw = params.w_sol.clone();
    for c in 0..r.d {
        let nu = params.nu_sol[c];
        if nu != 0.0 {
            raw.axpy(nu, &r.channels[c]);
        }
    }
    let mut a_perp = raw.skew();
    let asc_norm_sq = a_asc.frob_norm_sq();
    if asc_norm_sq > 1e-20 {
        let proj = a_perp.frob_dot(a_asc) / asc_norm_sq;
        a_perp.axpy(-proj, a_asc);
    }
    a_perp.scale(omega_perp);
    let p_norm = a_perp.frob_norm();
    if p_norm > max_norm && p_norm > 0.0 {
        a_perp.scale(max_norm / p_norm);
    }
    a_perp
}
pub fn cayley_rotation(a_skew: &Mat, dt: f64) -> Mat {
    let n = a_skew.n;
    if dt == 0.0 || a_skew.frob_norm_sq() <= 1e-30 {
        return Mat::identity(n);
    }
    let half_dt = 0.5 * dt;
    let mut lhs = Mat::identity(n);
    let mut rhs = Mat::identity(n);
    for i in 0..n {
        for j in 0..n {
            let aij = a_skew.get(i, j);
            lhs.add_assign(i, j, -half_dt * aij);
            rhs.add_assign(i, j, half_dt * aij);
        }
    }
    solve_linear_multi_rhs(&lhs, &rhs)
}
pub fn conjugate_isospectral(r: &RelTensor, q: &Mat) -> RelTensor {
    let mut out = RelTensor::zeros(r.n, r.d);
    for c in 0..r.d {
        let qr = q.matmul(&r.channels[c]);
        out.channels[c] = qr.matmul_t_right(q);
    }
    out
}
fn solve_linear_multi_rhs(a: &Mat, b: &Mat) -> Mat {
    let n = a.n;
    debug_assert_eq!(n, b.n);
    let mut m = a.data.clone();
    let mut x = b.data.clone();
    for col in 0..n {
        let mut pivot_row = col;
        let mut max_abs = m[col * n + col].abs();
        for row in (col + 1)..n {
            let val = m[row * n + col].abs();
            if val > max_abs {
                max_abs = val;
                pivot_row = row;
            }
        }
        if max_abs <= 1e-15 {
            continue;
        }
        if pivot_row != col {
            for k in col..n {
                m.swap(col * n + k, pivot_row * n + k);
            }
            for k in 0..n {
                x.swap(col * n + k, pivot_row * n + k);
            }
        }
        let diag = m[col * n + col];
        for row in (col + 1)..n {
            let factor = m[row * n + col] / diag;
            if factor == 0.0 {
                continue;
            }
            m[row * n + col] = 0.0;
            for k in (col + 1)..n {
                m[row * n + k] -= factor * m[col * n + k];
            }
            for k in 0..n {
                x[row * n + k] -= factor * x[col * n + k];
            }
        }
    }
    for col in (0..n).rev() {
        let diag = m[col * n + col];
        if diag.abs() <= 1e-15 {
            continue;
        }
        let inv_diag = 1.0 / diag;
        for k in 0..n {
            let mut sum = x[col * n + k];
            for j in (col + 1)..n {
                sum -= m[col * n + j] * x[j * n + k];
            }
            x[col * n + k] = sum * inv_diag;
        }
    }
    Mat { n, data: x }
}
pub fn run_abductive_flow(
    params: &ModelParams,
    effect: &EffectSummary,
    probes: &StabilityProbes,
    phi_cfg: &PhiSharpConfig,
    flow_cfg: &FlowConfig,
    r_init_override: Option<RelTensor>,
) -> FlowTrajectory {
    let mut r = r_init_override
        .unwrap_or_else(|| params.encode_regional_pole(effect, flow_cfg.ball_radius_k));
    r.clamp_to_ball(flow_cfg.ball_radius_k);
    let r_init = r.clone();
    let (mut lyap, mut r_abs, mut eval, mut grad_opt) =
        eval_lyapunov(params, &r, effect, probes, phi_cfg, flow_cfg, true);
    let mut t = 0.0f64;
    let mut dt = flow_cfg.dt_init;
    let mut trace = Vec::with_capacity(flow_cfg.max_steps + 1);
    let mut converged = false;
    for step in 0..flow_cfg.max_steps {
        let grad = grad_opt.take().unwrap();
        let proj_g = r.project_tangent_ball(&grad, flow_cfg.ball_radius_k);
        let dual = r.skew_commutator_dual(&grad);
        let mut a_asc = dual.clone();
        a_asc.scale(flow_cfg.omega);
        let rate_grad = flow_cfg.eta * proj_g.frob_norm_sq();
        let rate_lie = flow_cfg.omega * dual.frob_norm_sq();
        let lyap_rate = rate_grad + rate_lie;
        let ascent_scale = proj_g.frob_norm() + a_asc.frob_norm();
        let perp_cap = (flow_cfg.omega_perp * ascent_scale).min(flow_cfg.max_perp_norm);
        let a_perp = build_orthogonal_solenoidal(
            params,
            &r,
            &a_asc,
            flow_cfg.omega_perp,
            perp_cap,
        );
        let perp_norm = a_perp.frob_norm();
        if step == 0 {
            trace.push(FlowStepTrace {
                step: 0,
                t: 0.0,
                dt: 0.0,
                lyapunov: lyap,
                phi_sharp: eval.phi_sharp,
                lambda2: eval.lambda2,
                residual: eval.residual,
                v_stab: eval.v_stab,
                r_abs,
                active_causal_edges: eval.active_causal_edges,
                state_norm: r.frob_norm(),
                lyapunov_rate_pred: lyap_rate,
                perp_norm,
                backtracks: 0,
            });
        }
        if lyap_rate <= flow_cfg.lyapunov_tol {
            converged = true;
            break;
        }
        let mut a_total = a_asc.clone();
        a_total.axpy(1.0, &a_perp);
        let mut trial_dt = dt;
        let mut backtracks = 0usize;
        let mut accepted = None;
        while trial_dt >= flow_cfg.dt_min {
            let q = cayley_rotation(&a_total, trial_dt);
            let mut r_cand = conjugate_isospectral(&r, &q);
            r_cand.axpy(trial_dt * flow_cfg.eta, &proj_g);
            r_cand.clamp_to_ball(flow_cfg.ball_radius_k);
            let (cand_lyap, cand_r_abs, cand_eval, _) =
                eval_lyapunov(params, &r_cand, effect, probes, phi_cfg, flow_cfg, false);
            let armijo_rhs = lyap + 1e-4 * trial_dt * lyap_rate;
            if cand_lyap >= armijo_rhs || (cand_lyap >= lyap && trial_dt <= flow_cfg.dt_min * 4.0) {
                accepted = Some((r_cand, cand_lyap, cand_r_abs, cand_eval, trial_dt));
                break;
            }
            trial_dt *= 0.5;
            backtracks += 1;
        }
        if let Some((r_next, next_lyap, next_r_abs, _, used_dt)) = accepted {
            r = r_next;
            t += used_dt;
            let (l_ref, ra_ref, e_ref, g_ref) =
                eval_lyapunov(params, &r, effect, probes, phi_cfg, flow_cfg, true);
            lyap = l_ref.max(next_lyap);
            r_abs = ra_ref.min(next_r_abs).max(0.0);
            eval = e_ref;
            grad_opt = g_ref;
            trace.push(FlowStepTrace {
                step: step + 1,
                t,
                dt: used_dt,
                lyapunov: lyap,
                phi_sharp: eval.phi_sharp,
                lambda2: eval.lambda2,
                residual: eval.residual,
                v_stab: eval.v_stab,
                r_abs,
                active_causal_edges: eval.active_causal_edges,
                state_norm: r.frob_norm(),
                lyapunov_rate_pred: lyap_rate,
                perp_norm,
                backtracks,
            });
            if backtracks == 0 {
                dt = (used_dt * 1.25).min(flow_cfg.dt_max);
            } else {
                dt = used_dt;
            }
        } else {
            converged = true;
            break;
        }
    }
    FlowTrajectory {
        r_init,
        r_final: r,
        final_eval: eval,
        final_lyapunov: lyap,
        trace,
        converged_to_critical_manifold: converged,
    }
}

use crate::core::flow::FlowConfig;
use crate::core::phi_sharp::{ModelParams, PhiSharpConfig, BI_DIM, REL_DIM};
use crate::core::projection::ProjectionConfig;
use crate::core::tensor::Mat;
use std::fs::{self, File};
use std::io::{Read, Write};
use std::path::Path;
const MAGIC: &[u8; 8] = b"PHISHARP";
const FORMAT_VERSION: u32 = 2;
#[derive(Debug, Clone)]
pub struct PhisCheckpoint {
    pub params: ModelParams,
    pub phi_cfg: PhiSharpConfig,
    pub flow_cfg: FlowConfig,
    pub proj_cfg: ProjectionConfig,
}
impl PhisCheckpoint {
    pub fn save<P: AsRef<Path>>(&self, path: P) -> Result<(), String> {
        let path_ref = path.as_ref();
        if let Some(parent) = path_ref.parent() {
            if !parent.as_os_str().is_empty() {
                fs::create_dir_all(parent)
                    .map_err(|e| format!("Impossible de créer le dossier {:?}: {}", parent, e))?;
            }
        }
        let payload = self.encode_payload();
        let checksum = fnv1a_64(&payload);
        let mut file = File::create(path_ref)
            .map_err(|e| format!("Impossible de créer {:?}: {}", path_ref, e))?;
        file.write_all(MAGIC).map_err(|e| e.to_string())?;
        file.write_all(&FORMAT_VERSION.to_le_bytes())
            .map_err(|e| e.to_string())?;
        file.write_all(&(payload.len() as u64).to_le_bytes())
            .map_err(|e| e.to_string())?;
        file.write_all(&checksum.to_le_bytes())
            .map_err(|e| e.to_string())?;
        file.write_all(&payload).map_err(|e| e.to_string())?;
        Ok(())
    }
    pub fn load<P: AsRef<Path>>(path: P) -> Result<Self, String> {
        let path_ref = path.as_ref();
        let mut file = File::open(path_ref)
            .map_err(|e| format!("Impossible d'ouvrir {:?}: {}", path_ref, e))?;
        let mut bytes = Vec::new();
        file.read_to_end(&mut bytes).map_err(|e| e.to_string())?;
        if bytes.len() < 28 {
            return Err("Fichier .phis tronqué (en-tête incomplet)".to_string());
        }
        if &bytes[0..8] != MAGIC {
            return Err("En-tête magique invalide (attendu PHISHARP)".to_string());
        }
        let version = u32::from_le_bytes(bytes[8..12].try_into().unwrap());
        if version != FORMAT_VERSION {
            return Err(format!(
                "Version .phis non supportée: {} (attendue {})",
                version, FORMAT_VERSION
            ));
        }
        let payload_len = u64::from_le_bytes(bytes[12..20].try_into().unwrap()) as usize;
        let expected_crc = u64::from_le_bytes(bytes[20..28].try_into().unwrap());
        if bytes.len() != 28 + payload_len {
            return Err("Longueur de charge utile .phis incohérente".to_string());
        }
        let payload = &bytes[28..];
        let actual_crc = fnv1a_64(payload);
        if actual_crc != expected_crc {
            return Err(format!(
                "Checksum FNV-1a invalide (corrompu) : {:016X} != {:016X}",
                actual_crc, expected_crc
            ));
        }
        Self::decode_payload(payload)
    }
    fn encode_payload(&self) -> Vec<u8> {
        let mut buf = Vec::new();
        let p = &self.params;
        write_u64(&mut buf, p.n as u64);
        write_u64(&mut buf, p.d as u64);
        write_u64(&mut buf, p.h as u64);
        write_f64(&mut buf, self.phi_cfg.beta_iit);
        write_f64(&mut buf, self.phi_cfg.alpha);
        write_f64(&mut buf, self.phi_cfg.beta);
        write_f64(&mut buf, self.phi_cfg.lambda_bi);
        write_f64(&mut buf, self.phi_cfg.hinge_tau);
        write_f64(&mut buf, self.phi_cfg.cap_eps);
        write_u64(&mut buf, self.phi_cfg.max_krylov as u64);
        write_u64(&mut buf, self.phi_cfg.max_inv_power as u64);
        write_u64(&mut buf, self.flow_cfg.max_steps as u64);
        write_f64(&mut buf, self.flow_cfg.dt_init);
        write_f64(&mut buf, self.flow_cfg.dt_min);
        write_f64(&mut buf, self.flow_cfg.dt_max);
        write_f64(&mut buf, self.flow_cfg.eta);
        write_f64(&mut buf, self.flow_cfg.omega);
        write_f64(&mut buf, self.flow_cfg.omega_perp);
        write_f64(&mut buf, self.flow_cfg.max_perp_norm);
        write_f64(&mut buf, self.flow_cfg.gamma_abs);
        write_f64(&mut buf, self.flow_cfg.abs_eps);
        write_f64(&mut buf, self.flow_cfg.ball_radius_k);
        write_f64(&mut buf, self.flow_cfg.lyapunov_tol);
        write_u64(&mut buf, self.proj_cfg.out_len as u64);
        write_u64(&mut buf, self.proj_cfg.max_steps as u64);
        write_f64(&mut buf, self.proj_cfg.eta_init);
        write_f64(&mut buf, self.proj_cfg.eta_min);
        write_f64(&mut buf, self.proj_cfg.eta_max);
        write_f64(&mut buf, self.proj_cfg.lambda_bi);
        write_f64(&mut buf, self.proj_cfg.gamma_alphabet);
        write_f64(&mut buf, self.proj_cfg.gamma_anticopy);
        write_f64(&mut buf, self.proj_cfg.tau_sharp);
        write_f64(&mut buf, self.proj_cfg.tol_step_l12);
        write_vec_f64(&mut buf, &p.w_dir);
        write_vec_f64(&mut buf, &p.w_comp);
        write_vec_f64(&mut buf, &p.b_edge);
        write_vec_f64(&mut buf, &p.w_rel_uni);
        write_vec_f64(&mut buf, &p.b_rel_uni);
        write_vec_f64(&mut buf, &p.w_rel_bi);
        write_vec_f64(&mut buf, &p.b_rel_bi);
        write_vec_f64(&mut buf, &p.w_enc_uni);
        write_vec_f64(&mut buf, &p.w_enc_bi);
        write_vec_f64(&mut buf, &p.w_init_asym);
        write_vec_f64(&mut buf, &p.w_init_ov);
        write_vec_f64(&mut buf, &p.w_sol.data);
        write_vec_f64(&mut buf, &p.nu_sol);
        buf
    }
    fn decode_payload(data: &[u8]) -> Result<Self, String> {
        let mut cursor = 0usize;
        let n = read_u64(data, &mut cursor)? as usize;
        let d = read_u64(data, &mut cursor)? as usize;
        let h = read_u64(data, &mut cursor)? as usize;
        let phi_cfg = PhiSharpConfig {
            beta_iit: read_f64(data, &mut cursor)?,
            alpha: read_f64(data, &mut cursor)?,
            beta: read_f64(data, &mut cursor)?,
            lambda_bi: read_f64(data, &mut cursor)?,
            hinge_tau: read_f64(data, &mut cursor)?,
            cap_eps: read_f64(data, &mut cursor)?,
            max_krylov: read_u64(data, &mut cursor)? as usize,
            max_inv_power: read_u64(data, &mut cursor)? as usize,
        };
        let flow_cfg = FlowConfig {
            max_steps: read_u64(data, &mut cursor)? as usize,
            dt_init: read_f64(data, &mut cursor)?,
            dt_min: read_f64(data, &mut cursor)?,
            dt_max: read_f64(data, &mut cursor)?,
            eta: read_f64(data, &mut cursor)?,
            omega: read_f64(data, &mut cursor)?,
            omega_perp: read_f64(data, &mut cursor)?,
            max_perp_norm: read_f64(data, &mut cursor)?,
            gamma_abs: read_f64(data, &mut cursor)?,
            abs_eps: read_f64(data, &mut cursor)?,
            ball_radius_k: read_f64(data, &mut cursor)?,
            lyapunov_tol: read_f64(data, &mut cursor)?,
        };
        let proj_cfg = ProjectionConfig {
            out_len: read_u64(data, &mut cursor)? as usize,
            max_steps: read_u64(data, &mut cursor)? as usize,
            eta_init: read_f64(data, &mut cursor)?,
            eta_min: read_f64(data, &mut cursor)?,
            eta_max: read_f64(data, &mut cursor)?,
            lambda_bi: read_f64(data, &mut cursor)?,
            gamma_alphabet: read_f64(data, &mut cursor)?,
            gamma_anticopy: read_f64(data, &mut cursor)?,
            tau_sharp: read_f64(data, &mut cursor)?,
            tol_step_l12: read_f64(data, &mut cursor)?,
        };
        let w_dir = read_vec_f64(data, &mut cursor)?;
        let w_comp = read_vec_f64(data, &mut cursor)?;
        let b_edge = read_vec_f64(data, &mut cursor)?;
        let w_rel_uni = read_vec_f64(data, &mut cursor)?;
        let b_rel_uni = read_vec_f64(data, &mut cursor)?;
        let w_rel_bi = read_vec_f64(data, &mut cursor)?;
        let b_rel_bi = read_vec_f64(data, &mut cursor)?;
        let w_enc_uni = read_vec_f64(data, &mut cursor)?;
        let w_enc_bi = read_vec_f64(data, &mut cursor)?;
        let w_init_asym = read_vec_f64(data, &mut cursor)?;
        let w_init_ov = read_vec_f64(data, &mut cursor)?;
        let w_sol_data = read_vec_f64(data, &mut cursor)?;
        let nu_sol = read_vec_f64(data, &mut cursor)?;
        if w_rel_uni.len() != h * REL_DIM
            || b_rel_uni.len() != REL_DIM
            || w_rel_bi.len() != h * BI_DIM
            || b_rel_bi.len() != BI_DIM
            || w_init_ov.len() != d
            || w_sol_data.len() != n * n
        {
            return Err("Dimensions incohérentes dans le checkpoint .phis v2".to_string());
        }
        let params = ModelParams {
            n,
            d,
            h,
            w_dir,
            w_comp,
            b_edge,
            w_rel_uni,
            b_rel_uni,
            w_rel_bi,
            b_rel_bi,
            w_enc_uni,
            w_enc_bi,
            w_init_asym,
            w_init_ov,
            w_sol: Mat {
                n,
                data: w_sol_data,
            },
            nu_sol,
        };
        Ok(Self {
            params,
            phi_cfg,
            flow_cfg,
            proj_cfg,
        })
    }
}
fn fnv1a_64(bytes: &[u8]) -> u64 {
    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
    for &b in bytes {
        hash ^= b as u64;
        hash = hash.wrapping_mul(0x0000_0100_0000_01b3);
    }
    hash
}
fn write_u64(buf: &mut Vec<u8>, v: u64) {
    buf.extend_from_slice(&v.to_le_bytes());
}
fn write_f64(buf: &mut Vec<u8>, v: f64) {
    buf.extend_from_slice(&v.to_le_bytes());
}
fn write_vec_f64(buf: &mut Vec<u8>, slice: &[f64]) {
    write_u64(buf, slice.len() as u64);
    for &v in slice {
        write_f64(buf, v);
    }
}
fn read_u64(data: &[u8], cursor: &mut usize) -> Result<u64, String> {
    if *cursor + 8 > data.len() {
        return Err("Fin de fichier inattendue (u64)".to_string());
    }
    let v = u64::from_le_bytes(data[*cursor..*cursor + 8].try_into().unwrap());
    *cursor += 8;
    Ok(v)
}
fn read_f64(data: &[u8], cursor: &mut usize) -> Result<f64, String> {
    if *cursor + 8 > data.len() {
        return Err("Fin de fichier inattendue (f64)".to_string());
    }
    let v = f64::from_le_bytes(data[*cursor..*cursor + 8].try_into().unwrap());
    *cursor += 8;
    Ok(v)
}
fn read_vec_f64(data: &[u8], cursor: &mut usize) -> Result<Vec<f64>, String> {
    let len = read_u64(data, cursor)? as usize;
    if *cursor + len * 8 > data.len() {
        return Err("Fin de fichier inattendue (Vec<f64>)".to_string());
    }
    let mut out = Vec::with_capacity(len);
    for _ in 0..len {
        out.push(read_f64(data, cursor)?);
    }
    Ok(out)
}

    pub fn from_bytes(bytes: &[u8], n: usize) -> Self {
        let len = bytes.len();
        if len == 0 || n == 0 {
            return Self::empty(n);
        }
        let mut counts = [0usize; BYTE_VOCAB];
        let mut positions: Vec<Vec<usize>> = vec![Vec::new(); BYTE_VOCAB];
        for (pos, &b) in bytes.iter().enumerate() {
            let bi = b as usize;
            counts[bi] += 1;
            positions[bi].push(pos);
        }
        let mut alphabet = Vec::new();
        let mut byte_to_alpha = [None; BYTE_VOCAB];
        for b in 0..BYTE_VOCAB {
            if counts[b] > 0 {
                let idx = alphabet.len();
                alphabet.push(b as u8);
                byte_to_alpha[b] = Some(idx);
            }
        }
        let k_uni = alphabet.len();
        let max_count = counts.iter().copied().max().unwrap_or(1) as f64;
        let ln_len1 = ((len as f64) + 1.0).ln().max(1e-12);
        let ln_kmax = (k_uni.max(2) as f64).ln();
        let mut out_counts: Vec<BTreeMap<usize, usize>> = vec![BTreeMap::new(); k_uni];
        let mut in_counts: Vec<BTreeMap<usize, usize>> = vec![BTreeMap::new(); k_uni];
        let mut out_totals = vec![0usize; k_uni];
        let mut in_totals = vec![0usize; k_uni];
        let mut bigram_counts: BTreeMap<(usize, usize), usize> = BTreeMap::new();
        let mut bigram_pos_sum: BTreeMap<(usize, usize), f64> = BTreeMap::new();
        let mut bigram_pos_sq_sum: BTreeMap<(usize, usize), f64> = BTreeMap::new();
        for pos in 1..len {
            let u = byte_to_alpha[bytes[pos - 1] as usize].unwrap();
            let v = byte_to_alpha[bytes[pos] as usize].unwrap();
            *out_counts[u].entry(v).or_insert(0) += 1;
            *in_counts[v].entry(u).or_insert(0) += 1;
            out_totals[u] += 1;
            in_totals[v] += 1;
            *bigram_counts.entry((u, v)).or_insert(0) += 1;
            let x = ((pos as f64) + 0.5) / (len as f64);
            *bigram_pos_sum.entry((u, v)).or_insert(0.0) += x;
            *bigram_pos_sq_sum.entry((u, v)).or_insert(0.0) += x * x;
        }
        let mut win_neighbors: Vec<BTreeSet<usize>> = vec![BTreeSet::new(); k_uni];
        for pos in 0..len {
            let u = byte_to_alpha[bytes[pos] as usize].unwrap();
            let lo = pos.saturating_sub(2);
            let hi = (pos + 2).min(len - 1);
            for p2 in lo..=hi {
                if p2 != pos {
                    let v = byte_to_alpha[bytes[p2] as usize].unwrap();
                    win_neighbors[u].insert(v);
                }
            }
        }

        for a in 0..k_uni {

            let c = counts[b] as f64;
            let pos_list = &positions[b];
            let f0 = (c + 1.0).ln() / ln_len1;
            let f1 = c / max_count;
            let f2 = normalized_entropy(&in_counts[a], in_totals[a], ln_kmax);

            let mut ret_short = 0usize;
            let mut ret_med = 0usize;
            for &p in pos_list {

                {

                }
                let max_d = (p + 8).min(len - 1);

                    ret_med += 1;
                }

            let f5 = (ret_med as f64) / c;
            let mut mean_x = 0.0;
            for &p in pos_list {
                mean_x += ((p as f64) + 0.5) / (len as f64);
            }
            mean_x /= c;
            let mut var_x = 0.0;
            for &p in pos_list {
                let dx = ((p as f64) + 0.5) / (len as f64) - mean_x;
                var_x += dx * dx;
            }
            let std_x = (var_x / c).sqrt();

            rel_feats[off + 5] = f5;
            rel_feats[off + 6] = mean_x;
            rel_feats[off + 7] = std_x;
            rel_feats[off + 8] = span_x;

            }

                    let f = compute_pair_feat(u, v);

                }

            (k_bi, bi_feats, bi, bi_mass, h_emp_bi)
        } else {
            (0, Vec::new(), Vec::new(), vec![0.0; n], 0.0)
        };

        let mut node_bi_sig = vec![0.0; n * BI_DIM];

                        for m in 0..REL_DIM {
                            node_uni_sig[i * REL_DIM + m] += p * rel_feats[a * REL_DIM + m];
                        }
                    }

                            node_bi_sig[i * BI_DIM + m] += p * bi_feats[g * BI_DIM + m];
                        }
                    }
                }
            }
        }

        for i in 0..n {
            for j in i..n {
                let denom = node_norms[i] * node_norms[j];
                let cos = if denom > 1e-12 && uni_mass[i] > 1e-12 && uni_mass[j] > 1e-12 {
                    let inv_i = 1.0 / uni_mass[i];
                    let inv_j = 1.0 / uni_mass[j];
                    let qi = &uni[i * k_uni..(i + 1) * k_uni];
                    let qj = &uni[j * k_uni..(j + 1) * k_uni];
                    let mut dot = 0.0;
                    for a in 0..k_uni {
                        dot += (qi[a] * inv_i) * (qj[a] * inv_j);
                    }
                    (dot / denom).clamp(0.0, 1.0)
                } else {
                    0.0
                };
                node_overlap.set(i, j, cos);
                node_overlap.set(j, i, cos);
            }
        }
        Self {
            n,
            len,
            alphabet,
            k_uni,
            byte_to_alpha,
            rel_feats,
            uni,
            uni_mass,
            h_emp_uni,
            k_bi,
            bi_feats,
            bi,
            bi_mass,
            h_emp_bi,
            has_bigrams,
            node_uni_sig,
            node_bi_sig,
            node_overlap,
        }

    pub fn rho_background(&self, lambda_bi: f64) -> f64 {
        let (w_uni, w_bi) = residual_weights(self.has_bigrams, lambda_bi);
        let rho0_u = if self.k_uni > 0 {
            ((self.k_uni as f64).ln() - self.h_emp_uni).max(0.0)
        } else {
            0.0
        };
        let rho0_b = if w_bi > 0.0 && self.k_bi > 0 {
            ((self.k_bi as f64).ln() - self.h_emp_bi).max(0.0)
        } else {
            0.0
        };
        w_uni * rho0_u + w_bi * rho0_b
    }

fn normalized_entropy(counts: &BTreeMap<usize, usize>, total: usize, ln_kmax: f64) -> f64 {
    if total == 0 || ln_kmax <= 0.0 {
        return 0.0;
    }
    let inv_t = 1.0 / (total as f64);
    let mut h = 0.0;
    for &c in counts.values() {
        if c > 0 {
            let p = (c as f64) * inv_t;
            h -= p * p.ln();
        }
    }
    (h / ln_kmax).clamp(0.0, 1.0)
}
fn compute_empirical_entropy(dist: &[f64], mass: &[f64], n: usize, k: usize) -> f64 {
    if k == 0 {
        return 0.0;
    }
    let mut h_total = 0.0;
    for i in 0..n {
        let m = mass[i];
        if m <= 1e-15 {
            continue;
        }
        let inv_m = 1.0 / m;
        let row = &dist[i * k..(i + 1) * k];
        for &q in row {
            if q > 1e-15 {
                h_total -= q * (q * inv_m).ln();
            }
        }
    }
    h_total.max(0.0)
}
pub fn spatial_assignment(pos: usize, len: usize, n: usize, out: &mut [f64]) {
    let x = ((pos as f64) + 0.5) / (len as f64);
    let sigma = (0.65 / (n as f64)).max(1e-4);
    let inv_2sig2 = 0.5 / (sigma * sigma);
    let mut max_logit = f64::NEG_INFINITY;
    for i in 0..n {
        let center = ((i as f64) + 0.5) / (n as f64);
        let d = x - center;
        let v = -d * d * inv_2sig2;
        out[i] = v;
        if v > max_logit {
            max_logit = v;
        }
    }
    let mut sum = 0.0;
    for v in out.iter_mut() {
        *v = (*v - max_logit).exp();
        sum += *v;
    }
    let inv_sum = 1.0 / sum;
    for v in out.iter_mut() {
        *v *= inv_sum;
    }
}

            u0_uni,
            k_bi: effect.k_bi,
            u_bi,
            u0_bi,
        }
    }
    pub fn encode_regional_pole(&self, effect: &EffectSummary, radius_k: f64) -> RelTensor {
        debug_assert_eq!(effect.n, self.n);
        let n = self.n;
        let d = self.d;
        let mut sigs = vec![0.0; n * d];
        for i in 0..n {
            let u_sig = &effect.node_uni_sig[i * REL_DIM..(i + 1) * REL_DIM];
            let b_sig = &effect.node_bi_sig[i * BI_DIM..(i + 1) * BI_DIM];
            for c in 0..d {
                let mut acc = 0.0;
                let w_u = &self.w_enc_uni[c * REL_DIM..(c + 1) * REL_DIM];
                let w_b = &self.w_enc_bi[c * BI_DIM..(c + 1) * BI_DIM];
                for m in 0..REL_DIM {
                    acc += w_u[m] * u_sig[m];
                }
                if effect.has_bigrams {
                    for m in 0..BI_DIM {
                        acc += 0.5 * w_b[m] * b_sig[m];
                    }
                }
                sigs[i * d + c] = acc.tanh();
            }
        }
        let mut target_sigs = vec![0.0; n * d];
        for j in 0..n {
            for c in 0..d {
                let mut acc = 0.0;
                for a in 0..d {
                    acc += self.w_init_asym[c * d + a] * sigs[j * d + a];
                }
                target_sigs[j * d + c] = acc;
            }
        }
        let mut r0 = RelTensor::zeros(n, d);
        for c in 0..d {
            let w_ov = self.w_init_ov[c];
            for i in 0..n {
                for j in 0..n {
                    let ov = effect.node_overlap.get(i, j);
                    let raw = sigs[i * d + c] * target_sigs[j * d + c]
                        + 0.25 * (sigs[i * d + c] - sigs[j * d + c])
                        + w_ov * ov;
                    r0.set(i, j, c, raw.tanh() * 0.35);
                }
            }
        }
        r0.clamp_to_ball(radius_k);
        r0
    }
    pub fn residual_only(&self, r: &RelTensor, effect: &EffectSummary, lambda_bi: f64) -> f64 {
        let heads = self.precompute_heads(effect);
        self.residual_only_precomputed(r, effect, &heads, lambda_bi)
    }
    pub fn residual_only_precomputed(
        &self,
        r: &RelTensor,
        effect: &EffectSummary,
        heads: &PrecomputedHeads,
        lambda_bi: f64,
    ) -> f64 {
        if effect.k_uni == 0 {
            return 0.0;
        }
        let (w_uni, w_bi) = residual_weights(effect.has_bigrams, lambda_bi);
        let n = self.n;
        let d = self.d;
        let h = self.h;
        let inv_sqrt_n = 1.0 / (n as f64).sqrt();
        let comp: Vec<Mat> = (0..d)
            .map(|c| r.channels[c].matmul(&r.channels[c]))
            .collect();
        let mut h_node = vec![0.0; h];
        let mut logits = vec![0.0; heads.k_uni.max(heads.k_bi).max(1)];
        let mut ce_uni = 0.0;
        let mut ce_bi = 0.0;
        for i in 0..n {
            h_node.fill(0.0);
            for j in 0..n {
                for r_idx in 0..h {
                    let mut z = self.b_edge[r_idx];
                    let w_d = &self.w_dir[r_idx * d..(r_idx + 1) * d];
                    let w_c = &self.w_comp[r_idx * d..(r_idx + 1) * d];
                    for c in 0..d {
                        z += w_d[c] * r.get(i, j, c) + w_c[c] * comp[c].get(i, j);
                    }
                    h_node[r_idx] += z.tanh();
                }
            }
            for val in &mut h_node {
                *val *= inv_sqrt_n;
            }
            if effect.uni_mass[i] > 0.0 {
                ce_uni += bilinear_cross_entropy(
                    &h_node,
                    &heads.u_uni,
                    &heads.u0_uni,
                    &effect.uni[i * heads.k_uni..(i + 1) * heads.k_uni],
                    effect.uni_mass[i],
                    &mut logits[..heads.k_uni],
                );
            }
            if w_bi > 0.0 && heads.k_bi > 0 && effect.bi_mass[i] > 0.0 {
                ce_bi += bilinear_cross_entropy(
                    &h_node,
                    &heads.u_bi,
                    &heads.u0_bi,
                    &effect.bi[i * heads.k_bi..(i + 1) * heads.k_bi],
                    effect.bi_mass[i],
                    &mut logits[..heads.k_bi],
                );
            }
        }
        let kl_uni = (ce_uni - effect.h_emp_uni).max(0.0);
        let kl_bi = if w_bi > 0.0 {
            (ce_bi - effect.h_emp_bi).max(0.0)
        } else {
            0.0
        };
        w_uni * kl_uni + w_bi * kl_bi
    }
    pub fn residual_and_grad_r(
        &self,
        r: &RelTensor,
        effect: &EffectSummary,
        lambda_bi: f64,
    ) -> (f64, RelTensor) {
        let heads = self.precompute_heads(effect);
        self.residual_and_grad_r_precomputed(r, effect, &heads, lambda_bi)
    }
    pub fn residual_and_grad_r_precomputed(
        &self,
        r: &RelTensor,
        effect: &EffectSummary,
        heads: &PrecomputedHeads,
        lambda_bi: f64,
    ) -> (f64, RelTensor) {
        if effect.k_uni == 0 {
            return (0.0, RelTensor::zeros(self.n, self.d));
        }
        let (w_uni, w_bi) = residual_weights(effect.has_bigrams, lambda_bi);
        let n = self.n;
        let d = self.d;
        let h = self.h;
        let inv_sqrt_n = 1.0 / (n as f64).sqrt();
        let comp: Vec<Mat> = (0..d)
            .map(|c| r.channels[c].matmul(&r.channels[c]))
            .collect();
        let mut m_edge = vec![0.0; n * n * h];
        let mut h_nodes = vec![0.0; n * h];
        for i in 0..n {
            for j in 0..n {
                let base = (i * n + j) * h;
                for r_idx in 0..h {
                    let mut z = self.b_edge[r_idx];
                    let w_d = &self.w_dir[r_idx * d..(r_idx + 1) * d];
                    let w_c = &self.w_comp[r_idx * d..(r_idx + 1) * d];
                    for c in 0..d {
                        z += w_d[c] * r.get(i, j, c) + w_c[c] * comp[c].get(i, j);
                    }
                    let m = z.tanh();
                    m_edge[base + r_idx] = m;
                    h_nodes[i * h + r_idx] += m;
                }
            }
            for r_idx in 0..h {
                h_nodes[i * h + r_idx] *= inv_sqrt_n;
            }
        }
        let mut ce_uni = 0.0;
        let mut ce_bi = 0.0;
        let mut grad_h = vec![0.0; n * h];
        let mut probs = vec![0.0; heads.k_uni.max(heads.k_bi).max(1)];
        for i in 0..n {
            let hi = &h_nodes[i * h..(i + 1) * h];
            let ghi = &mut grad_h[i * h..(i + 1) * h];
            if effect.uni_mass[i] > 0.0 {
                ce_uni += bilinear_cross_entropy_and_grad_h(
                    hi,
                    &heads.u_uni,
                    &heads.u0_uni,
                    &effect.uni[i * heads.k_uni..(i + 1) * heads.k_uni],
                    effect.uni_mass[i],
                    w_uni,
                    &mut probs[..heads.k_uni],
                    ghi,
                );
            }
            if w_bi > 0.0 && heads.k_bi > 0 && effect.bi_mass[i] > 0.0 {
                ce_bi += bilinear_cross_entropy_and_grad_h(
                    hi,
                    &heads.u_bi,
                    &heads.u0_bi,
                    &effect.bi[i * heads.k_bi..(i + 1) * heads.k_bi],
                    effect.bi_mass[i],
                    w_bi,
                    &mut probs[..heads.k_bi],
                    ghi,
                );
            }
        }
        let mut grad_r = RelTensor::zeros(n, d);
        let mut grad_comp: Vec<Mat> = (0..d).map(|_| Mat::zeros(n)).collect();
        for i in 0..n {
            let ghi = &grad_h[i * h..(i + 1) * h];
            for j in 0..n {
                let base = (i * n + j) * h;
                for r_idx in 0..h {
                    let m = m_edge[base + r_idx];
                    let dz = inv_sqrt_n * ghi[r_idx] * (1.0 - m * m);
                    if dz == 0.0 {
                        continue;
                    }
                    let w_d = &self.w_dir[r_idx * d..(r_idx + 1) * d];
                    let w_c = &self.w_comp[r_idx * d..(r_idx + 1) * d];
                    for c in 0..d {
                        grad_r.channels[c].add_assign(i, j, dz * w_d[c]);
                        grad_comp[c].add_assign(i, j, dz * w_c[c]);
                    }
                }
            }
        }
        for c in 0..d {
            let term1 = grad_comp[c].matmul_t_right(&r.channels[c]);
            let term2 = r.channels[c].matmul_t_left(&grad_comp[c]);
            grad_r.channels[c].axpy(1.0, &term1);
            grad_r.channels[c].axpy(1.0, &term2);
        }
        let kl_uni = (ce_uni - effect.h_emp_uni).max(0.0);
        let kl_bi = if w_bi > 0.0 {
            (ce_bi - effect.h_emp_bi).max(0.0)
        } else {
            0.0
        };
        (w_uni * kl_uni + w_bi * kl_bi, grad_r)
    }
}
#[derive(Debug, Clone)]
pub struct StabilityProbes {
    pub rotations: Vec<Mat>,
    pub p_avg: Mat,
    pub abs_kappa: f64,
}
impl StabilityProbes {
    pub fn build(
        n: usize,
        num_rotations: usize,
        max_angle: f64,
        abs_kappa: f64,
        seed: u64,
    ) -> Self {
        let mut rng = Rng::from_seed(seed);
        let mut rotations = Vec::with_capacity(num_rotations);
        for _ in 0..num_rotations {
            let mut q = Mat::identity(n);
            let num_planes = n.max(2);
            for _ in 0..num_planes {
                if n < 2 {
                    break;
                }
                let p = rng.gen_range(n);
                let mut r = rng.gen_range(n - 1);
                if r >= p {
                    r += 1;
                }
                let theta = (rng.gen_f64() * 2.0 - 1.0) * max_angle;
                let c = theta.cos();
                let s = theta.sin();
                for col in 0..n {
                    let qp = q.get(p, col);
                    let qr = q.get(r, col);
                    q.set(p, col, c * qp - s * qr);
                    q.set(r, col, s * qp + c * qr);
                }
            }
            rotations.push(q);
        }
        let mu = 0.35;
        let off = mu / (n.max(1) as f64);
        let mut p_avg = Mat::zeros(n);
        for i in 0..n {
            for j in 0..n {
                p_avg.set(i, j, if i == j { (1.0 - mu) + off } else { off });
            }
        }
        Self {
            rotations,
            p_avg,
            abs_kappa,
        }
    }
    pub fn probe_count(&self) -> usize {
        self.rotations.len() + if self.abs_kappa > 0.0 { 1 } else { 0 }
    }
    pub fn apply_rotation(q: &Mat, r: &RelTensor) -> RelTensor {
        let mut out = RelTensor::zeros(r.n, r.d);
        for c in 0..r.d {
            out.channels[c] = q.matmul(&r.channels[c]).matmul_t_right(q);
        }
        out
    }
    pub fn adjoint_rotation(q: &Mat, grad_rot: &RelTensor) -> RelTensor {
        let mut out = RelTensor::zeros(grad_rot.n, grad_rot.d);
        for c in 0..grad_rot.d {
            out.channels[c] = q.matmul_t_left(&grad_rot.channels[c]).matmul(q);
        }
        out
    }
    pub fn apply_abstraction(&self, r: &RelTensor) -> RelTensor {
        let k = self.abs_kappa;
        let one_minus_k = 1.0 - k;
        let mut out = RelTensor::zeros(r.n, r.d);
        for c in 0..r.d {
            let prp = self.p_avg.matmul(&r.channels[c]).matmul(&self.p_avg);
            for idx in 0..(r.n * r.n) {
                out.channels[c].data[idx] =
                    one_minus_k * r.channels[c].data[idx] + k * prp.data[idx];
            }
        }
        out
    }
    pub fn adjoint_abstraction(&self, grad_abs: &RelTensor) -> RelTensor {
        self.apply_abstraction(grad_abs)
    }
}
#[derive(Debug, Clone)]
pub struct PhiSharpEval {
    pub phi_sharp: f64,
    pub lambda2: f64,
    pub residual: f64,
    pub rho_background: f64,
    pub v_stab: f64,
    pub causal_weights: Mat,
    pub active_causal_edges: usize,
    pub fiedler: FiedlerResult,
}
pub fn evaluate_phi_sharp(
    params: &ModelParams,
    r: &RelTensor,
    effect: &EffectSummary,
    probes: &StabilityProbes,
    cfg: &PhiSharpConfig,
    compute_grad: bool,
) -> (PhiSharpEval, Option<RelTensor>) {
    let n = r.n;
    let d = r.d;
    let heads = params.precompute_heads(effect);
    let rho_bg = effect.rho_background(cfg.lambda_bi);
    let (rho, grad_rho_opt) = if compute_grad {
        let (val, g) = params.residual_and_grad_r_precomputed(r, effect, &heads, cfg.lambda_bi);
        (val, Some(g))
    } else {
        (
            params.residual_only_precomputed(r, effect, &heads, cfg.lambda_bi),
            None,
        )
    };
    let p_bar = (-rho).exp();
    let mut causal_weights = Mat::zeros(n);
    let mut abl_deriv_rho = Mat::zeros(n);
    let mut abl_deriv_rho_ij = Mat::zeros(n);
    let mut active_causal_edges = 0usize;
    for i in 0..n {
        for j in 0..n {
            if i == j {
                continue;
            }
            if (0..d).all(|c| r.get(i, j, c) == 0.0) {
                continue;
            }
            let r_abl = r.with_ablated_edge(i, j);
            let rho_ij = params.residual_only_precomputed(&r_abl, effect, &heads, cfg.lambda_bi);
            let (s_ij, ds_drho_ij) = smooth_cap(rho_ij, rho_bg, cfg.cap_eps);
            let delta = s_ij - rho;
            let (h_val, h_prime) = smooth_hinge(delta, cfg.hinge_tau);
            if h_val > 0.0 {
                let wij = p_bar * h_val;
                causal_weights.set(i, j, wij);
                active_causal_edges += 1;
                if compute_grad {
                    abl_deriv_rho.set(i, j, -p_bar * (h_val + h_prime));
                    abl_deriv_rho_ij.set(i, j, p_bar * h_prime * ds_drho_ij);
                }
            }
        }
    }
    let lap = CausalLaplacian::from_directed_weights(&causal_weights);
    let fiedler = lap.compute_fiedler(cfg.max_krylov, cfg.max_inv_power);
    let lambda2 = fiedler.lambda2;
    let total_probes = probes.probe_count();
    let mut v_stab = 0.0;
    let mut grad_r = if compute_grad {
        Some(RelTensor::zeros(n, d))
    } else {
        None
    };
    let mut coeff_on_grad_rho = -cfg.alpha;
    if compute_grad {
        let g_w = CausalLaplacian::grad_directed_weights(&fiedler.v2);
        let grad_acc = grad_r.as_mut().unwrap();
        for i in 0..n {
            for j in 0..n {
                if i == j {
                    continue;
                }
                let gw = cfg.beta_iit * g_w.get(i, j);
                if gw <= 0.0 {
                    continue;
                }
                let dw_drho = abl_deriv_rho.get(i, j);
                let dw_drho_ij = abl_deriv_rho_ij.get(i, j);
                if dw_drho == 0.0 && dw_drho_ij == 0.0 {
                    continue;
                }
                coeff_on_grad_rho += gw * dw_drho;
                let mult = gw * dw_drho_ij;
                if mult.abs() > 0.0 {
                    let r_abl = r.with_ablated_edge(i, j);
                    let (_, mut g_abl) =
                        params.residual_and_grad_r_precomputed(&r_abl, effect, &heads, cfg.lambda_bi);
                    for c in 0..d {
                        g_abl.set(i, j, c, 0.0);
                    }
                    grad_acc.axpy(mult, &g_abl);
                }
            }
        }
    }
    if total_probes > 0 && cfg.beta > 0.0 {
        let inv_m = 1.0 / (total_probes as f64);
        for q in &probes.rotations {
            let r_rot = StabilityProbes::apply_rotation(q, r);
            let rho_rot =
                params.residual_only_precomputed(&r_rot, effect, &heads, cfg.lambda_bi);
            let (hv, hp) = smooth_hinge(rho_rot - rho, cfg.hinge_tau);
            v_stab += inv_m * hv;
            if compute_grad && hp > 0.0 {
                let scale = -cfg.beta * inv_m * hp;
                coeff_on_grad_rho -= scale;
                let (_, g_rot) = params.residual_and_grad_r_precomputed(
                    &r_rot,
                    effect,
                    &heads,
                    cfg.lambda_bi,
                );
                let g_back = StabilityProbes::adjoint_rotation(q, &g_rot);
                grad_r.as_mut().unwrap().axpy(scale, &g_back);
            }
        }
        if probes.abs_kappa > 0.0 {
            let r_abs = probes.apply_abstraction(r);
            let rho_abs =
                params.residual_only_precomputed(&r_abs, effect, &heads, cfg.lambda_bi);
            let (hv, hp) = smooth_hinge(rho_abs - rho, cfg.hinge_tau);
            v_stab += inv_m * hv;
            if compute_grad && hp > 0.0 {
                let scale = -cfg.beta * inv_m * hp;
                coeff_on_grad_rho -= scale;
                let (_, g_abs) = params.residual_and_grad_r_precomputed(
                    &r_abs,
                    effect,
                    &heads,
                    cfg.lambda_bi,
                );
                let g_back = probes.adjoint_abstraction(&g_abs);
                grad_r.as_mut().unwrap().axpy(scale, &g_back);
            }
        }
    }
    if let (Some(g_acc), Some(g_rho)) = (grad_r.as_mut(), grad_rho_opt.as_ref()) {
        g_acc.axpy(coeff_on_grad_rho, g_rho);
    }
    let phi_sharp = cfg.beta_iit * lambda2 - cfg.alpha * rho - cfg.beta * v_stab;
    let eval = PhiSharpEval {
        phi_sharp,
        lambda2,
        residual: rho,
        rho_background: rho_bg,
        v_stab,
        causal_weights,
        active_causal_edges,
        fiedler,
    };
    (eval, grad_r)
}
#[inline]
pub fn residual_weights(has_bigrams: bool, lambda_bi: f64) -> (f64, f64) {
    if !has_bigrams || lambda_bi <= 0.0 {
        (1.0, 0.0)
    } else {
        let denom = 1.0 + lambda_bi;
        (1.0 / denom, lambda_bi / denom)
    }
}
#[inline]
pub fn smooth_hinge(x: f64, tau: f64) -> (f64, f64) {
    if x <= 0.0 {
        (0.0, 0.0)
    } else if x < tau {
        (0.5 * x * x / tau, x / tau)
    } else {
        (x - 0.5 * tau, 1.0)
    }
}
#[inline]
pub fn smooth_cap(a: f64, b: f64, eps: f64) -> (f64, f64) {
    let (h, hp) = smooth_hinge(a - b, eps);
    (a - h, 1.0 - hp)
}
fn bilinear_cross_entropy(
    h_node: &[f64],
    u_mat: &[f64],
    u0: &[f64],
    target: &[f64],
    mass: f64,
    logits: &mut [f64],
) -> f64 {
    let k = u0.len();
    let h = h_node.len();
    let mut max_l = f64::NEG_INFINITY;
    for a in 0..k {
        let row = &u_mat[a * h..(a + 1) * h];
        let mut acc = u0[a];
        for r in 0..h {
            acc += row[r] * h_node[r];
        }
        logits[a] = acc;
        if acc > max_l {
            max_l = acc;
        }
    }
    let mut sum_exp = 0.0;
    for &l in logits.iter().take(k) {
        sum_exp += (l - max_l).exp();
    }
    let lse = max_l + sum_exp.ln();
    let mut dot_qt = 0.0;
    for a in 0..k {
        if target[a] > 0.0 {
            dot_qt += target[a] * logits[a];
        }
    }
    mass * lse - dot_qt
}
#[allow(clippy::too_many_arguments)]
fn bilinear_cross_entropy_and_grad_h(
    h_node: &[f64],
    u_mat: &[f64],
    u0: &[f64],
    target: &[f64],
    mass: f64,
    loss_weight: f64,
    probs: &mut [f64],
    grad_h_out: &mut [f64],
) -> f64 {
    let k = u0.len();
    let h = h_node.len();
    let mut max_l = f64::NEG_INFINITY;
    for a in 0..k {
        let row = &u_mat[a * h..(a + 1) * h];
        let mut acc = u0[a];
        for r in 0..h {
            acc += row[r] * h_node[r];
        }
        probs[a] = acc;
        if acc > max_l {
            max_l = acc;
        }
    }
    let mut sum_exp = 0.0;
    let mut dot_qt = 0.0;
    for a in 0..k {
        let l = probs[a];
        if target[a] > 0.0 {
            dot_qt += target[a] * l;
        }
        let ex = (l - max_l).exp();
        probs[a] = ex;
        sum_exp += ex;
    }
    let lse = max_l + sum_exp.ln();
    let inv_sum = 1.0 / sum_exp;
    for a in 0..k {
        let p_a = probs[a] * inv_sum;
        let dl = loss_weight * (mass * p_a - target[a]);
        if dl == 0.0 {
            continue;
        }
        let row = &u_mat[a * h..(a + 1) * h];
        for r in 0..h {
            grad_h_out[r] += dl * row[r];
        }
    }
    mass * lse - dot_qt
}

struct ProjectionField {
    out_len: usize,
    k_uni: usize,
    w_uni: f64,
    w_bi: f64,
    linear_pot: Vec<f64>,
    obs_pairs: Vec<(usize, usize)>,
    bi_pot: Vec<f64>,
    copy_ref_alpha: Vec<usize>,
    gamma_anticopy: f64,
    tau_sharp: f64,
}
impl ProjectionField {
    fn build(
        params: &ModelParams,
        r_star: &RelTensor,
        eval_star: &PhiSharpEval,
        effect: &EffectSummary,
        perceived_eff: &[u8],
        cfg: &ProjectionConfig,
    ) -> Self {
        let n = params.n;
        let d = params.d;
        let h = params.h;
        let k_uni = effect.k_uni;
        let out_len = cfg.out_len.max(1);
        let inv_sqrt_n = 1.0 / (n as f64).sqrt();
        let comp: Vec<Mat> = (0..d)
            .map(|c| r_star.channels[c].matmul(&r_star.channels[c]))
            .collect();
        let mut h_nodes = vec![0.0; n * h];
        for i in 0..n {
            for j in 0..n {
                for r_idx in 0..h {
                    let mut z = params.b_edge[r_idx];
                    let w_d = &params.w_dir[r_idx * d..(r_idx + 1) * d];
                    let w_c = &params.w_comp[r_idx * d..(r_idx + 1) * d];
                    for c in 0..d {
                        z += w_d[c] * r_star.get(i, j, c) + w_c[c] * comp[c].get(i, j);
                    }
                    h_nodes[i * h + r_idx] += z.tanh();
                }
            }
            for r_idx in 0..h {
                h_nodes[i * h + r_idx] *= inv_sqrt_n;
            }
        }
        let heads = params.precompute_heads(effect);
        let mut log_p_uni = vec![0.0; n * k_uni];
        for i in 0..n {
            let hi = &h_nodes[i * h..(i + 1) * h];
            compute_bilinear_log_softmax(
                hi,
                &heads.u_uni,
                &heads.u0_uni,
                &mut log_p_uni[i * k_uni..(i + 1) * k_uni],
            );
        }
        let has_bi = out_len >= 2 && cfg.lambda_bi > 0.0 && perceived_eff.len() >= 2;
        let (obs_pairs, bi_node_scores) = if has_bi {
            build_observed_transition_scores(params, &h_nodes, effect, perceived_eff)
        } else {
            (Vec::new(), Vec::new())
        };
        let num_obs = obs_pairs.len();
        let active_bi = has_bi && num_obs > 0;
        let mut causal_deg = vec![0.0; n];
        for i in 0..n {
            let mut sum = 0.0;
            for j in 0..n {
                if i != j {
                    sum += 0.5
                        * (eval_star.causal_weights.get(i, j)
                            + eval_star.causal_weights.get(j, i));
                }
            }
            causal_deg[i] = sum;
        }
        let max_deg = causal_deg
            .iter()
            .copied()
            .fold(0.0f64, f64::max)
            .max(1e-12);
        let mut node_imp = vec![1.0; n];
        for i in 0..n {
            let deg_norm = causal_deg[i] / max_deg;
            let fiedler_e = eval_star
                .fiedler
                .v2
                .get(i)
                .map_or(0.0, |v| v * v * (n as f64));
            node_imp[i] = 1.0 + deg_norm + 0.5 * fiedler_e;
        }
        let mut counts = vec![0usize; k_uni];
        for &b in perceived_eff {
            if let Some(a) = effect.byte_to_alpha[b as usize] {
                counts[a] += 1;
            }
        }
        let total = (perceived_eff.len().max(1)) as f64;
        let mut log_q_alph = vec![0.0; k_uni];
        for a in 0..k_uni {
            let p_unif = 0.7 / (k_uni as f64);
            let p_emp = 0.3 * (counts[a] as f64) / total;
            log_q_alph[a] = (p_unif + p_emp).max(1e-12).ln();
        }
        let (w_uni, w_bi) = if active_bi {
            let denom = 1.0 + cfg.lambda_bi;
            (1.0 / denom, cfg.lambda_bi / denom)
        } else {
            (1.0, 0.0)
        };
        let inv_l = 1.0 / (out_len as f64);
        let mut linear_pot = vec![0.0; out_len * k_uni];
        let mut bi_pot = if active_bi {
            vec![0.0; (out_len - 1) * num_obs]
        } else {
            Vec::new()
        };
        let mut assign = vec![0.0; n];
        for pos in 0..out_len {
            spatial_assignment(pos, out_len, n, &mut assign);
            let mut z_sum = 0.0;
            for i in 0..n {
                assign[i] *= node_imp[i];
                z_sum += assign[i];
            }
            let inv_z = 1.0 / z_sum.max(1e-12);
            for a in &mut assign {
                *a *= inv_z;
            }
            let lin_row = &mut linear_pot[pos * k_uni..(pos + 1) * k_uni];
            for a in 0..k_uni {
                let mut u_val = 0.0;
                for i in 0..n {
                    u_val += assign[i] * log_p_uni[i * k_uni + a];
                }
                lin_row[a] = inv_l * (w_uni * u_val + cfg.gamma_alphabet * log_q_alph[a]);
            }
            if active_bi && pos >= 1 {
                let bi_row = &mut bi_pot[(pos - 1) * num_obs..pos * num_obs];
                for e_idx in 0..num_obs {
                    let mut b_val = 0.0;
                    for i in 0..n {
                        b_val += assign[i] * bi_node_scores[i * num_obs + e_idx];
                    }
                    bi_row[e_idx] = b_val;
                }
            }
        }
        let copy_len = out_len.min(perceived_eff.len());
        let copy_ref_alpha: Vec<usize> = perceived_eff[..copy_len]
            .iter()
            .filter_map(|&b| effect.byte_to_alpha[b as usize])
            .collect();
        Self {
            out_len,
            k_uni,
            w_uni,
            w_bi,
            linear_pot,
            obs_pairs,
            bi_pot,
            copy_ref_alpha,
            gamma_anticopy: cfg.gamma_anticopy,
            tau_sharp: cfg.tau_sharp,
        }
    }
    fn eval_and_grad(&self, y: &[f64], grad_out: Option<&mut [f64]>) -> f64 {
        let l = self.out_len;
        let k = self.k_uni;
        let inv_l = 1.0 / (l as f64);
        let mut psi = 0.0;
        let mut grad = grad_out;
        if let Some(g) = grad.as_deref_mut() {
            g.fill(0.0);
        }
        for pos in 0..l {
            let off = pos * k;
            let y_row = &y[off..off + k];
            let c_row = &self.linear_pot[off..off + k];
            for a in 0..k {
                let ya = y_row[a];
                psi += ya * c_row[a] + 0.5 * self.tau_sharp * inv_l * ya * ya;
                if let Some(g) = grad.as_deref_mut() {
                    g[off + a] += c_row[a] + self.tau_sharp * inv_l * ya;
                }
            }
        }
        if self.gamma_anticopy > 0.0 {
            let coef = self.gamma_anticopy * inv_l;
            for (pos, &a) in self.copy_ref_alpha.iter().enumerate() {
                let idx = pos * k + a;
                let ya = y[idx];
                psi -= 0.5 * coef * ya * ya;
                if let Some(g) = grad.as_deref_mut() {
                    g[idx] -= coef * ya;
                }
            }
        }
        let num_obs = self.obs_pairs.len();
        if self.w_bi > 0.0 && l >= 2 && num_obs > 0 {
            let bi_scale = self.w_bi / ((l - 1) as f64);
            for pos in 1..l {
                let prev_off = (pos - 1) * k;
                let curr_off = pos * k;
                let b_row = &self.bi_pot[(pos - 1) * num_obs..pos * num_obs];
                for (e_idx, &(u, v)) in self.obs_pairs.iter().enumerate() {
                    let w = bi_scale * b_row[e_idx];
                    let yu = y[prev_off + u];
                    let yv = y[curr_off + v];
                    psi += yu * yv * w;
                    if let Some(g) = grad.as_deref_mut() {
                        g[prev_off + u] += yv * w;
                        g[curr_off + v] += yu * w;
                    }
                }
            }
        }
        psi
    }
}
fn build_observed_transition_scores(
    params: &ModelParams,
    h_nodes: &[f64],
    effect: &EffectSummary,
    bytes: &[u8],
) -> (Vec<(usize, usize)>, Vec<f64>) {
    let n = params.n;
    let h = params.h;
    let k_uni = effect.k_uni;
    let len = bytes.len();
    let ln_len1 = ((len as f64) + 1.0).ln().max(1e-12);
    let mut counts = vec![0usize; k_uni];
    let mut out_totals = vec![0usize; k_uni];
    let mut in_totals = vec![0usize; k_uni];
    let mut bi_counts: BTreeMap<(usize, usize), usize> = BTreeMap::new();
    let mut bi_pos_sum: BTreeMap<(usize, usize), f64> = BTreeMap::new();
    let mut bi_pos_sq: BTreeMap<(usize, usize), f64> = BTreeMap::new();
    for &b in bytes {
        counts[effect.byte_to_alpha[b as usize].unwrap()] += 1;
    }
    for pos in 1..len {
        let u = effect.byte_to_alpha[bytes[pos - 1] as usize].unwrap();
        let v = effect.byte_to_alpha[bytes[pos] as usize].unwrap();
        out_totals[u] += 1;
        in_totals[v] += 1;
        *bi_counts.entry((u, v)).or_insert(0) += 1;
        let x = ((pos as f64) + 0.5) / (len as f64);
        *bi_pos_sum.entry((u, v)).or_insert(0.0) += x;
        *bi_pos_sq.entry((u, v)).or_insert(0.0) += x * x;
    }
    let obs_pairs: Vec<(usize, usize)> = bi_counts.keys().copied().collect();
    let num_obs = obs_pairs.len();
    if num_obs == 0 {
        return (obs_pairs, Vec::new());
    }
    let mut u_obs = vec![0.0; num_obs * h];
    let mut u0_obs = vec![0.0; num_obs];
    for (idx, &(u, v)) in obs_pairs.iter().enumerate() {
        let c_uv = bi_counts[&(u, v)] as f64;
        let c_vu = bi_counts.get(&(v, u)).copied().unwrap_or(0) as f64;
        let c_u_tot = (counts[u] as f64).max(1.0);
        let c_v_tot = (counts[v] as f64).max(1.0);
        let c_u_out = (out_totals[u] as f64).max(1.0);
        let c_v_in = (in_totals[v] as f64).max(1.0);
        let g0 = (c_uv + 1.0).ln() / ln_len1;
        let pmi = ((c_uv * (len as f64)) / (c_u_tot * c_v_tot))
            .max(1e-12)
            .ln();
        let g1 = (pmi / 3.0).tanh();
        let g2 = c_uv / c_u_out;
        let g3 = c_uv / c_v_in;
        let g4 = c_vu / (c_uv + c_vu);
        let mean_p = bi_pos_sum[&(u, v)] / c_uv;
        let mean_sq = bi_pos_sq[&(u, v)] / c_uv;
        let std_p = (mean_sq - mean_p * mean_p).max(0.0).sqrt();
        let g7 = if u == v { 1.0 } else { 0.0 };
        let g8 = effect.rel_feats[v * REL_DIM + 3] - effect.rel_feats[u * REL_DIM + 3];
        let g9 = effect.rel_feats[v * REL_DIM] - effect.rel_feats[u * REL_DIM];
        let feat = [g0, g1, g2, g3, g4, mean_p, std_p, g7, g8, g9];
        let mut b0 = 0.0;
        for m in 0..BI_DIM {
            b0 += params.b_rel_bi[m] * feat[m];
        }
        u0_obs[idx] = b0;
        for r in 0..h {
            let w_row = &params.w_rel_bi[r * BI_DIM..(r + 1) * BI_DIM];
            let mut acc = 0.0;
            for m in 0..BI_DIM {
                acc += w_row[m] * feat[m];
            }
            u_obs[idx * h + r] = acc;
        }
    }
    let mut node_scores = vec![0.0; n * num_obs];
    for i in 0..n {
        let hi = &h_nodes[i * h..(i + 1) * h];
        let row = &mut node_scores[i * num_obs..(i + 1) * num_obs];
        compute_bilinear_log_softmax(hi, &u_obs, &u0_obs, row);
        let min_lp = row.iter().copied().fold(f64::INFINITY, f64::min);
        for v in row.iter_mut() {
            *v = (*v - min_lp) + 0.5;
        }
    }
    (obs_pairs, node_scores)
}
pub fn project_witness_mirror_ascent(
    params: &ModelParams,
    r_star: &RelTensor,
    eval_star: &PhiSharpEval,
    perceived_eff: &[u8],
    cfg: &ProjectionConfig,
) -> ProjectionResult {
    let l = cfg.out_len.max(1);
    let effect = EffectSummary::from_bytes(perceived_eff, params.n);
    let k = effect.k_uni;
    if k == 0 {
        return ProjectionResult {
            alphabet: Vec::new(),
            k_uni: 0,
            soft_bytes: Vec::new(),
            witness_bytes: Vec::new(),
            mean_confidence: 0.0,
            final_psi: 0.0,
            trace: Vec::new(),
            converged: true,
        };
    }
    let field = ProjectionField::build(params, r_star, eval_star, &effect, perceived_eff, cfg);
    let mut y = vec![1.0 / (k as f64); l * k];
    let mut y_cand = vec![0.0; l * k];
    let mut grad = vec![0.0; l * k];
    let mut psi = field.eval_and_grad(&y, Some(&mut grad));
    let mut eta = cfg.eta_init;
    let mut converged = false;
    let mut trace = Vec::with_capacity(cfg.max_steps + 1);
    trace.push(ProjectionStepTrace {
        step: 0,
        psi,
        eta: 0.0,
        kl_old_new: 0.0,
        step_l12: 0.0,
    });
    for step in 1..=cfg.max_steps {
        let mut trial_eta = eta;
        let mut accepted = None;
        while trial_eta >= cfg.eta_min {
            mirror_step(&y, &grad, l, k, trial_eta, &mut y_cand);
            let cand_psi = field.eval_and_grad(&y_cand, None);
            let kl_old_new = kl_divergence_simplex(&y, &y_cand);
            if cand_psi + 1e-12 >= psi + kl_old_new / trial_eta {
                let l12 = norm_l12_diff(&y, &y_cand, l, k);
                accepted = Some((cand_psi, kl_old_new, l12, trial_eta));
                break;
            }
            trial_eta *= 0.5;
        }
        if let Some((next_psi, kl_old_new, step_l12, used_eta)) = accepted {
            y.copy_from_slice(&y_cand);
            psi = field.eval_and_grad(&y, Some(&mut grad));
            if psi < next_psi {
                psi = next_psi;
            }
            trace.push(ProjectionStepTrace {
                step,
                psi,
                eta: used_eta,
                kl_old_new,
                step_l12,
            });
            if step_l12 <= cfg.tol_step_l12 {
                converged = true;
                break;
            }
            eta = (used_eta * 1.15).min(cfg.eta_max);
        } else {
            converged = true;
            break;
        }
    }
    let mut witness_bytes = Vec::with_capacity(l);
    let mut conf_sum = 0.0;
    for pos in 0..l {
        let row = &y[pos * k..(pos + 1) * k];
        let mut best_a = 0usize;
        let mut best_p = -1.0f64;
        for (a, &p) in row.iter().enumerate() {
            if p > best_p {
                best_p = p;
                best_a = a;
            }
        }
        witness_bytes.push(effect.alphabet[best_a]);
        conf_sum += best_p;
    }
    ProjectionResult {
        alphabet: effect.alphabet,
        k_uni: k,
        soft_bytes: y,
        witness_bytes,
        mean_confidence: conf_sum / (l as f64),
        final_psi: psi,
        trace,
        converged,
    }
}
fn mirror_step(y: &[f64], grad: &[f64], out_len: usize, k: usize, eta: f64, out: &mut [f64]) {
    const FLOOR: f64 = 1e-300;
    for pos in 0..out_len {
        let off = pos * k;
        let mut max_log = f64::NEG_INFINITY;
        for a in 0..k {
            let v = y[off + a].max(FLOOR).ln() + eta * grad[off + a];
            out[off + a] = v;
            if v > max_log {
                max_log = v;
            }
        }
        let mut sum = 0.0;
        for a in 0..k {
            let ex = (out[off + a] - max_log).exp();
            out[off + a] = ex;
            sum += ex;
        }
        let inv_sum = 1.0 / sum.max(FLOOR);
        for a in 0..k {
            out[off + a] *= inv_sum;
        }
    }
}
pub fn kl_divergence_simplex(p: &[f64], q: &[f64]) -> f64 {
    const FLOOR: f64 = 1e-300;
    let mut kl = 0.0;
    for (&pi, &qi) in p.iter().zip(q.iter()) {
        if pi > FLOOR {
            kl += pi * (pi.ln() - qi.max(FLOOR).ln());
        }
    }
    kl.max(0.0)
}
pub fn norm_l12_diff(a: &[f64], b: &[f64], out_len: usize, k: usize) -> f64 {
    let mut sum_sq = 0.0;
    for pos in 0..out_len {
        let off = pos * k;
        let mut l1 = 0.0;
        for j in 0..k {
            l1 += (a[off + j] - b[off + j]).abs();
        }
        sum_sq += l1 * l1;
    }
    sum_sq.sqrt()
}
fn compute_bilinear_log_softmax(h_node: &[f64], u_mat: &[f64], u0: &[f64], out: &mut [f64]) {
    let k = u0.len();
    let h = h_node.len();
    let mut max_l = f64::NEG_INFINITY;
    for a in 0..k {
        let row = &u_mat[a * h..(a + 1) * h];
        let mut acc = u0[a];
        for r in 0..h {
            acc += row[r] * h_node[r];
        }
        out[a] = acc;
        if acc > max_l {
            max_l = acc;
        }
    }
    let mut sum_exp = 0.0;
    for &v in out.iter().take(k) {
        sum_exp += (v - max_l).exp();
    }
    let lse = max_l + sum_exp.ln();
    for v in out.iter_mut().take(k) {
        *v -= lse;
    }
}

#[derive(Debug, Clone)]
pub struct CausalLaplacian {
    pub n: usize,
    pub deg: Vec<f64>,
    pub w_sym: Mat,
}
impl CausalLaplacian {
    pub fn from_directed_weights(w_dir: &Mat) -> Self {
        let n = w_dir.n;
        let mut w_sym = Mat::zeros(n);
        let mut deg = vec![0.0; n];
        for i in 0..n {
            for j in (i + 1)..n {
                let wij = w_dir.get(i, j).max(0.0);
                let wji = w_dir.get(j, i).max(0.0);
                let bar = 0.5 * (wij + wji);
                if bar > 0.0 {
                    w_sym.set(i, j, bar);
                    w_sym.set(j, i, bar);
                    deg[i] += bar;
                    deg[j] += bar;
                }
            }
        }
        Self { n, deg, w_sym }
    }
    pub fn apply(&self, x: &[f64]) -> Vec<f64> {
        let n = self.n;
        debug_assert_eq!(n, x.len());
        let mut y = vec![0.0; n];
        for i in 0..n {
            let mut acc = self.deg[i] * x[i];
            let row = i * n;
            for j in 0..n {
                let w = self.w_sym.data[row + j];
                if w != 0.0 {
                    acc -= w * x[j];
                }
            }
            y[i] = acc;
        }
        y
    }
    pub fn compute_fiedler(&self, max_krylov: usize, max_inv_power: usize) -> FiedlerResult {
        let n = self.n;
        if n <= 1 {
            return FiedlerResult {
                lambda2: 0.0,
                lambda3: 0.0,
                v2: vec![0.0; n],
                residual_norm: 0.0,
                lanczos_iters: 0,
                inverse_power_iters: 0,
            };
        }
        if n == 2 {
            let w01 = self.w_sym.get(0, 1);
            let s = std::f64::consts::FRAC_1_SQRT_2;
            return FiedlerResult {
                lambda2: 2.0 * w01,
                lambda3: 2.0 * w01,
                v2: vec![s, -s],
                residual_norm: 0.0,
                lanczos_iters: 1,
                inverse_power_iters: 0,
            };
        }
        if self.deg.iter().all(|&d| d <= 1e-15) {
            let mut v = deterministic_unit_perp_ones(n, 0xA5A5_1234_5678_90EF);
            normalize(&mut v);
            return FiedlerResult {
                lambda2: 0.0,
                lambda3: 0.0,
                v2: v,
                residual_norm: 0.0,
                lanczos_iters: 0,
                inverse_power_iters: 0,
            };
        }
        let m_max = max_krylov.clamp(2, n - 1);
        let mut basis: Vec<Vec<f64>> = Vec::with_capacity(m_max);
        let mut alpha: Vec<f64> = Vec::with_capacity(m_max);
        let mut beta: Vec<f64> = Vec::with_capacity(m_max);
        let mut seed_state: u64 = 0x9E37_79B9_7F4A_7C15;
        let q0 = deterministic_unit_perp_ones(n, next_seed(&mut seed_state));
        basis.push(q0);
        for j in 0..m_max {
            let mut w = self.apply(&basis[j]);
            project_perp_ones(&mut w);
            if j > 0 && beta[j - 1] > 0.0 {
                let b_prev = beta[j - 1];
                let q_prev = &basis[j - 1];
                for i in 0..n {
                    w[i] -= b_prev * q_prev[i];
                }
            }
            let a_j = dot(&basis[j], &w);
            alpha.push(a_j);
            for i in 0..n {
                w[i] -= a_j * basis[j][i];
            }
            reorthogonalize(&mut w, &basis);
            let mut b_j = norm2(&w);
            if j + 1 < m_max {
                if b_j < 1e-11 {
                    if let Some(fresh) = fresh_orthogonal_vector(n, &basis, &mut seed_state) {
                        beta.push(0.0);
                        basis.push(fresh);
                    } else {
                        break;
                    }
                } else {
                    for x in &mut w {
                        *x /= b_j;
                    }
                    beta.push(b_j);
                    basis.push(w);
                }
            }
        }
        let m = basis.len();
        let mut t_mat = Mat::zeros(m);
        for i in 0..m {
            t_mat.set(i, i, alpha[i]);
            if i + 1 < m && i < beta.len() {
                t_mat.set(i, i + 1, beta[i]);
                t_mat.set(i + 1, i, beta[i]);
            }
        }
        let (ritz_vals, ritz_vecs) = t_mat.jacobi_eigen_sym(60, 1e-14);
        let mut lambda2 = ritz_vals[0].max(0.0);
        let lambda3 = if m >= 2 {
            ritz_vals[1].max(0.0)
        } else {
            lambda2
        };
        let mut v2 = vec![0.0; n];
        for col in 0..m {
            let coeff = ritz_vecs.get(col, 0);
            for i in 0..n {
                v2[i] += coeff * basis[col][i];
            }
        }
        project_perp_ones(&mut v2);
        normalize(&mut v2);
        let mut res = self.eigen_residual(&v2, lambda2);
        let mut inv_iters = 0usize;
        if lambda2 > 1e-9 && res > 1e-11 && max_inv_power > 0 {
            for _ in 0..max_inv_power {
                if res <= 1e-11 {
                    break;
                }
                if let Some(mut z) = self.solve_cg_perp_ones(&v2, 2 * n, 1e-12) {
                    project_perp_ones(&mut z);
                    if normalize(&mut z) > 1e-15 {
                        let lz = self.apply(&z);
                        let new_lam = dot(&z, &lz).max(0.0);
                        let new_res = residual_from_lv(&z, &lz, new_lam);
                        if new_lam <= lambda2 + 1e-12 && new_res <= res + 1e-12 {
                            v2 = z;
                            lambda2 = new_lam;
                            res = new_res;
                            inv_iters += 1;
                            continue;
                        }
                    }
                }
                break;
            }
        }
        if let Some(&first_sig) = v2.iter().find(|x| x.abs() > 1e-9) {
            if first_sig < 0.0 {
                for x in &mut v2 {
                    *x = -*x;
                }
            }
        }
        FiedlerResult {
            lambda2,
            lambda3,
            v2,
            residual_norm: res,
            lanczos_iters: m,
            inverse_power_iters: inv_iters,
        }
    }
    pub fn grad_directed_weights(v2: &[f64]) -> Mat {
        let n = v2.len();
        let mut grad = Mat::zeros(n);
        for i in 0..n {
            for j in 0..n {
                if i != j {
                    let diff = v2[i] - v2[j];
                    grad.set(i, j, 0.5 * diff * diff);
                }
            }
        }
        grad
    }
    fn eigen_residual(&self, v: &[f64], lam: f64) -> f64 {
        let lv = self.apply(v);
        residual_from_lv(v, &lv, lam)
    }
    fn solve_cg_perp_ones(&self, b: &[f64], max_iter: usize, tol: f64) -> Option<Vec<f64>> {
        let n = self.n;
        let mut x = vec![0.0; n];
        let mut r = b.to_vec();
        project_perp_ones(&mut r);
        let mut p = r.clone();
        let mut rs_old = dot(&r, &r);
        let b_norm = rs_old.sqrt();
        if b_norm <= 1e-15 {
            return None;
        }
        for _ in 0..max_iter {
            let mut ap = self.apply(&p);
            project_perp_ones(&mut ap);
            let denom = dot(&p, &ap);
            if denom <= 1e-15 {
                return None;
            }
            let alpha = rs_old / denom;
            for i in 0..n {
                x[i] += alpha * p[i];
                r[i] -= alpha * ap[i];
            }
            project_perp_ones(&mut r);
            let rs_new = dot(&r, &r);
            if rs_new.sqrt() <= tol * b_norm {
                return Some(x);
            }
            let beta = rs_new / rs_old;
            for i in 0..n {
                p[i] = r[i] + beta * p[i];
            }
            project_perp_ones(&mut p);
            rs_old = rs_new;
        }
        Some(x)
    }
}
fn dot(a: &[f64], b: &[f64]) -> f64 {
    a.iter().zip(b.iter()).map(|(x, y)| x * y).sum()
}
fn norm2(a: &[f64]) -> f64 {
    dot(a, a).sqrt()
}
fn normalize(a: &mut [f64]) -> f64 {
    let n = norm2(a);
    if n > 0.0 {
        for x in a.iter_mut() {
            *x /= n;
        }
    }
    n
}
fn project_perp_ones(v: &mut [f64]) {
    let mean = v.iter().sum::<f64>() / (v.len() as f64);
    for x in v.iter_mut() {
        *x -= mean;
    }
}
fn residual_from_lv(v: &[f64], lv: &[f64], lam: f64) -> f64 {
    let mut sq = 0.0;
    for i in 0..v.len() {
        let d = lv[i] - lam * v[i];
        sq += d * d;
    }
    sq.sqrt()
}
fn reorthogonalize(w: &mut [f64], basis: &[Vec<f64>]) {
    for _ in 0..2 {
        project_perp_ones(w);
        for q in basis {
            let proj = dot(q, w);
            for i in 0..w.len() {
                w[i] -= proj * q[i];
            }
        }
    }
}
fn next_seed(state: &mut u64) -> u64 {
    *state = state.wrapping_add(0x9E37_79B9_7F4A_7C15);
    let mut z = *state;
    z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
    z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
    z ^ (z >> 31)
}
fn deterministic_unit_perp_ones(n: usize, mut seed: u64) -> Vec<f64> {
    let mut v = Vec::with_capacity(n);
    for _ in 0..n {
        let r = next_seed(&mut seed);
        let f = ((r >> 11) as f64) / ((1u64 << 53) as f64) * 2.0 - 1.0;
        v.push(f);
    }
    project_perp_ones(&mut v);
    normalize(&mut v);
    v
}
fn fresh_orthogonal_vector(n: usize, basis: &[Vec<f64>], seed_state: &mut u64) -> Option<Vec<f64>> {
    for _ in 0..8 {
        let mut cand = deterministic_unit_perp_ones(n, next_seed(seed_state));
        reorthogonalize(&mut cand, basis);
        if normalize(&mut cand) > 1e-10 {
            return Some(cand);
        }
    }
    None
}

#[derive(Debug, Clone, PartialEq)]
pub struct Mat {

    pub data: Vec<f64>,
}
impl Mat {
    pub fn zeros(n: usize) -> Self {
        Self {
            n,
            data: vec![0.0; n * n],
        }
    }
    pub fn identity(n: usize) -> Self {
        let mut m = Self::zeros(n);
        for i in 0..n {
            m.data[i * n + i] = 1.0;
        }
        m
    }
    #[inline]
    pub fn get(&self, i: usize, j: usize) -> f64 {
        self.data[i * self.n + j]
    }
    #[inline]
    pub fn set(&mut self, i: usize, j: usize, v: f64) {
        self.data[i * self.n + j] = v;
    }
    #[inline]
    pub fn add_assign(&mut self, i: usize, j: usize, v: f64) {
        self.data[i * self.n + j] += v;
    }
    pub fn frob_dot(&self, other: &Mat) -> f64 {
        debug_assert_eq!(self.n, other.n);
        self.data
            .iter()
            .zip(other.data.iter())
            .map(|(a, b)| a * b)
            .sum()
    }
    pub fn frob_norm_sq(&self) -> f64 {
        self.frob_dot(self)
    }
    pub fn frob_norm(&self) -> f64 {
        self.frob_norm_sq().sqrt()
    }
    pub fn scale(&mut self, s: f64) {
        for v in &mut self.data {
            *v *= s;
        }
    }
    pub fn axpy(&mut self, alpha: f64, other: &Mat) {
        debug_assert_eq!(self.n, other.n);
        for (a, b) in self.data.iter_mut().zip(other.data.iter()) {
            *a += alpha * *b;
        }
    }
    pub fn skew(&self) -> Mat {
        let n = self.n;
        let mut out = Mat::zeros(n);
        for i in 0..n {
            for j in (i + 1)..n {
                let v = 0.5 * (self.get(i, j) - self.get(j, i));
                out.set(i, j, v);
                out.set(j, i, -v);
            }
        }
        out
    }
    pub fn matmul(&self, rhs: &Mat) -> Mat {
        let n = self.n;
        debug_assert_eq!(n, rhs.n);
        let mut out = Mat::zeros(n);
        for i in 0..n {
            let row_a = i * n;
            let row_c = i * n;
            for k in 0..n {
                let aik = self.data[row_a + k];
                if aik == 0.0 {
                    continue;
                }
                let row_b = k * n;
                for j in 0..n {
                    out.data[row_c + j] += aik * rhs.data[row_b + j];
                }
            }
        }
        out
    }
    pub fn matmul_t_left(&self, rhs: &Mat) -> Mat {
        let n = self.n;
        debug_assert_eq!(n, rhs.n);
        let mut out = Mat::zeros(n);
        for k in 0..n {
            let row_a = k * n;
            let row_b = k * n;
            for i in 0..n {
                let aki = self.data[row_a + i];
                if aki == 0.0 {
                    continue;
                }
                let row_c = i * n;
                for j in 0..n {
                    out.data[row_c + j] += aki * rhs.data[row_b + j];
                }
            }
        }
        out
    }
    pub fn matmul_t_right(&self, rhs: &Mat) -> Mat {
        let n = self.n;
        debug_assert_eq!(n, rhs.n);
        let mut out = Mat::zeros(n);
        for i in 0..n {
            let row_a = i * n;
            for j in 0..n {
                let row_b = j * n;
                let mut acc = 0.0;
                for k in 0..n {
                    acc += self.data[row_a + k] * rhs.data[row_b + k];
                }
                out.data[i * n + j] = acc;
            }
        }
        out
    }

    }
    pub fn jacobi_eigen_sym(&self, max_sweeps: usize, tol: f64) -> (Vec<f64>, Mat) {
        let n = self.n;
        if n == 0 {
            return (Vec::new(), Mat::zeros(0));
        }
        if n == 1 {
            return (vec![self.data[0]], Mat::identity(1));
        }
        let mut a = Mat::zeros(n);
        for i in 0..n {
            a.set(i, i, self.get(i, i));
            for j in (i + 1)..n {
                let sym = 0.5 * (self.get(i, j) + self.get(j, i));
                a.set(i, j, sym);
                a.set(j, i, sym);
            }
        }
        let mut v = Mat::identity(n);
        let max_iters = max_sweeps * n * n;
        for _ in 0..max_iters {
            let mut p = 0usize;
            let mut q = 1usize;
            let mut max_off = 0.0f64;
            for i in 0..n {
                let row = i * n;
                for j in (i + 1)..n {
                    let val = a.data[row + j].abs();
                    if val > max_off {
                        max_off = val;
                        p = i;
                        q = j;
                    }
                }
            }
            if max_off <= tol {
                break;
            }
            let apq = a.get(p, q);
            let app = a.get(p, p);
            let aqq = a.get(q, q);
            let tau = (aqq - app) / (2.0 * apq);
            let t = if tau >= 0.0 {
                1.0 / (tau + (1.0 + tau * tau).sqrt())
            } else {
                -1.0 / (-tau + (1.0 + tau * tau).sqrt())
            };
            let c = 1.0 / (1.0 + t * t).sqrt();
            let s = t * c;
            a.set(p, p, app - t * apq);
            a.set(q, q, aqq + t * apq);
            a.set(p, q, 0.0);
            a.set(q, p, 0.0);
            for k in 0..n {
                if k != p && k != q {
                    let akp = a.get(k, p);
                    let akq = a.get(k, q);
                    let new_kp = c * akp - s * akq;
                    let new_kq = s * akp + c * akq;
                    a.set(k, p, new_kp);
                    a.set(p, k, new_kp);
                    a.set(k, q, new_kq);
                    a.set(q, k, new_kq);
                }
            }
            for k in 0..n {
                let vkp = v.get(k, p);
                let vkq = v.get(k, q);
                v.set(k, p, c * vkp - s * vkq);
                v.set(k, q, s * vkp + c * vkq);
            }
        }
        let mut order: Vec<usize> = (0..n).collect();
        order.sort_by(|&i, &j| {
            a.get(i, i)
                .partial_cmp(&a.get(j, j))
                .unwrap_or(std::cmp::Ordering::Equal)
        });
        let evals: Vec<f64> = order.iter().map(|&idx| a.get(idx, idx)).collect();
        let mut sorted_v = Mat::zeros(n);
        for (new_col, &old_col) in order.iter().enumerate() {
            for row in 0..n {
                sorted_v.set(row, new_col, v.get(row, old_col));
            }
        }
        (evals, sorted_v)
    }
}

        }
    }
    pub fn clamp_to_ball(&mut self, radius_k: f64) {
        let norm = self.frob_norm();
        if norm > radius_k && norm > 0.0 {
            self.scale(radius_k / norm);
        }
    }
    pub fn project_tangent_ball(&self, grad: &RelTensor, radius_k: f64) -> RelTensor {
        let norm_sq = self.frob_norm_sq();
        let k_sq = radius_k * radius_k;
        let mut out = grad.clone();
        if norm_sq >= k_sq * (1.0 - 1e-12) && norm_sq > 0.0 {
            let dot = grad.frob_dot(self);
            if dot > 0.0 {
                out.axpy(-dot / norm_sq, self);
            }
        }
        out
    }
    pub fn commutator_with(&self, a_skew: &Mat) -> RelTensor {
        debug_assert_eq!(self.n, a_skew.n);
        let mut out = RelTensor::zeros(self.n, self.d);
        for c in 0..self.d {
            let ar = a_skew.matmul(&self.channels[c]);
            let ra = self.channels[c].matmul(a_skew);
            for idx in 0..(self.n * self.n) {
                out.channels[c].data[idx] = ar.data[idx] - ra.data[idx];
            }
        }
        out
    }
    pub fn skew_commutator_dual(&self, grad: &RelTensor) -> Mat {
        debug_assert_eq!(self.n, grad.n);
        debug_assert_eq!(self.d, grad.d);
        let mut acc = Mat::zeros(self.n);
        for c in 0..self.d {
            let grt = grad.channels[c].matmul_t_right(&self.channels[c]);
            let rtg = self.channels[c].matmul_t_left(&grad.channels[c]);
            for idx in 0..(self.n * self.n) {
                acc.data[idx] += grt.data[idx] - rtg.data[idx];
            }
        }
        acc.skew()
    }
    pub fn spectral_abstraction_reg(&self, eps: f64) -> (f64, RelTensor) {
        let eps_sq = eps * eps;
        let mut total_val = 0.0;
        let mut grad = RelTensor::zeros(self.n, self.d);
        for c in 0..self.d {
            let m = &self.channels[c];
            let s = m.matmul_t_left(m); 
            let (evals, v) = s.jacobi_eigen_sym(50, 1e-13);
            let mut nuc_smooth = 0.0;
            let mut inv_sqrt = vec![0.0; self.n];
            for i in 0..self.n {
                let sq = (evals[i].max(0.0) + eps_sq).sqrt();
                nuc_smooth += sq;
                inv_sqrt[i] = 1.0 / sq;
            }
            let mut inv_mat = Mat::zeros(self.n);
            for i in 0..self.n {
                for j in i..self.n {
                    let mut sum = 0.0;
                    for k in 0..self.n {
                        sum += v.get(i, k) * inv_sqrt[k] * v.get(j, k);
                    }
                    inv_mat.set(i, j, sum);
                    inv_mat.set(j, i, sum);
                }
            }
            let grad_nuc = m.matmul(&inv_mat);
            let frob_smooth = (m.frob_norm_sq() + eps_sq).sqrt();
            total_val += nuc_smooth - frob_smooth;
            let inv_frob = 1.0 / frob_smooth;
            for idx in 0..(self.n * self.n) {
                grad.channels[c].data[idx] = grad_nuc.data[idx] - m.data[idx] * inv_frob;
            }
        }
        (total_val, grad)
    }

        let mut copy = self.clone();
        for c in 0..self.d {
            copy.set(i, j, c, 0.0);
        }

    }

    pub w_init_ov: Vec<f64>,
    pub w_sol: Mat,
    pub nu_sol: Vec<f64>,
}
impl ParamGrads {
    pub fn zeros_like(p: &ModelParams) -> Self {
        Self {
            w_dir: vec![0.0; p.w_dir.len()],
            w_comp: vec![0.0; p.w_comp.len()],
            b_edge: vec![0.0; p.b_edge.len()],
            w_rel_uni: vec![0.0; p.w_rel_uni.len()],
            b_rel_uni: vec![0.0; p.b_rel_uni.len()],
            w_rel_bi: vec![0.0; p.w_rel_bi.len()],
            b_rel_bi: vec![0.0; p.b_rel_bi.len()],
            w_enc_uni: vec![0.0; p.w_enc_uni.len()],
            w_enc_bi: vec![0.0; p.w_enc_bi.len()],
            w_init_asym: vec![0.0; p.w_init_asym.len()],
            w_init_ov: vec![0.0; p.w_init_ov.len()],
            w_sol: Mat::zeros(p.n),
            nu_sol: vec![0.0; p.nu_sol.len()],
        }
    }
    pub fn l2_norm(&self) -> f64 {
        let mut s = 0.0;
        for slice in [
            self.w_dir.as_slice(),
            self.w_comp.as_slice(),
            self.b_edge.as_slice(),
            self.w_rel_uni.as_slice(),
            self.b_rel_uni.as_slice(),
            self.w_rel_bi.as_slice(),
            self.b_rel_bi.as_slice(),
            self.w_enc_uni.as_slice(),
            self.w_enc_bi.as_slice(),
            self.w_init_asym.as_slice(),
            self.w_init_ov.as_slice(),
            self.w_sol.data.as_slice(),
            self.nu_sol.as_slice(),
        ] {
            for &v in slice {
                s += v * v;
            }
        }
        s.sqrt()
    }
    pub fn clip_norm(&mut self, max_norm: f64) -> f64 {
        let norm = self.l2_norm();
        if norm > max_norm && norm > 0.0 {
            let scale = max_norm / norm;
            for slice in [
                self.w_dir.as_mut_slice(),
                self.w_comp.as_mut_slice(),
                self.b_edge.as_mut_slice(),
                self.w_rel_uni.as_mut_slice(),
                self.b_rel_uni.as_mut_slice(),
                self.w_rel_bi.as_mut_slice(),
                self.b_rel_bi.as_mut_slice(),
                self.w_enc_uni.as_mut_slice(),
                self.w_enc_bi.as_mut_slice(),
                self.w_init_asym.as_mut_slice(),
                self.w_init_ov.as_mut_slice(),
                self.w_sol.data.as_mut_slice(),
                self.nu_sol.as_mut_slice(),
            ] {
                for v in slice {
                    *v *= scale;
                }
            }
        }
        norm
    }
}
struct AdamWState {
    step: usize,
    m: ParamGrads,
    v: ParamGrads,
}
impl AdamWState {
    fn new(p: &ModelParams) -> Self {
        Self {
            step: 0,
            m: ParamGrads::zeros_like(p),
            v: ParamGrads::zeros_like(p),
        }
    }
    fn step_ascent(
        &mut self,
        params: &mut ModelParams,
        grads: &ParamGrads,
        lr: f64,
        weight_decay: f64,
    ) {
        self.step += 1;
        let b1: f64 = 0.9;
        let b2: f64 = 0.999;
        let eps = 1e-8;
        let bc1 = 1.0 - b1.powi(self.step as i32);
        let bc2 = 1.0 - b2.powi(self.step as i32);
        let update = |p: &mut [f64], g: &[f64], m: &mut [f64], v: &mut [f64], wd: f64| {
            for i in 0..p.len() {
                let gi = g[i];
                m[i] = b1 * m[i] + (1.0 - b1) * gi;
                v[i] = b2 * v[i] + (1.0 - b2) * gi * gi;
                let m_hat = m[i] / bc1;
                let v_hat = v[i] / bc2;
                p[i] = p[i] * (1.0 - lr * wd) + lr * m_hat / (v_hat.sqrt() + eps);
            }
        };
        update(&mut params.w_dir, &grads.w_dir, &mut self.m.w_dir, &mut self.v.w_dir, weight_decay);
        update(&mut params.w_comp, &grads.w_comp, &mut self.m.w_comp, &mut self.v.w_comp, weight_decay);
        update(&mut params.b_edge, &grads.b_edge, &mut self.m.b_edge, &mut self.v.b_edge, 0.0);
        update(&mut params.w_rel_uni, &grads.w_rel_uni, &mut self.m.w_rel_uni, &mut self.v.w_rel_uni, weight_decay);
        update(&mut params.b_rel_uni, &grads.b_rel_uni, &mut self.m.b_rel_uni, &mut self.v.b_rel_uni, 0.0);
        update(&mut params.w_rel_bi, &grads.w_rel_bi, &mut self.m.w_rel_bi, &mut self.v.w_rel_bi, weight_decay);
        update(&mut params.b_rel_bi, &grads.b_rel_bi, &mut self.m.b_rel_bi, &mut self.v.b_rel_bi, 0.0);
        update(&mut params.w_enc_uni, &grads.w_enc_uni, &mut self.m.w_enc_uni, &mut self.v.w_enc_uni, weight_decay);
        update(&mut params.w_enc_bi, &grads.w_enc_bi, &mut self.m.w_enc_bi, &mut self.v.w_enc_bi, weight_decay);
        update(&mut params.w_init_asym, &grads.w_init_asym, &mut self.m.w_init_asym, &mut self.v.w_init_asym, weight_decay);
        update(&mut params.w_init_ov, &grads.w_init_ov, &mut self.m.w_init_ov, &mut self.v.w_init_ov, weight_decay);
        update(&mut params.w_sol.data, &grads.w_sol.data, &mut self.m.w_sol.data, &mut self.v.w_sol.data, weight_decay);
        update(&mut params.nu_sol, &grads.nu_sol, &mut self.m.nu_sol, &mut self.v.nu_sol, weight_decay);
    }
}
fn accumulate_residual_grad_theta(
    params: &ModelParams,
    r: &RelTensor,
    effect: &EffectSummary,
    heads: &PrecomputedHeads,
    lambda_bi: f64,
    scale: f64,
    grads: &mut ParamGrads,
) {
    if scale == 0.0 || effect.k_uni == 0 {
        return;
    }
    let (w_uni, w_bi) = residual_weights(effect.has_bigrams, lambda_bi);
    let n = params.n;
    let d = params.d;
    let h = params.h;
    let inv_sqrt_n = 1.0 / (n as f64).sqrt();
    let comp: Vec<Mat> = (0..d)
        .map(|c| r.channels[c].matmul(&r.channels[c]))
        .collect();
    let mut m_edge = vec![0.0; n * n * h];
    let mut h_nodes = vec![0.0; n * h];
    for i in 0..n {
        for j in 0..n {
            let base = (i * n + j) * h;
            for r_idx in 0..h {
                let mut z = params.b_edge[r_idx];
                let w_d = &params.w_dir[r_idx * d..(r_idx + 1) * d];
                let w_c = &params.w_comp[r_idx * d..(r_idx + 1) * d];
                for c in 0..d {
                    z += w_d[c] * r.get(i, j, c) + w_c[c] * comp[c].get(i, j);
                }
                let m = z.tanh();
                m_edge[base + r_idx] = m;
                h_nodes[i * h + r_idx] += m;
            }
        }
        for r_idx in 0..h {
            h_nodes[i * h + r_idx] *= inv_sqrt_n;
        }
    }
    let mut grad_h = vec![0.0; n * h];
    let mut probs = vec![0.0; heads.k_uni.max(heads.k_bi).max(1)];
    for i in 0..n {
        let hi = &h_nodes[i * h..(i + 1) * h];
        let ghi = &mut grad_h[i * h..(i + 1) * h];
        if effect.uni_mass[i] > 0.0 {
            bilinear_head_grad_theta(
                hi,
                &heads.u_uni,
                &heads.u0_uni,
                &effect.rel_feats,
                REL_DIM,
                &effect.uni[i * heads.k_uni..(i + 1) * heads.k_uni],
                effect.uni_mass[i],
                scale * w_uni,
                &mut probs[..heads.k_uni],
                ghi,
                &mut grads.w_rel_uni,
                &mut grads.b_rel_uni,
            );
        }
        if w_bi > 0.0 && heads.k_bi > 0 && effect.bi_mass[i] > 0.0 {
            bilinear_head_grad_theta(
                hi,
                &heads.u_bi,
                &heads.u0_bi,
                &effect.bi_feats,
                BI_DIM,
                &effect.bi[i * heads.k_bi..(i + 1) * heads.k_bi],
                effect.bi_mass[i],
                scale * w_bi,
                &mut probs[..heads.k_bi],
                ghi,
                &mut grads.w_rel_bi,
                &mut grads.b_rel_bi,
            );
        }
    }
    for i in 0..n {
        let ghi = &grad_h[i * h..(i + 1) * h];
        for j in 0..n {
            let base = (i * n + j) * h;
            for r_idx in 0..h {
                let m = m_edge[base + r_idx];
                let dz = inv_sqrt_n * ghi[r_idx] * (1.0 - m * m);
                if dz == 0.0 {
                    continue;
                }
                grads.b_edge[r_idx] += dz;
                for c in 0..d {
                    grads.w_dir[r_idx * d + c] += dz * r.get(i, j, c);
                    grads.w_comp[r_idx * d + c] += dz * comp[c].get(i, j);
                }
            }
        }
    }
}
#[allow(clippy::too_many_arguments)]
fn bilinear_head_grad_theta(
    h_node: &[f64],
    u_mat: &[f64],
    u0: &[f64],
    feats: &[f64],
    feat_dim: usize,
    target: &[f64],
    mass: f64,
    weight: f64,
    probs: &mut [f64],
    grad_h: &mut [f64],
    grad_w_rel: &mut [f64],
    grad_b_rel: &mut [f64],
) {
    let k = u0.len();
    let h = h_node.len();
    let mut max_l = f64::NEG_INFINITY;
    for a in 0..k {
        let row = &u_mat[a * h..(a + 1) * h];
        let mut acc = u0[a];
        for r in 0..h {
            acc += row[r] * h_node[r];
        }
        probs[a] = acc;
        if acc > max_l {
            max_l = acc;
        }
    }
    let mut sum_exp = 0.0;
    for v in probs.iter_mut().take(k) {
        *v = (*v - max_l).exp();
        sum_exp += *v;
    }
    let inv_sum = 1.0 / sum_exp;
    let mut g_feat = vec![0.0; feat_dim];
    for a in 0..k {
        let p_a = probs[a] * inv_sum;
        let dl = weight * (mass * p_a - target[a]);
        if dl == 0.0 {
            continue;
        }
        let u_row = &u_mat[a * h..(a + 1) * h];
        for r in 0..h {
            grad_h[r] += dl * u_row[r];
        }
        let f_row = &feats[a * feat_dim..(a + 1) * feat_dim];
        for m in 0..feat_dim {
            g_feat[m] += dl * f_row[m];
        }
    }
    for m in 0..feat_dim {
        let gm = g_feat[m];
        if gm == 0.0 {
            continue;
        }
        grad_b_rel[m] += gm;
        for r in 0..h {
            grad_w_rel[r * feat_dim + m] += h_node[r] * gm;
        }
    }
}
pub fn grad_theta_phi_sharp(
    params: &ModelParams,
    r: &RelTensor,
    effect: &EffectSummary,
    probes: &StabilityProbes,
    cfg: &PhiSharpConfig,
) -> (f64, ParamGrads) {
    let (eval, _) = evaluate_phi_sharp(params, r, effect, probes, cfg, false);
    let mut grads = ParamGrads::zeros_like(params);
    let heads = params.precompute_heads(effect);
    let n = r.n;
    let d = r.d;
    let rho = eval.residual;
    let rho_bg = eval.rho_background;
    let p_bar = (-rho).exp();
    let g_w = CausalLaplacian::grad_directed_weights(&eval.fiedler.v2);
    let mut coeff_on_rho = -cfg.alpha;
    for i in 0..n {
        for j in 0..n {
            if i == j {
                continue;
            }
            let gw = cfg.beta_iit * g_w.get(i, j);
            if gw <= 0.0 || (0..d).all(|c| r.get(i, j, c) == 0.0) {
                continue;
            }
            let r_abl = r.with_ablated_edge(i, j);
            let rho_ij = params.residual_only_precomputed(&r_abl, effect, &heads, cfg.lambda_bi);
            let (s_ij, ds_drho_ij) = smooth_cap(rho_ij, rho_bg, cfg.cap_eps);
            let (h_val, h_prime) = smooth_hinge(s_ij - rho, cfg.hinge_tau);
            if h_val > 0.0 {
                let dw_drho = -p_bar * (h_val + h_prime);
                let dw_drho_ij = p_bar * h_prime * ds_drho_ij;
                coeff_on_rho += gw * dw_drho;
                accumulate_residual_grad_theta(
                    params,
                    &r_abl,
                    effect,
                    &heads,
                    cfg.lambda_bi,
                    gw * dw_drho_ij,
                    &mut grads,
                );
            }
        }
    }
    let total_probes = probes.probe_count();
    if total_probes > 0 && cfg.beta > 0.0 {
        let inv_m = 1.0 / (total_probes as f64);
        for q in &probes.rotations {
            let r_rot = StabilityProbes::apply_rotation(q, r);
            let rho_rot =
                params.residual_only_precomputed(&r_rot, effect, &heads, cfg.lambda_bi);
            let (_, hp) = smooth_hinge(rho_rot - rho, cfg.hinge_tau);
            if hp > 0.0 {
                let scale = -cfg.beta * inv_m * hp;
                coeff_on_rho -= scale;
                accumulate_residual_grad_theta(
                    params,
                    &r_rot,
                    effect,
                    &heads,
                    cfg.lambda_bi,
                    scale,
                    &mut grads,
                );
            }
        }
        if probes.abs_kappa > 0.0 {
            let r_abs = probes.apply_abstraction(r);
            let rho_abs =
                params.residual_only_precomputed(&r_abs, effect, &heads, cfg.lambda_bi);
            let (_, hp) = smooth_hinge(rho_abs - rho, cfg.hinge_tau);
            if hp > 0.0 {
                let scale = -cfg.beta * inv_m * hp;
                coeff_on_rho -= scale;
                accumulate_residual_grad_theta(
                    params,
                    &r_abs,
                    effect,
                    &heads,
                    cfg.lambda_bi,
                    scale,
                    &mut grads,
                );
            }
        }
    }
    accumulate_residual_grad_theta(
        params,
        r,
        effect,
        &heads,
        cfg.lambda_bi,
        coeff_on_rho,
        &mut grads,
    );
    (eval.phi_sharp, grads)
}
fn accumulate_trajectory_es_grad(
    params: &ModelParams,
    effect: &EffectSummary,
    probes: &StabilityProbes,
    phi_cfg: &PhiSharpConfig,
    flow_cfg: &FlowConfig,
    sigma: f64,
    rng: &mut Rng,
    grads: &mut ParamGrads,
) {
    if sigma <= 0.0 {
        return;
    }
    let sample_dir = |len: usize, rng: &mut Rng| -> Vec<f64> {
        (0..len)
            .map(|_| if rng.gen_bool(0.5) { 1.0 } else { -1.0 })
            .collect()
    };
    let d_enc_u = sample_dir(params.w_enc_uni.len(), rng);
    let d_enc_b = sample_dir(params.w_enc_bi.len(), rng);
    let d_asym = sample_dir(params.w_init_asym.len(), rng);
    let d_ov = sample_dir(params.w_init_ov.len(), rng);
    let d_sol = sample_dir(params.w_sol.data.len(), rng);
    let d_nu = sample_dir(params.nu_sol.len(), rng);
    let make_perturbed = |sign: f64| -> ModelParams {
        let mut p = params.clone();
        let s = sign * sigma;
        for (w, &dv) in p.w_enc_uni.iter_mut().zip(d_enc_u.iter()) {
            *w += s * dv;
        }
        for (w, &dv) in p.w_enc_bi.iter_mut().zip(d_enc_b.iter()) {
            *w += s * dv;
        }
        for (w, &dv) in p.w_init_asym.iter_mut().zip(d_asym.iter()) {
            *w += s * dv;
        }
        for (w, &dv) in p.w_init_ov.iter_mut().zip(d_ov.iter()) {
            *w += s * dv;
        }
        for (w, &dv) in p.w_sol.data.iter_mut().zip(d_sol.iter()) {
            *w += s * dv;
        }
        for (w, &dv) in p.nu_sol.iter_mut().zip(d_nu.iter()) {
            *w += s * dv;
        }
        p
    };
    let mut short_flow = flow_cfg.clone();
    short_flow.max_steps = flow_cfg.max_steps.min(10);
    let p_plus = make_perturbed(1.0);
    let p_minus = make_perturbed(-1.0);
    let phi_plus = run_abductive_flow(&p_plus, effect, probes, phi_cfg, &short_flow, None)
        .final_eval
        .phi_sharp;
    let phi_minus = run_abductive_flow(&p_minus, effect, probes, phi_cfg, &short_flow, None)
        .final_eval
        .phi_sharp;
    let coeff = (phi_plus - phi_minus) / (2.0 * sigma);
    for (g, &dv) in grads.w_enc_uni.iter_mut().zip(d_enc_u.iter()) {
        *g += coeff * dv;
    }
    for (g, &dv) in grads.w_enc_bi.iter_mut().zip(d_enc_b.iter()) {
        *g += coeff * dv;
    }
    for (g, &dv) in grads.w_init_asym.iter_mut().zip(d_asym.iter()) {
        *g += coeff * dv;
    }
    for (g, &dv) in grads.w_init_ov.iter_mut().zip(d_ov.iter()) {
        *g += coeff * dv;
    }
    for (g, &dv) in grads.w_sol.data.iter_mut().zip(d_sol.iter()) {
        *g += coeff * dv;
    }
    for (g, &dv) in grads.nu_sol.iter_mut().zip(d_nu.iter()) {
        *g += coeff * dv;
    }
}
pub fn train_session<F>(
    params: &mut ModelParams,
    phi_cfg: &PhiSharpConfig,
    flow_cfg: &FlowConfig,
    train_cfg: &TrainConfig,
    mut on_episode: F,
) -> Vec<EpisodeTrainLog>
where
    F: FnMut(&EpisodeTrainLog),
{
    let mut adam = AdamWState::new(params);
    let mut es_rng = Rng::from_seed(train_cfg.seed ^ 0x55AA_33CC_11EE_7799);
    let mut logs = Vec::with_capacity(train_cfg.episodes);
    let total_eps = train_cfg.episodes.max(1);
    for ep_idx in 0..train_cfg.episodes {
        let progress = (ep_idx as f64) / ((total_eps.saturating_sub(1)).max(1) as f64);
        let cur_k = lerp_usize(train_cfg.start_horizon_k, train_cfg.end_horizon_k, progress);
        let cur_d = lerp_usize(
            train_cfg.start_distractors,
            train_cfg.end_distractors,
            progress,
        );
        let cur_noise =
            train_cfg.start_noise + progress * (train_cfg.end_noise - train_cfg.start_noise);
        let cur_cfg = CurriculumConfig {
            horizon_k: cur_k,
            num_distractors: cur_d,
            noise_ratio: cur_noise,
            max_attempts: 64,
        };
        let mut gen_ = HoPtgrsGenerator::new(
            train_cfg
                .seed
                .wrapping_add((ep_idx as u64).wrapping_mul(0x9E37_79B9)),
            cur_cfg,
        );
        let ep = match gen_.generate_episode(None) {
            Ok(e) => e,
            Err(_) => continue,
        };
        let effect = EffectSummary::from_bytes(&ep.perceived_eff, params.n);
        let probes = StabilityProbes::build(
            params.n,
            train_cfg.num_rot_probes,
            train_cfg.probe_angle,
            train_cfg.probe_abs_kappa,
            train_cfg.seed.wrapping_add((ep_idx as u64) * 101 + 7),
        );
        let traj = run_abductive_flow(params, &effect, &probes, phi_cfg, flow_cfg, None);
        let first = traj.trace.first().unwrap();
        let last = traj.trace.last().unwrap();
        let (_, mut grads) =
            grad_theta_phi_sharp(params, &traj.r_final, &effect, &probes, phi_cfg);
        if train_cfg.es_every > 0 && ep_idx % train_cfg.es_every == 0 {
            accumulate_trajectory_es_grad(
                params,
                &effect,
                &probes,
                phi_cfg,
                flow_cfg,
                train_cfg.es_sigma,
                &mut es_rng,
                &mut grads,
            );
        }
        let raw_grad_norm = grads.clip_norm(train_cfg.grad_clip_norm);
        let lr_t = train_cfg.lr
            * (0.1 + 0.9 * 0.5 * (1.0 + (std::f64::consts::PI * progress).cos()));
        adam.step_ascent(params, &grads, lr_t, train_cfg.weight_decay);
        let log = EpisodeTrainLog {
            episode: ep_idx + 1,
            horizon_k: cur_k,
            distractors: cur_d,
            eff_bytes: ep.perceived_eff.len(),
            phi_init: first.phi_sharp,
            phi_final: last.phi_sharp,
            lambda2_final: last.lambda2,
            residual_init: first.residual,
            residual_final: last.residual,
            v_stab_final: last.v_stab,
            active_edges: last.active_causal_edges,
            flow_steps: traj.trace.len().saturating_sub(1),
            grad_norm: raw_grad_norm,
        };
        on_episode(&log);
        logs.push(log);
    }
    logs
}
pub fn evaluate_post_hoc(
    params: &ModelParams,
    phi_cfg: &PhiSharpConfig,
    flow_cfg: &FlowConfig,
    proj_cfg: &ProjectionConfig,
    num_episodes: usize,
    seed: u64,
) -> PostHocEvalSummary {
    let mut gen_ = HoPtgrsGenerator::new(seed, CurriculumConfig::default());
    let mut tested = 0usize;
    let mut sum_phi_gain = 0.0;
    let mut sum_lam2 = 0.0;
    let mut sum_res_drop = 0.0;
    let mut sum_alph_ratio = 0.0;
    let mut sum_target_overlap = 0.0;
    for i in 0..num_episodes {
        let ep = match gen_.generate_episode(None) {
            Ok(e) => e,
            Err(_) => continue,
        };
        let effect = EffectSummary::from_bytes(&ep.perceived_eff, params.n);
        let probes = StabilityProbes::build(params.n, 2, 0.06, 0.12, seed + i as u64 + 13);
        let traj = run_abductive_flow(params, &effect, &probes, phi_cfg, flow_cfg, None);
        let proj = project_witness_mirror_ascent(
            params,
            &traj.r_final,
            &traj.final_eval,
            &ep.perceived_eff,
            proj_cfg,
        );
        let first = traj.trace.first().unwrap();
        let last = traj.trace.last().unwrap();
        sum_phi_gain += last.phi_sharp - first.phi_sharp;
        sum_lam2 += last.lambda2;
        sum_res_drop += first.residual - last.residual;
        let mut eff_present = [false; BYTE_VOCAB];
        for &b in &ep.perceived_eff {
            eff_present[b as usize] = true;
        }
        let in_alph = proj
            .witness_bytes
            .iter()
            .filter(|&&b| eff_present[b as usize])
            .count();
        sum_alph_ratio += (in_alph as f64) / (proj.witness_bytes.len().max(1) as f64);
        let target_bytes = &ep.hidden_cause.target_concept_bytes;
        let hits = target_bytes
            .iter()
            .filter(|b| proj.witness_bytes.contains(b))
            .count();
        sum_target_overlap += (hits as f64) / (target_bytes.len().max(1) as f64);
        tested += 1;
    }
    let denom = tested.max(1) as f64;
    PostHocEvalSummary {
        episodes_tested: tested,
        mean_phi_gain: sum_phi_gain / denom,
        mean_lambda2: sum_lam2 / denom,
        mean_residual_drop: sum_res_drop / denom,
        mean_witness_alphabet_ratio: sum_alph_ratio / denom,
        mean_target_byte_overlap: sum_target_overlap / denom,
    }
}
fn lerp_usize(a: usize, b: usize, t: f64) -> usize {
    let v = (a as f64) + t * ((b as f64) - (a as f64));
    v.round().max(1.0) as usize
}

#[derive(Debug, Clone)]
pub struct EpisodicCodec {
    def_sym_map: BTreeMap<SymbolId, Vec<u8>>,
    inst_sym_map: BTreeMap<SymbolId, Vec<u8>>,
    def_tags: BTreeMap<DefTag, Vec<u8>>,
    inst_tags: BTreeMap<InstTag, Vec<u8>>,
    def_idx_codes: Vec<Vec<u8>>,
    inst_idx_codes: Vec<Vec<u8>>,
}
impl EpisodicCodec {
    pub fn sample(
        rng: &mut Rng,
        def_syms: &BTreeSet<SymbolId>,
        inst_syms: &BTreeSet<SymbolId>,
        max_def_idx: usize,
        max_inst_idx: usize,
    ) -> Self {
        let total_needed = def_syms.len()
            + inst_syms.len()
            + 3 
            + 9 
            + max_def_idx.max(1)
            + max_inst_idx.max(1);
        let mut pool = generate_prefix_free_pool(rng, total_needed);
        rng.shuffle(&mut pool);
        let mut iter = pool.into_iter();
        let mut def_sym_map = BTreeMap::new();
        for &s in def_syms {
            def_sym_map.insert(s, iter.next().unwrap());
        }
        let mut inst_sym_map = BTreeMap::new();
        for &s in inst_syms {
            inst_sym_map.insert(s, iter.next().unwrap());
        }
        let mut def_tags = BTreeMap::new();
        for tag in [DefTag::BlockStart, DefTag::EdgeSep, DefTag::BlockEnd] {
            def_tags.insert(tag, iter.next().unwrap());
        }
        let mut inst_tags = BTreeMap::new();
        for tag in [
            InstTag::BlockStart,
            InstTag::TermStart,
            InstTag::NodeSep,
            InstTag::ChildStart,
            InstTag::ChildEnd,
            InstTag::RuleStart,
            InstTag::RuleArrow,
            InstTag::VarPrefix,
            InstTag::BlockEnd,
        ] {
            inst_tags.insert(tag, iter.next().unwrap());
        }
        let def_idx_codes: Vec<Vec<u8>> = (0..max_def_idx.max(1))
            .map(|_| iter.next().unwrap())
            .collect();
        let inst_idx_codes: Vec<Vec<u8>> = (0..max_inst_idx.max(1))
            .map(|_| iter.next().unwrap())
            .collect();
        Self {
            def_sym_map,
            inst_sym_map,
            def_tags,
            inst_tags,
            def_idx_codes,
            inst_idx_codes,
        }
    }
    pub fn def_code(&self, sym: SymbolId) -> Option<&[u8]> {
        self.def_sym_map.get(&sym).map(|v| v.as_slice())
    }
    pub fn verify_invariants(&self) -> bool {
        let mut def_words: BTreeSet<&[u8]> = BTreeSet::new();
        for v in self.def_sym_map.values() {
            def_words.insert(v.as_slice());
        }
        for v in self.def_tags.values() {
            def_words.insert(v.as_slice());
        }
        for v in &self.def_idx_codes {
            def_words.insert(v.as_slice());
        }
        let mut inst_words: BTreeSet<&[u8]> = BTreeSet::new();
        for v in self.inst_sym_map.values() {
            inst_words.insert(v.as_slice());
        }
        for v in self.inst_tags.values() {
            inst_words.insert(v.as_slice());
        }
        for v in &self.inst_idx_codes {
            inst_words.insert(v.as_slice());
        }
        if !def_words.is_disjoint(&inst_words) {
            return false;
        }
        let mut all: Vec<&[u8]> = def_words.into_iter().chain(inst_words).collect();
        all.sort_unstable();
        for pair in all.windows(2) {
            if pair[1].starts_with(pair[0]) {
                return false;
            }
        }
        true
    }
    pub fn serialize(&self, rng: &mut Rng, def: &DefGraph, inst: &Instance) -> Vec<u8> {
        let mut def_bytes = self.serialize_def(rng, def);
        let mut inst_bytes = self.serialize_inst(rng, inst);
        let mut out = Vec::with_capacity(def_bytes.len() + inst_bytes.len());
        if rng.gen_bool(0.5) {
            out.append(&mut def_bytes);
            out.append(&mut inst_bytes);
        } else {
            out.append(&mut inst_bytes);
            out.append(&mut def_bytes);
        }
        out
    }
    fn serialize_def(&self, rng: &mut Rng, def: &DefGraph) -> Vec<u8> {
        let mut buf = Vec::new();
        buf.extend_from_slice(&self.def_tags[&DefTag::BlockStart]);
        let mut shuffled_edges = def.edges.clone();
        rng.shuffle(&mut shuffled_edges);
        for (src, rel, dst) in shuffled_edges {
            buf.extend_from_slice(&self.def_tags[&DefTag::EdgeSep]);
            buf.extend_from_slice(&self.def_sym_map[&def.node_symbols[src]]);
            buf.extend_from_slice(&self.def_sym_map[&rel]);
            buf.extend_from_slice(&self.def_sym_map[&def.node_symbols[dst]]);
        }
        buf.extend_from_slice(&self.def_tags[&DefTag::BlockEnd]);
        buf
    }
    fn serialize_inst(&self, rng: &mut Rng, inst: &Instance) -> Vec<u8> {
        let mut buf = Vec::new();
        buf.extend_from_slice(&self.inst_tags[&InstTag::BlockStart]);
        let mut roots: Vec<&TermGraph> = inst.roots.iter().collect();
        rng.shuffle(&mut roots);
        let mut rules: Vec<&Rule> = inst.rules.iter().collect();
        rng.shuffle(&mut rules);
        let terms_first = rng.gen_bool(0.5);
        if terms_first {
            for g in &roots {
                self.encode_term_graph(&mut buf, g);
            }
            for r in &rules {
                self.encode_rule(&mut buf, r);
            }
        } else {
            for r in &rules {
                self.encode_rule(&mut buf, r);
            }
            for g in &roots {
                self.encode_term_graph(&mut buf, g);
            }
        }
        buf.extend_from_slice(&self.inst_tags[&InstTag::BlockEnd]);
        buf
    }
    fn encode_term_graph(&self, buf: &mut Vec<u8>, g: &TermGraph) {
        buf.extend_from_slice(&self.inst_tags[&InstTag::TermStart]);
        buf.extend_from_slice(&self.inst_idx_codes[g.root]);
        for (idx, node) in g.nodes.iter().enumerate() {
            buf.extend_from_slice(&self.inst_tags[&InstTag::NodeSep]);
            buf.extend_from_slice(&self.inst_idx_codes[idx]);
            buf.extend_from_slice(&self.inst_sym_map[&node.label]);
            buf.extend_from_slice(&self.inst_tags[&InstTag::ChildStart]);
            for &c in &node.succ {
                buf.extend_from_slice(&self.inst_idx_codes[c]);
            }
            buf.extend_from_slice(&self.inst_tags[&InstTag::ChildEnd]);
        }
    }
    fn encode_rule(&self, buf: &mut Vec<u8>, r: &Rule) {
        buf.extend_from_slice(&self.inst_tags[&InstTag::RuleStart]);
        self.encode_pattern(buf, &r.lhs);
        buf.extend_from_slice(&self.inst_tags[&InstTag::RuleArrow]);
        self.encode_pattern(buf, &r.rhs);
    }
    fn encode_pattern(&self, buf: &mut Vec<u8>, pat: &PatternGraph) {
        buf.extend_from_slice(&self.inst_idx_codes[pat.root]);
        for (idx, node) in pat.nodes.iter().enumerate() {
            buf.extend_from_slice(&self.inst_tags[&InstTag::NodeSep]);
            buf.extend_from_slice(&self.inst_idx_codes[idx]);
            match node {
                PatternNode::Var(vid) => {
                    buf.extend_from_slice(&self.inst_tags[&InstTag::VarPrefix]);
                    buf.extend_from_slice(&self.inst_idx_codes[*vid as usize]);
                }
                PatternNode::Sym(sym, succ) => {
                    buf.extend_from_slice(&self.inst_sym_map[sym]);
                    buf.extend_from_slice(&self.inst_tags[&InstTag::ChildStart]);
                    for &c in succ {
                        buf.extend_from_slice(&self.inst_idx_codes[c]);
                    }
                    buf.extend_from_slice(&self.inst_tags[&InstTag::ChildEnd]);
                }
            }
        }
    }
}
fn generate_prefix_free_pool(rng: &mut Rng, needed: usize) -> Vec<Vec<u8>> {
    let mut single_bytes: Vec<u8> = (0u8..=255).collect();
    rng.shuffle(&mut single_bytes);
    let mut ready: Vec<Vec<u8>> = Vec::with_capacity(needed * 2);
    let mut alphabet_buf: Vec<u8> = (0u8..=255).collect();
    while ready.len() < needed + 32 {
        if let Some(b0) = single_bytes.pop() {
            rng.shuffle(&mut alphabet_buf);
            let branch = rng.gen_range_inclusive(8, 24);
            for &b1 in alphabet_buf.iter().take(branch) {
                ready.push(vec![b0, b1]);
            }
        } else {
            let pick = rng.gen_range(ready.len());
            let base = ready.swap_remove(pick);
            rng.shuffle(&mut alphabet_buf);
            let branch = rng.gen_range_inclusive(4, 12);
            for &b in alphabet_buf.iter().take(branch) {
                let mut ext = base.clone();
                ext.push(b);
                ready.push(ext);
            }
        }
    }
    let deepen_steps = (ready.len() / 5).max(4);
    for _ in 0..deepen_steps {
        let pick = rng.gen_range(ready.len());
        if ready[pick].len() >= 4 {
            continue;
        }
        let base = ready.swap_remove(pick);
        rng.shuffle(&mut alphabet_buf);
        let branch = rng.gen_range_inclusive(2, 4);
        for &b in alphabet_buf.iter().take(branch) {
            let mut ext = base.clone();
            ext.push(b);
            ready.push(ext);
        }
    }
    ready
}

    pub fn generate_episode(&mut self, forced_target: Option<MetaOp>) -> Result<Episode, String> {
        let k = self.config.horizon_k.max(3);
        let catalog = MetaOp::catalog();
        let d_count = self.config.num_distractors.clamp(1, catalog.len() - 1);
        for _attempt in 0..self.config.max_attempts {
            let target = forced_target.clone().unwrap_or_else(|| {
                let idx = self.rng.gen_range(catalog.len());
                catalog[idx].clone()
            });
            let mut others: Vec<MetaOp> = catalog
                .iter()
                .filter(|op| **op != target)
                .cloned()
                .collect();
            self.rng.shuffle(&mut others);
            let distractors: Vec<MetaOp> = others.into_iter().take(d_count).collect();
            let mut all_ops = vec![target.clone()];
            all_ops.extend(distractors.iter().cloned());
            self.rng.shuffle(&mut all_ops);
            let def_graph = build_def_graph(&all_ops);
            let rigidity = verify_rigidity_2fwl(&def_graph);
            if !rigidity.is_rigid {
                continue;
            }
            let mut alloc = InstSymAlloc::new();
            let mut inst = synthesize_instance(&mut self.rng, &mut alloc, &target, k);
            if !evaluate_meta_op(&target, &inst, k) {
                continue;
            }
            if distractors.iter().any(|d| evaluate_meta_op(d, &inst, k)) {
                continue;
            }
            let noise_added =
                inject_verified_noise(&mut self.rng, &mut alloc, &mut inst, &target, &distractors, k, self.config.noise_ratio);
            let def_syms = def_graph.all_symbols();
            let inst_syms = inst.all_symbols();
            assert!(
                def_syms.is_disjoint(&inst_syms),
                "Violation de partition stricte Sigma_def / Sigma_inst"
            );
            let max_inst_idx = compute_max_inst_index(&inst);
            let codec = EpisodicCodec::sample(
                &mut self.rng,
                &def_syms,
                &inst_syms,
                def_graph.node_symbols.len(),
                max_inst_idx,
            );
            let codec_ok = codec.verify_invariants();
            if !codec_ok {
                continue;
            }
            let perceived_eff = codec.serialize(&mut self.rng, &def_graph, &inst);
            let target_node = def_graph.concept_nodes[&target];
            let target_sym = def_graph.node_symbols[target_node];
            let target_concept_bytes = codec.def_code(target_sym).unwrap().to_vec();
            let stats = EpisodeStats {
                total_bytes: perceived_eff.len(),
                def_nodes: def_graph.node_symbols.len(),
                def_edges: def_graph.edges.len(),
                inst_roots: inst.roots.len(),
                inst_rules_total: inst.rules.len(),
                noise_rules_added: noise_added,
                codec_invariants_ok: codec_ok,
            };
            return Ok(Episode {
                perceived_eff,
                hidden_cause: HiddenCause {
                    target_op: target,
                    distractors,
                    target_concept_bytes,
                },
                rigidity,
                stats,
            });
        }
        Err("Échec du rejection sampling dans max_attempts".to_string())
    }
}
fn compute_max_inst_index(inst: &Instance) -> usize {
    let mut m = 4usize;
    for g in &inst.roots {
        m = m.max(g.nodes.len() + 1);
    }
    for r in &inst.rules {
        m = m.max(r.lhs.nodes.len() + 1).max(r.rhs.nodes.len() + 1);
        for n in r.lhs.nodes.iter().chain(r.rhs.nodes.iter()) {
            if let PatternNode::Var(vid) = n {
                m = m.max(*vid as usize + 1);
            }
        }
    }
    m
}

fn synth_bif(rng: &mut Rng, alloc: &mut InstSymAlloc, k: usize) -> Instance {
    let leaf = alloc.fresh(0);
    let pre_len = rng.gen_range_inclusive(0, k.saturating_sub(2).min(2));
    let mut rules = Vec::new();
    let mut curr = alloc.fresh(1);
    let start = curr;
    for _ in 0..pre_len {
        let next = alloc.fresh(1);
        rules.push(unary_rule(alloc, curr, next, 1.0));
        curr = next;
    }
    let branches = rng.gen_range_inclusive(2, 3);
    for _ in 0..branches {
        let b_mid = alloc.fresh(1);
        rules.push(unary_rule(alloc, curr, b_mid, 1.0 / branches as f64));
        if pre_len + 2 <= k && rng.gen_bool(0.5) {
            let b_end = alloc.fresh(1);
            rules.push(unary_rule(alloc, b_mid, b_end, 1.0));
        }
    }
    Instance {
        roots: vec![unary_term(start, leaf)],
        rules,
        arities: alloc.arities.clone(),
    }
}

    let pre_len = rng.gen_range_inclusive(0, k.saturating_sub(cycle_len).min(1));
    let mut rules = Vec::new();
    let start = alloc.fresh(1);

    for _ in 0..pre_len {
        let next = alloc.fresh(1);
        rules.push(unary_rule(alloc, curr, next, 1.0));
        curr = next;

        let next = alloc.fresh(1);
        rules.push(unary_rule(alloc, curr, next, 1.0));
        curr = next;
    }
    rules.push(unary_rule(alloc, curr, cycle_entry, 1.0));
    Instance {
        roots: vec![unary_term(start, leaf)],
        rules,
        arities: alloc.arities.clone(),
    }
}

fn synth_iso(rng: &mut Rng, alloc: &mut InstSymAlloc, k: usize) -> Instance {

    let mut roots = Vec::new();
    let mut rules = Vec::new();
    for _domain in 0..2 {
        let leaf = alloc.fresh(0);
        if use_diamond {
            let s0 = alloc.fresh(1);
            let s1 = alloc.fresh(1);
            let s2 = alloc.fresh(1);
            let s3 = alloc.fresh(1);
            rules.push(unary_rule(alloc, s0, s1, 0.5));
            rules.push(unary_rule(alloc, s0, s2, 0.5));
            rules.push(unary_rule(alloc, s1, s3, 1.0));
            rules.push(unary_rule(alloc, s2, s3, 1.0));
            roots.push(unary_term(s0, leaf));
        } else {
            let start = alloc.fresh(1);
            let mut curr = start;
            for _ in 0..chain_len {
                let next = alloc.fresh(1);
                rules.push(unary_rule(alloc, curr, next, 1.0));
                curr = next;
            }
            roots.push(unary_term(start, leaf));
        }
    }
    Instance {
        roots,
        rules,
        arities: alloc.arities.clone(),
    }
}
fn synth_quot(rng: &mut Rng, alloc: &mut InstSymAlloc, k: usize) -> Instance {
    let outer = alloc.fresh(2);
    let inner = alloc.fresh(2);
    let cst = alloc.fresh(0);
    let m = rng.gen_range_inclusive(3, 4);
    let mut roots = Vec::new();
    let mut rules = Vec::new();
    for i in 0..m {
        let hole_val = alloc.fresh(0);
        let mut nf_rhs = PatternGraph::builder();
        let hv = nf_rhs.sym(hole_val, vec![]);
        let c1 = nf_rhs.sym(cst, vec![]);
        let c2 = nf_rhs.sym(cst, vec![]);
        let inn = nf_rhs.sym(inner, vec![hv, c1]);
        let out_r = nf_rhs.sym(outer, vec![inn, c2]);
        let target_rhs = nf_rhs.finish(out_r);
        let steps = (i % k.min(3)) + 1;
        let start = alloc.fresh(0);
        let mut curr = start;
        for step_idx in 0..steps {
            let mut lb = PatternGraph::builder();
            let lr = lb.sym(curr, vec![]);
            if step_idx + 1 == steps {
                rules.push(Rule {
                    id: alloc.rule_id(),
                    lhs: lb.finish(lr),
                    rhs: target_rhs.clone(),
                    prob: 1.0,
                });
            } else {
                let next = alloc.fresh(0);
                let mut rb = PatternGraph::builder();
                let rr = rb.sym(next, vec![]);
                rules.push(Rule {
                    id: alloc.rule_id(),
                    lhs: lb.finish(lr),
                    rhs: rb.finish(rr),
                    prob: 1.0,
                });
                curr = next;
            }
        }
        let mut tb = TermGraph::builder();
        let r = tb.add(start, vec![]);
        roots.push(tb.finish(r));
    }
    Instance {
        roots,
        rules,
        arities: alloc.arities.clone(),
    }
}
fn synth_comp(
    rng: &mut Rng,
    alloc: &mut InstSymAlloc,
    a: &MetaOp,
    b: &MetaOp,
    k: usize,
) -> Instance {
    match (a, b) {
        (MetaOp::Bif, MetaOp::Fix) | (MetaOp::Fix, MetaOp::Bif) => {
            let leaf = alloc.fresh(0);
            let start = alloc.fresh(1);
            let nf_sym = alloc.fresh(1);
            let c1 = alloc.fresh(1);
            let c2 = alloc.fresh(1);
            let mut rules = vec![
                unary_rule(alloc, start, nf_sym, 0.5),
                unary_rule(alloc, start, c1, 0.5),
                unary_rule(alloc, c1, c2, 1.0),
                unary_rule(alloc, c2, c1, 1.0),
            ];
            if k >= 4 && rng.gen_bool(0.5) {
                let nf2 = alloc.fresh(1);
                rules.push(unary_rule(alloc, nf_sym, nf2, 1.0));
            }
            Instance {
                roots: vec![unary_term(start, leaf)],
                rules,
                arities: alloc.arities.clone(),
            }
        }
        (MetaOp::Iso, MetaOp::Bif) | (MetaOp::Bif, MetaOp::Iso) => {
            let mut roots = Vec::new();
            let mut rules = Vec::new();
            for _ in 0..2 {
                let leaf = alloc.fresh(0);
                let s0 = alloc.fresh(1);
                let b1 = alloc.fresh(1);
                let b2 = alloc.fresh(1);
                rules.push(unary_rule(alloc, s0, b1, 0.5));
                rules.push(unary_rule(alloc, s0, b2, 0.5));
                roots.push(unary_term(s0, leaf));
            }
            Instance {
                roots,
                rules,
                arities: alloc.arities.clone(),
            }
        }
        (MetaOp::Iso, MetaOp::Fix) | (MetaOp::Fix, MetaOp::Iso) => {
            let cycle_len = rng.gen_range_inclusive(2, k.min(3));
            let mut roots = Vec::new();
            let mut rules = Vec::new();
            for _ in 0..2 {
                let leaf = alloc.fresh(0);
                let start = alloc.fresh(1);
                let mut curr = start;
                for _ in 1..cycle_len {
                    let next = alloc.fresh(1);
                    rules.push(unary_rule(alloc, curr, next, 1.0));
                    curr = next;
                }
                rules.push(unary_rule(alloc, curr, start, 1.0));
                roots.push(unary_term(start, leaf));
            }
            Instance {
                roots,
                rules,
                arities: alloc.arities.clone(),
            }
        }
        (MetaOp::Quot, MetaOp::Bif) | (MetaOp::Bif, MetaOp::Quot) => {
            let outer = alloc.fresh(2);
            let inner = alloc.fresh(2);
            let cst = alloc.fresh(0);
            let mut roots = Vec::new();
            let mut rules = Vec::new();
            for _ in 0..3 {
                let hv = alloc.fresh(0);
                let nf1 = alloc.fresh(0);
                let nf2 = alloc.fresh(0);
                for target_nf in [nf1, nf2] {
                    let mut lb = PatternGraph::builder();
                    let lr = lb.sym(hv, vec![]);
                    let mut rb = PatternGraph::builder();
                    let rr = rb.sym(target_nf, vec![]);
                    rules.push(Rule {
                        id: alloc.rule_id(),
                        lhs: lb.finish(lr),
                        rhs: rb.finish(rr),
                        prob: 0.5,
                    });
                }
                let mut tb = TermGraph::builder();
                let nh = tb.add(hv, vec![]);
                let nc1 = tb.add(cst, vec![]);
                let nc2 = tb.add(cst, vec![]);
                let ni = tb.add(inner, vec![nh, nc1]);
                let nr = tb.add(outer, vec![ni, nc2]);
                roots.push(tb.finish(nr));
            }
            Instance {
                roots,
                rules,
                arities: alloc.arities.clone(),
            }
        }
        _ => synth_bif(rng, alloc, k),
    }
}
fn inject_verified_noise(
    rng: &mut Rng,
    alloc: &mut InstSymAlloc,
    inst: &mut Instance,
    target: &MetaOp,
    distractors: &[MetaOp],
    k: usize,
    noise_ratio: f64,
) -> usize {
    let desired = ((inst.rules.len() as f64) * noise_ratio).round() as usize;
    let mut added = 0usize;
    for _ in 0..desired {
        let decoy_a = alloc.fresh(1);
        let decoy_b = alloc.fresh(1);
        let noise_rule = unary_rule(alloc, decoy_a, decoy_b, rng.gen_f64().clamp(0.1, 0.9));
        inst.rules.push(noise_rule);
        inst.arities = alloc.arities.clone();
        let still_valid = evaluate_meta_op(target, inst, k)
            && distractors.iter().all(|d| !evaluate_meta_op(d, inst, k));
        if still_valid {
            added += 1;
        } else {
            inst.rules.pop();
        }
    }
    added
}

fn reaches_absorber(dg: &DerivationGraph, absorbers: &BTreeSet<SymbolId>) -> bool {
    if absorbers.is_empty() {
        return false;
    }
    dg.normal_forms().into_iter().any(|nf_id| {
        let g = &dg.states[nf_id];
        g.nodes.len() == 1 && absorbers.contains(&g.nodes[g.root].label)
    })
}
pub fn evaluate_meta_op(op: &MetaOp, inst: &Instance, horizon_k: usize) -> bool {
    const MAX_STATES: usize = 64;
    if inst.roots.is_empty() {
        return false;
    }
    let dgs: Vec<DerivationGraph> = inst
        .roots
        .iter()
        .map(|g| DerivationGraph::build(g, &inst.rules, horizon_k, MAX_STATES))
        .collect();
    let absorbers = detect_absorber_symbols(&inst.rules, &inst.arities);
    match op {
        MetaOp::Bif => {
            dgs.len() == 1
                && !dgs[0].has_directed_cycle()
                && !reaches_absorber(&dgs[0], &absorbers)
                && dgs[0].normal_forms().len() >= 2
        }
        MetaOp::Fix => {
            dgs.len() == 1
                && dgs[0].has_directed_cycle()
                && dgs[0].normal_forms().is_empty()
                && !reaches_absorber(&dgs[0], &absorbers)
        }
        MetaOp::Bot => {
            !absorbers.is_empty()
                && dgs.iter().all(|dg| !dg.has_directed_cycle())
                && dgs.iter().any(|dg| reaches_absorber(dg, &absorbers))
        }
        MetaOp::Iso => {
            if dgs.len() != 2 {
                return false;
            }
            let s0 = dgs[0].reachable_symbols();
            let s1 = dgs[1].reachable_symbols();
            s0.is_disjoint(&s1)
                && dgs[0].states.len() >= 3
                && !dgs[0].has_directed_cycle()
                && !dgs[1].has_directed_cycle()
                && dgs[0].normal_forms().len() == 1
                && dgs[1].normal_forms().len() == 1
                && !reaches_absorber(&dgs[0], &absorbers)
                && !reaches_absorber(&dgs[1], &absorbers)
                && dgs[0].is_unlabeled_isomorphic(&dgs[1])
        }
        MetaOp::Quot => {
            if dgs.len() < 3 {
                return false;
            }
            let all_clean = dgs.iter().all(|dg| {
                !dg.has_directed_cycle()
                    && dg.normal_forms().len() == 1
                    && !reaches_absorber(dg, &absorbers)
            });
            if !all_clean {
                return false;
            }
            let all_iso = (1..dgs.len()).all(|i| dgs[0].is_unlabeled_isomorphic(&dgs[i]));
            if all_iso {
                return false;
            }
            let nfs: Vec<TermGraph> = dgs
                .iter()
                .map(|dg| dg.states[dg.normal_forms()[0]].clone())
                .collect();
            anti_unify(&nfs).map_or(false, |t| t.is_nontrivial())
        }
        MetaOp::Comp(a, b) => match (a.as_ref(), b.as_ref()) {
            (MetaOp::Bif, MetaOp::Fix) | (MetaOp::Fix, MetaOp::Bif) => {
                dgs.len() == 1
                    && dgs[0].has_directed_cycle()
                    && !dgs[0].normal_forms().is_empty()
                    && !reaches_absorber(&dgs[0], &absorbers)
            }
            (MetaOp::Iso, MetaOp::Bif) | (MetaOp::Bif, MetaOp::Iso) => {
                if dgs.len() != 2 {
                    return false;
                }
                let s0 = dgs[0].reachable_symbols();
                let s1 = dgs[1].reachable_symbols();
                s0.is_disjoint(&s1)
                    && !dgs[0].has_directed_cycle()
                    && !dgs[1].has_directed_cycle()
                    && dgs[0].normal_forms().len() >= 2
                    && dgs[1].normal_forms().len() >= 2
                    && !reaches_absorber(&dgs[0], &absorbers)
                    && !reaches_absorber(&dgs[1], &absorbers)
                    && dgs[0].is_unlabeled_isomorphic(&dgs[1])
            }
            (MetaOp::Iso, MetaOp::Fix) | (MetaOp::Fix, MetaOp::Iso) => {
                if dgs.len() != 2 {
                    return false;
                }
                let s0 = dgs[0].reachable_symbols();
                let s1 = dgs[1].reachable_symbols();
                s0.is_disjoint(&s1)
                    && dgs[0].states.len() >= 2
                    && dgs[0].has_directed_cycle()
                    && dgs[1].has_directed_cycle()
                    && dgs[0].normal_forms().is_empty()
                    && dgs[1].normal_forms().is_empty()
                    && dgs[0].is_unlabeled_isomorphic(&dgs[1])
            }
            (MetaOp::Quot, MetaOp::Bif) | (MetaOp::Bif, MetaOp::Quot) => {
                if dgs.len() < 3 {
                    return false;
                }
                let all_bif = dgs.iter().all(|dg| {
                    !dg.has_directed_cycle()
                        && dg.normal_forms().len() >= 2
                        && !reaches_absorber(dg, &absorbers)
                });
                all_bif && anti_unify(&inst.roots).map_or(false, |t| t.is_nontrivial())
            }
            _ => false,
        },
    }
}

pub struct DefGraph {
    pub node_symbols: Vec<SymbolId>,
    pub edges: Vec<(usize, SymbolId, usize)>,
    pub concept_nodes: BTreeMap<MetaOp, usize>,
    pub is_concept_node: Vec<bool>,
}
impl DefGraph {
    pub fn all_symbols(&self) -> BTreeSet<SymbolId> {
        let mut set: BTreeSet<SymbolId> = self.node_symbols.iter().copied().collect();
        for &(_, rel, _) in &self.edges {
            set.insert(rel);
        }
        set
    }

const REL_DEFINES: SymbolId = SymbolId(1);
const REL_STEP: SymbolId = SymbolId(2);
const REL_NORMAL_FORM: SymbolId = SymbolId(3);
const REL_DISTINCT: SymbolId = SymbolId(4);
const REL_CYCLE_EDGE: SymbolId = SymbolId(5);
const REL_NO_NF: SymbolId = SymbolId(6);
const REL_ABSORBS: SymbolId = SymbolId(7);
const REL_DISJOINT: SymbolId = SymbolId(8);
const REL_BIJECTION: SymbolId = SymbolId(9);
const REL_TEMPLATE_OF: SymbolId = SymbolId(10);
const REL_NON_ISO: SymbolId = SymbolId(11);
const REL_COMP_OUTER: SymbolId = SymbolId(12);
const REL_COMP_INNER: SymbolId = SymbolId(13);
const REL_ACYCLIC: SymbolId = SymbolId(14);
const HUB_ACYCLIC: SymbolId = SymbolId(100);
const HUB_CYCLE_PROP: SymbolId = SymbolId(101);
const HUB_ABSORBER_PROP: SymbolId = SymbolId(102);
const HUB_NO_NF_PROP: SymbolId = SymbolId(103);
pub fn build_def_graph(ops: &[MetaOp]) -> DefGraph {
    let mut node_symbols = Vec::new();
    let mut is_concept_node = Vec::new();
    let mut edges = Vec::new();
    let mut concept_nodes = BTreeMap::new();
    let mut next_sym_id: u32 = 200;
    let mut alloc_node = |sym: Option<SymbolId>, is_concept: bool| -> usize {
        let s = sym.unwrap_or_else(|| {
            let id = SymbolId(next_sym_id);
            next_sym_id += 1;
            id
        });
        let idx = node_symbols.len();
        node_symbols.push(s);
        is_concept_node.push(is_concept);
        idx
    };
    let h_acyclic = alloc_node(Some(HUB_ACYCLIC), false);
    let h_cycle = alloc_node(Some(HUB_CYCLE_PROP), false);
    let h_absorber = alloc_node(Some(HUB_ABSORBER_PROP), false);
    let h_no_nf = alloc_node(Some(HUB_NO_NF_PROP), false);
    #[allow(clippy::too_many_arguments)]
    fn build_schema(
        op: &MetaOp,
        h_acyclic: usize,
        h_cycle: usize,
        h_absorber: usize,
        h_no_nf: usize,
        alloc: &mut dyn FnMut(Option<SymbolId>, bool) -> usize,
        edges: &mut Vec<(usize, SymbolId, usize)>,
    ) -> usize {
        match op {
            MetaOp::Bif => {
                let r = alloc(None, false);
                let nf1 = alloc(None, false);
                let nf2 = alloc(None, false);
                edges.push((r, REL_STEP, nf1));
                edges.push((r, REL_STEP, nf2));
                edges.push((r, REL_NORMAL_FORM, nf1));
                edges.push((r, REL_NORMAL_FORM, nf2));
                edges.push((nf1, REL_DISTINCT, nf2));
                edges.push((nf2, REL_DISTINCT, nf1));
                edges.push((r, REL_ACYCLIC, h_acyclic));
                r
            }
            MetaOp::Fix => {
                let s1 = alloc(None, false);
                let s2 = alloc(None, false);
                edges.push((s1, REL_STEP, s2));
                edges.push((s2, REL_STEP, s1));
                edges.push((s1, REL_CYCLE_EDGE, h_cycle));
                edges.push((s1, REL_NO_NF, h_no_nf));
                s1
            }
            MetaOp::Bot => {
                let r = alloc(None, false);
                let z = alloc(None, false);
                let ctor = alloc(None, false);
                edges.push((r, REL_STEP, z));
                edges.push((r, REL_NORMAL_FORM, z));
                edges.push((z, REL_ABSORBS, ctor));
                edges.push((z, REL_ABSORBS, h_absorber));
                edges.push((r, REL_ACYCLIC, h_acyclic));
                r
            }
            MetaOp::Iso => {
                let hub = alloc(None, false);
                let d1 = alloc(None, false);
                let d2 = alloc(None, false);
                let nf1 = alloc(None, false);
                let nf2 = alloc(None, false);
                edges.push((hub, REL_STEP, d1));
                edges.push((hub, REL_STEP, d2));
                edges.push((d1, REL_DISJOINT, d2));
                edges.push((d2, REL_DISJOINT, d1));
                edges.push((d1, REL_BIJECTION, d2));
                edges.push((d2, REL_BIJECTION, d1));
                edges.push((d1, REL_NORMAL_FORM, nf1));
                edges.push((d2, REL_NORMAL_FORM, nf2));
                edges.push((hub, REL_ACYCLIC, h_acyclic));
                hub
            }
            MetaOp::Quot => {
                let tmpl = alloc(None, false);
                let i1 = alloc(None, false);
                let i2 = alloc(None, false);
                let i3 = alloc(None, false);
                edges.push((tmpl, REL_TEMPLATE_OF, i1));
                edges.push((tmpl, REL_TEMPLATE_OF, i2));
                edges.push((tmpl, REL_TEMPLATE_OF, i3));
                edges.push((i1, REL_NON_ISO, i2));
                edges.push((i2, REL_NON_ISO, i3));
                edges.push((tmpl, REL_ACYCLIC, h_acyclic));
                tmpl
            }
            MetaOp::Comp(a, b) => {
                let comp_root = alloc(None, false);
                let sa = build_schema(a, h_acyclic, h_cycle, h_absorber, h_no_nf, alloc, edges);
                let sb = build_schema(b, h_acyclic, h_cycle, h_absorber, h_no_nf, alloc, edges);
                edges.push((comp_root, REL_COMP_OUTER, sa));
                edges.push((comp_root, REL_COMP_INNER, sb));
                comp_root
            }
        }
    }
    for op in ops {
        let v_concept = alloc_node(None, true);
        let schema_root = build_schema(
            op,
            h_acyclic,
            h_cycle,
            h_absorber,
            h_no_nf,
            &mut alloc_node,
            &mut edges,
        );
        edges.push((v_concept, REL_DEFINES, schema_root));
        concept_nodes.insert(op.clone(), v_concept);
    }
    DefGraph {
        node_symbols,
        edges,
        concept_nodes,
        is_concept_node,
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PatternNode {
    Var(u32),
    Sym(SymbolId, Vec<usize>),
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatternGraph {
    pub nodes: Vec<PatternNode>,
    pub root: usize,
}
impl PatternGraph {
    pub fn builder() -> PatternBuilder {
        PatternBuilder { nodes: Vec::new() }
    }
    pub fn symbols(&self) -> BTreeSet<SymbolId> {
        self.nodes
            .iter()
            .filter_map(|n| match n {
                PatternNode::Sym(s, _) => Some(*s),
                PatternNode::Var(_) => None,
            })
            .collect()
    }
}
pub struct PatternBuilder {
    nodes: Vec<PatternNode>,
}
impl PatternBuilder {
    pub fn var(&mut self, id: u32) -> usize {
        let idx = self.nodes.len();
        self.nodes.push(PatternNode::Var(id));
        idx
    }
    pub fn sym(&mut self, s: SymbolId, succ: Vec<usize>) -> usize {
        let idx = self.nodes.len();
        self.nodes.push(PatternNode::Sym(s, succ));
        idx
    }
    pub fn finish(self, root: usize) -> PatternGraph {
        PatternGraph {
            nodes: self.nodes,
            root,
        }
    }
}

fn match_pattern(
    pat: &PatternGraph,
    graph: &TermGraph,
    target: usize,
) -> Option<HashMap<u32, usize>> {
    let mut var_bindings: HashMap<u32, usize> = HashMap::new();
    let mut pat_bindings: HashMap<usize, usize> = HashMap::new();
    fn recurse(
        p_idx: usize,
        g_idx: usize,
        pat: &PatternGraph,
        graph: &TermGraph,
        var_bindings: &mut HashMap<u32, usize>,
        pat_bindings: &mut HashMap<usize, usize>,
    ) -> bool {
        if let Some(&bound) = pat_bindings.get(&p_idx) {
            return bound == g_idx;
        }
        pat_bindings.insert(p_idx, g_idx);
        match &pat.nodes[p_idx] {
            PatternNode::Var(vid) => {
                if let Some(&bound) = var_bindings.get(vid) {
                    bound == g_idx
                } else {
                    var_bindings.insert(*vid, g_idx);
                    true
                }
            }
            PatternNode::Sym(sym, p_children) => {
                let g_node = &graph.nodes[g_idx];
                if g_node.label != *sym || g_node.succ.len() != p_children.len() {
                    return false;
                }
                for (&pc, &gc) in p_children.iter().zip(g_node.succ.iter()) {
                    if !recurse(pc, gc, pat, graph, var_bindings, pat_bindings) {
                        return false;
                    }
                }
                true
            }
        }
    }
    if recurse(pat.root, target, pat, graph, &mut var_bindings, &mut pat_bindings) {
        Some(var_bindings)
    } else {
        None
    }
}
fn instantiate_rhs(
    r_idx: usize,
    rhs: &PatternGraph,
    var_bindings: &HashMap<u32, usize>,
    nodes: &mut Vec<Node>,
    rhs_map: &mut [Option<usize>],
) -> usize {
    if let Some(existing) = rhs_map[r_idx] {
        return existing;
    }
    match &rhs.nodes[r_idx] {
        PatternNode::Var(vid) => {
            let bound = var_bindings[vid];
            rhs_map[r_idx] = Some(bound);
            bound
        }
        PatternNode::Sym(sym, r_children) => {
            let slot = nodes.len();
            nodes.push(Node {
                label: *sym,
                succ: Vec::new(),
            });
            rhs_map[r_idx] = Some(slot);
            let mut child_ids = Vec::with_capacity(r_children.len());
            for &rc in r_children {
                child_ids.push(instantiate_rhs(rc, rhs, var_bindings, nodes, rhs_map));
            }
            nodes[slot].succ = child_ids;
            slot
        }
    }
}
pub fn apply_rule_everywhere(graph: &TermGraph, rule: &Rule) -> Vec<TermGraph> {
    let mut results = Vec::new();
    for target in 0..graph.nodes.len() {
        if let Some(var_bindings) = match_pattern(&rule.lhs, graph, target) {
            let mut nodes = graph.nodes.clone();
            let orig_len = nodes.len();
            let mut rhs_map = vec![None; rule.rhs.nodes.len()];
            let new_subroot = instantiate_rhs(
                rule.rhs.root,
                &rule.rhs,
                &var_bindings,
                &mut nodes,
                &mut rhs_map,
            );
            for n in &mut nodes[..orig_len] {
                for child in &mut n.succ {
                    if *child == target {
                        *child = new_subroot;
                    }
                }
            }
            let new_root = if graph.root == target {
                new_subroot
            } else {
                graph.root
            };
            results.push(TermGraph { nodes, root: new_root }.canonicalize());
        }
    }
    results
}
#[derive(Debug, Clone)]
pub struct DerivationGraph {
    pub states: Vec<TermGraph>,
    pub edges: Vec<(usize, usize, u32)>,
    pub root: usize,
    pub used_rules: BTreeSet<u32>,
}
impl DerivationGraph {
    pub fn build(
        root_graph: &TermGraph,
        rules: &[Rule],
        horizon_k: usize,
        max_states: usize,
    ) -> Self {
        let canon_root = root_graph.canonicalize();
        let mut states: Vec<TermGraph> = vec![canon_root.clone()];
        let mut state_index: HashMap<TermGraph, usize> = HashMap::new();
        state_index.insert(canon_root, 0);
        let mut edges: Vec<(usize, usize, u32)> = Vec::new();
        let mut edge_set: BTreeSet<(usize, usize, u32)> = BTreeSet::new();
        let mut used_rules: BTreeSet<u32> = BTreeSet::new();
        let mut queue: VecDeque<(usize, usize)> = VecDeque::new();
        queue.push_back((0, 0));
        while let Some((u, depth)) = queue.pop_front() {
            if depth >= horizon_k {
                continue;
            }
            let current = states[u].clone();
            for rule in rules {
                for next_g in apply_rule_everywhere(&current, rule) {
                    let v = if let Some(&existing) = state_index.get(&next_g) {
                        existing
                    } else {
                        if states.len() >= max_states {
                            continue;
                        }
                        let id = states.len();
                        states.push(next_g.clone());
                        state_index.insert(next_g, id);
                        queue.push_back((id, depth + 1));
                        id
                    };
                    if edge_set.insert((u, v, rule.id)) {
                        edges.push((u, v, rule.id));
                        used_rules.insert(rule.id);
                    }
                }
            }
        }
        Self {
            states,
            edges,
            root: 0,
            used_rules,
        }
    }
    pub fn reachable_symbols(&self) -> BTreeSet<SymbolId> {
        let mut set = BTreeSet::new();
        for g in &self.states {
            set.extend(g.symbols());
        }
        set
    }
    pub fn normal_forms(&self) -> Vec<usize> {
        let mut out_deg = vec![0usize; self.states.len()];
        for &(u, _, _) in &self.edges {
            out_deg[u] += 1;
        }
        (0..self.states.len()).filter(|&u| out_deg[u] == 0).collect()
    }
    pub fn has_directed_cycle(&self) -> bool {
        let n = self.states.len();
        let mut adj = vec![Vec::new(); n];
        for &(u, v, _) in &self.edges {
            adj[u].push(v);
        }
        let mut color = vec![0u8; n];
        fn dfs(u: usize, adj: &[Vec<usize>], color: &mut [u8]) -> bool {
            color[u] = 1;
            for &v in &adj[u] {
                if color[v] == 1 {
                    return true;
                }
                if color[v] == 0 && dfs(v, adj, color) {
                    return true;
                }
            }
            color[u] = 2;
            false
        }
        for i in 0..n {
            if color[i] == 0 && dfs(i, &adj, &mut color) {
                return true;
            }
        }
        false
    }
    fn unlabeled_adj(&self) -> (Vec<BTreeSet<usize>>, Vec<usize>, Vec<usize>) {
        let n = self.states.len();
        let mut adj = vec![BTreeSet::new(); n];
        let mut in_deg = vec![0usize; n];
        let mut out_deg = vec![0usize; n];
        for &(u, v, _) in &self.edges {
            if adj[u].insert(v) {
                out_deg[u] += 1;
                in_deg[v] += 1;
            }
        }
        (adj, in_deg, out_deg)
    }
    pub fn is_unlabeled_isomorphic(&self, other: &DerivationGraph) -> bool {
        let n = self.states.len();
        if n != other.states.len() {
            return false;
        }
        let (adj1, in1, out1) = self.unlabeled_adj();
        let (adj2, in2, out2) = other.unlabeled_adj();
        if in1[self.root] != in2[other.root] || out1[self.root] != out2[other.root] {
            return false;
        }
        let mut sig1: Vec<(usize, usize)> = (0..n).map(|i| (in1[i], out1[i])).collect();
        let mut sig2: Vec<(usize, usize)> = (0..n).map(|i| (in2[i], out2[i])).collect();
        sig1.sort_unstable();
        sig2.sort_unstable();
        if sig1 != sig2 {
            return false;
        }
        let mut mapping: Vec<Option<usize>> = vec![None; n];
        let mut used = vec![false; n];
        mapping[self.root] = Some(other.root);
        used[other.root] = true;
        let mut order: Vec<usize> = (0..n).filter(|&i| i != self.root).collect();
        order.sort_by_key(|&i| std::cmp::Reverse(in1[i] + out1[i]));
        fn backtrack(
            pos: usize,
            order: &[usize],
            adj1: &[BTreeSet<usize>],
            adj2: &[BTreeSet<usize>],
            in1: &[usize],
            out1: &[usize],
            in2: &[usize],
            out2: &[usize],
            mapping: &mut [Option<usize>],
            used: &mut [bool],
        ) -> bool {
            if pos == order.len() {
                return true;
            }
            let u = order[pos];
            for v in 0..adj2.len() {
                if used[v] || in1[u] != in2[v] || out1[u] != out2[v] {
                    continue;
                }
                let mut ok = true;
                for (u_prev, m_opt) in mapping.iter().enumerate() {
                    if let Some(v_prev) = *m_opt {
                        if adj1[u].contains(&u_prev) != adj2[v].contains(&v_prev)
                            || adj1[u_prev].contains(&u) != adj2[v_prev].contains(&v)
                        {
                            ok = false;
                            break;
                        }
                    }
                }
                if ok {
                    mapping[u] = Some(v);
                    used[v] = true;
                    if backtrack(
                        pos + 1, order, adj1, adj2, in1, out1, in2, out2, mapping, used,
                    ) {
                        return true;
                    }
                    mapping[u] = None;
                    used[v] = false;
                }
            }
            false
        }
        backtrack(
            0, &order, &adj1, &adj2, &in1, &out1, &in2, &out2, &mut mapping, &mut used,
        )
    }
}
pub fn detect_absorber_symbols(
    rules: &[Rule],
    arities: &BTreeMap<SymbolId, usize>,
) -> BTreeSet<SymbolId> {
    let mut nullary_candidates: BTreeSet<SymbolId> = arities
        .iter()
        .filter_map(|(&s, &a)| if a == 0 { Some(s) } else { None })
        .collect();
    for r in rules {
        if let PatternNode::Sym(s, ch) = &r.lhs.nodes[r.lhs.root] {
            if ch.is_empty() {
                nullary_candidates.remove(s);
            }
        }
    }
    let mut absorbers = BTreeSet::new();
    for z in nullary_candidates {
        let mut absorbed_positions: HashMap<SymbolId, BTreeSet<usize>> = HashMap::new();
        for r in rules {
            let rhs_is_z = matches!(&r.rhs.nodes[r.rhs.root], PatternNode::Sym(s, ch) if *s == z && ch.is_empty());
            if !rhs_is_z {
                continue;
            }
            if let PatternNode::Sym(f, ch) = &r.lhs.nodes[r.lhs.root] {
                if !ch.is_empty() {
                    let mut z_pos = None;
                    let mut all_others_vars = true;
                    for (idx, &c) in ch.iter().enumerate() {
                        match &r.lhs.nodes[c] {
                            PatternNode::Sym(s, sub) if *s == z && sub.is_empty() && z_pos.is_none() => {
                                z_pos = Some(idx);
                            }
                            PatternNode::Var(_) => {}
                            _ => {
                                all_others_vars = false;
                                break;
                            }
                        }
                    }
                    if all_others_vars {
                        if let Some(pos) = z_pos {
                            absorbed_positions.entry(*f).or_default().insert(pos);
                        }
                    }
                }
            }
        }
        let is_absorber = absorbed_positions.iter().any(|(f, positions)| {
            arities.get(f).copied().unwrap_or(0) == positions.len() && !positions.is_empty()
        });
        if is_absorber {
            absorbers.insert(z);
        }
    }
    absorbers
}

#[derive(Debug, Clone)]
pub struct RigidityReport {
    pub is_rigid: bool,
    pub iterations: usize,
    pub total_color_classes: usize,
    pub concept_multiplicities: BTreeMap<MetaOp, usize>,
}
pub fn verify_rigidity_2fwl(def_graph: &DefGraph) -> RigidityReport {
    let n = def_graph.node_symbols.len();
    if n == 0 {
        return RigidityReport {
            is_rigid: false,
            iterations: 0,
            total_color_classes: 0,
            concept_multiplicities: BTreeMap::new(),
        };
    }
    let mut edge_labels: Vec<Vec<Vec<u32>>> = vec![vec![Vec::new(); n]; n];
    for &(u, rel, v) in &def_graph.edges {
        edge_labels[u][v].push(rel.0);
    }
    for row in &mut edge_labels {
        for cell in row {
            cell.sort_unstable();
        }
    }
    let mut init_keys: Vec<(u8, Vec<u32>, Vec<u32>)> = Vec::with_capacity(n * n);
    for i in 0..n {
        for j in 0..n {
            if i == j {
                let role = if def_graph.is_concept_node[i] { 0u8 } else { 1u8 };
                init_keys.push((role, Vec::new(), Vec::new()));
            } else {
                init_keys.push((2u8, edge_labels[i][j].clone(), edge_labels[j][i].clone()));
            }
        }
    }
    let (mut colors, mut num_classes) = compress_colors(&init_keys);
    let mut iterations = 0usize;
    loop {
        iterations += 1;
        let mut next_keys: Vec<(u32, Vec<(u32, u32)>)> = Vec::with_capacity(n * n);
        for i in 0..n {
            for j in 0..n {
                let mut multiset = Vec::with_capacity(n);
                for k in 0..n {
                    multiset.push((colors[i * n + k], colors[k * n + j]));
                }
                multiset.sort_unstable();
                next_keys.push((colors[i * n + j], multiset));
            }
        }
        let (next_colors, next_classes) = compress_colors(&next_keys);
        colors = next_colors;
        if next_classes == num_classes {
            break;
        }
        num_classes = next_classes;
        if iterations >= n * n {
            break;
        }
    }
    let mut diag_counts: HashMap<u32, usize> = HashMap::new();
    for v in 0..n {
        *diag_counts.entry(colors[v * n + v]).or_insert(0) += 1;
    }
    let mut concept_multiplicities = BTreeMap::new();
    let mut is_rigid = true;
    for (op, &node_idx) in &def_graph.concept_nodes {
        let mult = diag_counts[&colors[node_idx * n + node_idx]];
        if mult != 1 {
            is_rigid = false;
        }
        concept_multiplicities.insert(op.clone(), mult);
    }
    RigidityReport {
        is_rigid,
        iterations,
        total_color_classes: num_classes,
        concept_multiplicities,
    }
}
fn compress_colors<K: Ord>(keys: &[K]) -> (Vec<u32>, usize) {
    let mut indexed: Vec<(&K, usize)> = keys.iter().enumerate().map(|(i, k)| (k, i)).collect();
    indexed.sort_unstable_by(|a, b| a.0.cmp(b.0));
    let mut result = vec![0u32; keys.len()];
    let mut current_id = 0u32;
    for idx in 0..indexed.len() {
        if idx > 0 && indexed[idx].0 != indexed[idx - 1].0 {
            current_id += 1;
        }
        result[indexed[idx].1] = current_id;
    }
    let count = if keys.is_empty() {
        0
    } else {
        (current_id as usize) + 1
    };
    (result, count)
}

#[derive(Debug, Clone)]
pub struct Rng {
    s: [u64; 4],
}
impl Rng {
    pub fn from_seed(mut seed: u64) -> Self {
        let mut splitmix = || -> u64 {
            seed = seed.wrapping_add(0x9E37_79B9_7F4A_7C15);
            let mut z = seed;
            z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
            z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
            z ^ (z >> 31)
        };
        Self {
            s: [splitmix(), splitmix(), splitmix(), splitmix()],
        }
    }
    #[inline]
    pub fn next_u64(&mut self) -> u64 {
        let result = self.s[1].wrapping_mul(5).rotate_left(7).wrapping_mul(9);
        let t = self.s[1] << 17;
        self.s[2] ^= self.s[0];
        self.s[3] ^= self.s[1];
        self.s[1] ^= self.s[2];
        self.s[0] ^= self.s[3];
        self.s[2] ^= t;
        self.s[3] = self.s[3].rotate_left(45);
        result
    }
    pub fn gen_range(&mut self, upper: usize) -> usize {
        if upper <= 1 {
            return 0;
        }
        let u = upper as u64;
        let zone = u64::MAX - (u64::MAX % u);
        loop {
            let v = self.next_u64();
            if v < zone {
                return (v % u) as usize;
            }
        }
    }
    pub fn gen_range_inclusive(&mut self, low: usize, high: usize) -> usize {
        if high <= low {
            return low;
        }
        low + self.gen_range(high - low + 1)
    }
    pub fn gen_f64(&mut self) -> f64 {
        (self.next_u64() >> 11) as f64 * (1.0 / ((1u64 << 53) as f64))
    }
    pub fn gen_bool(&mut self, prob: f64) -> bool {
        self.gen_f64() < prob
    }
    pub fn shuffle<T>(&mut self, slice: &mut [T]) {
        for i in (1..slice.len()).rev() {
            let j = self.gen_range(i + 1);
            slice.swap(i, j);
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct SymbolId(pub u32);
#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct Node {
    pub label: SymbolId,
    pub succ: Vec<usize>,
}
#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct TermGraph {
    pub nodes: Vec<Node>,
    pub root: usize,
}
impl TermGraph {
    pub fn new(nodes: Vec<Node>, root: usize) -> Self {
        Self { nodes, root }.canonicalize()
    }
    pub fn canonicalize(&self) -> Self {
        if self.nodes.is_empty() {
            return self.clone();
        }
        let mut mapping: Vec<Option<usize>> = vec![None; self.nodes.len()];
        let mut order: Vec<usize> = Vec::with_capacity(self.nodes.len());
        fn dfs(
            u: usize,
            nodes: &[Node],
            mapping: &mut [Option<usize>],
            order: &mut Vec<usize>,
        ) {
            if mapping[u].is_some() {
                return;
            }
            let new_id = order.len();
            mapping[u] = Some(new_id);
            order.push(u);
            for &v in &nodes[u].succ {
                dfs(v, nodes, mapping, order);
            }
        }
        dfs(self.root, &self.nodes, &mut mapping, &mut order);
        let new_nodes = order
            .iter()
            .map(|&old_id| {
                let n = &self.nodes[old_id];
                Node {
                    label: n.label,
                    succ: n.succ.iter().map(|&c| mapping[c].unwrap()).collect(),
                }
            })
            .collect();
        Self {
            nodes: new_nodes,
            root: 0,
        }
    }
    pub fn symbols(&self) -> BTreeSet<SymbolId> {
        self.nodes.iter().map(|n| n.label).collect()
    }
    pub fn builder() -> TermGraphBuilder {
        TermGraphBuilder { nodes: Vec::new() }
    }
}

pub enum TemplateNode {
    Sym(SymbolId, Vec<usize>),
    Hole(usize),
}
#[derive(Debug, Clone)]
pub struct TemplateGraph {
    pub nodes: Vec<TemplateNode>,
    pub root: usize,
    pub hole_count: usize,
}
impl TemplateGraph {
    pub fn internal_constructor_count(&self) -> usize {
        self.nodes
            .iter()
            .filter(|n| matches!(n, TemplateNode::Sym(_, succ) if !succ.is_empty()))
            .count()
    }
    pub fn is_nontrivial(&self) -> bool {
        self.internal_constructor_count() >= 2 && self.hole_count >= 1
    }
}
pub fn anti_unify(graphs: &[TermGraph]) -> Option<TemplateGraph> {
    if graphs.is_empty() {
        return None;
    }
    let mut memo: HashMap<Vec<usize>, usize> = HashMap::new();
    let mut nodes: Vec<TemplateNode> = Vec::new();
    let mut hole_count: usize = 0;
    fn recurse(
        tuple: Vec<usize>,
        graphs: &[TermGraph],
        memo: &mut HashMap<Vec<usize>, usize>,
        nodes: &mut Vec<TemplateNode>,
        hole_count: &mut usize,
    ) -> usize {
        if let Some(&existing) = memo.get(&tuple) {
            return existing;
        }
        let first_node = &graphs[0].nodes[tuple[0]];
        let all_match = graphs.iter().enumerate().all(|(i, g)| {
            let n = &g.nodes[tuple[i]];
            n.label == first_node.label && n.succ.len() == first_node.succ.len()
        });
        if all_match {
            let arity = first_node.succ.len();
            let slot = nodes.len();
            nodes.push(TemplateNode::Sym(first_node.label, Vec::new()));
            memo.insert(tuple.clone(), slot);
            let mut children = Vec::with_capacity(arity);
            for arg_idx in 0..arity {
                let child_tuple: Vec<usize> = graphs
                    .iter()
                    .enumerate()
                    .map(|(i, g)| g.nodes[tuple[i]].succ[arg_idx])
                    .collect();
                let cid = recurse(child_tuple, graphs, memo, nodes, hole_count);
                children.push(cid);
            }
            if let TemplateNode::Sym(_, ref mut succ) = nodes[slot] {
                *succ = children;
            }
            slot
        } else {
            let hid = *hole_count;
            *hole_count += 1;
            let slot = nodes.len();
            nodes.push(TemplateNode::Hole(hid));
            memo.insert(tuple, slot);
            slot
        }
    }
    let roots: Vec<usize> = graphs.iter().map(|g| g.root).collect();
    let root = recurse(roots, graphs, &mut memo, &mut nodes, &mut hole_count);
    Some(TemplateGraph {
        nodes,
        root,
        hole_count,
    })
}

    eprintln!("  gen-eff --out perceived.eff [--op bif|iso|bot|fix|quot] [--horizon 4] [--distractors 4] [--noise 0.5] [--seed 42]");
}
fn cmd_ask(args: &[String]) {
    let mut model_path: Option<String> = None;
    let mut file_path: Option<String> = None;
    let mut steps_override: Option<usize> = None;
    let mut out_len_override: Option<usize> = None;
    let mut raw_output = false;
    let mut i = 0;
    while i < args.len() {
        match args[i].as_str() {
            "--model" if i + 1 < args.len() => {
                model_path = Some(args[i + 1].clone());
                i += 2;
            }
            "--file" if i + 1 < args.len() => {
                file_path = Some(args[i + 1].clone());
                i += 2;
            }
            "--steps" if i + 1 < args.len() => {
                steps_override = Some(args[i + 1].parse().expect("--steps invalide"));
                i += 2;
            }
            "--out-len" if i + 1 < args.len() => {
                out_len_override = Some(args[i + 1].parse().expect("--out-len invalide"));
                i += 2;
            }
            "--raw" => {
                raw_output = true;
                i += 1;
            }
            other => {
                eprintln!("Option ignorée : {}", other);
                i += 1;
            }
        }
    }
    let model_path = model_path.unwrap_or_else(|| {
        eprintln!("Erreur : --model <chemin.phis> requis.");
        process::exit(1);
    });
    let file_path = file_path.unwrap_or_else(|| {
        eprintln!("Erreur : --file <perceived.eff> requis.");
        process::exit(1);
    });
    let mut ckpt = PhisCheckpoint::load(&model_path).unwrap_or_else(|e| {
        eprintln!("Erreur de chargement du modèle {} : {}", model_path, e);
        process::exit(1);
    });
    let perceived_eff = fs::read(&file_path).unwrap_or_else(|e| {
        eprintln!("Erreur de lecture de {} : {}", file_path, e);
        process::exit(1);
    });
    if let Some(s) = steps_override {
        ckpt.flow_cfg.max_steps = s;
    }
    if let Some(l) = out_len_override {
        ckpt.proj_cfg.out_len = l;
    }
    let effect = EffectSummary::from_bytes(&perceived_eff, ckpt.params.n);
    let probe_seed = deterministic_content_seed(&perceived_eff);
    let probes = StabilityProbes::build(ckpt.params.n, 2, 0.06, 0.12, probe_seed);
    let traj = run_abductive_flow(
        &ckpt.params,
        &effect,
        &probes,
        &ckpt.phi_cfg,
        &ckpt.flow_cfg,
        None,
    );
    let proj = project_witness_mirror_ascent(
        &ckpt.params,
        &traj.r_final,
        &traj.final_eval,
        &perceived_eff,
        &ckpt.proj_cfg,
    );
    if raw_output {
        io::stdout()
            .write_all(&proj.witness_bytes)
            .expect("Erreur d'écriture stdout");
        return;
    }
    let first = traj.trace.first().unwrap();
    let last = traj.trace.last().unwrap();
    println!("=== Phisharp :: Trajectoire Abductive ===");
    println!(
        "Entrée        : {} ({} octets) | Graphe N={}, d={}, h={}",
        file_path,
        perceived_eff.len(),
        ckpt.params.n,
        ckpt.params.d,
        ckpt.params.h
    );
    println!(
        "Trajectoire   : {} étapes (t={:.3}, variété critique={}, LaSalle monotone={})",
        traj.trace.len().saturating_sub(1),
        last.t,
        traj.converged_to_critical_manifold,
        traj.is_monotone_lyapunov(1e-9)
    );
    println!(
        "phi_sharp     : {:.6} -> {:.6}  (lambda_2={:.6}, arêtes causales={})",
        first.phi_sharp, last.phi_sharp, last.lambda2, last.active_causal_edges
    );
    println!(
        "Résidu / Stab : rho {:.5} -> {:.5} | V_stab={:.5} | R_abs={:.5}",
        first.residual, last.residual, last.v_stab, last.r_abs
    );
    println!(
        "Projection    : {} itérations miroir (Psi={:.5}, confiance={:.1}%, monotone={})",
        proj.trace.len().saturating_sub(1),
        proj.final_psi,
        proj.mean_confidence * 100.0,
        proj.is_monotone_psi(1e-9)
    );
    println!("Témoin (hex)  : {:02X?}", proj.witness_bytes);
    println!("Témoin (NL)   : {}", proj.display_lossy_nl());
}
fn cmd_train(args: &[String]) {
    let mut out_path = String::from("out/sharp.phis");
    let mut n: usize = 6;
    let mut d: usize = 2;
    let mut h: usize = 8;
    let mut train_cfg = TrainConfig::default();
    let mut flow_steps: usize = 16;
    let mut i = 0;
    while i < args.len() {
        match args[i].as_str() {
            "--out" if i + 1 < args.len() => {
                out_path = args[i + 1].clone();
                i += 2;
            }
            "--episodes" if i + 1 < args.len() => {
                train_cfg.episodes = args[i + 1].parse().expect("--episodes invalide");
                i += 2;
            }
            "--n" if i + 1 < args.len() => {
                n = args[i + 1].parse().expect("--n invalide");
                i += 2;
            }
            "--d" if i + 1 < args.len() => {
                d = args[i + 1].parse().expect("--d invalide");
                i += 2;
            }
            "--h" if i + 1 < args.len() => {
                h = args[i + 1].parse().expect("--h invalide");
                i += 2;
            }
            "--lr" if i + 1 < args.len() => {
                train_cfg.lr = args[i + 1].parse().expect("--lr invalide");
                i += 2;
            }
            "--seed" if i + 1 < args.len() => {
                train_cfg.seed = args[i + 1].parse().expect("--seed invalide");
                i += 2;
            }
            "--flow-steps" if i + 1 < args.len() => {
                flow_steps = args[i + 1].parse().expect("--flow-steps invalide");
                i += 2;
            }
            other => {
                eprintln!("Option ignorée : {}", other);
                i += 1;
            }
        }
    }
    let mut rng = Rng::from_seed(train_cfg.seed);
    let mut params = ModelParams::random_init(n, d, h, &mut rng);
    let phi_cfg = PhiSharpConfig::default();
    let flow_cfg = FlowConfig {
        max_steps: flow_steps,
        dt_init: 0.8,
        dt_max: 8.0,
        ..FlowConfig::default()
    };
    let proj_cfg = ProjectionConfig::default();
    println!(
        "=== Phisharp :: Séance unique de méta-entraînement HO-PTGRS ({} épisodes) ===",
        train_cfg.episodes
    );
    println!("Architecture : N={}, d={}, h={} | signal exclusif : phi_sharp(E)", n, d, h);
    let _logs = train_session(&mut params, &phi_cfg, &flow_cfg, &train_cfg, |log| {
        println!(
            "[ep {:>3} | k={} D={}] bytes={:<4} | phi {:+.4} -> {:+.4} | lam2={:.5} | rho {:.3}->{:.3} | edges={:<2} | |g|={:.3}",
            log.episode,
            log.horizon_k,
            log.distractors,
            log.eff_bytes,
            log.phi_init,
            log.phi_final,
            log.lambda2_final,
            log.residual_init,
            log.residual_final,
            log.active_edges,
            log.grad_norm
        );
    });
    let eval = evaluate_post_hoc(
        &params,
        &phi_cfg,
        &flow_cfg,
        &proj_cfg,
        6,
        train_cfg.seed ^ 0xDEAD_BEEF,
    );
    println!("=== Évaluation post-hoc ({} épisodes HO-PTGRS frais) ===", eval.episodes_tested);
    println!(
        "Gain moyen phi_sharp : {:+.5} | lambda_2 moyen : {:.5} | chute rho : {:.4}",
        eval.mean_phi_gain, eval.mean_lambda2, eval.mean_residual_drop
    );
    println!(
        "Témoin dans alphabet(E) : {:.1}% | recouvrement concept caché : {:.1}%",
        eval.mean_witness_alphabet_ratio * 100.0,
        eval.mean_target_byte_overlap * 100.0
    );
    let ckpt = PhisCheckpoint {
        params,
        phi_cfg,
        flow_cfg,
        proj_cfg,
    };
    ckpt.save(&out_path).unwrap_or_else(|e| {
        eprintln!("Échec de sauvegarde {} : {}", out_path, e);
        process::exit(1);
    });
    println!("Modèle sauvegardé dans : {}", out_path);
}
fn cmd_gen_eff(args: &[String]) {
    let mut out_path = String::from("perceived.eff");
    let mut seed: u64 = 42;
    let mut horizon_k: usize = 4;
    let mut num_distractors: usize = 4;
    let mut noise_ratio: f64 = 0.5;
    let mut forced_op: Option<MetaOp> = None;
    let mut i = 0;
    while i < args.len() {
        match args[i].as_str() {
            "--out" if i + 1 < args.len() => {
                out_path = args[i + 1].clone();
                i += 2;
            }
            "--seed" if i + 1 < args.len() => {
                seed = args[i + 1].parse().expect("--seed invalide");
                i += 2;
            }
            "--horizon" if i + 1 < args.len() => {
                horizon_k = args[i + 1].parse().expect("--horizon invalide");
                i += 2;
            }
            "--distractors" if i + 1 < args.len() => {
                num_distractors = args[i + 1].parse().expect("--distractors invalide");
                i += 2;
            }
            "--noise" if i + 1 < args.len() => {
                noise_ratio = args[i + 1].parse().expect("--noise invalide");
                i += 2;
            }
            "--op" if i + 1 < args.len() => {
                forced_op = Some(parse_op(&args[i + 1]));
                i += 2;
            }
            _ => {
                i += 1;
            }
        }
    }
    let mut gen_ = HoPtgrsGenerator::new(
        seed,
        CurriculumConfig {
            horizon_k,
            num_distractors,
            noise_ratio,
            max_attempts: 64,
        },
    );
    let ep = gen_
        .generate_episode(forced_op)
        .expect("Échec génération HO-PTGRS");
    fs::write(&out_path, &ep.perceived_eff).expect("Échec écriture perceived.eff");
    println!(
        "Écrit {} ({} octets) | cause cachée post-hoc={:?} | code cible={:02X?}",
        out_path,
        ep.perceived_eff.len(),
        ep.hidden_cause.target_op,
        ep.hidden_cause.target_concept_bytes
    );
}
fn parse_op(s: &str) -> MetaOp {
    match s.to_lowercase().as_str() {
        "bif" => MetaOp::Bif,
        "iso" => MetaOp::Iso,
        "bot" => MetaOp::Bot,
        "fix" => MetaOp::Fix,
        "quot" => MetaOp::Quot,
        "comp_bif_fix" => MetaOp::comp(MetaOp::Bif, MetaOp::Fix),
        "comp_iso_bif" => MetaOp::comp(MetaOp::Iso, MetaOp::Bif),
        "comp_iso_fix" => MetaOp::comp(MetaOp::Iso, MetaOp::Fix),
        "comp_quot_bif" => MetaOp::comp(MetaOp::Quot, MetaOp::Bif),
        other => panic!("Opérateur inconnu : {}", other),
    }
}
fn deterministic_content_seed(bytes: &[u8]) -> u64 {
    let mut h: u64 = 0xcbf2_9ce4_8422_2325;
    for &b in bytes {
        h ^= b as u64;
        h = h.wrapping_mul(0x0000_0100_0000_01b3);
    }
    h
}
