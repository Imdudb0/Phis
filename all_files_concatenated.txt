#[derive(Clone, Debug)]
pub struct RawSubstrate {
    pub len: usize,
    pub offsets: Vec<usize>,
    pub edges: Vec<[f32; EDGE_IN]>,
    pub edge_targets: Vec<u32>,
    pub deg: Vec<f32>,
}
impl RawSubstrate {
    pub fn from_bytes(bytes: &[u8]) -> Self {
        let n = bytes.len();
        let win_cap = 30usize;
        let deg_rad = 12usize;
        let max_edges = 12usize;
        let log_cap = (win_cap as f32).ln();
        let m_nb = |a: isize, b: isize, ti: isize, ji: isize| -> f32 {
            if a >= 0 && b >= 0 && a != b && !(a == ji && b == ti) {
                let au = a as usize;
                let bu = b as usize;
                if au < n && bu < n && bytes[au] == bytes[bu] {
                    return 1.0;
                }
            }
            0.0
        };
        let mut offsets = Vec::with_capacity(n + 1);
        let mut edges = Vec::with_capacity(n * 6);
        let mut edge_targets = Vec::with_capacity(n * 6);
        let mut deg = Vec::with_capacity(n);
        let mut candidates: Vec<(usize, usize, f32)> = Vec::with_capacity(32);
        offsets.push(0);
        for t in 0..n {
            let b = bytes[t];
            let lo = t.saturating_sub(win_cap);
            let hi = (t + win_cap + 1).min(n);
            let deg_lo = t.saturating_sub(deg_rad);
            let deg_hi = (t + deg_rad + 1).min(n);
            let deg_span = (deg_hi - deg_lo).saturating_sub(1).max(1);
            let ti = t as isize;
            candidates.clear();
            let mut local_matches = 0usize;
            for j in lo..hi {
                if j != t && bytes[j] == b {
                    let dist = t.abs_diff(j);
                    if dist <= deg_rad {
                        local_matches += 1;
                    }
                    let ji = j as isize;
                    let neigh = m_nb(ti - 1, ji - 1, ti, ji)
                        + m_nb(ti + 1, ji + 1, ti, ji)
                        + m_nb(ti - 2, ji - 2, ti, ji)
                        + m_nb(ti + 2, ji + 2, ti, ji)
                        + m_nb(ti - 1, ji + 1, ti, ji)
                        + m_nb(ti + 1, ji - 1, ti, ji)
                        + m_nb(ti - 2, ji + 2, ti, ji)
                        + m_nb(ti + 2, ji - 2, ti, ji);
                    candidates.push((j, dist, neigh));
                }
            }
            deg.push(2.0 * (local_matches as f32 / deg_span as f32) - 1.0);
            let match_count = candidates.len();
            let mut selected: Vec<usize> = Vec::with_capacity(max_edges.min(match_count));
            if match_count <= max_edges {
                for &(j, _, _) in &candidates {
                    selected.push(j);
                }
            } else {
                let mut by_dist = candidates.clone();
                by_dist.sort_unstable_by_key(|c| c.1);
                for &(j, _, _) in by_dist.iter().take(6) {
                    selected.push(j);
                }
                candidates.sort_by(|a, b| b.2.total_cmp(&a.2).then_with(|| a.1.cmp(&b.1)));
                for &(j, _, _) in &candidates {
                    if selected.len() >= max_edges {
                        break;
                    }
                    if !selected.contains(&j) {
                        selected.push(j);
                    }
                }
            }
            for j in selected {
                let ji = j as isize;
                let dist = t.abs_diff(j) as f32;
                let log_dist = (dist.ln() / log_cap).clamp(0.0, 1.0);
                let dir = if j < t { -1.0 } else { 1.0 };
                edges.push([
                    m_nb(ti - 1, ji - 1, ti, ji),
                    m_nb(ti + 1, ji + 1, ti, ji),
                    m_nb(ti - 2, ji - 2, ti, ji),
                    m_nb(ti + 2, ji + 2, ti, ji),
                    m_nb(ti - 1, ji + 1, ti, ji),
                    m_nb(ti + 1, ji - 1, ti, ji),
                    m_nb(ti - 2, ji + 2, ti, ji),
                    m_nb(ti + 2, ji - 2, ti, ji),
                    m_nb(ti - 1, ji, ti, ji),
                    m_nb(ti, ji - 1, ti, ji),
                    log_dist,
                    dir,
                ]);
                edge_targets.push(j as u32);
            }
            offsets.push(edges.len());
        }
        Self {
            len: n,
            offsets,
            edges,
            edge_targets,
            deg,
        }
    }
}

pub struct AbductionTrace {

impl PhiGraph {
    pub fn random(rng: &mut Rng) -> Self {
        let mut weights = Vec::with_capacity(NUM_PARAMS);
        for _ in 0..NUM_PARAMS {
            weights.push(rng.next_normal() * 0.22);
        }
        Self { weights }
    }
    pub fn from_weights(weights: Vec<f32>) -> Self {
        assert_eq!(weights.len(), NUM_PARAMS);
        Self { weights }
    }
    pub fn encode_with_gate(&self, sub: &RawSubstrate) -> (Vec<[f32; REL_DIM]>, Vec<f32>, Vec<f32>) {
        let w = &self.weights;
        let n = sub.len;
        if n == 0 {
            return (Vec::new(), Vec::new(), Vec::new());
        }
        let mut pre = Vec::with_capacity(n);
        let mut e_max_all = Vec::with_capacity(n);
        let mut has_all = Vec::with_capacity(n);
        let mut energy = 0.0f32;
        for t in 0..n {
            let slice = &sub.edges[sub.offsets[t]..sub.offsets[t + 1]];
            let mut e_max = [-1.0f32; EDGE_HID];
            let mut e_mean = [-1.0f32; EDGE_HID];
            let has_edge;
            if !slice.is_empty() {
                has_edge = 1.0f32;
                e_max = [f32::NEG_INFINITY; EDGE_HID];
                e_mean = [0.0f32; EDGE_HID];
                let inv_c = 1.0 / (slice.len() as f32);
                for feat in slice {
                    for h in 0..EDGE_HID {
                        let base = OFF_REL_EDGE_W + h * EDGE_IN;
                        let mut acc = w[OFF_REL_EDGE_B + h];
                        for k in 0..EDGE_IN {
                            acc += w[base + k] * feat[k];
                        }
                        let val = acc.tanh();
                        if val > e_max[h] {
                            e_max[h] = val;
                        }
                        e_mean[h] += val * inv_c;
                    }
                }
            } else {
                has_edge = -1.0f32;
            }
            e_max_all.push(e_max);
            has_all.push(has_edge);
            let node_in = [
                e_max[0],
                e_max[1],
                e_max[2],
                e_max[3],
                e_mean[0],
                e_mean[1],
                e_mean[2],
                e_mean[3],
                sub.deg[t],
                has_edge,
            ];
            let mut row = [0.0f32; REL_DIM];
            for r in 0..REL_DIM {
                let base = OFF_REL_NODE_W + r * NODE_IN;
                let mut acc = 0.25 * w[OFF_REL_NODE_B + r];
                for k in 0..NODE_IN {
                    acc += w[base + k] * node_in[k];
                }
                row[r] = acc;
                energy += acc * acc;
            }
            pre.push(row);
        }
        let rms = (energy / ((n * REL_DIM) as f32) + 1e-4).sqrt();
        let inv = 1.25 / rms;
        for row in &mut pre {
            for val in row.iter_mut() {
                *val = (*val * inv).tanh();
            }
        }
        let eval_gate = |g_in: &[f32; GATE_IN]| -> f32 {
            let mut logit = w[OFF_GATE_B2];
            for h in 0..GATE_HID {
                let base = OFF_GATE_W1 + h * GATE_IN;
                let mut acc = w[OFF_GATE_B1 + h];
                for k in 0..GATE_IN {
                    acc += w[base + k] * g_in[k];
                }
                logit += w[OFF_GATE_W2 + h] * acc.tanh();
            }
            1.0 / (1.0 + (-logit.clamp(-12.0, 12.0)).exp())
        };
        let mut gates_ctx = Vec::with_capacity(n);
        let mut gates_pt = Vec::with_capacity(n);
        for t in 0..n {
            let lo = t.saturating_sub(2);
            let hi = (t + 2).min(n - 1);
            let inv_w = 1.0 / ((hi - lo + 1) as f32);
            let mut loc_max = [0.0f32; EDGE_HID];
            let mut loc_deg = 0.0f32;
            let mut loc_has = 0.0f32;
            for j in lo..=hi {
                for h in 0..EDGE_HID {
                    loc_max[h] += e_max_all[j][h] * inv_w;
                }
                loc_deg += sub.deg[j] * inv_w;
                loc_has += has_all[j] * inv_w;
            }
            let g_in_ctx = [
                loc_max[0],
                loc_max[1],
                loc_max[2],
                loc_max[3],
                e_max_all[t][0],
                e_max_all[t][1],
                e_max_all[t][2],
                e_max_all[t][3],
                loc_deg,
                loc_has,
            ];
            let g_in_pt = [
                e_max_all[t][0],
                e_max_all[t][1],
                e_max_all[t][2],
                e_max_all[t][3],
                e_max_all[t][0],
                e_max_all[t][1],
                e_max_all[t][2],
                e_max_all[t][3],
                sub.deg[t],
                has_all[t],
            ];
            gates_ctx.push(eval_gate(&g_in_ctx));
            gates_pt.push(eval_gate(&g_in_pt));
        }
        (pre, gates_ctx, gates_pt)
    }
    pub fn encode_relations(&self, sub: &RawSubstrate) -> Vec<[f32; REL_DIM]> {
        self.encode_with_gate(sub).0
    }
    pub fn encode_bytes(&self, bytes: &[u8]) -> Vec<[f32; REL_DIM]> {
        let sub = RawSubstrate::from_bytes(bytes);
        self.encode_relations(&sub)
    }
    pub fn evaluate_hypothesis(
        &self,
        state: &[f32; STATE_DIM],
        rels: &[[f32; REL_DIM]],
    ) -> (Vec<[f32; REL_DIM]>, Vec<f32>, f32) {
        let w = &self.weights;
        let mut g1 = [0.0f32; DIM];
        let mut g2 = [0.0f32; DIM];
        let mut bh = [0.0f32; DIM];
        for d in 0..DIM {
            let mut s1 = 0.0f32;
            let mut s2 = 0.0f32;
            let mut s3 = 0.0f32;
            for k in 0..(2 * DIM) {
                s1 += w[OFF_G1 + d * (2 * DIM) + k] * state[k];
                s2 += w[OFF_G2 + d * (2 * DIM) + k] * state[2 * DIM + k];
                s3 += w[OFF_BH + d * (2 * DIM) + k] * state[4 * DIM + k];
            }
            g1[d] = s1.tanh();
            g2[d] = s2.tanh();
            bh[d] = s3.tanh();
        }
        let mut u = [0.0f32; DIM];
        let mut preds = Vec::with_capacity(rels.len());
        let mut pos_err = Vec::with_capacity(rels.len());
        let mut total_err = 0.0f32;
        for t in 0..rels.len() {
            let prev = if t == 0 {
                [0.0f32; REL_DIM]
            } else {
                rels[t - 1]
            };
            let mut u_next = [0.0f32; DIM];
            for d in 0..DIM {
                let mut ctx = 0.0f32;
                for r in 0..REL_DIM {
                    ctx += w[OFF_CTX + d * REL_DIM + r] * prev[r];
                }
                let mut rec = 0.0f32;
                for k in 0..DIM {
                    rec += w[OFF_U + d * DIM + k] * u[k];
                }
                u_next[d] = (g1[d] * ctx + g2[d] * rec + bh[d]).tanh();
            }
            u = u_next;
            let mut pred = [0.0f32; REL_DIM];
            let mut err_t = 0.0f32;
            for r in 0..REL_DIM {
                let mut acc = 0.0f32;
                for d in 0..DIM {
                    acc += w[OFF_OUT + r * DIM + d] * u[d];
                }
                pred[r] = acc.tanh();
                let diff = rels[t][r] - pred[r];
                err_t += diff * diff;
            }
            err_t /= REL_DIM as f32;
            preds.push(pred);
            pos_err.push(err_t);
            total_err += err_t;
        }
        let mean_err = if rels.is_empty() {
            0.0
        } else {
            total_err / (rels.len() as f32)
        };
        (preds, pos_err, mean_err)
    }
    pub fn abduce_with_steps(
        &self,
        rels: &[[f32; REL_DIM]],
        cut_mask: Option<u8>,
        steps: usize,
    ) -> AbductionTrace {
        let w = &self.weights;
        let mut state = [0.0f32; STATE_DIM];
        let mut step_residuals = Vec::with_capacity(steps + 1);
        let mid = (rels.len() / 2).max(1);
        let inv_a = 1.0 / (mid as f32);
        let inv_b = 1.0 / (rels.len().saturating_sub(mid).max(1) as f32);
        let len_inv = 1.0 / (rels.len().max(1) as f32);
        for _ in 0..steps {
            let (preds, _, mean_err) = self.evaluate_hypothesis(&state, rels);
            step_residuals.push(mean_err);
            let mut inp = [0.0f32; INPUT_DIM];
            for t in 0..rels.len() {
                let prev = if t == 0 {
                    [0.0f32; REL_DIM]
                } else {
                    rels[t - 1]
                };
                for r in 0..REL_DIM {
                    let err = rels[t][r] - preds[t][r];
                    if t < mid {
                        inp[r] += err * inv_a;
                    } else {
                        inp[REL_DIM + r] += err * inv_b;
                    }
                    inp[2 * REL_DIM + r] += (prev[r] * err) * len_inv;
                }
            }
            let mut next_state = [0.0f32; STATE_DIM];
            for i in 0..NODES {
                let part_i = cut_mask.map(|m| (m >> i) & 1);
                for d in 0..DIM {
                    let mut acc = w[OFF_BIAS + i * DIM + d];
                    for j in 0..NODES {
                        if let Some(pi) = part_i {
                            let pj = (cut_mask.unwrap() >> j) & 1;
                            if i != j && pi != pj {
                                continue;
                            }
                        }
                        let base = OFF_EDGE + (i * NODES + j) * DIM * DIM + d * DIM;
                        for k in 0..DIM {
                            acc += w[base + k] * state[j * DIM + k];
                        }
                    }
                    let in_base = OFF_IN + (i * DIM + d) * INPUT_DIM;
                    for m in 0..INPUT_DIM {
                        acc += w[in_base + m] * inp[m];
                    }
                    let idx = i * DIM + d;
                    next_state[idx] = 0.35 * state[idx] + 0.65 * acc.tanh();
                }
            }
            state = next_state;
        }
        let (_, final_pos_err, final_mean_err) = self.evaluate_hypothesis(&state, rels);
        step_residuals.push(final_mean_err);
        AbductionTrace {
            final_state: state,
            step_residuals,
            final_pos_err,
            final_mean_err,
        }
    }
    pub fn abduce(&self, rels: &[[f32; REL_DIM]], cut_mask: Option<u8>) -> AbductionTrace {
        self.abduce_with_steps(rels, cut_mask, SEARCH_STEPS)
    }
    pub fn save<P: AsRef<Path>>(&self, path: P) -> io::Result<()> {
        if let Some(parent) = path.as_ref().parent()
            && !parent.as_os_str().is_empty()
        {
            fs::create_dir_all(parent)?;
        }
        let mut buf = Vec::with_capacity(12 + NUM_PARAMS * 4);
        buf.extend_from_slice(MAGIC);
        buf.extend_from_slice(&VERSION.to_le_bytes());
        buf.extend_from_slice(&(NUM_PARAMS as u32).to_le_bytes());
        for &val in &self.weights {
            buf.extend_from_slice(&val.to_le_bytes());
        }
        fs::write(path, buf)
    }
    pub fn load<P: AsRef<Path>>(path: P) -> io::Result<Self> {
        let data = fs::read(path)?;
        if data.len() < 12 || &data[0..4] != MAGIC {
            return Err(io::Error::new(
                ErrorKind::InvalidData,
                "Fichier .phis invalide (en-tête manquant)",
            ));
        }
        let version = u32::from_le_bytes(data[4..8].try_into().unwrap());
        let count = u32::from_le_bytes(data[8..12].try_into().unwrap()) as usize;
        if version != VERSION || count != NUM_PARAMS || data.len() != 12 + count * 4 {
            return Err(io::Error::new(
                ErrorKind::InvalidData,
                "Version ou taille du modèle .phis incompatible (ré-entraîne avec v0.6.1)",
            ));
        }
        let mut weights = Vec::with_capacity(count);
        for chunk in data[12..].chunks_exact(4) {
            weights.push(f32::from_le_bytes(chunk.try_into().unwrap()));
        }
        Ok(Self { weights })
    }

fn arg_val(args: &[String], flag: &str) -> Option<String> {
    args.windows(2)
        .find(|w| w[0] == flag)
        .map(|w| w[1].clone())
}
fn printable_preview(bytes: &[u8]) -> String {
    bytes
        .iter()
        .map(|&b| {
            if (32..=126).contains(&b) {
                b as char
            } else {
                '.'
            }
        })
        .collect()
}
fn trim_ctrl(bytes: &[u8]) -> &[u8] {
    let mut s = 0usize;
    let mut e = bytes.len();
    while s < e && bytes[s] < 32 {
        s += 1;
    }
    while e > s && bytes[e - 1] < 32 {
        e -= 1;
    }
    &bytes[s..e]
}
fn hex_preview(bytes: &[u8]) -> String {
    bytes
        .iter()
        .map(|b| format!("{:02x}", b))
        .collect::<Vec<_>>()
        .join(" ")
}

            2.min(matches.len())

                m.cross_err,

                m.byte_overlap * 100.0
            );
            println!(

                printable_preview(&bytes[m.core_start..m.core_end])
            );
            if let Some((rs, re)) = m.referent {
                println!(

                    printable_preview(trim_ctrl(&bytes[rs..re]))
                );
            }
        }
    }

        .hop1_per_part
        .first()
        .and_then(|v| v.first());
    let best_hop2 = ana.hop2_matches.first();

        (Some(h2), Some(h1)) => ana.target_spans.len() >= 2 || h2.score > h1.score,
        (Some(_), None) => true,
        _ => false,
    };

        let margin = ana
            .hop2_matches
            .get(1)
            .map(|s| h2.score - s.score)
            .unwrap_or(h2.score);
        println!(
            "\nExplication projetée (multi-sauts Hop-{}, alphabet de E, marge={:+.4}) :",
            h2.concept.hop_depth, margin
        );
        println!("  {}", printable_preview(&h2.concept.name_bytes));
    } else if let Some(best) = best_hop1 {
        let margin = ana.hop1_per_part[0]
            .get(1)
            .map(|s| best.score - s.score)
            .unwrap_or(best.score);
        let out_slice = match best.referent {
            Some((rs, re)) => trim_ctrl(&bytes[rs..re]),
            None => &bytes[best.core_start..best.core_end],
        };
        println!(
            "\nExplication projetée (Hop-1, alphabet de E, marge={:+.4}) :",
            margin
        );
        println!("  {}", printable_preview(out_slice));
    }
}

fn is_word_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'_'
}
fn trim_edge_ws(bytes: &[u8]) -> &[u8] {
    let mut s = 0usize;
    let mut e = bytes.len();
    while s < e && bytes[s] <= b' ' {
        s += 1;
    }
    while e > s && bytes[e - 1] <= b' ' {
        e -= 1;
    }
    &bytes[s..e]
}
fn contains_slice(haystack: &[u8], needle: &[u8]) -> bool {
    if needle.is_empty() || needle.len() > haystack.len() {
        return false;
    }
    haystack.windows(needle.len()).any(|w| w == needle)
}
fn find_standalone_positions(haystack: &[u8], needle: &[u8]) -> Vec<usize> {
    let mut out = Vec::new();
    if needle.is_empty() || needle.len() > haystack.len() {
        return out;
    }
    let word_like = needle.iter().all(|&b| is_word_byte(b));
    let mut i = 0usize;
    while i + needle.len() <= haystack.len() {
        if &haystack[i..i + needle.len()] == needle {
            let left_ok = !word_like || i == 0 || !is_word_byte(haystack[i - 1]);
            let right_ok = !word_like
                || (i + needle.len() == haystack.len())
                || !is_word_byte(haystack[i + needle.len()]);
            if left_ok && right_ok {
                out.push(i);
                i += needle.len();
                continue;
            }
        }
        i += 1;
    }
    out
}
pub fn decompose_by_residual(graph: &PhiGraph, bytes: &[u8]) -> ResidualSplit {
    let n = bytes.len();
    let sub = RawSubstrate::from_bytes(bytes);
    let rels = graph.encode_relations(&sub);
    let tr0 = graph.abduce(&rels, None);
    let single_err = tr0.final_mean_err;
    if n < 30 {
        return ResidualSplit {
            parts: vec![(0, n)],
            single_err,
            split_err: single_err,
            cross_cos: 1.0,
        };
    }
    let mut pref_err = vec![0.0f32; n + 1];
    for t in 0..n {
        pref_err[t + 1] = pref_err[t] + tr0.final_pos_err[t];
    }
    let mut proposals: Vec<(usize, f32)> = Vec::new();
    let mut p = 14usize;
    while p + 14 <= n {
        let ml = pref_err[p] / (p as f32);
        let mr = (pref_err[n] - pref_err[p]) / ((n - p) as f32);
        let res_jump = (ml - mr).abs();
        let cut_d = cut_discontinuity(&sub, 0, n, p);
        let abd_d = if p % 4 == 0 {
            let l0 = p.saturating_sub(16);
            let r1 = (p + 16).min(n);
            abductive_discontinuity(graph, &rels, (l0, p), (p, r1), 4)
        } else {
            0.0
        };
        proposals.push((p, res_jump + 0.45 * cut_d + 0.55 * abd_d));
        p += 2;
    }
    proposals.sort_by(|a, b| b.1.total_cmp(&a.1));
    let mut cands: Vec<usize> = Vec::new();
    for (cp, _) in proposals {
        if cands.iter().all(|&x| x.abs_diff(cp) >= 6) {
            cands.push(cp);
            if cands.len() >= 4 {
                break;
            }
        }
    }
    if cands.iter().all(|&x| x.abs_diff(n / 2) >= 6) {
        cands.push(n / 2);
    }
    let eval_cut = |cp: usize| -> (f32, f32, f32, f32) {
        let rl = graph.encode_bytes(&bytes[0..cp]);
        let rr = graph.encode_bytes(&bytes[cp..n]);
        let tl = graph.abduce(&rl, None);
        let tr = graph.abduce(&rr, None);
        let split_e =
            ((cp as f32) * tl.final_mean_err + ((n - cp) as f32) * tr.final_mean_err) / (n as f32);
        let cos_lr = cosine_sim(&tl.final_state, &tr.final_state);
        let (_, _, e_lr) = graph.evaluate_hypothesis(&tl.final_state, &rr);
        let (_, _, e_rl) = graph.evaluate_hypothesis(&tr.final_state, &rl);
        let cross_e = 0.5 * (e_lr + e_rl);
        let cross_gap = ((cross_e - split_e) / (cross_e + 0.08)).clamp(0.0, 1.0);
        let obj = (single_err - split_e) + 0.30 * cross_gap * (1.0 - cos_lr.max(0.0));
        (obj, split_e, cos_lr, cross_gap)
    };
    let mut best_p = n / 2;
    let mut best_obj = f32::NEG_INFINITY;
    let mut best_res = (single_err, 1.0f32, 0.0f32);
    for cp in cands {
        let (obj, se, cos_lr, cg) = eval_cut(cp);
        if obj > best_obj {
            best_obj = obj;
            best_p = cp;
            best_res = (se, cos_lr, cg);
        }
    }
    let lo = best_p.saturating_sub(2).max(12);
    let hi = (best_p + 2).min(n.saturating_sub(12));
    for cp in lo..=hi {
        if cp == best_p {
            continue;
        }
        let (obj, se, cos_lr, cg) = eval_cut(cp);
        if obj > best_obj {
            best_obj = obj;
            best_p = cp;
            best_res = (se, cos_lr, cg);
        }
    }
    let (split_err, cross_cos, cross_gap) = best_res;
    let significant_drop = (single_err - split_err) > 0.025 || split_err < 0.88 * single_err;
    if cross_cos < 0.55 && cross_gap > 0.22 && significant_drop {
        ResidualSplit {
            parts: vec![(0, best_p), (best_p, n)],
            single_err,
            split_err,
            cross_cos,
        }
    } else {
        ResidualSplit {
            parts: vec![(0, n)],
            single_err,
            split_err: single_err,
            cross_cos: 1.0,
        }
    }
}
fn extract_word_tokens(slice: &[u8]) -> Vec<Vec<u8>> {
    let mut out = Vec::new();
    let mut i = 0usize;
    while i < slice.len() {
        while i < slice.len() && !is_word_byte(slice[i]) {
            i += 1;
        }
        let start = i;
        while i < slice.len() && is_word_byte(slice[i]) {
            i += 1;
        }
        if i - start >= 3 {
            let tok = slice[start..i].to_vec();
            if !out.contains(&tok) {
                out.push(tok);
            }
        }
    }
    out
}
fn demote_metalanguage_regimes(
    bytes: &[u8],
    raw: &[DiscoveredRegime],
) -> (Vec<DiscoveredRegime>, usize) {
    if raw.len() < 3 {
        return (raw.to_vec(), 0);
    }
    let mut kept: Vec<DiscoveredRegime> = Vec::with_capacity(raw.len());
    let mut kept_tag_tokens: Vec<Vec<Vec<u8>>> = Vec::with_capacity(raw.len());
    let mut demoted = 0usize;
    for (idx, &reg) in raw.iter().enumerate() {
        let prev_end = if kept.is_empty() {
            0
        } else {
            kept.last().unwrap().core_end
        };
        let core_slice = &bytes[reg.core_start..reg.core_end];
        let mut echoed_molds = 0usize;
        let mut echoed_bytes = 0usize;
        if kept.len() >= 2 && idx + 1 < raw.len() {
            for m_i in 0..kept.len() {
                let mut hit_len = 0usize;
                for tok in &kept_tag_tokens[m_i] {
                    let unique_to_mi = (0..kept.len())
                        .all(|m_j| m_j == m_i || !kept_tag_tokens[m_j].contains(tok));
                    if unique_to_mi {
                        let cnt = find_standalone_positions(core_slice, tok).len();
                        if cnt > 0 {
                            hit_len += cnt * tok.len();
                        }
                    }
                }
                if hit_len > 0 {
                    echoed_molds += 1;
                    echoed_bytes += hit_len;
                }
            }
        }
        if echoed_molds >= 2 || (echoed_molds >= 1 && echoed_bytes * 4 >= core_slice.len()) {
            demoted += 1;
            continue;
        }
        let tag_start = prev_end.max(reg.core_start.saturating_sub(28));
        let tag_slice = &bytes[tag_start..reg.core_start];
        kept_tag_tokens.push(extract_word_tokens(tag_slice));
        kept.push(reg);
    }
    let k = kept.len();
    if k == 0 {
        return (raw.to_vec(), 0);
    }
    let lead_gap = kept[0].core_start;
    let trail_gap = bytes.len().saturating_sub(kept[k - 1].core_end);
    let suffix_mode = trail_gap >= 3 && lead_gap < 3;
    for i in 0..k {
        let prev_end = if i == 0 { 0 } else { kept[i - 1].core_end };
        let next_start = if i + 1 == k {
            bytes.len()
        } else {
            kept[i + 1].core_start
        };
        kept[i].referent = if suffix_mode {
            if next_start.saturating_sub(kept[i].core_end) >= 3 {
                Some((kept[i].core_end, next_start))
            } else {
                None
            }
        } else if kept[i].core_start.saturating_sub(prev_end) >= 3 {
            Some((prev_end, kept[i].core_start))
        } else {
            None
        };
    }
    (kept, demoted)
}
fn project_span_to_molds(

    bytes: &[u8],
    q_span: (usize, usize),

    for (idx, reg) in molds.iter().enumerate() {
        let core_bytes = &bytes[reg.core_start..reg.core_end];
        let core_rels = graph.encode_bytes(core_bytes);
        let core_trace = graph.abduce(&core_rels, None);
        let (_, _, fwd_err) = graph.evaluate_hypothesis(&q_trace.final_state, &core_rels);
        let (_, _, bwd_err) = graph.evaluate_hypothesis(&core_trace.final_state, &q_rels);
        let cross_err = 0.5 * (fwd_err + bwd_err);
        let cross_gen = (-2.2 * cross_err).exp().clamp(0.0, 1.0);
        let cos_sim = cosine_sim(&q_trace.final_state, &core_trace.final_state);
        let cos_pos = 0.5 * (1.0 + cos_sim);
        let score = (cross_gen * cos_pos).sqrt();
        let byte_overlap = byte_overlap_ratio(q_bytes, core_bytes);
        matches.push(ProjectionMatch {
            dst_idx: idx,
            score,
            cross_err,
            cos_sim,
            core_start: reg.core_start,
            core_end: reg.core_end,
            referent: reg.referent,
            byte_overlap,
        });
    }
    matches.sort_by(|a, b| b.score.total_cmp(&a.score));
    matches
}
fn find_discriminative_key(
    bytes: &[u8],
    tag_spans: &[(usize, usize)],
    mold_idx: usize,
    def_zones: &[(usize, usize)],
) -> Option<Vec<u8>> {
    let (ts, te) = tag_spans[mold_idx];
    let tag = &bytes[ts..te];
    if tag.len() < 3 || def_zones.is_empty() {
        return None;
    }
    let mut best: Option<(usize, usize, Vec<u8>)> = None;
    for s in 0..tag.len() {
        if tag[s] <= b' ' {
            continue;
        }
        for e in (s + 3)..=tag.len() {
            if tag[e - 1] <= b' ' {
                continue;
            }
            let cand = &tag[s..e];
            if cand.iter().any(|&b| b < 32) {
                continue;
            }
            let unique = tag_spans.iter().enumerate().all(|(j, &(os, oe))| {
                j == mold_idx || !contains_slice(&bytes[os..oe], cand)
            });
            if !unique {
                continue;
            }
            let mut hits = 0usize;
            for &(ds, de) in def_zones {
                hits += find_standalone_positions(&bytes[ds..de], cand).len();
            }
            if hits == 0 {
                continue;
            }
            let better = match &best {
                None => true,
                Some((blen, bhits, _)) => {
                    cand.len() > *blen || (cand.len() == *blen && hits > *bhits)
                }
            };
            if better {
                best = Some((cand.len(), hits, cand.to_vec()));
            }
        }
    }
    best.map(|(_, _, k)| k)
}

fn strip_common_head_affixes(raw_heads: &[&[u8]]) -> Vec<Vec<u8>> {
    let mut cleaned: Vec<Vec<u8>> = raw_heads
        .iter()
        .map(|h| {
            let slice = match h.iter().rposition(|&b| b == b'\n') {
                Some(nl) if trim_edge_ws(&h[nl + 1..]).len() >= 2 => &h[nl + 1..],
                _ => *h,
            };
            trim_edge_ws(slice).to_vec()
        })
        .collect();
    if cleaned.len() == 1 {
        let h = &mut cleaned[0];
        while h.len() > 2 && !is_word_byte(*h.last().unwrap()) {
            h.pop();
        }
        *h = trim_edge_ws(h).to_vec();
        return cleaned;
    }
    let min_len = cleaned.iter().map(|h| h.len()).min().unwrap_or(0);
    let mut suf = 0usize;
    while suf + 2 < min_len {
        let b = cleaned[0][cleaned[0].len() - 1 - suf];
        if !is_word_byte(b) && cleaned.iter().all(|h| h[h.len() - 1 - suf] == b) {
            suf += 1;
        } else {
            break;
        }
    }
    if suf > 0 {
        for h in &mut cleaned {
            let new_len = h.len().saturating_sub(suf);
            h.truncate(new_len);
            *h = trim_edge_ws(h).to_vec();
        }
    }
    let tail_min = cleaned[1..].iter().map(|h| h.len()).min().unwrap_or(0);
    let mut pre = 0usize;
    while pre + 2 < tail_min {
        let b = cleaned[1][pre];
        if !is_word_byte(b) && cleaned[1..].iter().all(|h| h[pre] == b) {
            pre += 1;
        } else {
            break;
        }
    }
    if pre > 0 {
        let sep_pre = cleaned[1][0..pre].to_vec();
        if let Some(pos) = cleaned[0]
            .windows(sep_pre.len())
            .rposition(|w| w == sep_pre.as_slice())
        {
            let tail = trim_edge_ws(&cleaned[0][pos + sep_pre.len()..]);
            if tail.len() >= 2 {
                cleaned[0] = tail.to_vec();
            }
        }
        for h in &mut cleaned[1..] {
            *h = trim_edge_ws(&h[pre..]).to_vec();
        }
    }
    let all_min = cleaned.iter().map(|h| h.len()).min().unwrap_or(0);
    let mut ap = 0usize;
    while ap + 2 < all_min {
        let b = cleaned[0][ap];
        if !is_word_byte(b) && cleaned.iter().all(|h| h[ap] == b) {
            ap += 1;
        } else {
            break;
        }
    }
    if ap > 0 {
        for h in &mut cleaned {
            *h = trim_edge_ws(&h[ap..]).to_vec();
        }
    }
    cleaned
}
fn extract_definitions_in_zone(
    zone_bytes: &[u8],
    zone_offset: usize,
    entries: &[KeyEntry],
) -> Vec<ComposedConcept> {
    let mut order: Vec<usize> = (0..entries.len()).collect();
    order.sort_by(|&a, &b| entries[b].key.len().cmp(&entries[a].key.len()));
    let mut used = vec![false; zone_bytes.len()];
    let mut hits: Vec<KeyHit> = Vec::new();
    for ei in order {
        let key = &entries[ei].key;
        let def_in = entries[ei].defined_in;
        for pos in find_standalone_positions(zone_bytes, key) {
            let end = pos + key.len();
            let abs_pos = zone_offset + pos;
            let abs_end = zone_offset + end;
            if let Some((ds, de)) = def_in
                && abs_pos >= ds
                && abs_end <= de
            {
                continue;
            }
            if used[pos..end].iter().any(|&u| u) {
                continue;
            }
            for u in &mut used[pos..end] {
                *u = true;
            }
            hits.push(KeyHit {
                pos,
                end,
                entry_idx: ei,
            });
        }
    }
    hits.sort_by_key(|h| h.pos);
    if hits.len() < 2 {
        return Vec::new();
    }
    let mut gaps: Vec<&[u8]> = Vec::with_capacity(hits.len() - 1);
    for w in hits.windows(2) {
        gaps.push(&zone_bytes[w[0].end..w[1].pos]);
    }
    let min_gap = gaps.iter().map(|g| g.len()).min().unwrap_or(0);
    let mut dom_conn: Option<&[u8]> = None;
    let mut dom_count = 1usize;
    for &g in &gaps {
        if g.len() <= 8 && !g.contains(&b'\n') {
            let cnt = gaps.iter().filter(|&&other| other == g).count();
            if cnt > dom_count
                || (cnt == dom_count && cnt >= 2 && dom_conn.is_some_and(|d| g.len() < d.len()))
            {
                dom_count = cnt;
                dom_conn = Some(g);
            }
        }
    }
    let mut clusters: Vec<Vec<usize>> = Vec::new();
    let mut cur = vec![0usize];
    for g_idx in 0..gaps.len() {
        let gap = gaps[g_idx];
        let h_prev = &hits[g_idx];
        let h_next = &hits[g_idx + 1];
        let same_ref = h_prev.entry_idx == h_next.entry_idx;
        let is_break = if same_ref || gap.contains(&b'\n') {
            true
        } else if let Some(dc) = dom_conn {
            gap != dc && gap.len() > dc.len() + 1
        } else {
            gap.len() > min_gap + 3
        };
        if is_break {
            clusters.push(cur);
            cur = vec![g_idx + 1];
        } else {
            cur.push(g_idx + 1);
        }
    }
    clusters.push(cur);
    let first_pos = hits[clusters[0][0]].pos;
    let last_end = hits[*clusters.last().unwrap().last().unwrap()].end;
    let lead_len = trim_edge_ws(&zone_bytes[0..first_pos]).len();
    let trail_len = trim_edge_ws(&zone_bytes[last_end..]).len();
    let suffix_mode = trail_len >= 3 && lead_len < 3;
    struct ValidCluster<'a> {
        raw_head: &'a [u8],
        clause_span: (usize, usize),
        mold_set: Vec<usize>,
        hop_depth: usize,
        ref_labels: Vec<Vec<u8>>,
    }
    let mut valid: Vec<ValidCluster<'_>> = Vec::new();
    for (ci, cl) in clusters.iter().enumerate() {
        let mut mold_set: Vec<usize> = Vec::new();
        let mut max_child_hop = 1usize;
        let mut ref_labels: Vec<Vec<u8>> = Vec::new();
        for &hi in cl {
            let entry = &entries[hits[hi].entry_idx];
            if entry.hop > max_child_hop {
                max_child_hop = entry.hop;
            }
            ref_labels.push(entry.key.clone());
            for &m in &entry.molds {
                if !mold_set.contains(&m) {
                    mold_set.push(m);
                }
            }
        }
        mold_set.sort_unstable();
        if mold_set.len() < 2 {
            continue;
        }
        let c_start = hits[cl[0]].pos;
        let c_end = hits[*cl.last().unwrap()].end;
        let prev_end = if ci == 0 {
            0
        } else {
            hits[*clusters[ci - 1].last().unwrap()].end
        };
        let next_start = if ci + 1 == clusters.len() {
            zone_bytes.len()
        } else {
            hits[clusters[ci + 1][0]].pos
        };
        let (raw_head, clause_span) = if suffix_mode {
            (
                &zone_bytes[c_end..next_start],
                (zone_offset + c_start, zone_offset + next_start),
            )
        } else {
            (
                &zone_bytes[prev_end..c_start],
                (zone_offset + prev_end, zone_offset + c_end),
            )
        };
        if trim_edge_ws(raw_head).len() >= 2 {
            valid.push(ValidCluster {
                raw_head,
                clause_span,
                mold_set,
                hop_depth: max_child_hop + 1,
                ref_labels,
            });
        }
    }
    if valid.is_empty() {
        return Vec::new();
    }
    let raw_heads: Vec<&[u8]> = valid.iter().map(|v| v.raw_head).collect();
    let clean_heads = strip_common_head_affixes(&raw_heads);
    let mut out = Vec::with_capacity(valid.len());
    for (vc, name_bytes) in valid.into_iter().zip(clean_heads.into_iter()) {
        if name_bytes.len() >= 2 {
            out.push(ComposedConcept {
                name_bytes,
                clause_span: vc.clause_span,
                mold_set: vc.mold_set,
                hop_depth: vc.hop_depth,
                ref_labels: vc.ref_labels,
            });
        }
    }
    out
}
pub fn analyze_multihop(graph: &PhiGraph, bytes: &[u8]) -> Option<MultiHopAnalysis> {
    let raw_regimes = discover_causal_regimes(graph, bytes, true);
    if raw_regimes.len() < 2 {
        return None;
    }
    let (regimes, demoted_meta_count) = demote_metalanguage_regimes(bytes, &raw_regimes);
    if regimes.len() < 2 {
        return None;
    }
    let k = regimes.len();
    let mut split_t = k - 1;
    while split_t >= 3 && (k - split_t) < 4 {
        let cand_idx = split_t - 1;
        let gap_to_next = regimes[split_t]
            .core_start
            .saturating_sub(regimes[cand_idx].core_end);
        if gap_to_next > 8 {
            break;
        }
        let prev_end = regimes[cand_idx - 1].core_end;
        let gap_before = regimes[cand_idx].core_start.saturating_sub(prev_end);
        if gap_before > 8 && gap_before < 20 {
            break;
        }
        let earlier = &regimes[0..cand_idx];
        let proj = project_span_to_molds(
            graph,
            bytes,
            (regimes[cand_idx].core_start, regimes[cand_idx].core_end),
            earlier,
        );
        let strong_match = proj
            .first()
            .is_some_and(|m| m.score >= 0.62 && m.cos_sim >= 0.40);
        if strong_match {
            split_t = cand_idx;
        } else {
            break;
        }
    }
    let mold_regimes = regimes[0..split_t].to_vec();
    let mut target_spans: Vec<(usize, usize)> = regimes[split_t..]
        .iter()
        .map(|r| (r.core_start, r.core_end))
        .collect();
    if target_spans.len() >= 2 {
        let mut merged_t: Vec<(usize, usize)> = vec![target_spans[0]];
        for &(s, e) in &target_spans[1..] {
            let prev = *merged_t.last().unwrap();
            let h_prev = graph
                .abduce(&graph.encode_bytes(&bytes[prev.0..prev.1]), None)
                .final_state;
            let h_cur = graph
                .abduce(&graph.encode_bytes(&bytes[s..e]), None)
                .final_state;
            let ov = byte_overlap_ratio(&bytes[prev.0..prev.1], &bytes[s..e]);
            if cosine_sim(&h_prev, &h_cur) > 0.72 && ov > 0.50 {
                merged_t.last_mut().unwrap().1 = e;
            } else {
                merged_t.push((s, e));
            }
        }
        target_spans = merged_t;
    }
    let mut residual_info: Option<(f32, f32, f32)> = None;
    if target_spans.len() == 1 {
        let (ts, te) = target_spans[0];
        let dec = decompose_by_residual(graph, &bytes[ts..te]);
        if dec.parts.len() >= 2 {
            target_spans = dec
                .parts
                .into_iter()
                .map(|(a, b)| (ts + a, ts + b))
                .collect();
            residual_info = Some((dec.single_err, dec.split_err, dec.cross_cos));
        }
    } else if target_spans.len() >= 2 {
        let (s0, e0) = target_spans[0];
        let (s1, e1) = target_spans[1];
        let full_err = graph
            .abduce(&graph.encode_bytes(&bytes[s0..e1]), None)
            .final_mean_err;
        let t0 = graph.abduce(&graph.encode_bytes(&bytes[s0..e0]), None);
        let t1 = graph.abduce(&graph.encode_bytes(&bytes[s1..e1]), None);
        let l0 = (e0 - s0) as f32;
        let l1 = (e1 - s1) as f32;
        let split_err = (l0 * t0.final_mean_err + l1 * t1.final_mean_err) / (l0 + l1).max(1.0);
        let cos_lr = cosine_sim(&t0.final_state, &t1.final_state);
        residual_info = Some((full_err, split_err, cos_lr));
    }
    let mut hop1_per_part: Vec<Vec<ProjectionMatch>> = Vec::with_capacity(target_spans.len());
    for &span in &target_spans {
        hop1_per_part.push(project_span_to_molds(graph, bytes, span, &mold_regimes));
    }
    let mut tag_spans: Vec<(usize, usize)> = Vec::with_capacity(mold_regimes.len());
    let mut def_zones: Vec<(usize, usize)> = Vec::new();
    for i in 0..mold_regimes.len() {
        let prev_end = if i == 0 {
            0
        } else {
            mold_regimes[i - 1].core_end
        };
        let cs = mold_regimes[i].core_start;
        if cs.saturating_sub(prev_end) > 28 {
            def_zones.push((prev_end, cs - 24));
            tag_spans.push((cs - 24, cs));
        } else {
            tag_spans.push((prev_end, cs));
        }
    }
    let after_molds = mold_regimes.last().unwrap().core_end;
    let before_target = target_spans[0].0;
    if before_target.saturating_sub(after_molds) >= 10 {
        def_zones.push((after_molds, before_target));
    }
    let after_target = target_spans.last().unwrap().1;
    if bytes.len().saturating_sub(after_target) >= 10 {
        def_zones.push((after_target, bytes.len()));
    }
    let mut mold_keys: Vec<Option<Vec<u8>>> = Vec::with_capacity(mold_regimes.len());
    let mut entries: Vec<KeyEntry> = Vec::new();
    for i in 0..mold_regimes.len() {
        let key_opt = find_discriminative_key(bytes, &tag_spans, i, &def_zones);
        if let Some(ref k) = key_opt {
            entries.push(KeyEntry {
                key: k.clone(),
                molds: vec![i],
                hop: 1,
                defined_in: None,
            });
        }
        mold_keys.push(key_opt);
    }
    let mut all_concepts: Vec<ComposedConcept> = Vec::new();
    if entries.len() >= 2 {
        for &(ds, de) in &def_zones {
            let found = extract_definitions_in_zone(&bytes[ds..de], ds, &entries);
            all_concepts.extend(found);
        }
        if !all_concepts.is_empty() {
            let mut ext_entries = entries.clone();
            for c in &all_concepts {
                if c.name_bytes.len() >= 3
                    && ext_entries.iter().all(|e| e.key != c.name_bytes)
                {
                    ext_entries.push(KeyEntry {
                        key: c.name_bytes.clone(),
                        molds: c.mold_set.clone(),
                        hop: c.hop_depth,
                        defined_in: Some(c.clause_span),
                    });
                }
            }
            if ext_entries.len() > entries.len() {
                for &(ds, de) in &def_zones {
                    let found3 = extract_definitions_in_zone(&bytes[ds..de], ds, &ext_entries);
                    for c3 in found3 {
                        if c3.hop_depth >= 3
                            && all_concepts
                                .iter()
                                .all(|old| old.name_bytes != c3.name_bytes)
                        {
                            all_concepts.push(c3);
                        }
                    }
                }
            }
        }
    }
    let m_count = mold_regimes.len();
    let mut res_mat: Vec<Vec<f32>> = vec![vec![0.0f32; m_count]; target_spans.len()];
    for (c_idx, matches) in hop1_per_part.iter().enumerate() {
        for m in matches {
            res_mat[c_idx][m.dst_idx] = m.score;
        }
    }
    let mut hop2_matches: Vec<MultiHopMatch> = Vec::with_capacity(all_concepts.len());
    for concept in all_concepts {
        let mut cov_sum = 0.0f32;
        for row in &res_mat {
            let best_m = concept
                .mold_set
                .iter()
                .map(|&mi| row[mi])
                .fold(0.0f32, f32::max);
            cov_sum += best_m;
        }
        let coverage = cov_sum / (res_mat.len().max(1) as f32);
        let mut prec_sum = 0.0f32;
        for &mi in &concept.mold_set {
            let best_c = res_mat.iter().map(|row| row[mi]).fold(0.0f32, f32::max);
            prec_sum += best_c;
        }
        let precision = prec_sum / (concept.mold_set.len().max(1) as f32);
        let score = (coverage * precision).sqrt();
        hop2_matches.push(MultiHopMatch {
            concept,
            score,
            coverage,
            precision,
        });
    }
    hop2_matches.sort_by(|a, b| b.score.total_cmp(&a.score));
    let definitions_found = hop2_matches.len();
    Some(MultiHopAnalysis {
        mold_regimes,
        mold_keys,
        target_spans,
        residual_info,
        hop1_per_part,
        hop2_matches,
        definitions_found,
        demoted_meta_count,
    })
}

pub fn cosine_sim(a: &[f32; STATE_DIM], b: &[f32; STATE_DIM]) -> f32 {
    let mut dot = 0.0f32;
    let mut na = 0.0f32;
    let mut nb = 0.0f32;
    for i in 0..STATE_DIM {
        dot += a[i] * b[i];
        na += a[i] * a[i];
        nb += b[i] * b[i];
    }
    let denom = (na * nb).sqrt();
    if denom < 1e-8 {
        0.0
    } else {
        (dot / denom).clamp(-1.0, 1.0)
    }
}

            idx += 1;
        }
    }
    rng.shuffle(&mut balanced);
    [
        sing[0],
        sing[1],
        sing[2],
        balanced[0],
        balanced[1],
        balanced[2],
    ]
}
impl PreparedEpisode {
    pub fn from_bytes(bytes: &[u8], rng: &mut Rng) -> Self {
        let orig = RawSubstrate::from_bytes(bytes);
        let mut neg_bytes = bytes.to_vec();
        rng.shuffle(&mut neg_bytes);
        let neg = RawSubstrate::from_bytes(&neg_bytes);
        let mut pool: [u8; 256] = std::array::from_fn(|i| i as u8);
        rng.shuffle(&mut pool);
        let mut passive_bytes = Vec::with_capacity(bytes.len());
        let mut cursor = 0usize;
        for _ in 0..bytes.len() {
            if cursor >= 6 && rng.next_usize(100) < 15 {
                passive_bytes.push(pool[rng.next_usize(cursor.min(14))]);
            } else {
                passive_bytes.push(pool[cursor % 256]);
                cursor += 1;
            }
        }
        let passive = RawSubstrate::from_bytes(&passive_bytes);
        let mut ref_bytes = bytes.to_vec();
        let corrupt_n = (bytes.len() / 6).max(1);
        for _ in 0..corrupt_n {
            let pos = rng.next_usize(ref_bytes.len());
            ref_bytes[pos] = ref_bytes[pos].wrapping_add(1 + rng.next_u8() % 250);
        }
        let refut = RawSubstrate::from_bytes(&ref_bytes);
        let mut shift_bytes = bytes.to_vec();
        let shift = (bytes.len() / 3).max(1) % bytes.len();
        shift_bytes.rotate_left(shift);
        let shift_sub = RawSubstrate::from_bytes(&shift_bytes);
        let mut perm: [u8; 256] = std::array::from_fn(|i| i as u8);
        rng.shuffle(&mut perm);
        let sub_len = ((bytes.len() * 3) / 4).max(8).min(bytes.len());
        let start_off = rng.next_usize(bytes.len() - sub_len + 1);
        let abs_bytes: Vec<u8> = bytes[start_off..start_off + sub_len]
            .iter()
            .map(|&b| perm[b as usize])
            .collect();
        let abs = RawSubstrate::from_bytes(&abs_bytes);
        let cuts = sample_training_cuts(rng);
        Self {
            orig,
            neg,
            passive,
            refut,
            shift: shift_sub,
            abs,
            cuts,
        }
    }
}
impl PreparedComposite {
    pub fn from_bytes(bytes: &[u8]) -> Self {
        Self {
            stream: RawSubstrate::from_bytes(bytes),
        }
    }
}
pub fn byte_overlap_ratio(a: &[u8], b: &[u8]) -> f32 {
    if a.is_empty() {
        return 0.0;
    }
    let mut seen = [false; 256];
    for &x in b {
        seen[x as usize] = true;
    }
    let shared = a.iter().filter(|&&x| seen[x as usize]).count();
    (shared as f32) / (a.len() as f32)
}
pub fn cut_discontinuity(sub: &RawSubstrate, s: usize, e: usize, p: usize) -> f32 {
    let l_lo = p.saturating_sub(10).max(s);
    let r_hi = (p + 10).min(e);
    if p <= l_lo || r_hi <= p {
        return 0.0;
    }
    let mut l_cross = 0usize;
    for t in l_lo..p {
        let mut has_cross = false;
        for idx in sub.offsets[t]..sub.offsets[t + 1] {
            let j = sub.edge_targets[idx] as usize;
            if j >= p && j < e {
                let f = &sub.edges[idx];
                if f[0] + f[1] + f[4] + f[5] >= 1.0 {
                    has_cross = true;
                    break;
                }
            }
        }
        if has_cross {
            l_cross += 1;
        }
    }
    let mut r_cross = 0usize;
    for t in p..r_hi {
        let mut has_cross = false;
        for idx in sub.offsets[t]..sub.offsets[t + 1] {
            let j = sub.edge_targets[idx] as usize;
            if j >= s && j < p {
                let f = &sub.edges[idx];
                if f[0] + f[1] + f[4] + f[5] >= 1.0 {
                    has_cross = true;
                    break;
                }
            }
        }
        if has_cross {
            r_cross += 1;
        }
    }
    let f_left = (l_cross as f32) / ((p - l_lo) as f32);
    let f_right = (r_cross as f32) / ((r_hi - p) as f32);
    (1.0 - 0.5 * (f_left + f_right)).clamp(0.0, 1.0)
}
pub fn abductive_discontinuity(
    graph: &PhiGraph,
    rels: &[[f32; REL_DIM]],
    left: (usize, usize),
    right: (usize, usize),
    steps: usize,
) -> f32 {
    let l_rels = &rels[left.0..left.1];
    let r_rels = &rels[right.0..right.1];
    let hl = graph.abduce_with_steps(l_rels, None, steps);
    let hr = graph.abduce_with_steps(r_rels, None, steps);
    let cos_lr = cosine_sim(&hl.final_state, &hr.final_state);
    let state_disc = 0.5 * (1.0 - cos_lr);
    let (_, _, e_lr) = graph.evaluate_hypothesis(&hl.final_state, r_rels);
    let (_, _, e_rl) = graph.evaluate_hypothesis(&hr.final_state, l_rels);
    let self_e = 0.5 * (hl.final_mean_err + hr.final_mean_err);
    let cross_e = 0.5 * (e_lr + e_rl);
    let err_disc = ((cross_e - self_e) / (cross_e + 0.08)).clamp(0.0, 1.0);
    0.55 * state_disc + 0.45 * err_disc
}
fn split_run_by_abduction(
    graph: &PhiGraph,
    sub: &RawSubstrate,
    rels: &[[f32; REL_DIM]],
    s: usize,
    e: usize,
    exact: bool,
) -> Vec<(usize, usize)> {
    let len = e.saturating_sub(s);
    if len < 32 {
        return vec![(s, e)];
    }
    let step = if exact { 4 } else { 6 };
    let win = if exact { 16 } else { 18 };
    let steps = if exact { SEARCH_STEPS } else { 4 };
    let offset = win / step;
    let mut anchors = Vec::new();
    let mut pos = s;
    while pos + win <= e {
        anchors.push(pos);
        pos += step;
    }
    if anchors.len() <= offset {
        return vec![(s, e)];
    }
    let traces: Vec<AbductionTrace> = anchors
        .iter()
        .map(|&a| graph.abduce_with_steps(&rels[a..a + win], None, steps))
        .collect();
    let mut candidates: Vec<(usize, f32)> = Vec::new();
    for k in 0..(anchors.len() - offset) {
        let grid_p = anchors[k] + win;
        if grid_p < s + 12 || e.saturating_sub(grid_p) < 12 {
            continue;
        }
        let hl = &traces[k];
        let hr = &traces[k + offset];
        let l_rels = &rels[anchors[k]..anchors[k] + win];
        let r_rels = &rels[anchors[k + offset]..anchors[k + offset] + win];
        let state_disc = 0.5 * (1.0 - cosine_sim(&hl.final_state, &hr.final_state));
        let (_, _, e_lr) = graph.evaluate_hypothesis(&hl.final_state, r_rels);
        let (_, _, e_rl) = graph.evaluate_hypothesis(&hr.final_state, l_rels);
        let self_e = 0.5 * (hl.final_mean_err + hr.final_mean_err);
        let cross_e = 0.5 * (e_lr + e_rl);
        let err_disc = ((cross_e - self_e) / (cross_e + 0.08)).clamp(0.0, 1.0);
        let d_abd = 0.55 * state_disc + 0.45 * err_disc;
        let p_lo = grid_p.saturating_sub(step / 2).max(s + 12);
        let p_hi = (grid_p + step / 2).min(e.saturating_sub(12));
        let mut best_p = grid_p;
        let mut best_cut = -1.0f32;
        for cp in p_lo..=p_hi {
            let c = cut_discontinuity(sub, s, e, cp);
            if c > best_cut {
                best_cut = c;
                best_p = cp;
            }
        }
        let mut score = 0.45 * d_abd + 0.55 * best_cut;
        if best_cut >= 0.85 {
            score = score.max(0.75 * best_cut);
        }
        candidates.push((best_p, score));
    }
    let mut peaks: Vec<(usize, f32)> = Vec::new();
    for i in 0..candidates.len() {
        let (p, sc) = candidates[i];
        if sc <= 0.52 {
            continue;
        }
        let left_ok = i == 0 || sc >= candidates[i - 1].1;
        let right_ok = i + 1 == candidates.len() || sc >= candidates[i + 1].1;
        if left_ok && right_ok {
            peaks.push((p, sc));
        }
    }
    peaks.sort_by(|a, b| b.1.total_cmp(&a.1));
    let mut chosen: Vec<usize> = Vec::new();
    for (p, _) in peaks {
        if chosen.iter().all(|&c| c.abs_diff(p) >= 16) {
            chosen.push(p);
        }
    }
    if chosen.is_empty() {
        return vec![(s, e)];
    }
    chosen.sort_unstable();
    if exact {
        for p in &mut chosen {
            let mut best_p = *p;
            let mut best_sc = f32::NEG_INFINITY;
            let lo = (*p).saturating_sub(3).max(s + 12);
            let hi = (*p + 3).min(e.saturating_sub(12));
            for cand in lo..=hi {
                let l_start = cand.saturating_sub(16).max(s);
                let r_end = (cand + 16).min(e);
                let d_abd = abductive_discontinuity(
                    graph,
                    rels,
                    (l_start, cand),
                    (cand, r_end),
                    SEARCH_STEPS,
                );
                let d_cut = cut_discontinuity(sub, s, e, cand);
                let sc = 0.35 * d_abd + 0.65 * d_cut;
                if sc > best_sc {
                    best_sc = sc;
                    best_p = cand;
                }
            }
            *p = best_p;
        }
    }
    let mut segs = Vec::with_capacity(chosen.len() + 1);
    let mut cur = s;
    for p in chosen {
        if p >= cur + 12 && e >= p + 12 {
            segs.push((cur, p));
            cur = p;
        }
    }
    segs.push((cur, e));
    segs
}
fn attach_passive_referents(cores: &[(usize, usize)], gates_ctx: &[f32], n: usize) -> Vec<DiscoveredRegime> {
    let k = cores.len();
    if k == 0 {
        return Vec::new();
    }
    let lead_gap = cores[0].0;
    let trail_gap = n.saturating_sub(cores[k - 1].1);
    let suffix_mode = trail_gap >= 3 && lead_gap < 3;
    let mut out = Vec::with_capacity(k);
    for i in 0..k {
        let (cs, ce) = cores[i];
        let mg = gates_ctx[cs..ce].iter().sum::<f32>() / ((ce - cs).max(1) as f32);
        let prev_end = if i == 0 { 0 } else { cores[i - 1].1 };
        let next_start = if i + 1 == k { n } else { cores[i + 1].0 };
        let referent = if suffix_mode {
            if next_start.saturating_sub(ce) >= 3 {
                Some((ce, next_start))
            } else {
                None
            }
        } else if cs.saturating_sub(prev_end) >= 3 {
            Some((prev_end, cs))
        } else {
            None
        };
        out.push(DiscoveredRegime {
            core_start: cs,
            core_end: ce,
            referent,
            mean_gate: mg,
        });
    }
    out
}
fn tighten_regime_isolated(graph: &PhiGraph, bytes: &[u8], mut cs: usize, mut ce: usize) -> (usize, usize) {
    for _ in 0..2 {
        if ce.saturating_sub(cs) < 12 {
            break;
        }
        let iso_sub = RawSubstrate::from_bytes(&bytes[cs..ce]);
        let (_, _, iso_pt) = graph.encode_with_gate(&iso_sub);
        let len = iso_pt.len();
        let mut l = 0usize;
        while l + 10 < len && iso_pt[l] < 0.44 {
            l += 1;
        }
        let mut r = len;
        while l + 10 < r && iso_pt[r - 1] < 0.44 {
            r -= 1;
        }
        if l == 0 && r == len {
            break;
        }
        cs += l;
        ce = cs + (r - l);
    }
    (cs, ce)
}
pub fn discover_causal_regimes(
    graph: &PhiGraph,
    bytes: &[u8],
    exact: bool,
) -> Vec<DiscoveredRegime> {
    let sub = RawSubstrate::from_bytes(bytes);
    let (rels, gates_ctx, gates_pt) = graph.encode_with_gate(&sub);
    let raw = discover_causal_regimes_from_encoded(graph, &sub, &rels, &gates_ctx, &gates_pt, exact);
    if !exact || raw.is_empty() {
        return raw;
    }
    let mut tightened: Vec<(usize, usize)> = Vec::with_capacity(raw.len());
    for reg in raw {
        let (cs, ce) = tighten_regime_isolated(graph, bytes, reg.core_start, reg.core_end);
        if ce.saturating_sub(cs) >= 10 {
            tightened.push((cs, ce));
        }
    }
    if tightened.is_empty() {
        return vec![DiscoveredRegime {
            core_start: 0,
            core_end: bytes.len(),
            referent: None,
            mean_gate: gates_ctx.iter().sum::<f32>() / (bytes.len().max(1) as f32),
        }];
    }
    attach_passive_referents(&tightened, &gates_ctx, bytes.len())
}
pub fn compute_phi_sharp_prepared(
    graph: &PhiGraph,
    prep: &PreparedEpisode,
    exact_mip: bool,
) -> PhiReport {
    let (rels, gates_orig_ctx, gates_orig_pt) = graph.encode_with_gate(&prep.orig);
    let trace = graph.abduce(&rels, None);
    let residual = trace.final_mean_err;
    let r0 = trace.step_residuals[0].max(1e-4);
    let traj_gain = ((r0 - residual) / r0).clamp(0.0, 1.0);
    let mut bounce_pen = 0.0f32;
    for pair in trace.step_residuals.windows(2) {
        if pair[1] > pair[0] {
            bounce_pen += pair[1] - pair[0];
        }
    }
    let generativity = (-2.2 * residual).exp().clamp(1e-4, 1.0);
    let (neg_rels, gates_neg_ctx, gates_neg_pt) = graph.encode_with_gate(&prep.neg);
    let (_, gates_pass_ctx, gates_pass_pt) = graph.encode_with_gate(&prep.passive);
    let (_, _, cross_err) = graph.evaluate_hypothesis(&trace.final_state, &neg_rels);
    let neg_trace = graph.abduce(&neg_rels, None);
    let err_gap = ((cross_err - residual) / (cross_err + 0.05)).clamp(0.0, 1.0);
    let state_gap = (1.0 - cosine_sim(&trace.final_state, &neg_trace.final_state)).clamp(0.0, 1.0);
    let discrimination = (0.65 * err_gap + 0.35 * state_gap).clamp(1e-4, 1.0);
    let inv_orig = 1.0 / (gates_orig_ctx.len().max(1) as f32);
    let inv_neg = 1.0 / (gates_neg_ctx.len().max(1) as f32);
    let inv_pass = 1.0 / (gates_pass_ctx.len().max(1) as f32);
    let mean_g_orig = 0.5
        * (gates_orig_ctx.iter().sum::<f32>() * inv_orig
            + gates_orig_pt.iter().sum::<f32>() * inv_orig);
    let mean_g_neg = 0.5
        * (gates_neg_ctx.iter().sum::<f32>() * inv_neg
            + gates_neg_pt.iter().sum::<f32>() * inv_neg);
    let mean_g_pass = 0.5
        * (gates_pass_ctx.iter().sum::<f32>() * inv_pass
            + gates_pass_pt.iter().sum::<f32>() * inv_pass);
    let worst_bad_g = mean_g_neg.max(mean_g_pass);
    let gate_contrast = (mean_g_orig * (1.0 - worst_bad_g)).clamp(0.0, 1.0);
    let ref_rels = graph.encode_relations(&prep.refut);
    let ref_trace = graph.abduce(&ref_rels, None);
    let (_, _, ref_on_orig) = graph.evaluate_hypothesis(&ref_trace.final_state, &rels);
    let ref_cos = 0.5 * (1.0 + cosine_sim(&trace.final_state, &ref_trace.final_state));
    let refutation_stab = (ref_cos * (-1.5 * ref_on_orig).exp()).clamp(0.0, 1.0);
    let shift_rels = graph.encode_relations(&prep.shift);
    let shift_trace = graph.abduce(&shift_rels, None);
    let displacement_stab =
        (0.5 * (1.0 + cosine_sim(&trace.final_state, &shift_trace.final_state))).clamp(0.0, 1.0);
    let abs_rels = graph.encode_relations(&prep.abs);
    let abs_trace = graph.abduce(&abs_rels, None);
    let abstraction_stab =
        (0.5 * (1.0 + cosine_sim(&trace.final_state, &abs_trace.final_state))).clamp(0.0, 1.0);
    let projection =
        ((refutation_stab + displacement_stab + abstraction_stab) / 3.0).clamp(1e-4, 1.0);
    let masks: Vec<u8> = if exact_mip {
        (1u8..(1u8 << (NODES - 1))).collect()
    } else {
        prep.cuts.to_vec()
    };
    let mut mip_loss = f32::INFINITY;
    let mut mip_mask = masks[0];
    for mask in masks {
        let cut_trace = graph.abduce(&rels, Some(mask));
        let err_drop = (cut_trace.final_mean_err - residual).max(0.0);
        let state_drop = (1.0 - cosine_sim(&trace.final_state, &cut_trace.final_state)).max(0.0);
        let loss = err_drop + 0.25 * state_drop;
        if loss < mip_loss {
            mip_loss = loss;
            mip_mask = mask;
        }
    }
    let integration = (1.0 - (-3.5 * mip_loss).exp()).clamp(0.0, 1.0);
    let conj = (generativity * discrimination * projection * (0.2 + 0.8 * integration)).powf(0.25);
    let phi_sharp =
        0.85 * conj + 0.10 * traj_gain + 0.15 * gate_contrast - 0.30 * residual - 0.30 * bounce_pen;
    PhiReport {
        phi_sharp,
        generativity,
        discrimination,
        projection,
        integration,
        gate_contrast,
        mean_gate: mean_g_orig,
        mip_loss,
        mip_mask,
        residual,
        traj_gain,
        bounce_pen,
        refutation_stab,
        displacement_stab,
        abstraction_stab,
        trace,
    }
}
pub fn compute_phi_sharp(
    graph: &PhiGraph,
    bytes: &[u8],
    rng: &mut Rng,
    exact_mip: bool,
) -> PhiReport {
    let prep = PreparedEpisode::from_bytes(bytes, rng);
    compute_phi_sharp_prepared(graph, &prep, exact_mip)
}
pub fn compute_composite_phi_prepared(
    graph: &PhiGraph,
    prep: &PreparedComposite,
) -> CompositePhi {
    let (rels, gates_ctx, gates_pt) = graph.encode_with_gate(&prep.stream);
    let regimes = discover_causal_regimes_from_encoded(
        graph,
        &prep.stream,
        &rels,
        &gates_ctx,
        &gates_pt,
        false,
    );
    let m = regimes.len();
    if m < 2 {
        return CompositePhi {
            score: -0.25,
            r_top: 0.0,
            r_rest: 0.0,
            selectivity: 0.0,
            top_pair: (0, 0),
        };
    }
    let mut traces = Vec::with_capacity(m);
    let mut sum_res = 0.0f32;
    for reg in &regimes {
        let tr = graph.abduce(&rels[reg.core_start..reg.core_end], None);
        sum_res += tr.final_mean_err;
        traces.push(tr);
    }
    let mean_residual = sum_res / (m as f32);
    let mut pairs: Vec<(f32, usize, usize)> = Vec::with_capacity(m * (m - 1) / 2);
    for i in 0..m {
        let ri = &rels[regimes[i].core_start..regimes[i].core_end];
        for j in (i + 1)..m {
            let rj = &rels[regimes[j].core_start..regimes[j].core_end];
            let (_, _, e_ij) = graph.evaluate_hypothesis(&traces[i].final_state, rj);
            let (_, _, e_ji) = graph.evaluate_hypothesis(&traces[j].final_state, ri);
            let cross_gen = (-2.2 * 0.5 * (e_ij + e_ji)).exp().clamp(0.0, 1.0);
            let cos_res = 0.5 * (1.0 + cosine_sim(&traces[i].final_state, &traces[j].final_state));
            let r_ij = (cross_gen * cos_res).sqrt();
            pairs.push((r_ij, i, j));
        }
    }
    pairs.sort_by(|a, b| b.0.total_cmp(&a.0));
    let (r_top, top_i, top_j) = pairs[0];
    let r_rest = if pairs.len() > 1 {
        pairs[1..].iter().map(|p| p.0).sum::<f32>() / ((pairs.len() - 1) as f32)
    } else {
        0.0
    };
    let selectivity = r_top * (1.0 - r_rest);
    let score = selectivity - 0.18 * mean_residual;
    CompositePhi {
        score,
        r_top,
        r_rest,
        selectivity,
        top_pair: (top_i, top_j),
    }
}
pub fn compute_composite_phi(graph: &PhiGraph, bytes: &[u8]) -> CompositePhi {
    let prep = PreparedComposite::from_bytes(bytes);
    compute_composite_phi_prepared(graph, &prep)
}
pub fn project_query_to_regimes(
    graph: &PhiGraph,
    bytes: &[u8],
    regimes: &[DiscoveredRegime],
    query_idx: usize,
) -> Vec<ProjectionMatch> {
    let q_reg = regimes[query_idx];
    let q_bytes = &bytes[q_reg.core_start..q_reg.core_end];
    let q_rels = graph.encode_bytes(q_bytes);
    let q_trace = graph.abduce(&q_rels, None);
    let mut matches = Vec::with_capacity(regimes.len().saturating_sub(1));
    for (idx, reg) in regimes.iter().enumerate() {
        if idx == query_idx {
            continue;
        }
        let core_bytes = &bytes[reg.core_start..reg.core_end];
        let core_rels = graph.encode_bytes(core_bytes);
        let core_trace = graph.abduce(&core_rels, None);
        let (_, _, fwd_err) = graph.evaluate_hypothesis(&q_trace.final_state, &core_rels);
        let (_, _, bwd_err) = graph.evaluate_hypothesis(&core_trace.final_state, &q_rels);
        let cross_err = 0.5 * (fwd_err + bwd_err);
        let cross_gen = (-2.2 * cross_err).exp().clamp(0.0, 1.0);
        let cos_sim = cosine_sim(&q_trace.final_state, &core_trace.final_state);
        let cos_pos = 0.5 * (1.0 + cos_sim);
        let score = (cross_gen * cos_pos).sqrt();
        let byte_overlap = byte_overlap_ratio(q_bytes, core_bytes);
        matches.push(ProjectionMatch {
            dst_idx: idx,
            score,
            cross_err,
            cos_sim,
            core_start: reg.core_start,
            core_end: reg.core_end,
            referent: reg.referent,
            byte_overlap,
        });
    }
    matches.sort_by(|a, b| b.score.total_cmp(&a.score));
    matches
}
pub fn extract_abductive_core(
    graph: &PhiGraph,
    bytes: &[u8],
    rels: &[[f32; REL_DIM]],
    global_state: &[f32; STATE_DIM],
) -> (usize, Vec<u8>, f32) {
    let n = bytes.len();
    let win = (n / 4).clamp(10, 24).min(n);
    if n <= win {
        let (_, _, err) = graph.evaluate_hypothesis(global_state, rels);
        return (0, bytes.to_vec(), err);
    }
    let stride = (win / 4).max(1);
    let mut best_start = 0usize;
    let mut best_score = f32::NEG_INFINITY;
    let mut best_err = f32::INFINITY;
    let mut s = 0usize;
    while s + win <= n {
        let sub_rels = graph.encode_bytes(&bytes[s..s + win]);
        let sub_trace = graph.abduce(&sub_rels, None);
        let (_, _, err_on_all) = graph.evaluate_hypothesis(&sub_trace.final_state, rels);
        let sim = 0.5 * (1.0 + cosine_sim(&sub_trace.final_state, global_state));
        let score = (-2.2 * err_on_all).exp() * sim;
        if score > best_score {
            best_score = score;
            best_start = s;
            best_err = err_on_all;
        }
        s += stride;
    }
    (best_start, bytes[best_start..best_start + win].to_vec(), best_err)
}
pub fn format_partition(mask: u8) -> String {
    let mut a = Vec::new();
    let mut b = Vec::new();
    for i in 0..NODES {
        if ((mask >> i) & 1) == 0 {
            a.push(i.to_string());
        } else {
            b.push(i.to_string());
        }
    }
    format!("{{{}}} | {{{}}}", a.join(","), b.join(","))
}
fn has_supported_bridge(sub: &RawSubstrate, left: (usize, usize), right: (usize, usize)) -> bool {
    let mut bridges = 0usize;
    for t in left.0..left.1 {
        for idx in sub.offsets[t]..sub.offsets[t + 1] {
            let j = sub.edge_targets[idx] as usize;
            if j >= right.0 && j < right.1 {
                let f = &sub.edges[idx];
                if f[0] + f[1] + f[4] + f[5] >= 1.0 {
                    bridges += 1;
                    if bridges >= 2 {
                        return true;
                    }
                    break;
                }
            }
        }
    }
    false
}
pub fn discover_causal_regimes_from_encoded(
    graph: &PhiGraph,
    sub: &RawSubstrate,
    rels: &[[f32; REL_DIM]],
    gates_ctx: &[f32],
    gates_pt: &[f32],
    exact: bool,
) -> Vec<DiscoveredRegime> {
    let n = rels.len();
    if n < 12 {
        let mg = if n == 0 {
            0.0
        } else {
            gates_ctx.iter().sum::<f32>() / (n as f32)
        };
        return vec![DiscoveredRegime {
            core_start: 0,
            core_end: n,
            referent: None,
            mean_gate: mg,
        }];
    }
    let mut raw_runs: Vec<(usize, usize)> = Vec::new();
    let mut in_run = false;
    let mut run_start = 0usize;
    for t in 0..n {
        if gates_ctx[t] >= 0.46 {
            if !in_run {
                in_run = true;
                run_start = t;
            }
        } else if in_run {
            raw_runs.push((run_start, t));
            in_run = false;
        }
    }
    if in_run {
        raw_runs.push((run_start, n));
    }
    let mut merged: Vec<(usize, usize)> = Vec::new();
    for (s, e) in raw_runs {
        if let Some(last) = merged.last_mut() {
            let gap = s.saturating_sub(last.1);
            let both_solid = last.1.saturating_sub(last.0) >= 8 && e.saturating_sub(s) >= 8;
            let bridged = gap <= 6
                && last.1.saturating_sub(last.0) >= 6
                && e.saturating_sub(s) >= 6
                && has_supported_bridge(sub, *last, (s, e));
            if gap <= 1 || (gap <= 3 && both_solid) || bridged {
                last.1 = e;
                continue;
            }
        }
        merged.push((s, e));
    }
    let mut active_runs: Vec<(usize, usize)> = merged
        .into_iter()
        .filter(|&(s, e)| e.saturating_sub(s) >= 10)
        .collect();
    if active_runs.is_empty() {
        let mg = gates_ctx.iter().sum::<f32>() / (n as f32);
        return vec![DiscoveredRegime {
            core_start: 0,
            core_end: n,
            referent: None,
            mean_gate: mg,
        }];
    }
    for i in 0..active_runs.len() {
        let min_s = if i == 0 { 0 } else { active_runs[i - 1].1 };
        let max_e = if i + 1 == active_runs.len() {
            n
        } else {
            active_runs[i + 1].0
        };
        let (mut s, mut e) = active_runs[i];
        let mut trim_l = 0usize;
        while s + 10 < e && trim_l < 14 && gates_pt[s] < 0.44 {
            s += 1;
            trim_l += 1;
        }
        let mut trim_r = 0usize;
        while s + 10 < e && trim_r < 14 && gates_pt[e - 1] < 0.44 {
            e -= 1;
            trim_r += 1;
        }
        if trim_l == 0 {
            let mut ext_l = 0usize;
            while s > min_s && ext_l < 2 && gates_pt[s - 1] >= 0.68 {
                s -= 1;
                ext_l += 1;
            }
        }
        if trim_r == 0 {
            let mut ext_r = 0usize;
            while e < max_e && ext_r < 2 && gates_pt[e] >= 0.68 {
                e += 1;
                ext_r += 1;
            }
        }
        active_runs[i] = (s, e);
    }
    let mut cores: Vec<(usize, usize)> = Vec::new();
    for (s, e) in active_runs {
        for (mut cs, mut ce) in split_run_by_abduction(graph, sub, rels, s, e, exact) {
            let mut tl = 0usize;
            while cs + 10 < ce && tl < 4 && gates_pt[cs] < 0.40 {
                cs += 1;
                tl += 1;
            }
            let mut tr = 0usize;
            while cs + 10 < ce && tr < 4 && gates_pt[ce - 1] < 0.40 {
                ce -= 1;
                tr += 1;
            }
            cores.push((cs, ce));
        }
    }
    let max_mg = cores
        .iter()
        .map(|&(cs, ce)| gates_ctx[cs..ce].iter().sum::<f32>() / ((ce - cs).max(1) as f32))
        .fold(0.0f32, f32::max);
    if max_mg >= 0.72 && cores.len() >= 2 {
        let filtered: Vec<(usize, usize)> = cores
            .iter()
            .copied()
            .filter(|&(cs, ce)| {
                let mg = gates_ctx[cs..ce].iter().sum::<f32>() / ((ce - cs).max(1) as f32);
                let len = ce.saturating_sub(cs);
                mg >= 0.62 * max_mg && (len >= 14 || mg >= 0.68)
            })
            .collect();
        if !filtered.is_empty() {
            cores = filtered;
        }
    }
    attach_passive_referents(&cores, gates_ctx, n)
}

struct TermNode {
    terminal: Option<u8>,
    children: Vec<usize>,
}
struct TermGraph {
    nodes: Vec<TermNode>,
}
impl TermGraph {
    fn new() -> Self {
        Self { nodes: Vec::new() }
    }
    fn add_leaf(&mut self, byte: u8) -> usize {
        let id = self.nodes.len();
        self.nodes.push(TermNode {
            terminal: Some(byte),
            children: Vec::new(),
        });
        id
    }
    fn add_branch(&mut self, children: Vec<usize>) -> usize {
        let id = self.nodes.len();
        self.nodes.push(TermNode {
            terminal: None,
            children,
        });
        id
    }
    fn linearize(&self, root: usize, max_len: usize) -> Vec<u8> {
        let mut out = Vec::with_capacity(max_len);
        self.dfs(root, 0, max_len, &mut out);
        out
    }
    fn dfs(&self, id: usize, depth: usize, max_len: usize, out: &mut Vec<u8>) {
        if out.len() >= max_len || depth > 28 {
            return;
        }
        let node = &self.nodes[id];
        if let Some(b) = node.terminal {
            out.push(b);
            return;
        }
        for &child in &node.children {
            self.dfs(child, depth + 1, max_len, out);
            if out.len() >= max_len {
                break;
            }
        }
    }
}

pub fn sample_episode_in_range(rng: &mut Rng, family: CauseFamily, lo: u8, hi: u8) -> Episode {
    let mut tg = TermGraph::new();
    let max_len = 28 + rng.next_usize(21);
    let root = match family {
        CauseFamily::Periodic => {
            let period = 2 + rng.next_usize(4);
            let alpha = sample_alphabet(rng, period, lo, hi);
            let leaves: Vec<usize> = alpha.into_iter().map(|b| tg.add_leaf(b)).collect();
            let motif = tg.add_branch(leaves);
            let reps = (max_len / period).max(6) + 1;
            tg.add_branch(vec![motif; reps])
        }
        CauseFamily::Symmetric => {
            let half = 3 + rng.next_usize(4);
            let odd = rng.next_usize(2) == 1;
            let alpha = sample_alphabet(rng, half + usize::from(odd), lo, hi);
            let fwd: Vec<usize> = alpha[..half].iter().map(|&b| tg.add_leaf(b)).collect();
            let rev: Vec<usize> = fwd.iter().rev().copied().collect();
            let mut kids = fwd;
            if odd {
                kids.push(tg.add_leaf(alpha[half]));
            }
            kids.extend(rev);
            let pal = tg.add_branch(kids);
            tg.add_branch(vec![pal, pal, pal, pal])
        }
        CauseFamily::Nested => {
            let depth = 4 + rng.next_usize(3);
            let alpha = sample_alphabet(rng, depth * 2 + 2, lo, hi);
            let core_a = tg.add_leaf(alpha[0]);
            let core_b = tg.add_leaf(alpha[1]);
            let mut cur = tg.add_branch(vec![core_a, core_b, core_a]);
            for d in 0..depth {
                let open = tg.add_leaf(alpha[2 + d * 2]);
                let close = tg.add_leaf(alpha[3 + d * 2]);
                cur = tg.add_branch(vec![open, cur, close]);
            }
            let r = cur;
            tg.add_branch(vec![r, r, r])
        }
        CauseFamily::SharedMotif => {
            let alpha = sample_alphabet(rng, 8, lo, hi);
            let m0 = tg.add_leaf(alpha[0]);
            let m1 = tg.add_leaf(alpha[1]);
            let m2 = tg.add_leaf(alpha[2]);
            let shared = tg.add_branch(vec![m0, m1, m2, m1]);
            let b0 = tg.add_leaf(alpha[3]);
            let b1 = tg.add_leaf(alpha[4]);
            let b2 = tg.add_leaf(alpha[5]);
            let b3 = tg.add_leaf(alpha[6]);
            let b4 = tg.add_leaf(alpha[7]);
            tg.add_branch(vec![
                shared, b0, shared, b1, shared, b2, shared, b3, shared, b4, shared,
            ])
        }
        CauseFamily::Substitution => {
            let alpha = sample_alphabet(rng, 5, lo, hi);
            let a0 = tg.add_leaf(alpha[0]);
            let a1 = tg.add_leaf(alpha[1]);
            let a2 = tg.add_leaf(alpha[2]);
            let a3 = tg.add_leaf(alpha[3]);
            let a4 = tg.add_leaf(alpha[4]);
            let mut prev = tg.add_branch(vec![a0, a1]);
            let mut curr = tg.add_branch(vec![a2, a3, a4]);
            for _ in 0..6 {
                let next = tg.add_branch(vec![curr, prev]);
                prev = curr;
                curr = next;
            }
            curr
        }
        CauseFamily::Grammar => {
            let alpha = sample_alphabet(rng, 14, lo, hi);
            let sep = tg.add_leaf(alpha[0]);
            let r0 = tg.add_leaf(alpha[1]);
            let r1 = tg.add_leaf(alpha[2]);
            let r2 = tg.add_leaf(alpha[3]);
            let stem = tg.add_branch(vec![r0, r1, r2]);
            let s0 = tg.add_leaf(alpha[4]);
            let s1 = tg.add_leaf(alpha[5]);
            let w0 = tg.add_branch(vec![stem, s0]);
            let w1 = tg.add_branch(vec![stem, s1]);
            let a0 = tg.add_leaf(alpha[6]);
            let a1 = tg.add_leaf(alpha[7]);
            let a2 = tg.add_leaf(alpha[8]);
            let w2 = tg.add_branch(vec![a0, a1, a2]);
            let b0 = tg.add_leaf(alpha[9]);
            let b1 = tg.add_leaf(alpha[10]);
            let b2 = tg.add_leaf(alpha[11]);
            let w3 = tg.add_branch(vec![b0, b1, b2]);
            let clause_a = tg.add_branch(vec![w0, sep, w2, sep, w1]);
            let clause_b = tg.add_branch(vec![w2, sep, w3, sep, w0]);
            tg.add_branch(vec![clause_a, sep, clause_b, sep, clause_a])
        }
    };
    let mut effect = tg.linearize(root, max_len);
    while effect.len() < 24 {
        let copy = effect.clone();
        effect.extend_from_slice(&copy);
    }
    effect.truncate(max_len);
    Episode {
        effect,
        hidden_cause: family,
    }
}

pub fn sample_composite_episode(
    rng: &mut Rng,
    num_molds: usize,
    with_tags: bool,
    printable_query: bool,
    query_last: bool,
    sep_mode: SepMode,
) -> CompositeEpisode {
    let mut fams = CauseFamily::ALL.to_vec();
    rng.shuffle(&mut fams);
    let k = num_molds.clamp(2, fams.len());
    fams.truncate(k);
    let target_fam = fams[0];
    let sep = choose_sep(rng, sep_mode);
    let mut entries: Vec<(Option<Vec<u8>>, Vec<u8>, CauseFamily, bool, bool)> =
        Vec::with_capacity(k + 1);
    for (idx, &fam) in fams.iter().enumerate() {
        let body = sample_episode_in_range(rng, fam, 33, 90).effect;
        let tag = if with_tags {
            let prefix = if printable_query {
                format!("MOLD_{}: ", fam.name().to_uppercase()).into_bytes()
            } else {
                let tag_len = 8 + rng.next_usize(5);
                sample_alphabet(rng, tag_len, 33, 90)
            };
            Some(prefix)
        } else {
            None
        };
        entries.push((tag, body, fam, false, idx == 0));
    }
    let (q_lo, q_hi) = if printable_query {
        (97, 122)
    } else {
        (140, 220)
    };
    let query = sample_episode_in_range(rng, target_fam, q_lo, q_hi).effect;
    if query_last {
        rng.shuffle(&mut entries);
        entries.push((None, query, target_fam, true, false));
    } else {
        entries.push((None, query, target_fam, true, false));
        rng.shuffle(&mut entries);
    }
    let mut effect = Vec::new();
    let mut hidden_query_idx = 0;
    let mut hidden_match_idx = 0;
    let mut hidden_families = Vec::with_capacity(entries.len());
    let mut hidden_core_spans = Vec::with_capacity(entries.len());
    let mut hidden_tag_spans = Vec::with_capacity(entries.len());
    for (i, (tag_opt, body, fam, is_query, is_match)) in entries.into_iter().enumerate() {
        if i > 0 {
            effect.extend_from_slice(sep);
        }
        let tag_span = if let Some(tag) = tag_opt {
            let ts = effect.len();
            effect.extend_from_slice(&tag);
            let te = effect.len();
            Some((ts, te))
        } else {
            None
        };
        let cs = effect.len();
        effect.extend_from_slice(&body);
        let ce = effect.len();
        if is_query {
            hidden_query_idx = i;
        }
        if is_match {
            hidden_match_idx = i;
        }
        hidden_families.push(fam);
        hidden_core_spans.push((cs, ce));
        hidden_tag_spans.push(tag_span);
    }
    CompositeEpisode {
        effect,
        hidden_query_idx,
        hidden_match_idx,
        hidden_families,
        hidden_core_spans,
        hidden_tag_spans,
    }
}

pub fn sample_multihop_episode(
    rng: &mut Rng,
    flat: bool,
    readable: bool,
) -> MultiHopEpisode {
    let mut fams = CauseFamily::ALL.to_vec();
    rng.shuffle(&mut fams);
    fams.truncate(4);
    let words = sample_unique_words(rng, 8, readable);
    let prim_names: Vec<Vec<u8>> = words[0..4].to_vec();
    let comp_names: Vec<Vec<u8>> = words[4..8].to_vec();
    let mut pairs: Vec<(usize, usize)> = vec![(0, 1), (0, 2), (0, 3), (1, 2), (1, 3), (2, 3)];
    let target_pair = pairs.remove(0);

        (pairs[0].0, pairs[0].1, comp_names[1].clone(), false),
        (pairs[1].0, pairs[1].1, comp_names[2].clone(), false),
        (pairs[2].0, pairs[2].1, comp_names[3].clone(), false),
    ];
    rng.shuffle(&mut chosen_pairs);
    let line_sep: &[u8] = if flat { b" | " } else { b"\n" };
    let assign_op: &[u8] = match rng.next_usize(3) {
        0 => b" = ",
        1 => b" : ",
        _ => b" -> ",
    };
    let join_op: &[u8] = match rng.next_usize(3) {
        0 => b" + ",
        1 => b" & ",
        _ => b" , ",
    };
    let mut effect = Vec::new();
    for i in 0..4 {
        if i > 0 {
            effect.extend_from_slice(line_sep);
        }
        effect.extend_from_slice(&prim_names[i]);
        effect.extend_from_slice(b": ");
        let body = sample_episode_in_range(rng, fams[i], 33, 63).effect;
        effect.extend_from_slice(&body);
    }
    let mut hidden_concept_name = Vec::new();
    for (u, v, cname, is_target) in chosen_pairs {
        effect.extend_from_slice(line_sep);
        effect.extend_from_slice(&cname);
        effect.extend_from_slice(assign_op);
        effect.extend_from_slice(&prim_names[u]);
        effect.extend_from_slice(join_op);
        effect.extend_from_slice(&prim_names[v]);
        if is_target {
            hidden_concept_name = cname;
        }
    }
    effect.extend_from_slice(line_sep);
    let q1 = sample_episode_in_range(rng, fams[target_pair.0], b'a', b'z').effect;
    let q2 = sample_episode_in_range(rng, fams[target_pair.1], b'a', b'z').effect;
    effect.extend_from_slice(&q1);
    effect.extend_from_slice(&q2);
    MultiHopEpisode {
        effect,
        hidden_concept_name,
        hidden_cause_pair: (fams[target_pair.0], fams[target_pair.1]),
        hidden_prim_names: prim_names,
    }
}

use std::f32::consts::TAU;
#[derive(Clone, Debug)]
pub struct Rng {
    s: [u64; 4],
}
impl Rng {
    pub fn new(seed: u64) -> Self {
        let mut z = seed.wrapping_add(0x9E37_79B9_7F4A_7C15);
        let mut s = [0u64; 4];
        for slot in &mut s {
            z = z.wrapping_add(0x9E37_79B9_7F4A_7C15);
            let mut x = z;
            x = (x ^ (x >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
            x = (x ^ (x >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
            *slot = x ^ (x >> 31);
        }
        Self { s }
    }
    pub fn next_u64(&mut self) -> u64 {
        let result = self.s[1].wrapping_mul(5).rotate_left(7).wrapping_mul(9);
        let t = self.s[1] << 17;
        self.s[2] ^= self.s[0];
        self.s[3] ^= self.s[1];
        self.s[1] ^= self.s[2];
        self.s[0] ^= self.s[3];
        self.s[2] ^= t;
        self.s[3] = self.s[3].rotate_left(45);
        result
    }
    pub fn next_u8(&mut self) -> u8 {
        (self.next_u64() >> 56) as u8
    }
    pub fn next_usize(&mut self, bound: usize) -> usize {
        if bound == 0 {
            return 0;
        }
        (self.next_u64() as usize) % bound
    }
    pub fn next_f32(&mut self) -> f32 {
        ((self.next_u64() >> 40) as f32) / ((1u32 << 24) as f32)
    }
    pub fn next_normal(&mut self) -> f32 {
        let u1 = self.next_f32().max(1e-7);
        let u2 = self.next_f32();
        (-2.0 * u1.ln()).sqrt() * (TAU * u2).cos()
    }
    pub fn shuffle<T>(&mut self, slice: &mut [T]) {
        for i in (1..slice.len()).rev() {
            let j = self.next_usize(i + 1);
            slice.swap(i, j);
        }
    }
}

fn span_iou(a: (usize, usize), b: (usize, usize)) -> f32 {
    let inter_s = a.0.max(b.0);
    let inter_e = a.1.min(b.1);
    if inter_e <= inter_s {
        return 0.0;
    }
    let inter = (inter_e - inter_s) as f32;
    let union = ((a.1 - a.0) + (b.1 - b.0)) as f32 - inter;
    if union <= 0.0 { 0.0 } else { inter / union }
}
fn batch_collapse_penalty(states: &[[f32; STATE_DIM]]) -> f32 {
    let mut sum = 0.0f32;
    let mut max_sq = 0.0f32;
    let mut pairs = 0usize;
    for i in 0..states.len() {
        for j in (i + 1)..states.len() {
            let c = cosine_sim(&states[i], &states[j]).max(0.0);
            let sq = c * c;
            sum += sq;
            if sq > max_sq {
                max_sq = sq;
            }
            pairs += 1;
        }
    }
    if pairs == 0 {
        0.0
    } else {
        0.5 * (sum / (pairs as f32)) + 0.5 * max_sq
    }
}
fn cross_specificity(
    graph: &PhiGraph,
    singles: &[PreparedEpisode],
    states: &[[f32; STATE_DIM]],
    residuals: &[f32],
) -> f32 {
    let n = singles.len();
    if n < 2 {
        return 0.0;
    }
    let rels_list: Vec<_> = singles
        .iter()
        .map(|ep| graph.encode_relations(&ep.orig))
        .collect();
    let mut spec_sum = 0.0f32;
    for i in 0..n {
        let mut min_other = f32::INFINITY;
        for j in 0..n {
            if i == j {
                continue;
            }
            let (_, _, err_ij) = graph.evaluate_hypothesis(&states[i], &rels_list[j]);
            if err_ij < min_other {
                min_other = err_ij;
            }
        }
        spec_sum += ((min_other - residuals[i]) / (min_other + 0.05)).clamp(0.0, 1.0);
    }
    spec_sum / (n as f32)
}
fn eval_batch(
    graph: &PhiGraph,
    singles: &[PreparedEpisode],
    composites: &[PreparedComposite],
    comp_w: f32,
) -> f32 {
    let mut sum_phi = 0.0f32;
    let mut sum_gate_c = 0.0f32;
    let mut min_gate_c = 1.0f32;
    let mut states = Vec::with_capacity(singles.len());
    let mut residuals = Vec::with_capacity(singles.len());
    for ep in singles {
        let rep = compute_phi_sharp_prepared(graph, ep, false);
        sum_phi += rep.phi_sharp;
        sum_gate_c += rep.gate_contrast;
        if rep.gate_contrast < min_gate_c {
            min_gate_c = rep.gate_contrast;
        }
        states.push(rep.trace.final_state);
        residuals.push(rep.residual);
    }
    let n = singles.len() as f32;
    let mean_phi = sum_phi / n;
    let mean_gate_c = sum_gate_c / n;
    let collapse = batch_collapse_penalty(&states);
    let spec = cross_specificity(graph, singles, &states, &residuals);
    let mut comp_sum = 0.0f32;
    for comp in composites {
        comp_sum += compute_composite_phi_prepared(graph, comp).score;
    }
    let mean_comp = if composites.is_empty() {
        0.0
    } else {
        comp_sum / (composites.len() as f32)
    };
    mean_phi
        + 0.55 * spec
        + 0.20 * mean_gate_c
        + 0.30 * min_gate_c
        - 0.40 * collapse
        + comp_w * mean_comp
}
pub fn train_phisharp(cfg: &TrainConfig) -> PhiGraph {
    let start = Instant::now();
    let mut rng = Rng::new(cfg.seed);
    let mut theta = PhiGraph::random(&mut rng).weights;
    let mut m = vec![0.0f32; NUM_PARAMS];
    let mut v = vec![0.0f32; NUM_PARAMS];
    let beta1 = 0.9f32;
    let beta2 = 0.999f32;
    let eps_adam = 1e-8f32;
    let n_threads = thread::available_parallelism()
        .map(|n| n.get())
        .unwrap_or(4)
        .min(cfg.pop_pairs.max(1));
    println!(
        "Entraînement Phisharp v0.6 ({} paramètres, {} étapes, {} paires ES, {} threads)...",
        NUM_PARAMS, cfg.steps, cfg.pop_pairs, n_threads
    );
    for step in 1..=cfg.steps {
        let comp_w = 0.10 + 0.45 * ((step as f32) / 120.0).min(1.0);
        let mut prep_rng = Rng::new(rng.next_u64());
        let singles: Vec<PreparedEpisode> = CauseFamily::ALL
            .iter()
            .map(|&fam| {
                let eff = sample_episode(&mut rng, fam).effect;
                PreparedEpisode::from_bytes(&eff, &mut prep_rng)
            })
            .collect();
        let composites = vec![
            PreparedComposite::from_bytes(
                &sample_composite_episode(&mut rng, 3, false, false, false, SepMode::None).effect,
            ),
            PreparedComposite::from_bytes(
                &sample_composite_episode(&mut rng, 3, true, false, false, SepMode::Random).effect,
            ),
        ];
        let pair_seeds: Vec<u64> = (0..cfg.pop_pairs).map(|_| rng.next_u64()).collect();
        let results: Vec<(u64, f32, f32)> = thread::scope(|s| {
            let chunk_size = cfg.pop_pairs.div_ceil(n_threads);
            let mut handles = Vec::new();
            for chunk in pair_seeds.chunks(chunk_size) {
                let theta_ref = &theta;
                let singles_ref = &singles;
                let comp_ref = &composites;
                let sigma = cfg.sigma;
                handles.push(s.spawn(move || {
                    let mut local = Vec::with_capacity(chunk.len());
                    for &pseed in chunk {
                        let mut prng = Rng::new(pseed);
                        let mut w_pos = Vec::with_capacity(NUM_PARAMS);
                        let mut w_neg = Vec::with_capacity(NUM_PARAMS);
                        for &w in theta_ref {
                            let noise = prng.next_normal() * sigma;
                            w_pos.push(w + noise);
                            w_neg.push(w - noise);
                        }
                        let f_pos = eval_batch(
                            &PhiGraph::from_weights(w_pos),
                            singles_ref,
                            comp_ref,
                            comp_w,
                        );
                        let f_neg = eval_batch(
                            &PhiGraph::from_weights(w_neg),
                            singles_ref,
                            comp_ref,
                            comp_w,
                        );
                        local.push((pseed, f_pos, f_neg));
                    }
                    local
                }));
            }
            let mut merged = Vec::with_capacity(cfg.pop_pairs);
            for h in handles {
                merged.extend(h.join().unwrap());
            }
            merged
        });
        let diffs: Vec<f32> = results.iter().map(|&(_, fp, fn_)| fp - fn_).collect();
        let var: f32 = diffs.iter().map(|d| d * d).sum::<f32>() / (diffs.len() as f32);
        let std = var.sqrt().max(1e-6);
        let mut grad = vec![0.0f32; NUM_PARAMS];
        let scale = 1.0 / (2.0 * (cfg.pop_pairs as f32) * cfg.sigma * std);
        for (&(pseed, _, _), &diff) in results.iter().zip(diffs.iter()) {
            let mut prng = Rng::new(pseed);
            let coeff = diff * scale;
            for g in &mut grad {
                *g += coeff * prng.next_normal();
            }
        }
        let t = step as i32;
        let b1_corr = 1.0 - beta1.powi(t);
        let b2_corr = 1.0 - beta2.powi(t);
        for i in 0..NUM_PARAMS {
            let g = grad[i] - 1e-4 * theta[i];
            m[i] = beta1 * m[i] + (1.0 - beta1) * g;
            v[i] = beta2 * v[i] + (1.0 - beta2) * g * g;
            let m_hat = m[i] / b1_corr;
            let v_hat = v[i] / b2_corr;
            theta[i] += cfg.lr * m_hat / (v_hat.sqrt() + eps_adam);
        }
        if step == 1 || step % 25 == 0 || step == cfg.steps {
            let model = PhiGraph::from_weights(theta.clone());
            let mut phi = 0.0;
            let mut gen_sum = 0.0;
            let mut disc = 0.0;
            let mut integ = 0.0;
            let mut gate_c = 0.0;
            let mut min_gate_c = 1.0f32;
            let mut res = 0.0;
            let mut states = Vec::with_capacity(singles.len());
            let mut residuals = Vec::with_capacity(singles.len());
            for ep in &singles {
                let r = compute_phi_sharp_prepared(&model, ep, false);
                phi += r.phi_sharp;
                gen_sum += r.generativity;
                disc += r.discrimination;
                integ += r.integration;
                gate_c += r.gate_contrast;
                if r.gate_contrast < min_gate_c {
                    min_gate_c = r.gate_contrast;
                }
                res += r.residual;
                states.push(r.trace.final_state);
                residuals.push(r.residual);
            }
            let n = singles.len() as f32;
            let coll = batch_collapse_penalty(&states);
            let spec = cross_specificity(&model, &singles, &states, &residuals);
            let sel: f32 = composites
                .iter()
                .map(|c| compute_composite_phi_prepared(&model, c).selectivity)
                .sum::<f32>()
                / (composites.len() as f32);
            println!(
                "step {:>4}/{:<4} | φ={:+.4} | Gen={:.3} Disc={:.3} Spec={:.3} Gate={:.2}/{:.2} IIT={:.3} RefSel={:.3} | res={:.3} coll={:.3} ({:.1}s)",
                step,
                cfg.steps,
                phi / n,
                gen_sum / n,
                disc / n,
                spec,
                gate_c / n,
                min_gate_c,
                integ / n,
                sel,
                res / n,
                coll,
                start.elapsed().as_secs_f32()
            );
        }
    }
    let final_model = PhiGraph::from_weights(theta);
    evaluate_post_hoc(&final_model, cfg.seed ^ 0xDEAD_BEEF);
    final_model
}
fn evaluate_post_hoc(model: &PhiGraph, seed: u64) {
    let mut rng = Rng::new(seed);
    let mut same_sum = 0.0f32;
    let mut diff_sum = 0.0f32;
    let trials = 48;
    for _ in 0..trials {
        let f1 = CauseFamily::ALL[rng.next_usize(CauseFamily::ALL.len())];
        let mut f2 = CauseFamily::ALL[rng.next_usize(CauseFamily::ALL.len())];
        while f2 == f1 {
            f2 = CauseFamily::ALL[rng.next_usize(CauseFamily::ALL.len())];
        }
        let e_a = sample_episode_in_range(&mut rng, f1, 33, 90);
        let e_b = sample_episode_in_range(&mut rng, f1, 140, 220);
        let e_c = sample_episode_in_range(&mut rng, f2, 140, 220);
        let h_a = model.abduce(&model.encode_bytes(&e_a.effect), None).final_state;
        let h_b = model.abduce(&model.encode_bytes(&e_b.effect), None).final_state;
        let h_c = model.abduce(&model.encode_bytes(&e_c.effect), None).final_state;
        same_sum += cosine_sim(&h_a, &h_b);
        diff_sum += cosine_sim(&h_a, &h_c);
    }
    let same_avg = same_sum / (trials as f32);
    let diff_avg = diff_sum / (trials as f32);
    let proj_trials = 50;
    let mut hits_glued = 0usize;
    let mut hits_tagged = 0usize;
    let mut ref_hits = 0usize;
    for _ in 0..proj_trials {
        let ep_raw = sample_composite_episode(&mut rng, 4, false, false, true, SepMode::None);
        let regs_raw = discover_causal_regimes(model, &ep_raw.effect, true);
        if regs_raw.len() >= 2 {
            let q_idx = regs_raw.len() - 1;
            let q_reg = regs_raw[q_idx];
            let true_q = ep_raw.hidden_core_spans[ep_raw.hidden_query_idx];
            let true_m = ep_raw.hidden_core_spans[ep_raw.hidden_match_idx];
            if span_iou((q_reg.core_start, q_reg.core_end), true_q) >= 0.45 {
                let matches = project_query_to_regimes(model, &ep_raw.effect, &regs_raw, q_idx);
                if let Some(top) = matches.first()
                    && span_iou((top.core_start, top.core_end), true_m) >= 0.45
                {
                    hits_glued += 1;
                }
            }
        }
        let ep_tag = sample_composite_episode(&mut rng, 4, true, true, true, SepMode::Random);
        let regs_tag = discover_causal_regimes(model, &ep_tag.effect, true);
        if regs_tag.len() >= 2 {
            let q_idx = regs_tag.len() - 1;
            let q_reg = regs_tag[q_idx];
            let true_q = ep_tag.hidden_core_spans[ep_tag.hidden_query_idx];
            let true_m = ep_tag.hidden_core_spans[ep_tag.hidden_match_idx];
            let true_tag = ep_tag.hidden_tag_spans[ep_tag.hidden_match_idx];
            if span_iou((q_reg.core_start, q_reg.core_end), true_q) >= 0.45 {
                let matches = project_query_to_regimes(model, &ep_tag.effect, &regs_tag, q_idx);
                if let Some(top) = matches.first()
                    && span_iou((top.core_start, top.core_end), true_m) >= 0.45
                {
                    hits_tagged += 1;
                    if let (Some(ext_ref), Some(gt_tag)) = (top.referent, true_tag)
                        && span_iou(ext_ref, gt_tag) >= 0.50
                    {
                        ref_hits += 1;
                    }
                }
            }
        }
    }
    let mh_trials = 50;
    let mut mh_decomposed = 0usize;
    let mut mh_hits = 0usize;
    let mut mh_flat_hits = 0usize;
    let mut mh_margin = 0.0f32;
    for _ in 0..mh_trials {
        let ep = sample_multihop_episode(&mut rng, false, false);
        if let Some(ana) = analyze_multihop(model, &ep.effect) {
            if ana.target_spans.len() >= 2 {
                mh_decomposed += 1;
            }
            if let Some(top) = ana.hop2_matches.first()
                && top.concept.name_bytes == ep.hidden_concept_name
            {
                mh_hits += 1;
                if ana.hop2_matches.len() >= 2 {
                    mh_margin += top.score - ana.hop2_matches[1].score;
                }
            }
        }
        let ep_flat = sample_multihop_episode(&mut rng, true, false);
        if let Some(ana) = analyze_multihop(model, &ep_flat.effect)
            && let Some(top) = ana.hop2_matches.first()
            && top.concept.name_bytes == ep_flat.hidden_concept_name
        {
            mh_flat_hits += 1;
        }
    }
    println!("\n--- Évaluation post-hoc 1 : Abstraction Méta-relationnelle (0 octet commun) ---");
    println!("  Même cause cachée : cos = {:+.4}", same_avg);
    println!("  Causes différentes: cos = {:+.4}", diff_avg);
    println!("  Marge             : {:+.4}", same_avg - diff_avg);
    println!("\n--- Évaluation post-hoc 2 : Flux continu 1-saut auto-segmenté (4 moules, hasard = 25%) ---");
    println!(
        "  Moules collés SANS séparateur (0 octet commun)     : {}/{} ({:.1}%)",
        hits_glued,
        proj_trials,
        100.0 * (hits_glued as f32) / (proj_trials as f32)
    );
    println!(
        "  Moules étiquetés, sép. aléatoire (0 octet commun)  : {}/{} ({:.1}%)  [référent exact={}/{}]",
        hits_tagged,
        proj_trials,
        100.0 * (hits_tagged as f32) / (proj_trials as f32),
        ref_hits,
        hits_tagged
    );
    println!("\n--- Évaluation post-hoc 3 : Multi-Sauts sur Définitions Composées (4 concepts, hasard = 25%) ---");
    println!(
        "  Décomposition 2 causes (cible hybride)             : {}/{} ({:.1}%)",
        mh_decomposed,
        mh_trials,
        100.0 * (mh_decomposed as f32) / (mh_trials as f32)
    );
    println!(
        "  Concept composé exact (multiligne, 0 octet commun) : {}/{} ({:.1}%)  [marge Top1-Top2 = {:+.4}]",
        mh_hits,
        mh_trials,
        100.0 * (mh_hits as f32) / (mh_trials as f32),
        if mh_hits > 0 {
            mh_margin / (mh_hits as f32)
        } else {
            0.0
        }
    );
    println!(
        "  Concept composé exact (flux plat, 0 octet commun)  : {}/{} ({:.1}%)",
        mh_flat_hits,
        mh_trials,
        100.0 * (mh_flat_hits as f32) / (mh_trials as f32)
    );
}
