#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub enum Origin {
    Gold,
    WrongEntity,    
    WrongPredicate, 
    Insufficient,   
    Redundant,      
    RivalCause,     
}
impl Origin {
    pub fn label(self) -> &'static str {
        match self {
            Origin::Gold => "gold",
            Origin::WrongEntity => "wrong-entity",
            Origin::WrongPredicate => "wrong-predicate",
            Origin::Insufficient => "insufficient",
            Origin::Redundant => "redundant",
            Origin::RivalCause => "rival-cause",
        }
    }
}
#[derive(Clone, Debug)]
pub struct Candidate {
    pub facts: Vec<NodeId>, 
    pub origin: Origin,
    pub hits: usize,   
    pub loglik: f64,   
    pub logprior: f64, 
    pub score: f64,    
}
pub struct Problem {
    pub seed: u64,
    pub params: Params,
    pub world: World,               
    pub disclosed: Vec<NodeId>,     
    pub observed: Vec<NodeId>,      
    pub candidates: Vec<Candidate>, 
    pub gold: usize,                
    pub null: Candidate,            
    pub certificate: Vec<Step>,     
    pub truth: Vec<NodeId>,         
    pub trace_len: usize,           
    pub mc_seed: u64,               
    pub attempts: usize,

}
fn judge(
    w: &mut World,
    base: &[NodeId],
    hyp: &[NodeId],
    origin: Origin,
    obs: &[NodeId],
    mc: u64,
    p: &Params,
) -> Candidate {
    let mut init = State::default();
    for &f in base.iter().chain(hyp.iter()) {
        init.facts.insert(f);
    }
    let mut hits = 0usize;
    for t in 0..p.trials {
        let mut rng = Rng::new(mix(mc, t as u64));
        let (fin, _) = simulate(&mut w.g, &w.rules, &init, &mut rng, p.horizon, p.max_steps);
        if obs.iter().all(|f| fin.facts.contains(f)) {
            hits += 1;
        }
    }
    let ll = ((hits as f64 + 0.5) / (p.trials as f64 + 1.0)).ln();
    let lp: f64 = hyp.iter().map(|&f| w.logit_prior(f)).sum();
    Candidate {
        facts: hyp.to_vec(),
        origin,
        hits,
        loglik: round(ll, 4),
        logprior: round(lp, 4),
        score: round(ll + lp, 4),
    }
}
fn atom(w: &mut World, p: usize, args: &[NodeId]) -> NodeId {
    let s = w.preds[p].sym;
    w.g.mk(s, args)
}
fn sorted(mut v: Vec<NodeId>) -> Vec<NodeId> {
    v.sort_unstable();
    v.dedup();
    v
}
fn backchain(
    w: &mut World,
    rng: &mut Rng,
    goal: NodeId,
    depth: usize,
    out: &mut Vec<NodeId>,
    budget: &mut usize,
) -> bool {
    if depth > 6 || *budget == 0 {
        return false;
    }
    *budget -= 1;
    let pi = match w.pred_of(goal) {
        Some(i) => i,
        None => return false,
    };
    if w.preds[pi].layer == 0 {
        let a = w.g.args(goal);
        if a.len() == 2 && a[0] == a[1] {
            return false;
        }
        out.push(goal);
        return true;
    }
    let head = w.g.head(goal);
    let mut opts: Vec<(usize, usize, usize)> = Vec::new();
    for (ri, r) in w.rules.iter().enumerate() {
        for (ai, alt) in r.alts.iter().enumerate() {
            for (k, &f) in alt.1.iter().enumerate() {
                if w.g.head(f) == head {
                    opts.push((ri, ai, k));
                }
            }
        }
    }
    rng.shuffle(&mut opts);
    for (ri, ai, k) in opts {
        let nv = w.rules[ri].nvars;
        let ns = w.rules[ri].skolem.len();
        let pat = w.rules[ri].alts[ai].1[k];
        let mut sub: Sub = vec![None; nv + ns];
        if !w.g.match_term(pat, goal, &mut sub) {
            continue;
        }
        for v in 0..nv {
            if sub[v].is_none() {
                sub[v] = Some(*rng.pick(&w.consts));
            }
        }
        let binding: Vec<NodeId> = sub[..nv].iter().map(|x| x.unwrap_or(0)).collect();
        let mut full = sub.clone();
        let mut sound = true;
        for s in 0..ns {
            let sk = w.rules[ri].skolem[s];
            let t = w.g.mk(sk, &binding);
            if let Some(b) = sub[nv + s] {
                if b != t {
                    sound = false;
                    break;
                }
            }
            full[nv + s] = Some(t);
        }
        if !sound {
            continue;
        }
        let lhs: Vec<NodeId> = w.rules[ri].lhs.clone();
        let mut prem: Vec<NodeId> = Vec::with_capacity(lhs.len());
        for q in lhs {
            prem.push(w.g.subst(q, &full));
        }
        let mark = out.len();
        let mut good = true;
        for q in prem {
            if !backchain(w, rng, q, depth + 1, out, budget) {
                good = false;
                break;
            }
        }
        if good {
            return true;
        }
        out.truncate(mark);
    }
    false
}
#[allow(clippy::too_many_arguments)]
fn mutate(
    w: &mut World,
    rng: &mut Rng,
    kind: Origin,
    p: &Params,
    disclosed: &BTreeSet<NodeId>,
    gold: &[NodeId],
    obs: &[NodeId],
    per_obs: &[BTreeSet<NodeId>],
) -> Option<Vec<NodeId>> {
    match kind {
        Origin::Gold => None,
        Origin::RivalCause => {
            let n = obs.len();
            let mut redo: Vec<bool> = (0..n).map(|_| rng.chance(0.5)).collect();
            if !redo.iter().any(|&b| b) {
                let k = rng.below(n);
                redo[k] = true;
            }
            let mut leaves: BTreeSet<NodeId> = BTreeSet::new();
            for i in 0..n {
                if redo[i] {
                    let mut acc: Vec<NodeId> = Vec::new();
                    let mut budget = 40usize;
                    if !backchain(w, rng, obs[i], 0, &mut acc, &mut budget) {
                        return None;
                    }
                    leaves.extend(acc);
                } else {
                    leaves.extend(per_obs[i].iter().cloned());
                }
            }
            let h: Vec<NodeId> = leaves.into_iter().filter(|f| !disclosed.contains(f)).collect();
            if h.is_empty() || h.len() > p.max_expl + 2 {
                return None;
            }
            Some(h)
        }
        Origin::WrongEntity | Origin::WrongPredicate => {
            let f = *rng.pick(gold);
            let pi = w.pred_of(f)?;
            let mut args: Vec<NodeId> = w.g.args(f).to_vec();
            let mut q = pi;
            if kind == Origin::WrongEntity {
                let k = rng.below(args.len());
                let c = *rng.pick(&w.consts);
                if c == args[k] {
                    return None;
                }
                args[k] = c;
            } else {
                let same: Vec<usize> = (0..w.preds.len())
                    .filter(|&i| i != pi && w.preds[i].layer == 0 && w.preds[i].arity == args.len())
                    .collect();
                if same.is_empty() {
                    return None;
                }
                q = same[rng.below(same.len())];
            }
            if args.len() == 2 && args[0] == args[1] {
                return None;
            }
            let nf = atom(w, q, &args);
            if disclosed.contains(&nf) || gold.contains(&nf) {
                return None;
            }
            let mut h: Vec<NodeId> = gold.iter().cloned().filter(|&x| x != f).collect();
            h.push(nf);
            Some(sorted(h))
        }
        Origin::Insufficient => {
            if gold.len() < 2 {
                return None;
            }
            let f = *rng.pick(gold);
            let h: Vec<NodeId> = gold.iter().cloned().filter(|&x| x != f).collect();
            Some(h)
        }
        Origin::Redundant => {
            let l0: Vec<usize> = (0..w.preds.len()).filter(|&i| w.preds[i].layer == 0).collect();
            let q = l0[rng.below(l0.len())];
            let mut args: Vec<NodeId> = Vec::new();
            for _ in 0..w.preds[q].arity {
                args.push(*rng.pick(&w.consts));
            }
            if args.len() == 2 && args[0] == args[1] {
                return None;
            }
            let nf = atom(w, q, &args);
            if disclosed.contains(&nf) || gold.contains(&nf) || w.logit_prior(nf) > -(p.margin + 0.5) {
                return None;
            }
            let mut h = gold.to_vec();
            h.push(nf);
            Some(sorted(h))
        }
    }
}
fn make_rivals(
    w: &mut World,
    rng: &mut Rng,
    p: &Params,
    disclosed: &BTreeSet<NodeId>,
    gold: &[NodeId],
    obs: &[NodeId],
    per_obs: &[BTreeSet<NodeId>],
) -> Vec<(Origin, Vec<NodeId>)> {
    const CYCLE: [Origin; 5] = [
        Origin::RivalCause,
        Origin::WrongEntity,
        Origin::WrongPredicate,
        Origin::Insufficient,
        Origin::Redundant,
    ];
    let mut seen: BTreeSet<Vec<NodeId>> = BTreeSet::new();
    seen.insert(gold.to_vec());
    let mut out: Vec<(Origin, Vec<NodeId>)> = Vec::new();
    let mut tries = 0usize;
    while out.len() < p.distractors && tries < 12 * p.distractors.max(1) {
        let kind = CYCLE[tries % CYCLE.len()];
        tries += 1;
        if let Some(h) = mutate(w, rng, kind, p, disclosed, gold, obs, per_obs) {
            if seen.insert(h.clone()) {
                out.push((kind, h));
            }
        }
    }
    out
}
struct Draft {
    disclosed: Vec<NodeId>,
    observed: Vec<NodeId>,
    candidates: Vec<Candidate>, 
    null: Candidate,
    certificate: Vec<Step>,
    truth: Vec<NodeId>,
    trace_len: usize,
}
fn draft(w: &mut World, rng: &mut Rng, p: &Params, mc: u64) -> Option<Draft> {
    let init = w.sample_init(rng);
    let (fin, steps) = simulate(&mut w.g, &w.rules, &init, rng, p.horizon, p.max_steps);
    if steps.is_empty() {
        return None;
    }
    let mut bg: BTreeSet<NodeId> = BTreeSet::new();
    for &f in init.facts.iter() {
        if !w.is_latent(f) && rng.chance(p.known) {
            bg.insert(f);
        }
    }
    let need_latent = w.preds.iter().any(|d| d.latent);
    let mut pool: Vec<(NodeId, BTreeSet<NodeId>)> = Vec::new();
    for &f in fin.facts.iter() {
        if init.facts.contains(&f) {
            continue;
        }
        let (_, leaves) = slice_trace(&steps, &[f]);
        let hidden: BTreeSet<NodeId> = leaves.difference(&bg).cloned().collect();
        if hidden.is_empty() || hidden.len() > p.max_expl {
            continue;
        }
        pool.push((f, hidden));
    }
    let seeds: Vec<usize> = (0..pool.len())
        .filter(|&i| !need_latent || pool[i].1.iter().any(|&h| w.is_latent(h)))
        .collect();
    if seeds.is_empty() {
        return None;
    }
    let first = seeds[rng.below(seeds.len())];
    let mut chosen: Vec<usize> = vec![first];
    let mut hidden: BTreeSet<NodeId> = pool[first].1.clone();
    let mut rest: Vec<usize> = (0..pool.len()).filter(|&i| i != first).collect();
    rng.shuffle(&mut rest);
    rest.sort_by_key(|&i| std::cmp::Reverse(pool[i].1.intersection(&pool[first].1).count()));
    for &i in rest.iter() {
        if chosen.len() >= p.obs.max(1) {
            break;
        }
        let merged: BTreeSet<NodeId> = hidden.union(&pool[i].1).cloned().collect();
        if merged.len() <= p.max_expl {
            hidden = merged;
            chosen.push(i);
        }
    }
    let mut obs: Vec<NodeId> = chosen.iter().map(|&i| pool[i].0).collect();
    obs.sort_unstable();
    let (used, leaves) = slice_trace(&steps, &obs);
    let gold: Vec<NodeId> = leaves.iter().cloned().filter(|f| !bg.contains(f)).collect();
    if gold.is_empty() || gold.len() > p.max_expl {
        return None;
    }
    if need_latent && !gold.iter().any(|&f| w.is_latent(f)) {
        return None;
    }
    let per_obs: Vec<BTreeSet<NodeId>> = obs.iter().map(|&o| slice_trace(&steps, &[o]).1).collect();
    let rivals = make_rivals(w, rng, p, &bg, &gold, &obs, &per_obs);
    if rivals.len() < p.distractors.min(3) {
        return None;
    }
    let base: Vec<NodeId> = bg.iter().cloned().collect();
    let top = judge(w, &base, &gold, Origin::Gold, &obs, mc, p);
    if top.hits * 6 < p.trials {
        return None; 
    }
    let null = judge(w, &base, &[], Origin::Insufficient, &obs, mc, p);
    if top.score - null.score < p.margin {
        return None; 
    }
    let mut cands: Vec<Candidate> = vec![top];
    for (origin, h) in rivals.iter() {
        let c = judge(w, &base, h, *origin, &obs, mc, p);
        if cands[0].score - c.score < p.margin {
            return None;
        }
        cands.push(c);
    }
    let certificate: Vec<Step> = used.iter().map(|&i| steps[i].clone()).collect();
    Some(Draft {
        disclosed: base,
        observed: obs,
        candidates: cands,
        null,
        certificate,
        truth: init.facts.iter().cloned().collect(),
        trace_len: steps.len(),
    })
}
fn mask_latent(w: &mut World) {
    let mut k = 0usize;
    for i in 0..w.preds.len() {
        if w.preds[i].latent {
            k += 1;
            let s = w.preds[i].sym;
            w.g.rename_sym(s, &format!("H{k}"));
        }
    }
}
pub fn build(seed: u64, p: &Params) -> Result<Problem, String> {
    let mut rng = Rng::new(mix(seed, 0xAB5D));
    let mut world = World::generate(&mut rng, p);
    let mut worlds = 1usize;
    let total = p.attempts.max(1);
    for attempt in 1..=total {
        if attempt > 1 && (attempt - 1) % PER_WORLD == 0 {
            world = World::generate(&mut rng, p);
            worlds += 1;
        }
        let mc = mix(seed, 0x4D43_0000 + attempt as u64);
        let Some(d) = draft(&mut world, &mut rng, p, mc) else {
            continue;
        };
        let mut order: Vec<usize> = (0..d.candidates.len()).collect();
        rng.shuffle(&mut order);
        let gold = order.iter().position(|&i| i == 0).unwrap_or(0);
        let candidates: Vec<Candidate> = order.iter().map(|&i| d.candidates[i].clone()).collect();
        mask_latent(&mut world);
        return Ok(Problem {
            seed,
            params: p.clone(),
            world,
            disclosed: d.disclosed,
            observed: d.observed,
            candidates,
            gold,
            null: d.null,
            certificate: d.certificate,
            truth: d.truth,
            trace_len: d.trace_len,
            mc_seed: mc,
            attempts: attempt,
            worlds,
        });
    }
    Err(format!(
        "no instance met the acceptance rules after {total} attempts (seed {seed})"
    ))
}
impl Problem {
    pub fn text(&self, f: NodeId) -> String {
        self.world.g.show(f)
    }
    pub fn gold_candidate(&self) -> &Candidate {
        &self.candidates[self.gold]
    }
    pub fn posterior(&self) -> Vec<f64> {
        let top = self.candidates.iter().map(|c| c.score).fold(f64::NEG_INFINITY, f64::max);
        let e: Vec<f64> = self.candidates.iter().map(|c| (c.score - top).exp()).collect();
        let z: f64 = e.iter().sum();
        e.into_iter().map(|x| round(x / z, 4)).collect()
    }
    pub fn verify(&mut self) -> Result<(), String> {
        self.world.validate()?;
        if self.observed.is_empty() {
            return Err("no observations".to_string());
        }
        if self.gold >= self.candidates.len()
            || self.candidates[self.gold].origin != Origin::Gold
            || self.candidates.iter().filter(|c| c.origin == Origin::Gold).count() != 1
        {
            return Err("exactly one candidate must be the gold one".to_string());
        }
        let gold_facts: Vec<NodeId> = self.candidates[self.gold].facts.clone();
        for &f in self.disclosed.iter() {
            if self.world.is_latent(f) {
                return Err("a hidden-cause fact was disclosed".to_string());
            }
        }
        for (i, c) in self.candidates.iter().enumerate() {
            if c.facts.is_empty() {
                return Err(format!("candidate {i}: empty hypothesis"));
            }
            for &f in c.facts.iter() {
                let abducible = match self.world.pred_of(f) {
                    Some(pi) => self.world.preds[pi].layer == 0,
                    None => false,
                };
                if !abducible || self.disclosed.contains(&f) {
                    return Err(format!(
                        "candidate {i}: {} is not an undisclosed layer-0 fact",
                        self.world.g.show(f)
                    ));
                }
            }
        }
        for &f in self.disclosed.iter().chain(gold_facts.iter()) {
            if !self.truth.contains(&f) {
                return Err(format!(
                    "{} is not part of the hidden initial state",
                    self.world.g.show(f)
                ));
            }
        }
        for &o in self.observed.iter() {
            if self.truth.contains(&o) {
                return Err(format!("observation {} is an initial fact", self.world.g.show(o)));
            }
        }
        let mut init = State::default();
        for &f in self.disclosed.iter().chain(gold_facts.iter()) {
            init.facts.insert(f);
        }
        let fin = replay(&mut self.world.g, &self.world.rules, &init, &self.certificate, false)?;
        for &o in self.observed.iter() {
            if !fin.facts.contains(&o) {
                return Err(format!("certificate does not derive {}", self.world.g.show(o)));
            }
        }
        for &skip in gold_facts.iter() {
            let mut reduced = State::default();
            for &f in self.disclosed.iter().chain(gold_facts.iter()) {
                if f != skip {
                    reduced.facts.insert(f);
                }
            }
            let ok = replay(&mut self.world.g, &self.world.rules, &reduced, &self.certificate, false).is_ok();
            if ok {
                return Err(format!(
                    "gold fact {} is not needed by the certificate",
                    self.world.g.show(skip)
                ));
            }
        }
        let best = self.candidates[self.gold].score;
        for (i, c) in self.candidates.iter().enumerate() {
            if i != self.gold && best - c.score < self.params.margin - 1e-9 {
                return Err(format!(
                    "candidate {i} ({}) is within the margin of the gold one",
                    c.origin.label()
                ));
            }
        }
        if best - self.null.score < self.params.margin - 1e-9 {
            return Err("the empty hypothesis is within the margin of the gold one".to_string());
        }
        if self.world.preds.iter().any(|d| d.latent) {
            if !gold_facts.iter().any(|&f| self.world.is_latent(f)) {
                return Err("the gold hypothesis postulates no hidden cause".to_string());
            }
            for (i, d) in self.world.preds.iter().enumerate() {
                if d.latent && !self.world.pred_name(i).starts_with('H') {
                    return Err(format!("hidden predicate {i} is not masked"));
                }
            }
        }
        Ok(())
    }
    pub fn verify_scores(&mut self) -> Result<(), String> {
        for i in 0..=self.candidates.len() {
            let old = if i < self.candidates.len() {
                self.candidates[i].clone()
            } else {
                self.null.clone()
            };
            let c = judge(
                &mut self.world,
                &self.disclosed,
                &old.facts,
                old.origin,
                &self.observed,
                self.mc_seed,
                &self.params,
            );
            if c.hits != old.hits || (c.score - old.score).abs() > 1e-9 {
                return Err(format!("candidate {i}: Monte-Carlo score is not reproducible"));
            }
        }
        Ok(())
    }
}

#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub enum Armor {
    Raw,    
    Hex,    
    B64,    
    B64Url, 
    B32,    
    Z85,    
    B58,    
    Words,

impl Armor {
    pub const ALL: [Armor; 8] = [
        Armor::Raw,
        Armor::Hex,
        Armor::B64,
        Armor::B64Url,
        Armor::B32,
        Armor::Z85,
        Armor::B58,
        Armor::Words,
    ];
    pub fn name(self) -> &'static str {
        match self {
            Armor::Raw => "raw",
            Armor::Hex => "hex",
            Armor::B64 => "b64",
            Armor::B64Url => "b64u",
            Armor::B32 => "b32",
            Armor::Z85 => "z85",
            Armor::B58 => "b58",
            Armor::Words => "words",
        }
    }
    pub fn parse(s: &str) -> Option<Armor> {
        Armor::ALL.iter().copied().find(|a| a.name() == s)
    }
    pub fn encode(self, data: &[u8], wrap: usize) -> Result<String, String> {
        let flat = match self {
            Armor::Raw => {
                return String::from_utf8(data.to_vec())
                    .map_err(|_| "raw armor: data is not valid UTF-8".to_string());
            }
            Armor::Words => return Ok(wrap_words(&words_enc(data), wrap)),
            Armor::Hex => hex_enc(data),
            Armor::B64 => b64_enc(data, B64, true),
            Armor::B64Url => b64_enc(data, B64U, false),
            Armor::B32 => b32_enc(data),
            Armor::Z85 => z85_enc(data),
            Armor::B58 => b58_enc(data),
        };
        Ok(wrap_fixed(&flat, wrap))
    }
    pub fn decode(self, text: &str) -> Result<Vec<u8>, String> {
        if self == Armor::Raw {
            return Ok(text.as_bytes().to_vec());
        }
        if self == Armor::Words {
            return words_dec(text);
        }
        let c = squash(text);
        match self {
            Armor::Hex => hex_dec(&c),
            Armor::B64 => b64_dec(&c, B64),
            Armor::B64Url => b64_dec(&c, B64U),
            Armor::B32 => b32_dec(&c),
            Armor::Z85 => z85_dec(&c),
            Armor::B58 => b58_dec(&c),
            Armor::Raw | Armor::Words => Ok(Vec::new()), 
        }
    }
}
fn squash(t: &str) -> Vec<u8> {
    t.bytes().filter(|b| !b.is_ascii_whitespace()).collect()
}
fn wrap_fixed(s: &str, width: usize) -> String {
    if width == 0 || s.len() <= width {
        return s.to_string();
    }
    let mut out = String::with_capacity(s.len() + s.len() / width + 1);
    for (i, ch) in s.as_bytes().chunks(width).enumerate() {
        if i > 0 {
            out.push('\n');
        }
        out.push_str(std::str::from_utf8(ch).unwrap_or(""));
    }
    out
}
fn wrap_words(words: &[String], width: usize) -> String {
    let mut out = String::new();
    let mut col = 0usize;
    for (i, w) in words.iter().enumerate() {
        if i > 0 {
            if width > 0 && col + 1 + w.len() > width {
                out.push('\n');
                col = 0;
            } else {
                out.push(' ');
                col += 1;
            }
        }
        out.push_str(w);
        col += w.len();
    }
    out
}
fn hex_enc(d: &[u8]) -> String {
    const H: &[u8] = b"0123456789abcdef";
    let mut s = String::with_capacity(d.len() * 2);
    for &b in d {
        s.push(H[(b >> 4) as usize] as char);
        s.push(H[(b & 15) as usize] as char);
    }
    s
}
fn hex_dec(c: &[u8]) -> Result<Vec<u8>, String> {
    if c.len() % 2 != 0 {
        return Err("hex: odd number of digits".to_string());
    }
    let mut out = Vec::with_capacity(c.len() / 2);
    for p in c.chunks(2) {
        match ((p[0] as char).to_digit(16), (p[1] as char).to_digit(16)) {
            (Some(h), Some(l)) => out.push((h * 16 + l) as u8),
            _ => return Err("hex: bad digit".to_string()),
        }
    }
    Ok(out)
}
fn b64_enc(d: &[u8], tab: &[u8], pad: bool) -> String {
    let mut s = String::with_capacity(d.len() / 3 * 4 + 4);
    for ch in d.chunks(3) {
        let b = |k: usize| ch.get(k).copied().unwrap_or(0) as u32;
        let n = (b(0) << 16) | (b(1) << 8) | b(2);
        s.push(tab[(n >> 18) as usize & 63] as char);
        s.push(tab[(n >> 12) as usize & 63] as char);
        if ch.len() > 1 {
            s.push(tab[(n >> 6) as usize & 63] as char);
        } else if pad {
            s.push('=');
        }
        if ch.len() > 2 {
            s.push(tab[n as usize & 63] as char);
        } else if pad {
            s.push('=');
        }
    }
    s
}
fn b64_dec(c: &[u8], tab: &[u8]) -> Result<Vec<u8>, String> {
    let mut rev = [255u8; 256];
    for (i, &x) in tab.iter().enumerate() {
        rev[x as usize] = i as u8;
    }
    let mut vals: Vec<u8> = Vec::with_capacity(c.len());
    let mut pads = 0usize;
    for &x in c {
        if x == b'=' {
            pads += 1;
            continue;
        }
        if pads > 0 {
            return Err("base64: data after padding".to_string());
        }
        let v = rev[x as usize];
        if v == 255 {
            return Err(format!("base64: bad character {:?}", x as char));
        }
        vals.push(v);
    }
    if pads > 2 || vals.len() % 4 == 1 || (pads > 0 && (vals.len() + pads) % 4 != 0) {
        return Err("base64: bad length".to_string());
    }
    let mut out = Vec::with_capacity(vals.len() * 3 / 4 + 3);
    for ch in vals.chunks(4) {
        let mut n = 0u32;
        for (k, &v) in ch.iter().enumerate() {
            n |= (v as u32) << (18 - 6 * k as u32);
        }
        out.push((n >> 16) as u8);
        if ch.len() > 2 {
            out.push((n >> 8) as u8);
        }
        if ch.len() > 3 {
            out.push(n as u8);
        }
    }
    Ok(out)
}
fn b32_enc(d: &[u8]) -> String {
    let mut s = String::with_capacity((d.len() + 4) / 5 * 8);
    let (mut acc, mut bits) = (0u32, 0u32);
    for &b in d {
        acc = (acc << 8) | b as u32;
        bits += 8;
        while bits >= 5 {
            bits -= 5;
            s.push(B32[((acc >> bits) & 31) as usize] as char);
        }
        acc &= (1u32 << bits) - 1;
    }
    if bits > 0 {
        s.push(B32[((acc << (5 - bits)) & 31) as usize] as char);
    }
    while s.len() % 8 != 0 {
        s.push('=');
    }
    s
}
fn b32_dec(c: &[u8]) -> Result<Vec<u8>, String> {
    let mut out = Vec::with_capacity(c.len() * 5 / 8 + 1);
    let (mut acc, mut bits) = (0u32, 0u32);
    for &x in c {
        let v = match x {
            b'=' => continue,
            b'A'..=b'Z' => x - b'A',
            b'a'..=b'z' => x - b'a',
            b'2'..=b'7' => x - b'2' + 26,
            _ => return Err(format!("base32: bad character {:?}", x as char)),
        };
        acc = (acc << 5) | v as u32;
        bits += 5;
        if bits >= 8 {
            bits -= 8;
            out.push((acc >> bits) as u8);
            acc &= (1u32 << bits) - 1;
        }
    }
    Ok(out)
}
fn z85_enc(d: &[u8]) -> String {
    let pad = (4 - d.len() % 4) % 4;
    let mut buf = d.to_vec();
    buf.resize(d.len() + pad, 0);
    let mut s = String::with_capacity(buf.len() / 4 * 5 + 1);
    for ch in buf.chunks(4) {
        let mut v = u32::from_be_bytes([ch[0], ch[1], ch[2], ch[3]]);
        let mut g = [0u8; 5];
        for k in (0..5).rev() {
            g[k] = Z85[(v % 85) as usize];
            v /= 85;
        }
        for &c in g.iter() {
            s.push(c as char);
        }
    }
    s.push(Z85[pad] as char);
    s
}
fn z85_dec(c: &[u8]) -> Result<Vec<u8>, String> {
    if c.len() % 5 != 1 {
        return Err("z85: bad length".to_string());
    }
    let mut rev = [255u8; 256];
    for (i, &x) in Z85.iter().enumerate() {
        rev[x as usize] = i as u8;
    }
    let val = |x: u8| -> Result<u64, String> {
        match rev[x as usize] {
            255 => Err(format!("z85: bad character {:?}", x as char)),
            r => Ok(r as u64),
        }
    };
    let pad = val(c[c.len() - 1])? as usize;
    if pad > 3 {
        return Err("z85: bad padding".to_string());
    }
    let mut out = Vec::with_capacity(c.len() / 5 * 4);
    for g in c[..c.len() - 1].chunks(5) {
        let mut v = 0u64;
        for &x in g {
            v = v * 85 + val(x)?;
        }
        if v > u32::MAX as u64 {
            return Err("z85: group out of range".to_string());
        }
        out.extend_from_slice(&(v as u32).to_be_bytes());
    }
    if pad > out.len() {
        return Err("z85: padding exceeds data".to_string());
    }
    out.truncate(out.len() - pad);
    Ok(out)
}
fn b58_enc(d: &[u8]) -> String {
    let mut s = String::with_capacity(d.len() / 8 * 11 + 11);
    for blk in d.chunks(8) {
        let mut v: u64 = 0;
        for &b in blk {
            v = (v << 8) | b as u64;
        }
        let n = B58_CHARS[blk.len()];
        let mut tmp = [0u8; 11];
        for k in (0..n).rev() {
            tmp[k] = B58[(v % 58) as usize];
            v /= 58;
        }
        for &c in &tmp[..n] {
            s.push(c as char);
        }
    }
    s
}
fn b58_dec(c: &[u8]) -> Result<Vec<u8>, String> {
    let mut rev = [255u8; 256];
    for (i, &x) in B58.iter().enumerate() {
        rev[x as usize] = i as u8;
    }
    let mut out = Vec::with_capacity(c.len() / 11 * 8 + 8);
    for blk in c.chunks(11) {
        let k = match B58_CHARS.iter().position(|&n| n == blk.len()) {
            Some(k) if k > 0 => k,
            _ => return Err("base58: bad block length".to_string()),
        };
        let mut v: u128 = 0;
        for &x in blk {
            let d = rev[x as usize];
            if d == 255 {
                return Err(format!("base58: bad character {:?}", x as char));
            }
            v = v * 58 + d as u128;
        }
        if (v >> (8 * k)) != 0 {
            return Err("base58: block out of range".to_string());
        }
        for j in (0..k).rev() {
            out.push((v >> (8 * j)) as u8);
        }
    }
    Ok(out)
}
fn words_enc(d: &[u8]) -> Vec<String> {
    d.iter()
        .map(|&b| format!("{}{}", SYL_HI[(b >> 4) as usize], SYL_LO[(b & 15) as usize]))
        .collect()
}
fn words_dec(t: &str) -> Result<Vec<u8>, String> {
    let mut out = Vec::new();
    for w in t.split_whitespace() {
        let (a, b) = match (w.get(..2), w.get(2..)) {
            (Some(a), Some(b)) => (a, b),
            _ => return Err(format!("words: bad word '{w}'")),
        };
        match (
            SYL_HI.iter().position(|&s| s == a),
            SYL_LO.iter().position(|&s| s == b),
        ) {
            (Some(h), Some(l)) => out.push((h * 16 + l) as u8),
            _ => return Err(format!("words: bad word '{w}'")),
        }
    }
    Ok(out)
}
#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub enum Stage {
    Rle,      
    Lzss,     
    Delta,    
    Rev,      
    Mtf,      
    Xor(u64), 
}
impl Stage {
    pub fn name(self) -> String {
        match self {
            Stage::Rle => "rle".to_string(),
            Stage::Lzss => "lzss".to_string(),
            Stage::Delta => "delta".to_string(),
            Stage::Rev => "rev".to_string(),
            Stage::Mtf => "mtf".to_string(),
            Stage::Xor(k) => format!("xor:{k:016x}"),
        }
    }
    pub fn parse(s: &str) -> Option<Stage> {
        match s {
            "rle" => Some(Stage::Rle),
            "lzss" => Some(Stage::Lzss),
            "delta" => Some(Stage::Delta),
            "rev" => Some(Stage::Rev),
            "mtf" => Some(Stage::Mtf),
            _ => {
                let k = s.strip_prefix("xor:")?;
                if k.is_empty() || k.len() > 16 || !k.bytes().all(|b| b.is_ascii_hexdigit()) {
                    return None;
                }
                u64::from_str_radix(k, 16).ok().map(Stage::Xor)
            }
        }
    }
    pub fn kind(self) -> usize {
        match self {
            Stage::Rle => 0,
            Stage::Lzss => 1,
            Stage::Delta => 2,
            Stage::Rev => 3,
            Stage::Mtf => 4,
            Stage::Xor(_) => 5,
        }
    }
    pub fn is_compressor(self) -> bool {
        matches!(self, Stage::Rle | Stage::Lzss)
    }
    pub fn random(rng: &mut Rng) -> Stage {
        match rng.below(6) {
            0 => Stage::Rle,
            1 => Stage::Lzss,
            2 => Stage::Delta,
            3 => Stage::Rev,
            4 => Stage::Mtf,
            _ => Stage::Xor(rng.next_u64()),
        }
    }
    pub fn apply(self, data: &[u8]) -> Vec<u8> {
        match self {
            Stage::Rle => rle_enc(data),
            Stage::Lzss => lzss_enc(data),
            Stage::Delta => delta_enc(data),
            Stage::Rev => data.iter().rev().copied().collect(),
            Stage::Mtf => mtf_enc(data),
            Stage::Xor(key) => xor_stream(data, key),
        }
    }
    pub fn undo(self, data: &[u8]) -> Result<Vec<u8>, String> {
        match self {
            Stage::Rle => rle_dec(data),
            Stage::Lzss => lzss_dec(data),
            Stage::Delta => Ok(delta_dec(data)),
            Stage::Rev => Ok(data.iter().rev().copied().collect()),
            Stage::Mtf => Ok(mtf_dec(data)),
            Stage::Xor(key) => Ok(xor_stream(data, key)),
        }
    }
}
fn rle_enc(d: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(d.len() + d.len() / 128 + 2);
    let mut i = 0usize;
    while i < d.len() {
        let mut run = 1usize;
        while run < 128 && i + run < d.len() && d[i + run] == d[i] {
            run += 1;
        }
        if run >= 2 {
            out.push((257 - run) as u8);
            out.push(d[i]);
            i += run;
        } else {
            let start = i;
            i += 1;
            while i < d.len() && i - start < 128 && !(i + 1 < d.len() && d[i] == d[i + 1]) {
                i += 1;
            }
            out.push((i - start - 1) as u8);
            out.extend_from_slice(&d[start..i]);
        }
    }
    out
}
fn rle_dec(d: &[u8]) -> Result<Vec<u8>, String> {
    let mut out: Vec<u8> = Vec::new();
    let mut i = 0usize;
    while i < d.len() {
        let n = d[i] as usize;
        i += 1;
        if n < 128 {
            let len = n + 1;
            if i + len > d.len() {
                return Err("rle: literal runs past the end".to_string());
            }
            out.extend_from_slice(&d[i..i + len]);
            i += len;
        } else if n > 128 {
            if i >= d.len() {
                return Err("rle: run without a byte".to_string());
            }
            let b = d[i];
            i += 1;
            out.resize(out.len() + (257 - n), b);
        } else {
            return Err("rle: reserved control byte".to_string());
        }
        if out.len() > LIMIT {
            return Err("rle: output exceeds the size limit".to_string());
        }
    }
    Ok(out)
}
fn delta_enc(d: &[u8]) -> Vec<u8> {
    let mut prev = 0u8;
    d.iter()
        .map(|&b| {
            let r = b.wrapping_sub(prev);
            prev = b;
            r
        })
        .collect()
}
fn delta_dec(d: &[u8]) -> Vec<u8> {
    let mut acc = 0u8;
    d.iter()
        .map(|&r| {
            acc = acc.wrapping_add(r);
            acc
        })
        .collect()
}
fn mtf_enc(d: &[u8]) -> Vec<u8> {
    let mut tab: Vec<u8> = (0..=255u8).collect();
    let mut out = Vec::with_capacity(d.len());
    for &b in d {
        let pos = tab.iter().position(|&x| x == b).unwrap_or(0);
        out.push(pos as u8);
        tab.copy_within(0..pos, 1);
        tab[0] = b;
    }
    out
}
fn mtf_dec(d: &[u8]) -> Vec<u8> {
    let mut tab: Vec<u8> = (0..=255u8).collect();
    let mut out = Vec::with_capacity(d.len());
    for &i in d {
        let pos = i as usize;
        let b = tab[pos];
        out.push(b);
        tab.copy_within(0..pos, 1);
        tab[0] = b;
    }
    out
}
fn xor_stream(d: &[u8], key: u64) -> Vec<u8> {
    let mut rng = Rng::new(key);
    let mut out = Vec::with_capacity(d.len());
    for chunk in d.chunks(8) {
        let ks = rng.next_u64().to_le_bytes();
        for (k, &b) in chunk.iter().enumerate() {
            out.push(b ^ ks[k]);
        }
    }
    out
}
const LZ_WIN: usize = 4095; 
const LZ_MIN: usize = 3;
const LZ_MAX: usize = 18;
const LZ_DEPTH: usize = 24; 
const LZ_HASH_BITS: u32 = 15;
const LZ_NONE: usize = usize::MAX;
fn lz_hash(d: &[u8], i: usize) -> usize {
    let v = (d[i] as u32) | ((d[i + 1] as u32) << 8) | ((d[i + 2] as u32) << 16);
    (v.wrapping_mul(0x9E37_79B1) >> (32 - LZ_HASH_BITS)) as usize
}
fn lzss_enc(d: &[u8]) -> Vec<u8> {
    let n = d.len();
    let mut head = vec![LZ_NONE; 1 << LZ_HASH_BITS];
    let mut prev = vec![LZ_NONE; LZ_WIN + 1];
    let mut out: Vec<u8> = Vec::with_capacity(n / 2 + 16);
    let mut flag_at = 0usize;
    let mut bit = 8u32; 
    let mut i = 0usize;
    while i < n {
        let mut best_len = 0usize;
        let mut best_dist = 0usize;
        if i + LZ_MIN <= n {
            let limit = LZ_MAX.min(n - i);
            let mut cand = head[lz_hash(d, i)];
            let mut depth = 0usize;
            while cand != LZ_NONE && i - cand <= LZ_WIN && depth < LZ_DEPTH {
                let mut l = 0usize;
                while l < limit && d[cand + l] == d[i + l] {
                    l += 1;
                }
                if l > best_len {
                    best_len = l;
                    best_dist = i - cand;
                    if l == limit {
                        break;
                    }
                }
                cand = prev[cand & LZ_WIN];
                depth += 1;
            }
        }
        if bit == 8 {
            flag_at = out.len();
            out.push(0);
            bit = 0;
        }
        let step = if best_len >= LZ_MIN {
            let v = (best_dist << 4) | (best_len - LZ_MIN);
            out.push((v >> 8) as u8);
            out.push(v as u8);
            best_len
        } else {
            out[flag_at] |= 1u8 << bit;
            out.push(d[i]);
            1
        };
        bit += 1;
        for j in i..i + step {
            if j + LZ_MIN <= n {
                let h = lz_hash(d, j);
                prev[j & LZ_WIN] = head[h];
                head[h] = j;
            }
        }
        i += step;
    }
    out
}
fn lzss_dec(d: &[u8]) -> Result<Vec<u8>, String> {
    let mut out: Vec<u8> = Vec::with_capacity(d.len().saturating_mul(2).min(LIMIT));
    let mut i = 0usize;
    while i < d.len() {
        let flags = d[i];
        i += 1;
        for bit in 0..8u32 {
            if i >= d.len() {
                break;
            }
            if flags & (1u8 << bit) != 0 {
                out.push(d[i]);
                i += 1;
            } else {
                if i + 1 >= d.len() {
                    return Err("lzss: truncated match".to_string());
                }
                let v = ((d[i] as usize) << 8) | d[i + 1] as usize;
                i += 2;
                let dist = v >> 4;
                let len = (v & 15) + LZ_MIN;
                if dist == 0 || dist > out.len() {
                    return Err("lzss: match reaches before the start".to_string());
                }
                let from = out.len() - dist;
                for k in 0..len {
                    let b = out[from + k]; 
                    out.push(b);
                }
            }
            if out.len() > LIMIT {
                return Err("lzss: output exceeds the size limit".to_string());
            }
        }
    }
    Ok(out)
}

pub const MAX_DEPTH: usize = 128;
#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub enum Format {
    Json,
    JsonPretty,
    Sexp,
    Bencode,
    Cbor,
    MsgPack,
}
impl Format {
    pub const ALL: [Format; 6] = [
        Format::Json,
        Format::JsonPretty,
        Format::Sexp,
        Format::Bencode,
        Format::Cbor,
        Format::MsgPack,
    ];
    pub fn name(self) -> &'static str {
        match self {
            Format::Json => "json",
            Format::JsonPretty => "jsonp",
            Format::Sexp => "sexp",
            Format::Bencode => "bencode",
            Format::Cbor => "cbor",
            Format::MsgPack => "msgpack",
        }
    }
    pub fn parse(s: &str) -> Option<Format> {
        Format::ALL.iter().copied().find(|f| f.name() == s)
    }
    pub fn random(rng: &mut Rng) -> Format {
        *rng.pick(&Format::ALL)
    }
    pub fn is_text(self) -> bool {
        !matches!(self, Format::Cbor | Format::MsgPack)
    }
    pub fn encode(self, v: &Value) -> Result<Vec<u8>, String> {
        if v.depth() > MAX_DEPTH {
            return Err(format!("value nests deeper than {MAX_DEPTH} levels"));
        }
        match self {
            Format::Json => Ok(text::json_encode(v, 0).into_bytes()),
            Format::JsonPretty => Ok(text::json_encode(v, 2).into_bytes()),
            Format::Sexp => Ok(text::sexp_encode(v).into_bytes()),
            Format::Bencode => Ok(bin::bencode_encode(v)),
            Format::Cbor => Ok(bin::cbor_encode(v)),
            Format::MsgPack => bin::msgpack_encode(v),
        }
    }
    pub fn decode(self, data: &[u8]) -> Result<Value, String> {
        match self {
            Format::Json | Format::JsonPretty | Format::Sexp => {
                let s = std::str::from_utf8(data)
                    .map_err(|_| "text is not valid UTF-8".to_string())?;
                if self == Format::Sexp {
                    text::sexp_decode(s)
                } else {
                    text::json_decode(s)
                }
            }
            Format::Bencode => bin::bencode_decode(data),
            Format::Cbor => bin::cbor_decode(data),
            Format::MsgPack => bin::msgpack_decode(data),
        }
    }
}

struct Rd<'a> {
    b: &'a [u8],
    i: usize,
}
impl<'a> Rd<'a> {
    fn new(b: &'a [u8]) -> Rd<'a> {
        Rd { b, i: 0 }
    }
    fn left(&self) -> usize {
        self.b.len() - self.i
    }
    fn peek(&self) -> Option<u8> {
        self.b.get(self.i).copied()
    }
    fn fail<T>(&self, what: &str) -> Result<T, String> {
        Err(format!("{what} at byte {}", self.i))
    }
    fn byte(&mut self) -> Result<u8, String> {
        let Some(c) = self.peek() else {
            return self.fail("unexpected end of input");
        };
        self.i += 1;
        Ok(c)
    }
    fn take(&mut self, n: usize) -> Result<&'a [u8], String> {
        if n > self.left() {
            return self.fail("unexpected end of input");
        }
        let b = self.b;
        let s = &b[self.i..self.i + n];
        self.i += n;
        Ok(s)
    }
    fn be(&mut self, n: usize) -> Result<u64, String> {
        let s = self.take(n)?;
        Ok(s.iter().fold(0u64, |acc, &x| (acc << 8) | x as u64))
    }
    fn until(&mut self, term: u8) -> Result<&'a [u8], String> {
        let b = self.b;
        let rest = &b[self.i..];
        match rest.iter().position(|&x| x == term) {
            Some(k) => {
                self.i += k + 1;
                Ok(&rest[..k])
            }
            None => self.fail("unterminated token"),
        }
    }
    fn text(&mut self, n: u64) -> Result<String, String> {
        let len = match usize::try_from(n) {
            Ok(l) if l <= self.left() => l,
            _ => return self.fail("claimed length exceeds the data"),
        };
        let at = self.i;
        let raw = self.take(len)?;
        match std::str::from_utf8(raw) {
            Ok(s) => Ok(s.to_string()),
            Err(_) => Err(format!("invalid UTF-8 at byte {at}")),
        }
    }
    fn count(&self, n: u64, each: usize) -> Result<usize, String> {
        match usize::try_from(n) {
            Ok(c) if c <= self.left() / each => Ok(c),
            _ => self.fail("claimed count exceeds the data"),
        }
    }
    fn float(&self, x: f64) -> Result<Value, String> {
        if x.is_finite() {
            Ok(Value::Float(x))
        } else {
            self.fail("non-finite float")
        }
    }
    fn finish(&self) -> Result<(), String> {
        if self.i < self.b.len() {
            return self.fail("trailing data");
        }
        Ok(())
    }
}

fn ben_put_str(s: &str, out: &mut Vec<u8>) {
    out.extend_from_slice(s.len().to_string().as_bytes());
    out.push(b':');
    out.extend_from_slice(s.as_bytes());

fn ben_write(v: &Value, out: &mut Vec<u8>) {
    match v {
        Value::Null => out.push(b'n'),
        Value::Bool(b) => out.push(if *b { b't' } else { b'f' }),
        Value::Int(i) => {
            out.push(b'i');
            out.extend_from_slice(i.to_string().as_bytes());
            out.push(b'e');
        }
        Value::Float(x) => out.extend_from_slice(format!("r{:016x}e", x.to_bits()).as_bytes()),
        Value::Str(s) => ben_put_str(s, out),
        Value::List(l) => {
            out.push(b'l');
            for x in l {
                ben_write(x, out);
            }
            out.push(b'e');
        }
        Value::Map(m) => {
            out.push(b'd');
            for (k, x) in m {
                ben_put_str(k, out);
                ben_write(x, out);
            }
            out.push(b'e');
        }
    }
}
fn ben_int(tok: &[u8]) -> Option<i64> {
    let (neg, digits) = match tok.split_first() {
        Some((b'-', rest)) => (true, rest),
        _ => (false, tok),
    };
    let canonical = !digits.is_empty()
        && digits.iter().all(|c| c.is_ascii_digit())
        && (digits[0] != b'0' || (digits.len() == 1 && !neg));
    if !canonical {
        return None;
    }
    std::str::from_utf8(tok).ok()?.parse().ok()
}
fn ben_len(tok: &[u8]) -> Option<u64> {
    let plain = !tok.is_empty()
        && tok.iter().all(|c| c.is_ascii_digit())
        && (tok[0] != b'0' || tok.len() == 1);
    if !plain {
        return None;
    }
    std::str::from_utf8(tok).ok()?.parse().ok()
}
fn ben_str(r: &mut Rd<'_>) -> Result<String, String> {
    let tok = r.until(b':')?;
    match ben_len(tok) {
        Some(n) => r.text(n),
        None => r.fail("bad string length"),
    }
}
fn ben_read(r: &mut Rd<'_>, depth: usize) -> Result<Value, String> {
    if depth > MAX_DEPTH {
        return r.fail("nesting too deep");
    }
    let Some(c) = r.peek() else {
        return r.fail("unexpected end of input");
    };
    if c.is_ascii_digit() {
        return ben_str(r).map(Value::Str);
    }
    r.i += 1;
    match c {
        b'n' => Ok(Value::Null),
        b't' => Ok(Value::Bool(true)),
        b'f' => Ok(Value::Bool(false)),
        b'i' => match ben_int(r.until(b'e')?) {
            Some(n) => Ok(Value::Int(n)),
            None => r.fail("bad integer"),
        },
        b'r' => {
            let mut bits = 0u64;
            for &d in r.take(16)? {
                match (d as char).to_digit(16) {
                    Some(h) => bits = (bits << 4) | h as u64,
                    None => return r.fail("bad hex digit in a float"),
                }
            }
            if r.byte()? != b'e' {
                return r.fail("a float must end with 'e'");
            }
            r.float(f64::from_bits(bits))
        }
        b'l' => {
            let mut items = Vec::new();
            loop {
                match r.peek() {
                    Some(b'e') => {
                        r.i += 1;
                        return Ok(Value::List(items));
                    }
                    Some(_) => items.push(ben_read(r, depth + 1)?),
                    None => return r.fail("unterminated list"),
                }
            }
        }
        b'd' => {
            let mut items = Vec::new();
            loop {
                match r.peek() {
                    Some(b'e') => {
                        r.i += 1;
                        return Ok(Value::Map(items));
                    }
                    Some(k) if k.is_ascii_digit() => {
                        let key = ben_str(r)?;
                        items.push((key, ben_read(r, depth + 1)?));
                    }
                    Some(_) => return r.fail("dictionary keys must be strings"),
                    None => return r.fail("unterminated dictionary"),
                }
            }
        }
        _ => r.fail("unexpected byte"),
    }
}

fn cbor_head(out: &mut Vec<u8>, major: u8, n: u64) {
    let m = major << 5;
    if n < 24 {
        out.push(m | n as u8);
    } else if n <= 0xff {
        out.push(m | 24);
        out.push(n as u8);
    } else if n <= 0xffff {
        out.push(m | 25);
        out.extend_from_slice(&(n as u16).to_be_bytes());
    } else if n <= 0xffff_ffff {
        out.push(m | 26);
        out.extend_from_slice(&(n as u32).to_be_bytes());
    } else {
        out.push(m | 27);
        out.extend_from_slice(&n.to_be_bytes());
    }
}
fn cbor_put_str(s: &str, out: &mut Vec<u8>) {
    cbor_head(out, 3, s.len() as u64);
    out.extend_from_slice(s.as_bytes());
}
fn cbor_write(v: &Value, out: &mut Vec<u8>) {
    match v {
        Value::Null => out.push(0xf6),
        Value::Bool(b) => out.push(if *b { 0xf5 } else { 0xf4 }),
        Value::Int(i) if *i >= 0 => cbor_head(out, 0, *i as u64),
        Value::Int(i) => cbor_head(out, 1, (!*i) as u64),
        Value::Float(x) => {
            out.push(0xfb);
            out.extend_from_slice(&x.to_bits().to_be_bytes());
        }
        Value::Str(s) => cbor_put_str(s, out),
        Value::List(l) => {
            cbor_head(out, 4, l.len() as u64);
            for x in l {
                cbor_write(x, out);
            }
        }
        Value::Map(m) => {
            cbor_head(out, 5, m.len() as u64);
            for (k, x) in m {
                cbor_put_str(k, out);
                cbor_write(x, out);
            }
        }
    }
}
fn cbor_arg(r: &mut Rd<'_>, info: u8) -> Result<u64, String> {
    match info {
        0..=23 => Ok(info as u64),
        24 => r.be(1),
        25 => r.be(2),
        26 => r.be(4),
        27 => r.be(8),
        31 => r.fail("indefinite lengths are not supported"),
        _ => r.fail("reserved additional information"),
    }
}
fn half(h: u16) -> f64 {
    let sign = if h & 0x8000 != 0 { -1.0 } else { 1.0 };
    let exp = ((h >> 10) & 31) as i32;
    let man = (h & 0x3ff) as f64;
    let mag = match exp {
        0 => man * 2f64.powi(-24),
        31 if man == 0.0 => f64::INFINITY,
        31 => f64::NAN,
        e => (1.0 + man / 1024.0) * 2f64.powi(e - 15),
    };
    sign * mag
}
fn cbor_read(r: &mut Rd<'_>, depth: usize) -> Result<Value, String> {
    if depth > MAX_DEPTH {
        return r.fail("nesting too deep");
    }
    let ib = r.byte()?;
    let (major, info) = (ib >> 5, ib & 31);
    match major {
        0 | 1 => {
            let n = cbor_arg(r, info)?;
            match i64::try_from(n) {
                Ok(i) if major == 0 => Ok(Value::Int(i)),
                Ok(i) => Ok(Value::Int(-1 - i)),
                Err(_) => r.fail("integer beyond the i64 range"),
            }
        }
        2 => r.fail("byte strings are not supported"),
        3 => {
            let n = cbor_arg(r, info)?;
            r.text(n).map(Value::Str)
        }
        4 => {
            let n = cbor_arg(r, info)?;
            let n = r.count(n, 1)?;
            let mut items = Vec::with_capacity(n.min(256));
            for _ in 0..n {
                items.push(cbor_read(r, depth + 1)?);
            }
            Ok(Value::List(items))
        }
        5 => {
            let n = cbor_arg(r, info)?;
            let n = r.count(n, 2)?;
            let mut items = Vec::with_capacity(n.min(256));
            for _ in 0..n {
                let kb = r.byte()?;
                if kb >> 5 != 3 {
                    return r.fail("map keys must be text strings");
                }
                let kn = cbor_arg(r, kb & 31)?;
                let key = r.text(kn)?;
                items.push((key, cbor_read(r, depth + 1)?));
            }
            Ok(Value::Map(items))
        }
        6 => r.fail("tags are not supported"),
        _ => match info {
            20 => Ok(Value::Bool(false)),
            21 => Ok(Value::Bool(true)),
            22 => Ok(Value::Null),
            25 => {
                let h = r.be(2)? as u16;
                r.float(half(h))
            }
            26 => {
                let x = f32::from_bits(r.be(4)? as u32);
                r.float(f64::from(x))
            }
            27 => {
                let x = f64::from_bits(r.be(8)?);
                r.float(x)
            }
            _ => r.fail("unsupported simple value"),
        },
    }
}

fn mp_head(
    out: &mut Vec<u8>,
    n: usize,
    fix: u8,
    max: u64,
    t8: Option<u8>,
    t16: u8,
    t32: u8,
) -> Result<(), String> {
    let n = n as u64;
    if n <= max {
        out.push(fix | n as u8);
    } else if let (Some(t), true) = (t8, n <= 0xff) {
        out.push(t);
        out.push(n as u8);
    } else if n <= 0xffff {
        out.push(t16);
        out.extend_from_slice(&(n as u16).to_be_bytes());
    } else if n <= 0xffff_ffff {
        out.push(t32);
        out.extend_from_slice(&(n as u32).to_be_bytes());
    } else {
        return Err("string, array or map of 2^32 or more entries".to_string());
    }
    Ok(())
}
fn mp_put_str(s: &str, out: &mut Vec<u8>) -> Result<(), String> {
    mp_head(out, s.len(), 0xa0, 31, Some(0xd9), 0xda, 0xdb)?;
    out.extend_from_slice(s.as_bytes());
    Ok(())
}
fn mp_int(i: i64, out: &mut Vec<u8>) {
    if i >= 0 {
        let n = i as u64;
        if n < 128 {
            out.push(n as u8);
        } else if n <= 0xff {
            out.push(0xcc);
            out.push(n as u8);
        } else if n <= 0xffff {
            out.push(0xcd);
            out.extend_from_slice(&(n as u16).to_be_bytes());
        } else if n <= 0xffff_ffff {
            out.push(0xce);
            out.extend_from_slice(&(n as u32).to_be_bytes());
        } else {
            out.push(0xcf);
            out.extend_from_slice(&n.to_be_bytes());
        }
    } else if i >= -32 {
        out.push(i as i8 as u8);
    } else if i >= i8::MIN as i64 {
        out.push(0xd0);
        out.push(i as i8 as u8);
    } else if i >= i16::MIN as i64 {
        out.push(0xd1);
        out.extend_from_slice(&(i as i16).to_be_bytes());
    } else if i >= i32::MIN as i64 {
        out.push(0xd2);
        out.extend_from_slice(&(i as i32).to_be_bytes());
    } else {
        out.push(0xd3);
        out.extend_from_slice(&i.to_be_bytes());
    }
}
fn mp_write(v: &Value, out: &mut Vec<u8>) -> Result<(), String> {
    match v {
        Value::Null => out.push(0xc0),
        Value::Bool(b) => out.push(if *b { 0xc3 } else { 0xc2 }),
        Value::Int(i) => mp_int(*i, out),
        Value::Float(x) => {
            out.push(0xcb);
            out.extend_from_slice(&x.to_bits().to_be_bytes());
        }
        Value::Str(s) => mp_put_str(s, out)?,
        Value::List(l) => {
            mp_head(out, l.len(), 0x90, 15, None, 0xdc, 0xdd)?;
            for x in l {
                mp_write(x, out)?;
            }
        }
        Value::Map(m) => {
            mp_head(out, m.len(), 0x80, 15, None, 0xde, 0xdf)?;
            for (k, x) in m {
                mp_put_str(k, out)?;
                mp_write(x, out)?;
            }
        }
    }
    Ok(())
}
fn mp_list(r: &mut Rd<'_>, n: u64, depth: usize) -> Result<Value, String> {
    let n = r.count(n, 1)?;
    let mut items = Vec::with_capacity(n.min(256));
    for _ in 0..n {
        items.push(mp_read(r, depth + 1)?);
    }
    Ok(Value::List(items))
}
fn mp_map(r: &mut Rd<'_>, n: u64, depth: usize) -> Result<Value, String> {
    let n = r.count(n, 2)?;
    let mut items = Vec::with_capacity(n.min(256));
    for _ in 0..n {
        let key = mp_key(r)?;
        items.push((key, mp_read(r, depth + 1)?));
    }
    Ok(Value::Map(items))
}
fn mp_key(r: &mut Rd<'_>) -> Result<String, String> {
    let t = r.byte()?;
    let n = match t {
        0xa0..=0xbf => (t & 31) as u64,
        0xd9 => r.be(1)?,
        0xda => r.be(2)?,
        0xdb => r.be(4)?,
        _ => return r.fail("map keys must be strings"),
    };
    r.text(n)
}
fn mp_read(r: &mut Rd<'_>, depth: usize) -> Result<Value, String> {
    if depth > MAX_DEPTH {
        return r.fail("nesting too deep");
    }
    let t = r.byte()?;
    match t {
        0x00..=0x7f => Ok(Value::Int(t as i64)),
        0xe0..=0xff => Ok(Value::Int(t as i8 as i64)),
        0x80..=0x8f => mp_map(r, (t & 15) as u64, depth),
        0x90..=0x9f => mp_list(r, (t & 15) as u64, depth),
        0xa0..=0xbf => r.text((t & 31) as u64).map(Value::Str),
        0xc0 => Ok(Value::Null),
        0xc1 => r.fail("reserved type byte 0xc1"),
        0xc2 => Ok(Value::Bool(false)),
        0xc3 => Ok(Value::Bool(true)),
        0xc4..=0xc6 => r.fail("binary data is not supported"),
        0xc7..=0xc9 | 0xd4..=0xd8 => r.fail("extension types are not supported"),
        0xca => {
            let x = f32::from_bits(r.be(4)? as u32);
            r.float(f64::from(x))
        }
        0xcb => {
            let x = f64::from_bits(r.be(8)?);
            r.float(x)
        }
        0xcc => Ok(Value::Int(r.be(1)? as i64)),
        0xcd => Ok(Value::Int(r.be(2)? as i64)),
        0xce => Ok(Value::Int(r.be(4)? as i64)),
        0xcf => {
            let n = r.be(8)?;
            match i64::try_from(n) {
                Ok(i) => Ok(Value::Int(i)),
                Err(_) => r.fail("integer beyond the i64 range"),
            }
        }
        0xd0 => Ok(Value::Int(r.be(1)? as u8 as i8 as i64)),
        0xd1 => Ok(Value::Int(r.be(2)? as u16 as i16 as i64)),
        0xd2 => Ok(Value::Int(r.be(4)? as u32 as i32 as i64)),
        0xd3 => Ok(Value::Int(r.be(8)? as i64)),
        0xd9 => {
            let n = r.be(1)?;
            r.text(n).map(Value::Str)
        }
        0xda => {
            let n = r.be(2)?;
            r.text(n).map(Value::Str)
        }
        0xdb => {
            let n = r.be(4)?;
            r.text(n).map(Value::Str)
        }
        0xdc => {
            let n = r.be(2)?;
            mp_list(r, n, depth)
        }
        0xdd => {
            let n = r.be(4)?;
            mp_list(r, n, depth)
        }
        0xde => {
            let n = r.be(2)?;
            mp_map(r, n, depth)
        }
        0xdf => {
            let n = r.be(4)?;

}

struct Tx<'a> {
    s: &'a str,
    i: usize,
}
impl<'a> Tx<'a> {
    fn new(s: &'a str) -> Tx<'a> {
        Tx { s, i: 0 }
    }
    fn peek(&self) -> Option<u8> {
        self.s.as_bytes().get(self.i).copied()
    }
    fn ws(&mut self) {
        while self.peek().is_some_and(|c| c.is_ascii_whitespace()) {
            self.i += 1;
        }
    }
    fn eat(&mut self, c: u8) -> bool {
        let hit = self.peek() == Some(c);
        if hit {
            self.i += 1;
        }
        hit
    }
    fn fail<T>(&self, what: &str) -> Result<T, String> {
        Err(format!("{what} at byte {}", self.i))
    }
    fn finish(&mut self) -> Result<(), String> {
        self.ws();
        if self.i < self.s.len() {
            return self.fail("trailing data");
        }
        Ok(())
    }
    fn word(&mut self, w: &str) -> Result<(), String> {
        if !self.s[self.i..].starts_with(w) {
            return self.fail("unexpected token");
        }
        self.i += w.len();
        Ok(())
    }
    fn atom(&mut self) -> &'a str {
        let s = self.s;
        let start = self.i;
        while self
            .peek()
            .is_some_and(|c| !c.is_ascii_whitespace() && !matches!(c, b'(' | b')' | b'"'))
        {
            self.i += 1;
        }
        &s[start..self.i]
    }
    /// A JSON number token.
    fn number(&mut self) -> Result<Value, String> {
        let start = self.i;
        while self
            .peek()
            .is_some_and(|c| c.is_ascii_digit() || matches!(c, b'+' | b'-' | b'.' | b'e' | b'E'))
        {
            self.i += 1;
        }
        num(&self.s[start..self.i]).map_err(|e| format!("{e} at byte {start}"))
    }
    /// The string that starts at the opening quote under the cursor.
    fn string(&mut self) -> Result<String, String> {
        let s = self.s;
        let b = s.as_bytes();
        self.i += 1; // the opening quote
        let mut out = String::new();
        loop {
            let start = self.i;
            while self.i < b.len() && b[self.i] >= 0x20 && b[self.i] != b'"' && b[self.i] != b'\\' {
                self.i += 1;
            }
            out.push_str(&s[start..self.i]);
            match b.get(self.i).copied() {
                None => return self.fail("unterminated string"),
                Some(b'"') => {
                    self.i += 1;
                    return Ok(out);
                }
                Some(b'\\') => {
                    self.i += 1;
                    let Some(c) = b.get(self.i).copied() else {
                        return self.fail("unterminated escape");
                    };
                    self.i += 1;
                    match c {
                        b'"' => out.push('"'),
                        b'\\' => out.push('\\'),
                        b'/' => out.push('/'),
                        b'b' => out.push('\u{8}'),
                        b'f' => out.push('\u{c}'),
                        b'n' => out.push('\n'),
                        b'r' => out.push('\r'),
                        b't' => out.push('\t'),
                        b'u' => {
                            let ch = self.unicode()?;
                            out.push(ch);
                        }
                        _ => return self.fail("bad escape"),
                    }
                }
                Some(_) => return self.fail("control character in string"),
            }
        }
    }
    /// The part of `\uXXXX` after the `u`, including a trailing `\uXXXX` low surrogate.
    fn unicode(&mut self) -> Result<char, String> {
        let hi = self.hex4()?;
        let cp = if (0xD800..0xDC00).contains(&hi) {
            if !self.s[self.i..].starts_with("\\u") {
                return self.fail("lone surrogate");
            }
            self.i += 2;
            let lo = self.hex4()?;
            if !(0xDC00..0xE000).contains(&lo) {
                return self.fail("bad surrogate pair");
            }
            0x10000 + ((hi - 0xD800) << 10) + (lo - 0xDC00)
        } else {
            hi
        };
        match char::from_u32(cp) {
            Some(c) => Ok(c),
            None => self.fail("invalid code point"),
        }
    }
    fn hex4(&mut self) -> Result<u32, String> {
        let b = self.s.as_bytes();
        if self.i + 4 > b.len() {
            return self.fail("short \\u escape");
        }
        let mut v = 0u32;
        for k in 0..4 {
            match (b[self.i + k] as char).to_digit(16) {
                Some(d) => v = v * 16 + d,
                None => return self.fail("bad \\u escape"),
            }
        }
        self.i += 4;
        Ok(v)
    }
}
// -------------------------------------------------------- shared scalar rules
fn write_str(s: &str, out: &mut String) {
    out.push('"');
    for c in s.chars() {
        match c {
            '"' => out.push_str("\\\""),
            '\\' => out.push_str("\\\\"),
            '\n' => out.push_str("\\n"),
            '\r' => out.push_str("\\r"),
            '\t' => out.push_str("\\t"),
            c if (c as u32) < 0x20 => out.push_str(&format!("\\u{:04x}", c as u32)),
            c => out.push(c),
        }
    }
    out.push('"');
}
/// Shortest round-trip form; the safety net guarantees a `.` or an exponent.
fn write_float(x: f64, out: &mut String) {
    let t = format!("{x:?}");
    out.push_str(&t);
    if !t.contains(|c: char| matches!(c, '.' | 'e' | 'E')) {
        out.push_str(".0");
    }
}
/// A number token: `Float` if it has `.`, `e` or `E` (and is finite), else `Int`.
fn num(tok: &str) -> Result<Value, String> {
    if tok.contains(|c: char| matches!(c, '.' | 'e' | 'E')) {
        match tok.parse::<f64>() {
            Ok(x) if x.is_finite() => Ok(Value::Float(x)),
            _ => Err(format!("bad float {tok:?}")),
        }
    } else {
        tok.parse::<i64>()
            .map(Value::Int)
            .map_err(|_| format!("bad integer {tok:?}"))
    }
}
fn newline(out: &mut String, ind: usize, lvl: usize) {
    if ind > 0 {
        out.push('\n');
        for _ in 0..ind * lvl {
            out.push(' ');
        }
    }
}
// --------------------------------------------------------------------- JSON
fn json_write(v: &Value, ind: usize, lvl: usize, out: &mut String) {
    match v {
        Value::Null => out.push_str("null"),
        Value::Bool(b) => out.push_str(if *b { "true" } else { "false" }),
        Value::Int(i) => out.push_str(&i.to_string()),
        Value::Float(x) => write_float(*x, out),
        Value::Str(s) => write_str(s, out),
        Value::List(l) => {
            if l.is_empty() {
                out.push_str("[]");
                return;
            }
            out.push('[');
            for (k, x) in l.iter().enumerate() {
                if k > 0 {
                    out.push(',');
                }
                newline(out, ind, lvl + 1);
                json_write(x, ind, lvl + 1, out);
            }
            newline(out, ind, lvl);
            out.push(']');
        }
        Value::Map(m) => {
            if m.is_empty() {
                out.push_str("{}");
                return;
            }
            out.push('{');
            for (k, (key, x)) in m.iter().enumerate() {
                if k > 0 {
                    out.push(',');
                }
                newline(out, ind, lvl + 1);
                write_str(key, out);
                out.push(':');
                if ind > 0 {
                    out.push(' ');
                }
                json_write(x, ind, lvl + 1, out);
            }
            newline(out, ind, lvl);
            out.push('}');
        }
    }
}
fn json_read(p: &mut Tx<'_>, depth: usize) -> Result<Value, String> {
    if depth > MAX_DEPTH {
        return p.fail("nesting too deep");
    }
    p.ws();
    match p.peek() {
        None => p.fail("unexpected end of input"),
        Some(b'n') => p.word("null").map(|_| Value::Null),
        Some(b't') => p.word("true").map(|_| Value::Bool(true)),
        Some(b'f') => p.word("false").map(|_| Value::Bool(false)),
        Some(b'"') => p.string().map(Value::Str),
        Some(b'[') => {
            p.i += 1;
            let mut items = Vec::new();
            p.ws();
            if p.eat(b']') {
                return Ok(Value::List(items));
            }
            loop {
                items.push(json_read(p, depth + 1)?);
                p.ws();
                if p.eat(b',') {
                    continue;
                }
                if p.eat(b']') {
                    return Ok(Value::List(items));
                }
                return p.fail("expected ',' or ']'");
            }
        }
        Some(b'{') => {
            p.i += 1;
            let mut items = Vec::new();
            p.ws();
            if p.eat(b'}') {
                return Ok(Value::Map(items));
            }
            loop {
                p.ws();
                if p.peek() != Some(b'"') {
                    return p.fail("expected a string key");
                }
                let k = p.string()?;
                p.ws();
                if !p.eat(b':') {
                    return p.fail("expected ':'");
                }
                let x = json_read(p, depth + 1)?;
                items.push((k, x));
                p.ws();
                if p.eat(b',') {
                    continue;
                }
                if p.eat(b'}') {
                    return Ok(Value::Map(items));
                }
                return p.fail("expected ',' or '}'");
            }
        }
        Some(c) if c == b'-' || c.is_ascii_digit() => p.number(),
        Some(_) => p.fail("unexpected character"),
    }
}
// ------------------------------------------------------------ S-expressions
fn sexp_write(v: &Value, out: &mut String) {
    match v {
        Value::Null => out.push_str("nil"),
        Value::Bool(b) => out.push_str(if *b { "#t" } else { "#f" }),
        Value::Int(i) => out.push_str(&i.to_string()),
        Value::Float(x) => write_float(*x, out),
        Value::Str(s) => write_str(s, out),
        Value::List(l) => {
            out.push_str("(list");
            for x in l {
                out.push(' ');
                sexp_write(x, out);
            }
            out.push(')');
        }
        Value::Map(m) => {
            out.push_str("(map");
            for (k, x) in m {
                out.push_str(" (");
                write_str(k, out);
                out.push(' ');
                sexp_write(x, out);
                out.push(')');
            }
            out.push(')');
        }
    }
}
fn sexp_read(p: &mut Tx<'_>, depth: usize) -> Result<Value, String> {
    if depth > MAX_DEPTH {
        return p.fail("nesting too deep");
    }
    p.ws();
    match p.peek() {
        None => p.fail("unexpected end of input"),
        Some(b'"') => p.string().map(Value::Str),
        Some(b')') => p.fail("unexpected ')'"),
        Some(b'(') => {
            p.i += 1;
            p.ws();
            match p.atom() {
                "list" => {
                    let mut items = Vec::new();
                    loop {
                        p.ws();
                        if p.eat(b')') {
                            break;
                        }
                        items.push(sexp_read(p, depth + 1)?);
                    }
                    Ok(Value::List(items))
                }
                "map" => {
                    let mut items = Vec::new();
                    loop {
                        p.ws();
                        if p.eat(b')') {
                            break;
                        }
                        if !p.eat(b'(') {
                            return p.fail("expected '(' or ')'");
                        }
                        p.ws();
                        if p.peek() != Some(b'"') {
                            return p.fail("expected a string key");
                        }
                        let k = p.string()?;
                        let x = sexp_read(p, depth + 1)?;
                        p.ws();
                        if !p.eat(b')') {
                            return p.fail("expected ')'");
                        }
                        items.push((k, x));
                    }
                    Ok(Value::Map(items))
                }
                _ => p.fail("expected 'list' or 'map'"),
            }
        }
        Some(_) => {
            let at = p.i;
            match p.atom() {
                "nil" => Ok(Value::Null),
                "#t" => Ok(Value::Bool(true)),
                "#f" => Ok(Value::Bool(false)),
                a => num(a).map_err(|e| format!("{e} at byte {at}")),
            }
        }
    }
}

pub const MAX_STAGES: usize = 16;
pub const MAX_WRAP: usize = 4096;
const MAGIC: &str = "PTG1:";
const WRAPS: [usize; 6] = [48, 64, 72, 76, 80, 100];
fn checksum(format: Format, payload: &[u8]) -> u32 {
    let name = format.name().as_bytes();
    let mut buf = Vec::with_capacity(name.len() + 1 + payload.len());
    buf.extend_from_slice(name);
    buf.push(0);
    buf.extend_from_slice(payload);
    crc32(&buf)
}
pub fn seal(format: Format, payload: &[u8]) -> Vec<u8> {
    let head = format!("{}{:08x}:{}:", MAGIC, checksum(format, payload), payload.len());
    let mut out = Vec::with_capacity(head.len() + payload.len());
    out.extend_from_slice(head.as_bytes());
    out.extend_from_slice(payload);
    out
}
pub fn open(format: Format, data: &[u8]) -> Result<&[u8], String> {
    let Some(rest) = data.strip_prefix(MAGIC.as_bytes()) else {
        return Err("envelope: bad magic (wrong pipeline spec or damaged data)".to_string());
    };
    if rest.len() < 9 || rest[8] != b':' {
        return Err("envelope: bad checksum field".to_string());
    }
    let mut want = 0u32;
    for &d in &rest[..8] {
        match (d as char).to_digit(16) {
            Some(h) => want = (want << 4) | h,
            None => return Err("envelope: bad checksum field".to_string()),
        }
    }
    let rest = &rest[9..];
    let Some(colon) = rest.iter().take(21).position(|&b| b == b':') else {
        return Err("envelope: bad length field".to_string());
    };
    let digits = &rest[..colon];
    let canonical = !digits.is_empty()
        && digits.iter().all(u8::is_ascii_digit)
        && (digits[0] != b'0' || digits.len() == 1);
    let len = match std::str::from_utf8(digits).ok().and_then(|s| s.parse::<usize>().ok()) {
        Some(n) if canonical => n,
        _ => return Err("envelope: bad length field".to_string()),
    };
    let body = &rest[colon + 1..];
    if len > body.len() {
        return Err("envelope: truncated payload".to_string());
    }
    let (payload, tail) = body.split_at(len);
    if !tail.iter().all(u8::is_ascii_whitespace) {
        return Err("envelope: unexpected data after the payload".to_string());
    }
    if checksum(format, payload) != want {
        return Err("envelope: checksum mismatch (wrong pipeline spec or damaged data)".to_string());
    }
    Ok(payload)
}
#[derive(Clone, PartialEq, Eq, Debug)]
pub struct Pipeline {
    pub format: Format,
    pub stages: Vec<Stage>,
    pub armor: Armor,
    pub wrap: usize, 
}
fn parse_wrap(s: &str) -> Result<usize, String> {
    let plain = !s.is_empty() && !s.starts_with('0') && s.bytes().all(|b| b.is_ascii_digit());
    match s.parse::<usize>() {
        Ok(n) if plain && n <= MAX_WRAP => Ok(n),
        _ => Err(format!(
            "bad wrap {:?} (expected a decimal number from 1 to {})",
            s, MAX_WRAP
        )),
    }
}
impl Pipeline {
    pub fn new(
        format: Format,
        stages: Vec<Stage>,
        armor: Armor,
        wrap: usize,
    ) -> Result<Pipeline, String> {
        let p = Pipeline { format, stages, armor, wrap };
        p.check()?;
        Ok(p)
    }
    pub fn check(&self) -> Result<(), String> {
        if self.stages.len() > MAX_STAGES {
            return Err(format!(
                "too many stages ({}, at most {})",
                self.stages.len(),
                MAX_STAGES
            ));
        }
        if self.wrap > MAX_WRAP {
            return Err(format!("wrap {} is above {}", self.wrap, MAX_WRAP));
        }
        if self.armor == Armor::Raw {
            if !self.format.is_text() {
                let texts: Vec<&str> = Format::ALL
                    .iter()
                    .filter(|f| f.is_text())
                    .map(|f| f.name())
                    .collect();
                return Err(format!(
                    "armor raw needs a text format ({}), not {}",
                    texts.join(", "),
                    self.format.name()
                ));
            }
            if !self.stages.is_empty() {
                return Err("armor raw cannot follow stages (their output is not text)".to_string());
            }
            if self.wrap != 0 {
                return Err("armor raw takes no wrap width".to_string());
            }
        }
        Ok(())
    }
    pub fn parse(spec: &str) -> Result<Pipeline, String> {
        let spec = spec.trim();
        let parts: Vec<&str> = spec.split('|').collect();
        if parts.len() < 2 {
            return Err(format!(
                "bad spec {:?}: needs at least a format and an armor, like \"json|b64\"",
                spec
            ));
        }
        let Some(format) = Format::parse(parts[0]) else {
            let all: Vec<&str> = Format::ALL.iter().map(|f| f.name()).collect();
            return Err(format!(
                "unknown format {:?} (expected one of: {})",
                parts[0],
                all.join(", ")
            ));
        };
        let mut stages: Vec<Stage> = Vec::new();
        for &part in &parts[1..parts.len() - 1] {
            match Stage::parse(part) {
                Some(s) => stages.push(s),
                None => {
                    return Err(format!(
                        "unknown stage {:?} (expected rle, lzss, delta, rev, mtf or xor:<1 to 16 hex digits>)",
                        part
                    ));
                }
            }
        }
        let last = parts[parts.len() - 1];
        let (name, wrap) = match last.split_once('@') {
            Some((n, w)) => (n, parse_wrap(w)?),
            None => (last, 0),
        };
        let Some(armor) = Armor::parse(name) else {
            let all: Vec<&str> = Armor::ALL.iter().map(|a| a.name()).collect();
            return Err(format!(
                "unknown armor {:?} (expected one of: {})",
                name,
                all.join(", ")
            ));
        };
        Pipeline::new(format, stages, armor, wrap)
    }
    pub fn spec(&self) -> String {
        let mut s = String::from(self.format.name());
        for st in self.stages.iter() {
            s.push('|');
            s.push_str(&st.name());
        }
        s.push('|');
        s.push_str(self.armor.name());
        if self.wrap > 0 {
            s.push('@');
            s.push_str(&self.wrap.to_string());
        }
        s
    }
    pub fn random(rng: &mut Rng) -> Pipeline {
        let format = Format::random(rng);
        let n = *rng.pick(&[0usize, 0, 1, 1, 2, 2, 3, 4]);
        let mut stages: Vec<Stage> = Vec::with_capacity(n);
        while stages.len() < n {
            let s = Stage::random(rng);
            if stages.last().is_some_and(|l| l.kind() == s.kind()) {
                continue;
            }
            stages.push(s);
        }
        let raw_ok = format.is_text() && stages.is_empty();
        let armors: Vec<Armor> = Armor::ALL
            .iter()
            .copied()
            .filter(|&a| a != Armor::Raw || raw_ok)
            .collect();
        let armor = *rng.pick(&armors);
        let wrap = if armor == Armor::Raw || rng.chance(0.4) { 0 } else { *rng.pick(&WRAPS) };
        Pipeline { format, stages, armor, wrap }
    }
    pub fn encode(&self, v: &Value) -> Result<String, String> {
        self.check()?;
        let body = self
            .format
            .encode(v)
            .map_err(|e| format!("{}: {e}", self.format.name()))?;
        let mut buf = seal(self.format, &body);
        for s in self.stages.iter() {
            buf = s.apply(&buf);
        }
        self.armor.encode(&buf, self.wrap)
    }
    pub fn decode(&self, text: &str) -> Result<Value, String> {
        self.check()?;
        let mut buf = self.armor.decode(text)?;
        for s in self.stages.iter().rev() {
            buf = s.undo(&buf)?;
        }
        let payload = open(self.format, &buf)?;
        self.format
            .decode(payload)
            .map_err(|e| format!("{}: {e}", self.format.name()))
    }
}

pub fn letter(i: usize) -> String {
    let mut n = i;
    let mut s = String::new();
    loop {
        s.insert(0, (b'A' + (n % 26) as u8) as char);
        if n < 26 {
            return s;
        }
        n = n / 26 - 1;
    }
}

pub fn task(p: &Problem) -> Value {
    let w = &p.world;
    let mut preds = Value::arr();
    for (i, d) in w.preds.iter().enumerate() {
        let mut o = Value::obj()
            .with("name", w.pred_name(i))
            .with("arity", d.arity)
            .with("initial", d.layer == 0);
        if d.layer == 0 {
            o.put("prior", d.prior);
        }
        preds.push(o);
    }
    let mut options = Value::arr();
    for (i, c) in p.candidates.iter().enumerate() {
        options.push(
            Value::obj()
                .with("label", letter(i))
                .with("facts", list(p, &c.facts, true)),
        );
    }
    let entities: Vec<String> = w.consts.iter().map(|&c| text(p, c)).collect();
    let rules: Vec<String> = (0..w.rules.len()).map(|i| w.rule_text(i)).collect();
    let dynamics = Value::obj()
        .with("horizon", p.params.horizon)
        .with("max_steps", p.params.max_steps);
    Value::obj()
        .with("kind", "task")
        .with("version", VERSION)
        .with("id", hex(p.seed))
        .with("instructions", Value::seq(BRIEF.iter().copied()))
        .with("dynamics", dynamics)
        .with("entities", Value::seq(entities))
        .with("predicates", preds)
        .with("rules", Value::seq(rules))
        .with("disclosed", list(p, &p.disclosed, true))
        .with("observed", list(p, &p.observed, true))
        .with("options", options)
}

pub fn key(p: &Problem) -> Value {

    pub kept: Vec<NodeId>,
    pub added: Vec<NodeId>,
}
#[derive(Clone, Debug, Default)]
pub struct State {
    pub facts: BTreeSet<NodeId>,
}
fn heads_index(g: &Graph, st: &State) -> HashMap<SymId, Vec<NodeId>> {
    let mut m: HashMap<SymId, Vec<NodeId>> = HashMap::new();
    for &f in st.facts.iter() {
        m.entry(g.head(f)).or_default().push(f);
    }
    m
}
fn match_rec(
    g: &Graph,
    pats: &[NodeId],
    k: usize,
    idx: &HashMap<SymId, Vec<NodeId>>,
    used: &mut Vec<NodeId>,
    sub: &Sub,
    out: &mut Vec<Sub>,
    cap: usize,
) {
    if out.len() >= cap {
        return;
    }
    if k == pats.len() {
        out.push(sub.clone());
        return;
    }
    let cands = match idx.get(&g.head(pats[k])) {
        Some(c) => c,
        None => return,
    };
    for &f in cands.iter() {
        if used.contains(&f) {
            continue; 
        }
        let mut s2 = sub.clone();
        if g.match_term(pats[k], f, &mut s2) {
            used.push(f);
            match_rec(g, pats, k + 1, idx, used, &s2, out, cap);
            used.pop();
            if out.len() >= cap {
                return;
            }
        }
    }
}
pub fn find_matches(g: &Graph, r: &Rule, idx: &HashMap<SymId, Vec<NodeId>>, cap: usize) -> Vec<Sub> {
    let mut out: Vec<Sub> = Vec::new();
    let mut used: Vec<NodeId> = Vec::new();
    let sub: Sub = vec![None; r.nvars];
    match_rec(g, &r.lhs, 0, idx, &mut used, &sub, &mut out, cap);
    out
}
fn binding_of(r: &Rule, sub: &Sub) -> Vec<NodeId> {
    sub.iter().take(r.nvars).map(|x| x.unwrap_or(u32::MAX)).collect()
}
pub fn apply_binding(
    g: &mut Graph,
    r: &Rule,
    ri: usize,
    st: &mut State,
    binding: &[NodeId],
    alt: usize,
    t: f64,
) -> Result<Step, String> {
    if binding.len() != r.nvars {
        return Err("binding arity mismatch".to_string());
    }
    if alt >= r.alts.len() {
        return Err("bad outcome index".to_string());
    }
    let mut full: Sub = binding.iter().map(|&b| Some(b)).collect();
    for &sk in r.skolem.iter() {
        let n = g.mk(sk, binding);
        full.push(Some(n));
    }
    let matched: Vec<NodeId> = r.lhs.iter().map(|&p| g.subst(p, &full)).collect();
    for &f in matched.iter() {
        if !st.facts.contains(&f) {
            return Err(format!("missing precondition {}", g.show(f)));
        }
    }
    let consumed: Vec<NodeId> = matched[..r.take].to_vec();
    let kept: Vec<NodeId> = matched[r.take..].to_vec();
    for f in consumed.iter() {
        st.facts.remove(f);
    }
    let rhs: Vec<NodeId> = r.alts[alt].1.iter().map(|&p| g.subst(p, &full)).collect();
    let mut added: Vec<NodeId> = Vec::new();
    for f in rhs {
        if st.facts.insert(f) {
            added.push(f);
        }
    }
    Ok(Step { t, rule: ri, alt, binding: binding.to_vec(), consumed, kept, added })
}
pub fn simulate(
    g: &mut Graph,
    rules: &[Rule],
    init: &State,
    rng: &mut Rng,
    horizon: f64,
    max_steps: usize,
) -> (State, Vec<Step>) {
    let mut st = init.clone();
    let mut steps: Vec<Step> = Vec::new();
    let mut fired: HashSet<(usize, Vec<NodeId>)> = HashSet::new();
    let mut t = 0.0f64;
    let mut guard = 0usize;
    while steps.len() < max_steps && guard < max_steps * 4 {
        guard += 1;
        let idx = heads_index(g, &st);
        let mut chans: Vec<(usize, Sub)> = Vec::new();
        let mut rates: Vec<f64> = Vec::new();
        for (ri, r) in rules.iter().enumerate() {
            for sub in find_matches(g, r, &idx, 64) {
                if r.take == 0 && fired.contains(&(ri, binding_of(r, &sub))) {
                    continue;
                }
                rates.push(r.rate);
                chans.push((ri, sub));
            }
        }
        if chans.is_empty() {
            break;
        }
        let total: f64 = rates.iter().sum();
        t += rng.exp(total);
        if t > horizon {
            break;
        }
        let pick = rng.weighted(&rates);
        let (ri, sub) = chans.swap_remove(pick);
        let r = &rules[ri];
        let probs: Vec<f64> = r.alts.iter().map(|a| a.0).collect();
        let alt = rng.weighted(&probs);
        let binding = binding_of(r, &sub);
        if r.take == 0 {
            fired.insert((ri, binding.clone()));
        }
        if let Ok(step) = apply_binding(g, r, ri, &mut st, &binding, alt, t) {
            if !step.consumed.is_empty() || !step.added.is_empty() {
                steps.push(step);
            }
        }
    }
    (st, steps)
}
pub fn replay(
    g: &mut Graph,
    rules: &[Rule],
    init: &State,
    steps: &[Step],
    strict: bool,
) -> Result<State, String> {
    let mut st = init.clone();
    for (i, s) in steps.iter().enumerate() {
        let r = match rules.get(s.rule) {
            Some(r) => r,
            None => return Err(format!("step {i}: unknown rule")),
        };
        let new = apply_binding(g, r, s.rule, &mut st, &s.binding, s.alt, s.t)
            .map_err(|e| format!("step {i}: {e}"))?;
        if strict && (new.added != s.added || new.consumed != s.consumed || new.kept != s.kept) {
            return Err(format!("step {i}: effect mismatch"));
        }
    }
    Ok(st)
}
pub fn slice_trace(steps: &[Step], targets: &[NodeId]) -> (Vec<usize>, BTreeSet<NodeId>) {
    let mut need: BTreeSet<NodeId> = targets.iter().cloned().collect();
    let mut used: Vec<usize> = Vec::new();
    for i in (0..steps.len()).rev() {
        let s = &steps[i];
        if s.added.iter().any(|f| need.contains(f)) {
            for f in s.added.iter() {
                need.remove(f);
            }
            for f in s.consumed.iter().chain(s.kept.iter()) {
                need.insert(*f);
            }
            used.push(i);
        }
    }
    used.reverse();
    (used, need)
}

#[derive(Clone, Debug)]
pub struct Rng {
    s: [u64; 4],
}
fn splitmix(state: &mut u64) -> u64 {
    *state = state.wrapping_add(0x9E37_79B9_7F4A_7C15);
    let mut z = *state;
    z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
    z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
    z ^ (z >> 31)
}
impl Rng {
    pub fn new(seed: u64) -> Rng {
        let mut sm = seed;
        let mut s = [0u64; 4];
        for x in s.iter_mut() {
            *x = splitmix(&mut sm);
        }
        Rng { s }
    }
    pub fn next_u64(&mut self) -> u64 {
        let result = self.s[1].wrapping_mul(5).rotate_left(7).wrapping_mul(9);
        let t = self.s[1] << 17;
        self.s[2] ^= self.s[0];
        self.s[3] ^= self.s[1];
        self.s[1] ^= self.s[2];
        self.s[0] ^= self.s[3];
        self.s[2] ^= t;
        self.s[3] = self.s[3].rotate_left(45);
        result
    }
    pub fn below(&mut self, n: usize) -> usize {
        if n <= 1 {
            return 0;
        }
        let n64 = n as u64;
        let mut x = self.next_u64();
        let mut m = (x as u128) * (n64 as u128);
        let mut l = m as u64;
        if l < n64 {
            let t = n64.wrapping_neg() % n64;
            while l < t {
                x = self.next_u64();
                m = (x as u128) * (n64 as u128);
                l = m as u64;
            }
        }
        (m >> 64) as usize
    }
    pub fn range(&mut self, lo: usize, hi: usize) -> usize {
        if hi <= lo {
            return lo;
        }
        lo + self.below(hi - lo + 1)
    }
    pub fn unit(&mut self) -> f64 {
        (self.next_u64() >> 11) as f64 * (1.0 / 9007199254740992.0)
    }
    pub fn uniform(&mut self, lo: f64, hi: f64) -> f64 {
        lo + (hi - lo) * self.unit()
    }
    pub fn chance(&mut self, p: f64) -> bool {
        self.unit() < p
    }
    pub fn pick<'a, T>(&mut self, xs: &'a [T]) -> &'a T {
        &xs[self.below(xs.len())]
    }
    pub fn weighted(&mut self, w: &[f64]) -> usize {
        let total: f64 = w.iter().sum();
        if !(total > 0.0) {
            return self.below(w.len());
        }
        let mut x = self.unit() * total;
        for (i, &wi) in w.iter().enumerate() {
            if x < wi {
                return i;
            }
            x -= wi;
        }
        w.len() - 1
    }
    pub fn shuffle<T>(&mut self, xs: &mut [T]) {
        for i in (1..xs.len()).rev() {
            let j = self.below(i + 1);
            xs.swap(i, j);
        }
    }
    pub fn exp(&mut self, rate: f64) -> f64 {
        -(1.0 - self.unit()).ln() / rate
    }
    pub fn fork(&mut self) -> Rng {
        Rng::new(self.next_u64())
    }
}

use std::collections::HashMap;
pub type NodeId = u32;
pub type SymId = u32;
#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub enum Kind {
    Pred,
    Func, 
    Var,
}
#[derive(Clone, Debug)]
pub struct SymInfo {
    pub name: String,
    pub kind: Kind,
    pub arity: usize,
    pub aux: usize, 
}
#[derive(Clone, PartialEq, Eq, Hash, Debug)]
pub struct Node {
    pub sym: SymId,
    pub args: Box<[NodeId]>,
}
#[derive(Default)]
pub struct Graph {
    syms: Vec<SymInfo>,
    by_name: HashMap<String, SymId>,
    nodes: Vec<Node>,
    table: HashMap<Node, NodeId>,
    vars: Vec<NodeId>,
}
impl Graph {
    pub fn new() -> Graph {
        Graph::default()
    }
    fn new_sym(&mut self, name: String, kind: Kind, arity: usize, aux: usize) -> SymId {
        let id = self.syms.len() as SymId;
        self.by_name.insert(name.clone(), id);
        self.syms.push(SymInfo { name, kind, arity, aux });
        id
    }
    pub fn sym(&mut self, name: &str, kind: Kind, arity: usize) -> SymId {
        if let Some(&s) = self.by_name.get(name) {
            return s;
        }
        self.new_sym(name.to_string(), kind, arity, 0)
    }
    pub fn rename_sym(&mut self, s: SymId, new_name: &str) {
        let old = self.syms[s as usize].name.clone();
        self.by_name.remove(&old);
        self.syms[s as usize].name = new_name.to_string();
        self.by_name.insert(new_name.to_string(), s);
    }
    pub fn mk(&mut self, sym: SymId, args: &[NodeId]) -> NodeId {
        let node = Node { sym, args: args.to_vec().into_boxed_slice() };
        if let Some(&id) = self.table.get(&node) {
            return id;
        }
        let id = self.nodes.len() as NodeId;
        self.nodes.push(node.clone());
        self.table.insert(node, id);
        id
    }
    pub fn var(&mut self, i: usize) -> NodeId {
        while self.vars.len() <= i {
            let k = self.vars.len();
            let s = self.new_sym(format!("?{k}"), Kind::Var, 0, k);
            let n = self.mk(s, &[]);
            self.vars.push(n);
        }
        self.vars[i]
    }
    pub fn head(&self, id: NodeId) -> SymId {
        self.nodes[id as usize].sym
    }
    pub fn args(&self, id: NodeId) -> &[NodeId] {
        &self.nodes[id as usize].args
    }
    pub fn info(&self, s: SymId) -> &SymInfo {
        &self.syms[s as usize]
    }
    pub fn name(&self, s: SymId) -> &str {
        &self.syms[s as usize].name
    }
    pub fn var_index(&self, id: NodeId) -> Option<usize> {
        let s = &self.syms[self.nodes[id as usize].sym as usize];
        if s.kind == Kind::Var { Some(s.aux) } else { None }
    }
    pub fn max_var(&self, id: NodeId) -> Option<usize> {
        if let Some(i) = self.var_index(id) {
            return Some(i);
        }
        let mut best: Option<usize> = None;
        for &a in self.args(id).iter() {
            if let Some(m) = self.max_var(a) {
                best = Some(best.map_or(m, |b| b.max(m)));
            }
        }
        best
    }
    pub fn depth(&self, id: NodeId) -> usize {
        1 + self.args(id).iter().map(|&a| self.depth(a)).max().unwrap_or(0)
    }
    pub fn match_term(&self, pat: NodeId, tgt: NodeId, sub: &mut Vec<Option<NodeId>>) -> bool {
        if pat == tgt {
            return true;
        }
        let pn = &self.nodes[pat as usize];
        let ps = &self.syms[pn.sym as usize];
        if ps.kind == Kind::Var {
            let cur = sub[ps.aux];
            return match cur {
                Some(b) => b == tgt,
                None => {
                    sub[ps.aux] = Some(tgt);
                    true
                }
            };
        }
        let tn = &self.nodes[tgt as usize];
        if pn.sym != tn.sym || pn.args.len() != tn.args.len() {
            return false;
        }
        for k in 0..pn.args.len() {
            if !self.match_term(pn.args[k], tn.args[k], sub) {
                return false;
            }
        }
        true
    }
    pub fn subst(&mut self, pat: NodeId, sub: &[Option<NodeId>]) -> NodeId {
        let sym = self.nodes[pat as usize].sym;
        let info = &self.syms[sym as usize];
        if info.kind == Kind::Var {
            return sub[info.aux].expect("unbound variable in right-hand side");
        }
        let args: Vec<NodeId> = self.nodes[pat as usize].args.to_vec();
        if args.is_empty() {
            return pat;
        }
        let mut na: Vec<NodeId> = Vec::with_capacity(args.len());
        for a in args {
            na.push(self.subst(a, sub));
        }
        self.mk(sym, &na)
    }
    pub fn show(&self, id: NodeId) -> String {
        let mut s = String::new();
        self.show_into(id, &mut s);
        s
    }
    fn show_into(&self, id: NodeId, out: &mut String) {
        let n = &self.nodes[id as usize];
        out.push_str(self.name(n.sym));
        if !n.args.is_empty() {
            out.push('(');
            for (i, &a) in n.args.iter().enumerate() {
                if i > 0 {
                    out.push(',');
                }
                self.show_into(a, out);
            }
            out.push(')');
        }
    }
}

pub fn crc32(bytes: &[u8]) -> u32 {
    let mut crc: u32 = !0;
    for &b in bytes {
        crc ^= b as u32;
        for _ in 0..8 {
            let mask = (crc & 1).wrapping_neg();
            crc = (crc >> 1) ^ (0xEDB8_8320 & mask);
        }
    }
    !crc
}

    let mut z = a ^ b.wrapping_mul(0x9E37_79B9_7F4A_7C15);

    z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
    z ^ (z >> 31)
}

        match (self, o) {
            (Value::Null, Value::Null) => true,
            (Value::Bool(a), Value::Bool(b)) => a == b,
            (Value::Int(a), Value::Int(b)) => a == b,
            (Value::Float(a), Value::Float(b)) => a.to_bits() == b.to_bits(),
            (Value::Str(a), Value::Str(b)) => a == b,

            }

        }

fn random_int(rng: &mut Rng) -> i64 {
    const SPECIAL: &[i64] = &[
        0, 1, -1, 23, 24, -24, -25, 255, 256, 65_535, 65_536, 4_294_967_296, i64::MAX, i64::MIN,
    ];
    if rng.chance(0.4) {
        return SPECIAL[rng.below(SPECIAL.len())];
    }
    let bits = rng.range(1, 63) as u32;
    let x = (rng.next_u64() >> (64 - bits)) as i64; 
    if rng.chance(0.5) { -x } else { x }
}
fn random_float(rng: &mut Rng) -> f64 {
    match rng.below(7) {
        0 => 0.0,
        1 => -0.0,
        2 => f64::MIN_POSITIVE / 8.0, 
        3 => f64::MAX,
        4 => -f64::MAX / 3.0,
        5 => rng.uniform(-1.0, 1.0),
        _ => rng.uniform(-1.0, 1.0) * 10f64.powi(rng.range(0, 60) as i32 - 30),
    }
}
fn random_string(rng: &mut Rng) -> String {
    const PIECES: &[&str] = &[
        "a", "b", "zz", " ", "\"", "\\", "/", "\n", "\t", "\u{0}", "\u{1f}", "\u{7f}", "é", "ß",
        "漢", "😀", "\u{2028}", ",", ":", "(", ")", "{", "}", "[", "]", "'", ";", "#", "i", "e",
        "0", "-", ".", "nil", "@",
    ];
    let n = rng.below(7);
    let mut s = String::new();
    for _ in 0..n {
        s.push_str(rng.pick(PIECES));
    }
    s
}
pub fn random_value(rng: &mut Rng, depth: usize) -> Value {
    let kinds = if depth == 0 { 5 } else { 7 };
    match rng.below(kinds) {
        0 => Value::Null,
        1 => Value::Bool(rng.chance(0.5)),
        2 => Value::Int(random_int(rng)),
        3 => Value::Float(random_float(rng)),
        4 => Value::Str(random_string(rng)),
        5 => {
            let n = rng.below(5);
            let mut items = Vec::with_capacity(n);
            for _ in 0..n {
                items.push(random_value(rng, depth - 1));
            }
            Value::List(items)
        }
        _ => {
            let n = rng.below(5);
            let mut items = Vec::with_capacity(n);
            for _ in 0..n {
                let k = random_string(rng);
                items.push((k, random_value(rng, depth - 1)));
            }
            Value::Map(items)
        }
    }
}

    pub fn word(&mut self, rng: &mut Rng, lo: usize, hi: usize) -> String {
        let mut miss = 0usize;
        loop {
            let n = rng.range(lo.max(1), hi.max(1)) + miss / 16;
            let mut w = String::new();
            for i in 0..n {
                w.push_str(rng.pick(ONSETS));
                w.push_str(rng.pick(VOWELS));
                if i + 1 == n {
                    w.push_str(rng.pick(CODAS));
                }
            }
            if self.used.insert(w.clone()) {
                return w;
            }
            miss += 1;
        }
    }

fn pick_args(rng: &mut Rng, vs: &[NodeId], k: usize) -> Vec<NodeId> {
    let mut idx: Vec<usize> = (0..vs.len()).collect();

fn outcomes(rng: &mut Rng, leak: f64, fx: Vec<NodeId>) -> Vec<(f64, Vec<NodeId>)> {
    if rng.chance(leak) {
        let e = round(rng.uniform(0.05, 0.30), 2);
        vec![(round(1.0 - e, 2), fx), (e, Vec::new())]
    } else {
        vec![(1.0, fx)]
    }
}
fn probs(rng: &mut Rng, k: usize) -> Vec<f64> {
    let mut left = 1.0f64;
    let mut out: Vec<f64> = Vec::new();
    for i in 0..k - 1 {
        let hi = (left - 0.1 * ((k - 1 - i) as f64)).max(0.1);
        let x = round(rng.uniform(0.1, hi), 2);
        out.push(x);
        left = round(left - x, 2);
    }
    out.push(left);
    out
}
fn arities(rng: &mut Rng, n: usize, p2: f64, need2: bool) -> Vec<usize> {

        if !a.contains(&1) {

}

    fn add(
        &mut self,
        rng: &mut Rng,
        kind: RuleKind,
        lhs: Vec<NodeId>,
        take: usize,
        alts: Vec<(f64, Vec<NodeId>)>,
        nvars: usize,
        skolem: Vec<SymId>,
    ) -> bool {

            r.lhs == lhs
                && r.take == take
                && r.alts.len() == alts.len()
                && r.alts.iter().zip(alts.iter()).all(|(x, y)| x.1 == y.1)
        });
        if dup {
            return false;
        }
        let rate = round(rng.uniform(0.6, 2.5), 2);
        self.rules.push(Rule { lhs, take, alts, rate, nvars, skolem });
        self.kinds.push(kind);

    fn t_chain(&mut self, rng: &mut Rng, pr: &Params) -> bool {

        let (pl, pa) = (self.preds[p].layer, self.preds[p].arity);
        let dst = self.select(|q| q.layer > pl && q.arity <= pa && !q.latent);
        if dst.is_empty() {
            return false;
        }
        let q = dst[rng.below(dst.len())];
        let vs = self.vars(pa);
        let qargs = pick_args(rng, &vs, self.preds[q].arity);
        let lhs = self.atom(p, &vs);
        let rhs = self.atom(q, &qargs);
        let take: usize = if rng.chance(pr.consume) { 1 } else { 0 };
        let kind = if take > 0 { RuleKind::Consume } else { RuleKind::Chain };
        let alts = outcomes(rng, pr.leak, vec![rhs]);
        self.add(rng, kind, vec![lhs], take, alts, pa, Vec::new())
    }
    fn t_conj(&mut self, rng: &mut Rng, pr: &Params) -> bool {
        let p = rng.below(self.preds.len());

        let (pa, ra) = (self.preds[p].arity, self.preds[r].arity);
        if p == r && pa == 1 {
            return false;
        }
        let top = self.preds[p].layer.max(self.preds[r].layer);
        let nv = pa + ra - 1;
        let dst = self.select(|q| q.layer > top && q.arity <= nv && !q.latent);
        if dst.is_empty() {
            return false;
        }
        let q = dst[rng.below(dst.len())];
        let vs = self.vars(nv);
        let mut rargs: Vec<NodeId> = vec![vs[pa - 1]];
        rargs.extend_from_slice(&vs[pa..]);
        let qargs = pick_args(rng, &vs, self.preds[q].arity);
        let l1 = self.atom(p, &vs[..pa]);
        let l2 = self.atom(r, &rargs);
        let rhs = self.atom(q, &qargs);
        let take: usize = if rng.chance(pr.consume) { 1 } else { 0 };
        let kind = if take > 0 { RuleKind::Consume } else { RuleKind::Conj };
        let alts = outcomes(rng, pr.leak, vec![rhs]);
        self.add(rng, kind, vec![l1, l2], take, alts, nv, Vec::new())
    }

    fn t_consume(&mut self, rng: &mut Rng, pr: &Params) -> bool {

            return false;

            return false;

        let dst = self.select(|q| q.layer > top && q.arity <= pa && !q.latent);

    fn t_invent(&mut self, rng: &mut Rng, pr: &Params) -> bool {

        let two = self.select(|q| q.layer > pl && q.arity == 2 && !q.latent);
        let one = self.select(|q| q.layer > pl && q.arity == 1 && !q.latent);

        let lhs = self.atom(p, &vs);
        let take: usize = if rng.chance(pr.consume) { 1 } else { 0 };

        self.add(rng, RuleKind::Invent, vec![lhs], take, alts, pa, vec![sk])
    }
    fn t_func(&mut self, rng: &mut Rng, pr: &Params) -> bool {
        if self.funcs.is_empty() {
            return false;
        }
        let f = self.funcs[rng.below(self.funcs.len())];
        let uni = self.select(|q| q.arity == 1);
        if uni.is_empty() {
            return false;
        }
        let p = uni[rng.below(uni.len())];
        let pl = self.preds[p].layer;
        let mid = self.select(|q| q.arity == 1 && q.layer > pl && !q.latent);
        if mid.is_empty() {
            return false;
        }

            let ok2 = self.add(rng, RuleKind::Func, vec![l2], 0, alts2, 1, Vec::new());
            return ok || ok2;
        }
        ok
    }
    fn t_latent(&mut self, rng: &mut Rng, pr: &Params, h: usize) -> bool {
        let ha = self.preds[h].arity;
        let mut dst = self.select(|q| q.layer > 0 && q.arity <= ha && !q.latent);
        if dst.len() < 2 {
            return false;

        rng.shuffle(&mut dst);
        let k = rng.range(2, dst.len().min(3));

        let vs = self.vars(ha);

        let mut fx: Vec<NodeId> = Vec::new();

            let a = pick_args(rng, &vs, self.preds[q].arity);
            fx.push(self.atom(q, &a));

        let alts = if fx.len() > 1 && rng.chance(0.4) {

            outcomes(rng, pr.leak, fx)

        self.add(rng, RuleKind::Latent, vec![lhs], take, alts, ha, Vec::new())
    }

        let lhs = self.atom(p, &vs);
        let rhs = self.atom(q, &qargs);

    pub fn logit_prior(&self, f: NodeId) -> f64 {
        match self.pred_of(f) {
            Some(p) if self.preds[p].layer == 0 => {
                let x = self.preds[p].prior.clamp(1e-6, 1.0 - 1e-6);
                (x / (1.0 - x)).ln()
            }
            _ => -20.0,
        }
    }
    pub fn sample_init(&mut self, rng: &mut Rng) -> State {
        let mut st = State::default();
        let n = self.consts.len();
        for pi in 0..self.preds.len() {
            let (layer, arity, latent, prior) = {
                let d = &self.preds[pi];
                (d.layer, d.arity, d.latent, d.prior)
            };
            if layer != 0 {
                continue;
            }
            let before = st.facts.len();
            if arity == 1 {
                for i in 0..n {
                    if rng.chance(prior) {
                        let c = self.consts[i];
                        let f = self.atom(pi, &[c]);
                        st.facts.insert(f);
                    }
                }
            } else {
                for i in 0..n {
                    for j in 0..n {
                        if i != j && rng.chance(prior) {
                            let (a, b) = (self.consts[i], self.consts[j]);
                            let f = self.atom(pi, &[a, b]);
                            st.facts.insert(f);
                        }
                    }
                }
            }
            if latent && st.facts.len() == before {
                let a = self.consts[rng.below(n)];
                let mut args = vec![a];
                if arity == 2 {
                    let mut b = self.consts[rng.below(n)];
                    while b == a {
                        b = self.consts[rng.below(n)];
                    }
                    args.push(b);
                }
                let f = self.atom(pi, &args);
                st.facts.insert(f);
            }
        }
        st
    }
    pub fn validate(&self) -> Result<(), String> {
        for (i, r) in self.rules.iter().enumerate() {
            if r.lhs.is_empty() || r.take > r.lhs.len() {
                return Err(format!("rule {i}: bad lhs/take"));
            }
            if r.alts.is_empty() {
                return Err(format!("rule {i}: no outcomes"));
            }
            let total: f64 = r.alts.iter().map(|a| a.0).sum();
            if (total - 1.0).abs() > 1e-6 {
                return Err(format!("rule {i}: probabilities sum to {total}"));
            }
            if !(r.rate > 0.0) {
                return Err(format!("rule {i}: non-positive rate"));
            }
            let mut bound: Vec<usize> = Vec::new();
            for &p in r.lhs.iter() {
                collect_vars(&self.g, p, &mut bound);
            }
            for &v in bound.iter() {
                if v >= r.nvars {
                    return Err(format!("rule {i}: lhs uses ?{v} beyond nvars"));
                }
            }
            for v in 0..r.nvars {
                if !bound.contains(&v) {
                    return Err(format!("rule {i}: ?{v} does not occur in the lhs"));
                }
            }
            let mut used: Vec<usize> = Vec::new();
            for a in r.alts.iter() {
                for &f in a.1.iter() {
                    collect_vars(&self.g, f, &mut used);
                }
            }
            for &v in used.iter() {
                if v >= r.nvars + r.skolem.len() {
                    return Err(format!("rule {i}: rhs uses undeclared ?{v}"));
                }
            }
            for &s in r.skolem.iter() {
                if self.g.info(s).arity != r.nvars {
                    return Err(format!("rule {i}: skolem arity differs from nvars"));
                }
            }
        }
        Ok(())
    }
    pub fn rule_text(&self, i: usize) -> String {
        let r = &self.rules[i];
        let mut s = String::new();
        for (k, &pat) in r.lhs.iter().enumerate() {
            if k > 0 {
                s.push_str(" & ");
            }
            if k < r.take {
                s.push('-');
            }
            s.push_str(&self.g.show(pat));
        }
        s.push_str(" => ");
        for (a, alt) in r.alts.iter().enumerate() {
            if a > 0 {
                s.push_str(" | ");
            }
            if r.alts.len() > 1 {
                s.push_str(&format!("{}: ", alt.0));
            }
            if alt.1.is_empty() {
                s.push_str("nothing");
            }
            for (k, &f) in alt.1.iter().enumerate() {
                if k > 0 {
                    s.push_str(", ");
                }
                s.push_str(&self.g.show(f));
            }
        }
        if r.skolem.len() == 1 {
            s.push_str(&format!("  [new ?{}]", r.nvars));
        } else if r.skolem.len() > 1 {
            s.push_str(&format!("  [new ?{}..?{}]", r.nvars, r.nvars + r.skolem.len() - 1));
        }
        s.push_str(&format!("  @{}", r.rate));
        s
    }
}

fn dec(s: &str) -> Option<usize> {
    let plain = !s.is_empty()
        && s.bytes().all(|b| b.is_ascii_digit())
        && (s == "0" || !s.starts_with('0'));
    if plain { s.parse::<usize>().ok() } else { None }
}
fn knob(name: &str, val: &str) -> Result<(&'static str, f64), String> {
    let Some(&(key, lo, hi, _)) = KNOBS.iter().find(|k| k.0 == name) else {
        let all: Vec<&str> = KNOBS.iter().map(|k| k.0).collect();
        return Err(format!("unknown parameter {name:?} (expected one of: {})", all.join(", ")));
    };
    let x = if is_int(key) {
        match dec(val) {
            Some(n) => n as f64,
            None => return Err(format!("{key}: expected a plain decimal integer, got {val:?}")),
        }
    } else {
        match val.parse::<f64>() {
            Ok(x) if x.is_finite() => x,
            _ => return Err(format!("{key}: expected a number, got {val:?}")),
        }
    };
    if !(lo <= x && x <= hi) {
        return Err(format!("{key}: {val} is outside {lo}..={hi}"));
    }
    Ok((key, x))
}

            Some(Field::Real(r)) => *r = x,

    }
}

fn parse_seed(s: &str) -> Result<u64, String> {
    let r = match s.strip_prefix("0x").or_else(|| s.strip_prefix("0X")) {
        Some(h) if !h.is_empty() && h.bytes().all(|b| b.is_ascii_hexdigit()) => {
            u64::from_str_radix(h, 16).ok()
        }
        Some(_) => None,
        None if !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit()) => s.parse::<u64>().ok(),
        None => None,
    };
    r.ok_or_else(|| {
        format!("bad seed {s:?} (expected a decimal number or 0x and hex digits, below 2^64)")
    })
}
fn parse_level(s: &str) -> Result<(u32, u32), String> {
    let bad = || format!("bad level {s:?} (expected 1..=6 or a range such as 2-5)");
    let (a, b) = match s.split_once('-') {
        Some((a, b)) => (a, b),
        None => (s, s),
    };
    let (Some(lo), Some(hi)) = (dec(a), dec(b)) else {
        return Err(bad());
    };
    if lo < 1 || hi > 6 || lo > hi {
        return Err(bad());
    }
    Ok((lo as u32, hi as u32))
}
fn parse_opts(cmd: &str, args: &[String]) -> Result<Opts, String> {
    let allowed = accepts(cmd);
    let mut o = Opts {
        seed: None,
        level: None,
        codec: Codec::Random,
        key: false,
        count: None,
        set: Vec::new(),
    };
    let mut i = 0usize;
    while i < args.len() {
        let a = args[i].as_str();
        i += 1;
        let Some(body) = a.strip_prefix("--") else {
            return Err(format!("unexpected argument {a:?} for `{cmd}`"));
        };
        let (name, inline) = match body.split_once('=') {
            Some((n, v)) => (n, Some(v)),
            None => (body, None),
        };
        let Some(&(_, arg, _)) = OPTIONS.iter().find(|x| x.0 == name && allowed.contains(&x.0)) else {
            return Err(format!("unknown option --{name} for `{cmd}`"));
        };
        let v: &str = if arg.is_empty() {
            if inline.is_some() {
                return Err(format!("--{name} takes no value"));
            }
            ""
        } else if let Some(v) = inline {
            v
        } else {
            let Some(v) = args.get(i) else {
                return Err(format!("--{name} needs a value ({arg})"));
            };
            i += 1;
            v.as_str()
        };
        match name {
            "seed" => o.seed = Some(parse_seed(v)?),
            "level" => o.level = Some(parse_level(v)?),
            "codec" => {
                o.codec = if v == "random" {
                    Codec::Random
                } else {
                    Codec::Fixed(Pipeline::parse(v).map_err(|e| format!("--codec: {e}"))?)
                }
            }
            "key" => o.key = true,
            "count" => match dec(v) {
                Some(n) if (1..=MAX_COUNT).contains(&n) => o.count = Some(n),
                _ => return Err(format!("bad count {v:?} (expected 1 to {MAX_COUNT})")),
            },
            "set" => {
                let Some((k, x)) = v.split_once('=') else {
                    return Err(format!("--set expects K=V, got {v:?}"));
                };
                o.set.push(knob(k, x)?);
            }
            _ => return Err(format!("unknown option --{name} for `{cmd}`")),
        }
    }
    Ok(o)
}
fn parse(args: &[String]) -> Result<Cmd, String> {
    let Some(first) = args.first() else {
        return Ok(Cmd::Help);
    };
    let rest = &args[1..];
    Ok(match first.as_str() {
        "help" | "-h" | "--help" => Cmd::Help,
        "gen" => {
            let o = parse_opts("gen", rest)?;
            if let Some((lo, hi)) = o.level {
                if lo != hi {
                    return Err("gen takes a single level, not a range (use batch)".to_string());
                }
            }
            Cmd::Gen(o)
        }
        "batch" => Cmd::Batch(parse_opts("batch", rest)?),
        "verify" => Cmd::Verify(parse_opts("verify", rest)?),
        "selftest" => Cmd::SelfTest(parse_opts("selftest", rest)?),
        "codec" => match rest.split_first() {
            None => return Err("codec needs encode, decode or random".to_string()),
            Some((sub, more)) => match (sub.as_str(), more) {
                ("encode", [spec]) => Cmd::Encode(Pipeline::parse(spec)?),
                ("encode", _) => return Err("usage: codec encode SPEC".to_string()),
                ("decode", []) => Cmd::Decode(None),
                ("decode", [spec]) => Cmd::Decode(Some(Pipeline::parse(spec)?)),
                ("decode", _) => return Err("usage: codec decode [SPEC]".to_string()),
                ("random", _) => Cmd::Random(parse_opts("codec random", more)?),
                (other, _) => {
                    return Err(format!(
                        "unknown codec command {other:?} (expected encode, decode or random)"
                    ));
                }
            },
        },
        other => return Err(format!("unknown command {other:?} (try `ptgrs help`)")),
    })
}

fn make(
    seed: u64,
    level: u32,
    set: &[(&'static str, f64)],
    cap: Option<usize>,
) -> Result<Problem, String> {
    let mut p = params_for(level, set);
    if let Some(c) = cap {
        p.trials = p.trials.min(c);
    }
    let mut last = String::new();
    for k in 0..=RETRIES {
        let s = if k == 0 { seed } else { mix(seed, RETRY + k) };
        match build(s, &p) {
            Ok(a) => return Ok(a),
            Err(e) => last = e,
        }
    }
    Err(last)
}
fn frame(kind: &str, id: u64, level: u32, spec: &str, body: &str) -> String {
    format!(
        "{MAGIC} {kind} id={} level={level} bytes={} spec={spec}\n{body}\n\n",
        render::hex(id),
        body.len()
    )

struct Rec<'a> {
    kind: &'a str,
    id: &'a str,
    level: usize,
    spec: &'a str,
    body: &'a str,
}
fn unframe(text: &str) -> Result<Vec<Rec<'_>>, String> {
    let mut out: Vec<Rec<'_>> = Vec::new();
    let mut at = 0usize;
    loop {
        let rest = text[at..].trim_start();
        at = text.len() - rest.len();
        if rest.is_empty() {
            return Ok(out);
        }
        let n = out.len() + 1;
        let Some(eol) = rest.find('\n') else {
            return Err(format!("record {n}: the header has no end of line"));
        };
        let head = rest[..eol].trim_end();
        let f: Vec<&str> = head.split(' ').collect();
        if f.len() != 6 || f[0] != MAGIC {
            return Err(format!(
                "record {n}: expected `{MAGIC} <kind> id=.. level=.. bytes=.. spec=..`"
            ));
        }
        if f[1] != "task" && f[1] != "key" {
            return Err(format!("record {n}: unknown kind {:?}", f[1]));
        }
        let id = match f[2].strip_prefix("id=") {
            Some(h) if h.len() == 16 && h.bytes().all(|b| b.is_ascii_hexdigit()) => h,
            _ => return Err(format!("record {n}: bad id {:?}", f[2])),
        };
        let Some(level) = f[3].strip_prefix("level=").and_then(dec) else {
            return Err(format!("record {n}: bad level {:?}", f[3]));
        };
        let Some(len) = f[4].strip_prefix("bytes=").and_then(dec) else {
            return Err(format!("record {n}: bad byte count {:?}", f[4]));
        };
        let spec = match f[5].strip_prefix("spec=") {
            Some(s) if !s.is_empty() => s,
            _ => return Err(format!("record {n}: bad spec {:?}", f[5])),
        };
        let start = at + eol + 1;
        let end = match start.checked_add(len) {
            Some(e) if e <= text.len() => e,
            _ => {
                return Err(format!(
                    "record {n}: truncated body (the header claims {len} bytes)"
                ));
            }
        };
        let Some(body) = text.get(start..end) else {
            return Err(format!("record {n}: the body ends inside a character"));
        };
        out.push(Rec { kind: f[1], id, level, spec, body });
        at = end;
    }
}
fn decode_rec(r: &Rec<'_>) -> Result<Value, String> {
    let pipe = Pipeline::parse(r.spec)?;
    let v = pipe.decode(r.body)?;
    if v.get("kind").and_then(Value::as_str) != Some(r.kind) {
        return Err(format!("the {} record holds a different kind of value", r.kind));
    }
    if v.get("id").and_then(Value::as_str) != Some(r.id) {
        return Err(format!("the {} record holds another instance than its header says", r.kind));
    }
    Ok(v)
}
fn pipeline_for(id: u64, c: &Codec) -> Pipeline {
    match c {
        Codec::Fixed(p) => p.clone(),
        Codec::Random => Pipeline::random(&mut Rng::new(mix(id, CODEC))),
    }
}
fn key_pipe() -> Pipeline {
    Pipeline::new(Format::JsonPretty, Vec::new(), Armor::Raw, 0)
        .expect("jsonp|raw is a legal pipeline")
}
fn records(p: &Problem, level: u32, o: &Opts, with_key: bool) -> Result<String, String> {
    let pipe = pipeline_for(p.seed, &o.codec);
    let body = pipe.encode(&render::task(p)).map_err(|e| format!("task: {e}"))?;
    let mut out = frame("task", p.seed, level, &pipe.spec(), &body);
    if with_key {
        let kp = key_pipe();
        let kb = kp.encode(&render::key(p)).map_err(|e| format!("key: {e}"))?;
        out.push_str(&frame("key", p.seed, level, &kp.spec(), &kb));
    }
    Ok(out)
}
fn check_one(p: &mut Problem, level: u32, o: &Opts) -> Result<(), String> {
    p.verify()?;
    p.verify_scores()?;
    let text = records(p, level, o, true)?;
    let recs = unframe(&text)?;
    if recs.len() != 2 {
        return Err(format!("expected 2 records, read {}", recs.len()));
    }
    let want = [render::task(p), render::key(p)];
    for (r, w) in recs.iter().zip(want.iter()) {
        if r.id != render::hex(p.seed).as_str() || r.level != level as usize {
            return Err(format!("the {} record's header disagrees with the instance", r.kind));
        }
        let back = decode_rec(r)?;
        if !back.same(w) {
            return Err(format!("the {} record changed in transit through {}", r.kind, r.spec));
        }
    }
    Ok(())
}
fn rebuild(p: &Problem, level: u32, o: &Opts, cap: Option<usize>) -> Result<(), String> {
    let q = make(p.seed, level, &o.set, cap)?;
    if q.seed != p.seed {
        return Err("the seed does not rebuild the same instance".to_string());
    }
    if !render::task(&q).same(&render::task(p)) || !render::key(&q).same(&render::key(p)) {
        return Err("a rebuilt instance differs".to_string());
    }
    Ok(())
}
fn round_trips(n: usize, seed: u64) -> Result<(), String> {
    let mut rng = Rng::new(mix(seed, 0x7E57));
    for i in 0..n {
        let pipe = Pipeline::random(&mut rng);
        let depth = rng.below(5);
        let v = random_value(&mut rng, depth);
        let text = pipe.encode(&v).map_err(|e| format!("round trip {i}, {pipe}: {e}"))?;
        let back = pipe.decode(&text).map_err(|e| format!("round trip {i}, {pipe}: {e}"))?;
        if !back.same(&v) {
            return Err(format!("round trip {i}, {pipe}: the value changed"));
        }
    }
    Ok(())
}
fn put(s: &str) {
    if let Err(e) = io::stdout().write_all(s.as_bytes()) {
        if e.kind() != io::ErrorKind::BrokenPipe {
            eprintln!("ptgrs: cannot write to standard output: {e}");
            std::process::exit(1);
        }
        std::process::exit(0);
    }
}
fn fail(msg: &str) -> i32 {
    eprintln!("ptgrs: {msg}");
    1
}
fn read_stdin() -> Result<String, String> {
    let mut s = String::new();
    io::stdin()
        .read_to_string(&mut s)
        .map_err(|e| format!("cannot read standard input: {e}"))?;
    Ok(s)
}
fn json_line(v: &Value) -> Result<String, String> {
    let b = Format::Json.encode(v)?;
    String::from_utf8(b).map_err(|_| "JSON output is not UTF-8".to_string())
}
fn cmd_make(o: &Opts, single: bool) -> i32 {
    let base = base_seed(o, None);
    let levels = o.level.unwrap_or((3, 3));
    let n = if single { 1 } else { o.count.unwrap_or(10) };
    let mut bad = 0usize;
    for i in 0..n {
        let level = level_for(levels, i);
        let want = seed_for(base, i, single);
        let mut p = match make(want, level, &o.set, None) {
            Ok(p) => p,
            Err(e) => {
                eprintln!("ptgrs: instance {} (level {level}): {e}", i + 1);
                bad += 1;
                continue;
            }
        };
        if p.seed != want {
            eprintln!("ptgrs: seed 0x{want:016x} did not build; using 0x{:016x}", p.seed);
        }
        if let Err(e) = p.verify() {
            eprintln!(
                "ptgrs: instance {} ({}) failed verification: {e}",
                i + 1,
                render::hex(p.seed)
            );
            bad += 1;
            continue;
        }
        match records(&p, level, o, o.key) {
            Ok(text) => {
                put(&text);
                eprintln!("ptgrs: {}", render::summary(&p));
            }
            Err(e) => {
                eprintln!("ptgrs: instance {} ({}): {e}", i + 1, render::hex(p.seed));
                bad += 1;
            }
        }
    }
    if bad > 0 { 1 } else { 0 }
}
fn cmd_check(o: &Opts, deep: bool) -> i32 {
    let base = base_seed(o, if deep { Some(SELFTEST_SEED) } else { None });
    let levels = o.level.unwrap_or(if deep { (1, 6) } else { (3, 3) });
    let n = o.count.unwrap_or(if deep { 6 } else { 10 });
    let cap = if deep { Some(24) } else { None };
    let (mut ok, mut bad, mut skipped) = (0usize, 0usize, 0usize);
    for i in 0..n {
        let level = level_for(levels, i);
        let seed = seed_for(base, i, false);
        let mut p = match make(seed, level, &o.set, cap) {
            Ok(p) => p,
            Err(e) => {
                put(&format!("skip  {} level {level}: {e}\n", render::hex(seed)));
                skipped += 1;
                continue;
            }
        };
        let mut res = check_one(&mut p, level, o);
        if deep && res.is_ok() {
            res = rebuild(&p, level, o, cap);
        }
        match res {
            Ok(()) => {
                ok += 1;
                put(&format!("ok    {}\n", render::summary(&p)));
            }
            Err(e) => {
                bad += 1;
                put(&format!("FAIL  {} level {level}: {e}\n", render::hex(p.seed)));
            }
        }
    }
    if deep {
        match round_trips(300, base) {
            Ok(()) => put("ok    300 random value x pipeline round trips\n"),
            Err(e) => {
                bad += 1;
                put(&format!("FAIL  {e}\n"));
            }
        }
    }
    put(&format!("{ok} ok, {bad} failed, {skipped} skipped\n"));
    if bad > 0 || ok == 0 { 1 } else { 0 }
}
fn cmd_encode(pipe: &Pipeline) -> i32 {
    let input = match read_stdin() {
        Ok(s) => s,
        Err(e) => return fail(&e),
    };
    let v = match Format::Json.decode(input.as_bytes()) {
        Ok(v) => v,
        Err(e) => return fail(&format!("standard input is not JSON: {e}")),
    };
    match pipe.encode(&v) {
        Ok(text) => {
            put(&format!("{text}\n"));
            0
        }
        Err(e) => fail(&e),
    }
}
fn cmd_decode(pipe: Option<&Pipeline>) -> i32 {
    let input = match read_stdin() {
        Ok(s) => s,
        Err(e) => return fail(&e),
    };
    if let Some(p) = pipe {
        return match p.decode(&input).and_then(|v| json_line(&v)) {
            Ok(line) => {
                put(&format!("{line}\n"));
                0
            }
            Err(e) => fail(&e),
        };
    }
    let recs = match unframe(&input) {
        Ok(r) => r,
        Err(e) => return fail(&e),
    };
    if recs.is_empty() {
        return fail("no records on standard input");
    }
    let mut bad = 0usize;
    for (i, r) in recs.iter().enumerate() {
        match decode_rec(r).and_then(|v| json_line(&v)) {
            Ok(line) => put(&format!("{line}\n")),
            Err(e) => {
                eprintln!("ptgrs: record {}: {e}", i + 1);
                bad += 1;
            }
        }
    }
    if bad > 0 { 1 } else { 0 }
}
fn cmd_random(o: &Opts) -> i32 {
    let mut rng = Rng::new(base_seed(o, None));
    for _ in 0..o.count.unwrap_or(10) {
        put(&format!("{}\n", Pipeline::random(&mut rng)));
    }
    0
}
fn help() -> String {
    let formats: Vec<&str> = Format::ALL.iter().map(|f| f.name()).collect();
    let armors: Vec<&str> = Armor::ALL.iter().map(|a| a.name()).collect();
    let mut s = String::from(
        "ptgrs: synthetic creative-abduction tasks from stochastic rewriting worlds\n\nUSAGE\n",
    );
    for (name, _, about) in COMMANDS.iter() {
        s.push_str(&format!("  ptgrs {name:<12} [OPTIONS]  {about}\n"));
    }
    s.push_str("  ptgrs codec encode SPEC     JSON on stdin, encoded text on stdout\n");
    s.push_str("  ptgrs codec decode [SPEC]   text on stdin, one JSON line per value; without SPEC a stream of records\n");
    s.push_str("  ptgrs help\n\nOPTIONS\n");
    for (name, arg, about) in OPTIONS.iter() {
        let left = if arg.is_empty() { format!("--{name}") } else { format!("--{name} {arg}") };
        let users: Vec<&str> =
            COMMANDS.iter().filter(|c| c.1.contains(name)).map(|c| c.0).collect();
        s.push_str(&format!("  {left:<22}{about} [{}]\n", users.join(", ")));
    }
    s.push_str(&format!(
        "\nSPEC  format|stage|...|armor[@wrap]\n  format  {}\n  stage   rle, lzss, delta, rev, mtf, xor:<1 to 16 hex digits>\n  armor   {}\n  wrap    1 to {} columns; left out = one line\n  raw armor: text formats only, no stages, no wrap\n  example: cbor|delta|lzss|xor:00000000000000ff|z85@76\n",
        formats.join(", "),
        armors.join(", "),
        pipeline::MAX_WRAP
    ));
    s.push_str("\nRECORDS\n  #ptgrs <task|key> id=<16 hex> level=<n> bytes=<n> spec=<SPEC>\n  <exactly n bytes of encoded text>\n  <blank line>\n  The id is the seed that built the instance: `gen --seed 0x<id> --level <n>`, with the same\n  --set flags, rewrites the record byte for byte. `verify` takes the same --seed, --level,\n  --codec and --set as `batch` and checks the instances `batch` would write.\n");
    s.push_str("\nPARAMETERS  (--set K=V; the level gives the defaults)\n");
    for &(name, lo, hi, about) in KNOBS.iter() {
        s.push_str(&format!("  {name:<12}{:<14}{about}\n", format!("{lo}..={hi}")));
    }
    s.push_str("\nEXAMPLES\n  ptgrs batch --seed 7 --level 2-5 --count 20 --key > tasks.txt\n  ptgrs verify --seed 7 --level 2-5 --count 20\n  ptgrs gen --seed 0x1f --level 4 --codec 'cbor|lzss|b64@76' --set trials=24\n  ptgrs codec decode < tasks.txt\n\nEXIT STATUS  0 success, 1 failure (or nothing could be built), 2 usage error\n");
    s
}
fn run(cmd: Cmd) -> i32 {
    match cmd {
        Cmd::Help => {
            put(&help());
            0
        }
        Cmd::Gen(o) => cmd_make(&o, true),
        Cmd::Batch(o) => cmd_make(&o, false),
        Cmd::Verify(o) => cmd_check(&o, false),
        Cmd::SelfTest(o) => cmd_check(&o, true),
        Cmd::Random(o) => cmd_random(&o),
        Cmd::Encode(p) => cmd_encode(&p),
        Cmd::Decode(p) => cmd_decode(p.as_ref()),
    }
}
fn main() {
    let args: Vec<String> = std::env::args_os()
        .skip(1)
        .map(|a| a.to_string_lossy().into_owned())
        .collect();
    let code = match parse(&args) {
        Ok(cmd) => run(cmd),
        Err(e) => {
            eprintln!("ptgrs: {e}\ntry `ptgrs help`");
            2
        }
    };
    let _ = io::stdout().flush();
    std::process::exit(code);
}
